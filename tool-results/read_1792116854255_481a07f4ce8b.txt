    87→/** Store-themed button honoring the store's buttonStyle token. */
    88→function SfButton({
    89→  store,
    90→  children,
    91→  onClick,
    92→  outline = false,
    93→  small = false,
    94→  className,
    95→}: {
    96→  store: SfStore;
    97→  children: React.ReactNode;
    98→  onClick?: () => void;
    99→  outline?: boolean;
   100→  small?: boolean;
   101→  className?: string;
   102→}) {
   103→  const style = store.theme.buttonStyle ?? "solid";
   104→  const pill = style === "pill";
   105→  const base = cn(
   106→    "inline-flex items-center justify-center gap-1.5 font-semibold transition-all min-h-[44px] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2",
   107→    small ? "px-3.5 text-xs" : "px-5 py-2 text-sm",
   108→    pill ? "rounded-full" : "rounded-[var(--sf-radius)]",
   109→    className
   110→  );
   111→  if (outline || style === "outline") {
   112→    return (
   113→      <button type="button" onClick={onClick} className={cn(base, "border-2 bg-transparent hover:opacity-80")} style={{ borderColor: "var(--sf-primary)", color: "var(--sf-primary)" }}>
   114→        {children}
   115→      </button>
   116→    );
   117→  }
   118→  return (
   119→    <button type="button" onClick={onClick} className={cn(base, "text-white shadow-sm hover:opacity-90")} style={{ backgroundColor: "var(--sf-primary)" }}>
   120→      {children}
   121→    </button>
   122→  );
   123→}

   129→function resolveSfLink(url: string, store: SfStore, ctx: ShopperCtx) {
   130→  const u = (url ?? "").trim();
   131→  if (!u || u === "#") return;
   132→  if (u.startsWith("#")) {
   133→    const el = document.getElementById(u.slice(1));
   134→    el?.scrollIntoView({ behavior: "smooth", block: "start" });
   135→    return;
   136→  }
   137→  if (u.startsWith("/p/")) {
   138→    ctx.nav({ name: "store-page", slug: store.slug, pageSlug: u.slice(3) });
   139→    return;
   140→  }
   141→  if (u === "/" || u === "/home") {
   142→    ctx.nav({ name: "storefront", slug: store.slug });
   143→    return;
   144→  }
   145→  ctx.nav({ name: "search", q: "" });
   146→}

   161→function SectionRenderer({ sections, rctx }: { sections: Section[]; rctx: SfRenderCtx }) {
   162→  const firstGridId = sections.find((s) => s.type === "product_grid")?.id ?? null;
   163→  return (
   164→    <>
   165→      {sections.map((section) => {
   166→        const def = sectionDef(section.type);
   167→        const props = { ...(def?.defaultProps ?? {}), ...(section.props ?? {}) } as Record<string, unknown>;
   168→        // give the first product grid a stable id for #products anchors
   169→        const id = section.id === firstGridId ? rctx.productAnchorId || "products" : `sec-${section.id}`;
   170→        return (
   171→          <div key={section.id} id={id}>
   172→            <SectionErrorBoundary type={section.type}>
   173→              <SectionBody type={section.type} props={props} rctx={rctx} />
   174→            </SectionErrorBoundary>
   175→          </div>
   176→        );
   177→      })}
   178→    </>
   179→  );
   180→}

   182→/** A broken section must never take the whole storefront down. */
   183→class SectionErrorBoundary extends React.Component<{ type: string; children: React.ReactNode }, { hasError: boolean }> {
   184→  state = { hasError: false };
   185→  static getDerivedStateFromError() {
   186→    return { hasError: true };
   187→  }
   188→  render() {
   189→    if (this.state.hasError) return <UnsupportedSection type={this.props.type} />;
   190→    return this.props.children;
   191→  }
   192→}

   204→    case "product_grid":
   205→      return <ProductGridSection props={props} rctx={rctx} />;

   207→      // No public collections endpoint exists in this sandbox (only the
   208→      // seller-scoped /api/store/collections) — hide gracefully per spec.

   210→    case "rich_text":
   211→      return <RichText props={props} />;

   213→      return <ImageWithText props={props} rctx={rctx} />;
   214→    case "testimonials":
   215→      return <Testimonials props={props} />;
   216→    case "faq":
   217→      return <FaqSection props={props} />;

   219→      return <NewsletterSection props={props} rctx={rctx} />;
   220→    case "countdown":
   221→      return <CountdownSection props={props} />;

   244→function str(props: Record<string, unknown>, key: string, fallback = ""): string {
   245→  const v = props[key];
   246→  return typeof v === "string" ? v : v == null ? fallback : String(v);
   247→}
   248→function num(props: Record<string, unknown>, key: string, fallback: number): number {
   249→  const v = Number(props[key]);
   250→  return Number.isFinite(v) ? v : fallback;
   251→}
   252→function bool(props: Record<string, unknown>, key: string, fallback: boolean): boolean {
   253→  const v = props[key];
   254→  if (typeof v === "boolean") return v;
   255→  if (v === "true") return true;
   256→  if (v === "false") return false;
   257→  return fallback;
   258→}
   259→function list<T = Record<string, unknown>>(props: Record<string, unknown>, key: string): T[] {
   260→  const v = props[key];
   261→  if (Array.isArray(v)) return v as T[];
   262→  if (typeof v === "string" && v.trim()) {
   263→    try {
   264→      const parsed = JSON.parse(v);
   265→      return Array.isArray(parsed) ? (parsed as T[]) : [];
   266→    } catch {
   267→      return [];
   268→    }
   269→  }
   270→  return [];
   271→}

   441→function ProductGridSection({ props, rctx }: { props: Record<string, unknown>; rctx: SfRenderCtx }) {
   442→  const { store, sfSearch } = rctx;
   443→  const nav = rctx.nav.nav; // ShopperCtx → nav function
   444→  const source = str(props, "source", "latest");
   445→  const collectionSlug = str(props, "collectionSlug");
   446→  const limit = num(props, "limit", 8);
   447→  const showPrice = bool(props, "showPrice", true);
   448→  const showViewAll = bool(props, "showViewAll", true);
   449→  const columns = String(num(props, "columns", 4));
   450→  const [products, setProducts] = useState<ProductCardT[] | null>(null);
   451→  const [failed, setFailed] = useState(false);
   452→  const [expanded, setExpanded] = useState(false);
   453→
   454→  useEffect(() => {
   455→    let alive = true;
   456→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   457→    setProducts(null);
   458→    const params = new URLSearchParams({ source, limit: String(expanded ? 50 : limit) });
   459→    if (collectionSlug) params.set("collection", collectionSlug);
   460→    if (sfSearch.trim()) params.set("search", sfSearch.trim());
   461→    fetchJson<ProductCardT[]>(`/api/sf/${encodeURIComponent(store.slug)}/products?${params.toString()}`)
   462→      .then((d) => alive && setProducts(Array.isArray(d) ? d : []))
   463→      .catch(() => alive && setFailed(true));
   464→    return () => {
   465→      alive = false;
   466→    };
   467→  }, [store.slug, source, collectionSlug, limit, expanded, sfSearch]);
   468→
   469→  const title = str(props, "title");
   470→  const colClass =
   471→    columns === "2" ? "grid-cols-2" : columns === "3" ? "grid-cols-2 sm:grid-cols-3" : "grid-cols-2 sm:grid-cols-3 lg:grid-cols-4";
   472→
   473→  return (
   474→    <section className="mx-auto max-w-7xl px-4 py-10" style={{ fontFamily: "var(--sf-font-body)" }}>
   475→      {(title || sfSearch) && (
   476→        <div className="flex items-center justify-between gap-3 pb-5">
   477→          <h2 className="text-xl sm:text-2xl font-extrabold tracking-tight" style={{ fontFamily: "var(--sf-font-heading)" }}>
   478→            {sfSearch.trim() ? `Results for “${sfSearch.trim()}”` : title}
   479→          </h2>
   480→          {showViewAll && products && products.length > 0 ? (
   481→            <button
   482→              type="button"
   483→              onClick={() => setExpanded((e) => !e)}
   484→              className="inline-flex min-h-[44px] items-center gap-1 text-sm font-semibold transition-opacity hover:opacity-75"
   485→              style={{ color: "var(--sf-primary)" }}
   486→            >
   487→              {expanded ? "Show less" : "View all"} <ArrowRight className={cn("h-4 w-4 transition-transform", expanded && "rotate-90")} />
   488→            </button>
   489→          ) : null}
   490→        </div>
   491→      )}
   492→      {failed ? (

   779→// --- 12. countdown ---------------------------------------------------------------------
   780→
   781→function CountdownSection({ props }: { props: Record<string, unknown> }) {
   782→  const heading = str(props, "heading");
   783→  const endsRaw = str(props, "endsAt");
   784→  const style = str(props, "style", "banner");
   785→  const [now, setNow] = useState(() => Date.now());
   786→
   787→  const endsAt = useMemo(() => {
   788→    if (!endsRaw.trim()) return null;
   789→    const d = new Date(endsRaw);
   790→    return Number.isNaN(d.getTime()) ? null : d;
   791→  }, [endsRaw]);
   792→
   793→  useEffect(() => {
   794→    if (!endsAt) return;
   795→    const t = setInterval(() => setNow(Date.now()), 1000);
   796→    return () => clearInterval(t);
   797→  }, [endsAt]);
   798→
   799→  // hide gracefully when empty or already past
   800→  if (!endsAt || endsAt.getTime() <= now) return null;
   801→
   802→  const diff = Math.max(0, endsAt.getTime() - now);
   803→  const d = Math.floor(diff / 86400000);
   804→  const h = Math.floor((diff % 86400000) / 3600000);
   805→  const m = Math.floor((diff % 3600000) / 60000);
   806→  const s = Math.floor((diff % 60000) / 1000);
   807→  const cells: [number, string][] = [
   808→    [d, "days"],
   809→    [h, "hrs"],
   810→    [m, "min"],
   811→    [s, "sec"],
   812→  ];
   813→
   814→  if (style === "minimal") {
   815→    return (
   816→      <div className="mx-auto max-w-7xl px-4 py-4 text-center text-sm font-semibold" style={{ fontFamily: "var(--sf-font-heading)" }}>
   817→        {heading ? <span className="mr-2">{heading}</span> : null}
   818→        <span className="tabular-nums font-extrabold" style={{ color: "var(--sf-primary)" }}>
   819→          {d}d {h}h {m}m {s}s
   820→        </span>
   821→      </div>
   822→    );
   823→  }
   824→
   825→  return (
   826→    <div className="px-4 py-6 text-center text-white" style={{ backgroundColor: "var(--sf-primary)", fontFamily: "var(--sf-font-heading)" }}>
   827→      {heading ? <p className="pb-3 text-sm font-bold uppercase tracking-widest">{heading}</p> : null}
   828→      <div className="flex justify-center gap-3" role="timer" aria-label="Time remaining">
   829→        {cells.map(([v, label]) => (
   830→          <div key={label} className="rounded-lg bg-white/15 px-3 py-2 backdrop-blur min-w-[64px]">
   831→            <div className="text-2xl font-extrabold tabular-nums">{String(v).padStart(2, "0")}</div>
   832→            <div className="text-[10px] uppercase tracking-wider text-white/80">{label}</div>
   833→          </div>
   834→        ))}
   835→      </div>
   836→    </div>
   837→  );
   838→}

  1042→export default function StorefrontView({ ctx, slug }: { ctx: ShopperCtx; slug: string }) {
  1043→  const [data, setData] = useState<SfHomePayload | null>(null);
  1044→  const [notFound, setNotFound] = useState(false);
  1045→  const [sfSearch, setSfSearch] = useState("");
  1046→
  1047→  useEffect(() => {
  1048→    let alive = true;
  1049→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
  1050→    setData(null);
  1051→    setNotFound(false);
  1052→    setSfSearch("");
  1053→    fetchJson<SfHomePayload>(`/api/sf/${encodeURIComponent(slug)}`)
  1054→      .then((d) => alive && setData(d))
  1055→      .catch(() => alive && setNotFound(true));
  1056→    return () => {
  1057→      alive = false;
  1058→    };
  1059→  }, [slug]);
  1060→
  1061→  // SEO: store meta from real records (Phase 9)
  1062→  useEffect(() => {
  1063→    if (!data) return;
  1064→    const { store } = data;
  1065→    setMeta(
  1066→      `${store.name} — Official Store`,
  1067→      store.tagline ?? store.description ?? null,
  1068→      `/?sf=${store.slug}`
  1069→    );
  1070→    return () => {
  1071→      setMeta("SHOPMESH — Multi-vendor Marketplace");
  1072→    };
  1073→  }, [data]);
  1074→
  1075→  if (notFound) return <SfUnavailable onBack={() => ctx.nav({ name: "market" })} />;
  1076→  if (!data) return <SfLoading />;
  1077→
  1078→  const { store, page } = data;
  1079→  const productAnchorId = page.sections.find((s) => s.type === "product_grid")?.id ?? "products";
  1080→
  1081→  return (
  1082→    <StorefrontChrome store={store} title={page.title} ctx={ctx}>
  1083→      <SectionRenderer sections={page.sections} rctx={{ store, nav: ctx, openCart: ctx.openCart, sfSearch, onSearch: setSfSearch, productAnchorId }} />
  1084→    </StorefrontChrome>
  1085→  );
  1086→}
