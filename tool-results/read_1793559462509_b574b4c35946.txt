   123→   123→/**
   124→   124→ * Verify signature (caller may pre-compute it, as simulate does), log the raw
   125→   125→ * webhook, then process idempotently. Throws ApiError with proper status codes.
   126→   126→ *
   127→   127→ * RACE SAFETY (why claim-first): the success paths' FIRST write inside the
   128→   128→ * Serializable transaction is a CONDITIONAL updateMany (status flip guarded by
   129→   129→ * the pre-status). Two concurrent deliveries of the same event race on that
   130→   130→ * row lock; the loser matches zero rows, writes nothing and returns the
   131→   131→ * duplicate result — money movement, order transitions and notifications can
   132→   132→ * only ever run for the winner. The ledger dedupKeys (FIN-1) backstop this at
   133→   133→ * the posting layer, and the notification dedupKeys backstop the mail log.
   134→   134→ */
   135→   135→export async function processPaymentWebhook(input: {
   136→   136→  provider: string;
   137→   137→  body: { providerRef?: string; eventType?: string; amountMinor?: number; ts?: number; sig?: string; reason?: string; paymentId?: string };
   138→   138→  rawBody: string;
   139→   139→  headerSignature?: string | null;
   140→   140→  /** Phase 1 (E3): INTERNAL-ONLY recovery mode. Set exclusively by the
   141→   141→   *  authorized admin payout-ops sweep when re-feeding an event whose
   142→   142→   *  ORIGINAL delivery was signature-verified and persisted
   143→   143→   *  (PaymentWebhookLog.signatureValid = true). The HMAC is STILL fully
   144→   144→   *  verified against the stored payload — recovery NEVER forges or skips
   145→   145→   *  signature verification — only the ±10-minute freshness window is
   146→   146→   *  waived, because the event's age is precisely WHY it needs recovery.
   147→   147→   *  External webhook routes never set this flag. */
   148→   148→  opts?: { recovery?: boolean };
   149→   149→}): Promise<ProcessResult> {
   150→   150→  // (2b.2) Provider sandbox enforcement FIRST — fail closed. resolvePspProvider
   151→   151→  // falls back to the mock provider for unknown ids, so in production an
   152→   152→  // unregistered provider id can never sneak through: mock is sandboxOnly → 403.
   153→   153→  const psp = resolvePspProvider(input.provider);
   154→   154→  if (psp.sandboxOnly && process.env.NODE_ENV === "production") {
   155→   155→    throw new ApiError(403, "PROVIDER_SANDBOX_ONLY", `Provider "${input.provider}" is sandbox-only and cannot process webhooks in production`);
   156→   156→  }
   157→   157→
   158→   158→  const providerRef = String(input.body.providerRef ?? "");
   159→   159→  const eventType = String(input.body.eventType ?? "");
   160→   160→  const amountMinor = Math.round(Number(input.body.amountMinor ?? 0));
   161→   161→  const ts = Math.round(Number(input.body.ts ?? 0));
   162→   162→  if (!providerRef || !eventType) {
   163→   163→    throw new ApiError(400, "BAD_REQUEST", "providerRef and eventType are required");
   164→   164→  }
   165→   165→
   166→   166→  // (2b.1) Timing-safe compare + freshness. signatureValid in the log means
   167→   167→  // "authenticated as a LIVE event": correct MAC AND inside the replay window —
   168→   168→  // a stale-but-correctly-signed delivery is recorded as signatureValid=false
   169→   169→  // (replay protection is part of webhook authentication semantics) and
   170→   170→  // rejected with STALE_EVENT below, AFTER the row is logged.
   171→   171→  // Phase 1 (E3): in recovery mode the row records the HMAC result; the
   172→   172→  // freshness of the ORIGINAL receipt is what authorizes reprocessing.
   173→   173→  const providedSig = input.headerSignature ?? (input.body.sig ? String(input.body.sig) : null);
   174→   174→  const expected = expectedSignature(providerRef, eventType, amountMinor, ts);
   175→   175→  const sigMatches = !!providedSig && safeEqual(providedSig, expected);
   176→   176→  const fresh = Math.abs(Date.now() - ts) <= WEBHOOK_TS_WINDOW_MS;
   177→   177→  const recovery = input.opts?.recovery === true;
   178→   178→  const signatureValid = sigMatches && (fresh || recovery);
   179→   179→
   180→   180→  // OPTION A first: the ref may be a multi-seller payment group.
   181→   181→  const group = await db.paymentGroup.findUnique({
   182→   182→    where: { ref: providerRef },
   183→   183→    include: {
   184→   184→      orders: {
   185→   185→        orderBy: { createdAt: "asc" },
   186→   186→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   187→   187→      },
   188→   188→    },
   189→   189→  });
   190→   190→
   191→   191→  const intent = group ? null : await db.paymentIntent.findUnique({ where: { providerRef } });
   192→   192→
   193→   193→  // ALWAYS log the webhook first (even for invalid signature / unknown ref) —
   194→   194→  // invalid/stale rows stay in the log with their signatureValid flag.
   195→   195→  const log = await db.paymentWebhookLog.create({
   196→   196→    data: {
   197→   197→      provider: input.provider,
   198→   198→      providerRef,
   199→   199→      eventType,
   200→   200→      payloadJson: input.rawBody,
   201→   201→      signatureValid,
   202→   202→      processed: false,
   203→   203→      orderId: group?.orders[0]?.id ?? intent?.orderId ?? null,
   204→   204→    },
   205→   205→  });
   206→   206→
   207→   207→  if (!sigMatches) {
   208→   208→    throw new ApiError(400, "BAD_SIGNATURE", "Webhook signature verification failed");
   209→   209→  }
   210→   210→  if (!fresh && !recovery) {
   211→   211→    throw new ApiError(400, "STALE_EVENT", `Webhook timestamp is outside the ±${WEBHOOK_TS_WINDOW_MS / 60_000}-minute acceptance window (replay protection)`);
   212→   212→  }
   213→   213→
   214→   214→  // ------------------------------------------------------------------ FIN-2b: inbound payout events
   215→   215→  if (eventType === "payout.succeeded" || eventType === "payout.failed") {
   216→   216→    return processPayoutEvent({ provider: input.provider, providerRef, eventType, amountMinor, reason: input.body.reason ?? null, logId: log.id });
   217→   217→  }
   218→   218→
   219→   219→  // ------------------------------------------------------------------ OPTION A group path
   220→   220→  if (group) {
   221→   221→    // LATE PAYMENT AFTER EXPIRY — defined recovery policy (F3 remediation):
   222→   222→    // the expiry sweep has already claimed this group, transitioned its orders
   223→   223→    // to a terminal unpaid state and RESTOCKED the items. Fulfilling now would
   224→   224→    // sell unreserved stock; accepting the money without goods would steal.
   225→   225→    // The ONLY safe response is a definitive refusal: no money moves, no order
   226→   226→    // is resurrected, the PSP sees a permanent failure (mock PSP surfaces it
   227→   227→    // to the buyer as an expired payment window). Never silently fulfill.
   228→   228→    //
   229→   229→    // DURABLE REFUND OBLIGATION (mission: final operational gaps): a REAL PSP
   230→   230→    // has already CAPTURED the buyer's funds when this verified event arrives —
   231→   231→    // the 409 below protects the order/inventory transition, it does not return
   232→   232→    // those funds. Record the obligation BEFORE refusing so a crash between the
   233→   233→    // two leaves at worst "case exists, webhook answered 5xx" → PSP redelivery
   234→   234→    // converges on the unique (provider, providerRef) row. Re-reservation is
   235→   235→    // deliberately NOT attempted (business rule: released stock may be re-sold;
   236→   236→    // fulfilling a stale payment without buyer re-consent is not legitimate).
   237→   237→    if (eventType === "payment.succeeded" && group.status === "expired") {
   238→   238→      await recordLatePaymentObligation({
   239→   239→        provider: input.provider,
   240→   240→        providerRef,
   241→   241→        providerPaymentId: input.body.paymentId ?? null,
   242→   242→        amountMinor,
   243→   243→        groupTotalMinor: group.totalMinor,
   244→   244→        currency: group.currency,
   245→   245→      });
   246→   246→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   247→   247→      throw new ApiError(409, "GROUP_EXPIRED", `Payment group ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   248→   248→    }
   249→   249→    if (eventType === "payment.failed") {
   250→   250→      // No money moves on failure — a single conditional update is fine outside
   251→   251→      // a Serializable tx (an already-succeeded group ignores the failed event).
   252→   252→      if (group.status === "succeeded") {
   253→   253→        await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   254→   254→        return {
   255→   255→          ok: true,
   256→   256→          duplicate: true,
   257→   257→          eventType,
   258→   258→          orderNumber: group.orders[0]?.orderNumber ?? "",
   259→   259→          orderNumbers: group.orders.map((o) => o.orderNumber),
   260→   260→          orderStatus: "paid",
   261→   261→          intentStatus: "succeeded",
   262→   262→          isGroup: true,
   263→   263→        };
   264→   264→      }
   265→   265→      await db.paymentGroup.update({ where: { id: group.id }, data: { status: "failed" } });
   266→   266→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   267→   267→      return {
   268→   268→        ok: true,
   269→   269→        duplicate: false,
   270→   270→        eventType,
   271→   271→        orderNumber: group.orders[0]?.orderNumber ?? "",
   272→   272→        orderNumbers: group.orders.map((o) => o.orderNumber),
   273→   273→        orderStatus: group.orders[0]?.status ?? "payment_pending",
   274→   274→        intentStatus: "failed",
   275→   275→        isGroup: true,
   276→   276→      };
   277→   277→    }
   278→   278→
   279→   279→    // Unknown-event + amount checks BEFORE the claim (unchanged order) — a
   280→   280→    // malformed delivery must never even attempt the status flip.
   281→   281→    if (eventType !== "payment.succeeded") {
   282→   282→      throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   283→   283→    }
   284→   284→
   285→   285→    if (amountMinor !== group.totalMinor) {
   286→   286→      throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match payment group ${group.totalMinor}`);
   287→   287→    }
   288→   288→
   289→   289→    const settings = await getCommissionSettings();
   290→   290→    const paidAt = new Date();
   291→   291→
   292→   292→    // SUCCESS PATH — one Serializable transaction around claim + money.
   293→   293→    // The conditional claim is the FIRST write: count 0 ⇒ another delivery
   294→   294→    // already succeeded ⇒ nothing this tx wrote (the claim was the only
   295→   295→    // intended write) ⇒ return the duplicate sentinel; committing an empty tx
   296→   296→    // is equivalent to rollback-by-return.
   297→   297→    //
   298→   298→    // F1 REMEDIATION (measured): concurrent same-seller confirmations used to
   299→   299→    // mass-abort here (P2034 SSI conflicts on the wallet-cache row — 0/15 in
   300→   300→    // the measured storm). Two fixes, both evidence-driven:
   301→   301→    //   1. the wallet-cache refresh moved OUT of this tx (post-commit,
   302→   302→    //      best-effort) — the wallet row is no longer in this tx's conflict
   303→   303→    //      domain; payout authorization never trusted the cache anyway;
   304→   304→    //   2. bounded P2034 retry of the WHOLE transaction (2 retries, jittered
   305→   305→    //      backoff). SSI conflicts are transient by definition; the claim's
   306→   306→    //      conditional update + ledger dedupKeys make a full retry safe (no
   307→   307→    //      double-posting is possible). PSP calls stay outside — nothing
   308→   308→    //      non-transactional is ever retried.
   309→   309→    const walletIdsTouched = new Set<string>();
   310→   310→    // F1 [Measured]: notification/audit fan-out (notify, dispatchStoreWebhooks,
   311→   311→    // logActivity, shopper nudge) used to run INSIDE the Serializable tx — but
   312→   312→    // on the GLOBAL db client, i.e. on ADDITIONAL pool connections while the tx
   313→   313→    // holds its own. Under concurrent confirmations that multiplies connection
   314→   314→    // demand inside the tx window (15 webhooks × ≥2 connections vs pool of 10)
   315→   315→    // → measured P2024 connection-starvation + P2028 (dev.log). Money + order
   316→   316→    // state stay transactional; the non-transactional side effects fire
   317→   317→    // POST-COMMIT. dedupKeys make a redelivery unable to multiply mail; a
   318→   318→    // crash in the post-commit window can only delay/lose OBSERVABILITY rows,
   319→   319→    // never financial state — the correct trade for a bounded money tx.
   320→   320→    const postedOrders: { order: (typeof group.orders)[number]; sellerEmail: string | null }[] = [];
   321→   321→    let outcome: { claimed: boolean } | undefined;
   322→   322→    // C1.1 (Phase 2.2): set when the Serializable tx ABORTED because a member
   323→   323→    // order had already left `payment_pending` (expiry sweep released it) — the
   324→   324→    // group claim rolled back, the late-payment obligation path takes over.
   325→   325→    let ordersChanged = false;
   326→   326→    for (let attempt = 1; ; attempt++) {
   327→   327→      try {
   328→   328→        outcome = await db.$transaction(
   329→   329→      async (tx) => {
   330→   330→        // Bounded-failure policy (measured): a row lock held by another session
   331→   331→        // stalls statements past Prisma's interactive-tx timer (measured 120 s
   332→   332→        // = the server's 2-min statement_timeout). Session-scoped 10 s bound,
   333→   333→        // self-reverting at commit/rollback — lock contention fails FAST, the
   334→   334→        // PSP redelivers, and no pool connection is burned for 2 minutes.
   335→   335→        await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   336→   336→        const claim = await tx.paymentGroup.updateMany({
   337→   337→          where: { id: group.id, status: { notIn: ["succeeded", "expired"] } },
   338→   338→          data: { status: "succeeded", paidAt },
   339→   339→        });
   340→   340→        if (claim.count === 0) return { claimed: false as const };
   341→   341→
   342→   342→        // C1.1 (Phase 2.2) — per-order state verification INSIDE the claim tx:
   343→   343→        // re-read every member order and require `payment_pending`. A late
   344→   344→        // webhook must NEVER resurrect an order the expiry sweep already
   345→   345→        // cancelled/restocked (double-sell). Mismatch ⇒ THROW so the group
   346→   346→        // claim ROLLS BACK (a released order must not sit in a "succeeded"
   347→   347→        // group) and the caller records the durable refund obligation.
   348→   348→        // (Mirrors the legacy single-intent guard below; the group path
   349→   349→        // previously lacked it.)
   350→   350→        for (const order of group.orders) {
   351→   351→          const fresh = await tx.order.findUnique({ where: { id: order.id }, select: { status: true } });
   352→   352→          if (!fresh || fresh.status !== "payment_pending") {
   353→   353→            throw new OrderStateAbort(
   354→   354→              `Order ${order.orderNumber} is "${fresh?.status ?? "missing"}" — released before payment confirmed`
   355→   355→            );
   356→   356→          }
   357→   357→        }
   358→   358→
   359→   359→        for (const order of group.orders) {
   360→   360→          await transitionOrder({
   361→   361→            orderId: order.id,
   362→   362→            toStatus: "paid",
   363→   363→            actor: "system",
   364→   364→            note: `Payment ${providerRef} succeeded via ${group.provider} (group payment)`,
   365→   365→            timestamps: { paidAt },
   366→   366→            tx,
   367→   367→          });
   368→   368→
   369→   369→          // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   370→   370→          const seller = order.store.seller;
   371→   371→          const wallet = seller?.wallet;
   372→   372→          if (seller && wallet) {
   373→   373→            const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;

   753→   753→      await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   754→   754→      await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   755→   755→      await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: intent.provider, totalMinor: intent.amountMinor } });
   756→   756→    } catch (e) {
   757→   757→      console.error(`[post-commit] fan-out failed for order ${order.orderNumber} (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   758→   758→    }
   759→   759→  }
   760→   760→
   761→   761→  // F1 — post-commit wallet-cache refresh (same rationale as the group path).
   762→   762→  for (const wid of walletIdsTouched) {
   763→   763→    try {
   764→   764→      await refreshWalletCache(wid);
   765→   765→    } catch (e) {
   766→   766→      console.error(`[wallet-cache] post-commit refresh failed for ${wid} (benign, self-heals):`, e instanceof Error ? e.message : e);
   767→   767→    }
   768→   768→  }
   769→   769→
   770→   770→  // T2 auto-settle (see group path) — after commit, idempotent recipe.
   771→   771→  if (await autoSettleEnabled()) {
   772→   772→    await postGatewaySettlement({ provider: intent.provider, amountMinor: intent.amountMinor, ref: providerRef, occurredAt: paidAt });
   773→   773→  }
   774→   774→
   775→   775→  return {
   776→   776→    ok: true,
   777→   777→    duplicate: false,
   778→   778→    eventType,
   779→   779→    orderNumber: (await db.order.findUnique({ where: { id: intent.orderId } }))?.orderNumber ?? "",
   780→   780→    orderStatus: "paid",
   781→   781→    intentStatus: "succeeded",
   782→   782→  };
   783→   783→}
   784→   784→
   785→   785→/**
   786→   786→ * FIN-2b (2b.4) — inbound PAYOUT lifecycle events from the PSP.
   787→   787→ *
   788→   788→ * The provider confirms (or fails) a bank transfer we initiated. providerRef
   789→   789→ * resolution: try Payout.referenceNo first (the canonical platform reference —
   790→   790→ * what the mock PSP and payout-simulate echo); fall back to Payout.providerRef
   791→   791→ * (the transfer ref an admin recorded at mark-paid — some providers echo that
   792→   792→ * instead). Both lookups are documented PSP integration surface.
   793→   793→ *
   794→   794→ * Same claim-first race safety as the payment paths: assertTransition guards
   795→   795→ * the semantics, the conditional updateMany serializes concurrent deliveries,
   796→   796→ * and T4/T4f post inside the same Serializable tx as the status flip.
   797→   797→ */
   798→   798→async function processPayoutEvent(args: {
   799→   799→  provider: string;
   800→   800→  providerRef: string;
   801→   801→  eventType: "payout.succeeded" | "payout.failed";
   802→   802→  amountMinor: number;
   803→   803→  reason: string | null;
   804→   804→  logId: string;
   805→   805→}): Promise<ProcessResult> {
   806→   806→  const payout =
   807→   807→    (await db.payout.findUnique({ where: { referenceNo: args.providerRef } })) ??
   808→   808→    (await db.payout.findFirst({ where: { providerRef: args.providerRef } }));
   809→   809→  if (!payout) {
   810→   810→    throw new ApiError(404, "PAYOUT_NOT_FOUND", `No payout matches providerRef ${args.providerRef}`);
   811→   811→  }
   812→   812→
   813→   813→  const toStatus = args.eventType === "payout.succeeded" ? "paid" : "failed";
   814→   814→  const dedupKey = `${toStatus === "paid" ? "payout.paid" : "payout.failed"}:${payout.id}`;
   815→   815→
   816→   816→  // Idempotent replay: the provider re-delivered an event we already applied
   817→   817→  // (e.g. payout.succeeded for an already-paid payout) → ok + conflict flag,
   818→   818→  // never a raw 409 at the provider. Decision is recorded (processed=true).
   819→   819→  if (payout.status === toStatus) {
   820→   820→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   821→   821→    return {
   822→   822→      ok: true,
   823→   823→      duplicate: true,
   824→   824→      payoutConflict: true,
   825→   825→      eventType: args.eventType,
   826→   826→      orderNumber: "",
   827→   827→      orderStatus: payout.status,
   828→   828→      intentStatus: "",
   829→   829→      payoutStatus: payout.status,
   830→   830→    };
   831→   831→  }
   832→   832→
   833→   833→  // State machine guard (cheap, no locks): an illegal transition (e.g.
   834→   834→  // requested→paid — the money was never reserved by T3) is a hard 409. The
   835→   835→  // decision is recorded on the webhook row before throwing.
   836→   836→  assertTransition(payout.status, toStatus);
   837→   837→
   838→   838→  // Amount guard BEFORE the claim (mirror of the payment paths).
   839→   839→  if (args.amountMinor !== payout.amountMinor) {
   840→   840→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${args.amountMinor} does not match payout ${payout.amountMinor}`);
   841→   841→  }
   842→   842→
   843→   843→  const seller = await db.seller.findUnique({ where: { id: payout.sellerId } });
   844→   844→  if (!seller) throw new ApiError(404, "NOT_FOUND", "Seller for payout not found");
   845→   845→
   846→   846→  const at = new Date();
   847→   847→  // T4f reason: use the provider-supplied reason when present.
   848→   848→  const reason = args.reason ?? "provider reported failure";
   849→   849→  const subject = toStatus === "paid" ? `Payout ${payout.referenceNo} on the way` : `Payout ${payout.referenceNo} failed`;
   850→   850→
   851→   851→  // Phase 1 (E2): the SAME hardening the payment paths already carry —
   852→   852→  //   • withSerializableTxRetry: bounded P2034 retry of the WHOLE tx with
   853→   853→  //     jittered exponential backoff, and 503 PAYMENT_TX_RETRY_EXHAUSTED on
   854→   854→  //     exhaustion (the PSP retries later; the delivery stays
   855→   855→  //     processed=false for this sweep) instead of a raw 500.
   856→   856→  //   • SET LOCAL statement_timeout = 10 s bounds lock waits.
   857→   857→  //   • notification/audit fan-out moved POST-COMMIT (the F1 anti-pattern:
   858→   858→  //     extra round-trips on the global client INSIDE the Serializable tx
   859→   859→  //     multiply pool pressure and hot-row contention).
   860→   860→  //   • timeout/maxWait (10 s/10 s, env-tunable) replace Prisma's 5 s defaults.
   861→   861→  // Retry-safety: the claim is the first write and the only gate; a retried tx
   862→   862→  // whose claim matches 0 rows is a no-op. Ledger dedupKeys backstop T4/T4f.
   863→   863→  const outcome = await withSerializableTxRetry(async (tx) => {
   864→   864→    await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   865→   865→    // Conditional claim — expectedFrom is "processing" per the state machine
   866→   866→    // (only processing payouts are awaiting provider confirmation). count 0
   867→   867→    // ⇒ lost race / concurrent admin action ⇒ empty-tx-return (rollback-by-
   868→   868→    // return: the claim was the only intended write and it wrote nothing).
   869→   869→    const claim = await tx.payout.updateMany({
   870→   870→      where: { id: payout.id, status: "processing" },
   871→   871→      data: { status: toStatus },
   872→   872→    });
   873→   873→    if (claim.count === 0) return { claimed: false as const };
   874→   874→
   875→   875→    if (toStatus === "paid") {
   876→   876→      await tx.payout.update({ where: { id: payout.id }, data: { paidAt: at } });
   877→   877→      // T4 — clearing → bank; refreshWalletCache runs inside the recipe (tx).
   878→   878→      await postPayoutConfirmed({
   879→   879→        walletId: payout.walletId,
   880→   880→        payoutId: payout.id,
   881→   881→        referenceNo: payout.referenceNo,
   882→   882→        amountMinor: payout.amountMinor,
   883→   883→        tx,
   884→   884→      });
   885→   885→    } else {
   886→   886→      await tx.payout.update({ where: { id: payout.id }, data: { failReason: reason } });
   887→   887→      // T4f — reversal of T3 (clearing → back to seller payable).
   888→   888→      await postPayoutFailed({
   889→   889→        walletId: payout.walletId,
   890→   890→        payoutId: payout.id,
   891→   891→        referenceNo: payout.referenceNo,
   892→   892→        amountMinor: payout.amountMinor,
   893→   893→        reason,
   894→   894→        tx,
   895→   895→      });
   896→   896→    }
   897→   897→
   898→   898→    return { claimed: true as const };
   899→   899→  });
   900→   900→
   901→   901→  if (outcome.claimed) {
   902→   902→    // Phase 1 (E2): post-commit fan-out — seller notifications + audit live
   903→   903→    // OUTSIDE the Serializable tx. Stable per-event dedupKeys make a redelivery
   904→   904→    // converge; a failure here is logged and non-financial (never rolls back
   905→   905→    // committed ledger work).
   906→   906→    try {
   907→   907→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "email", recipient: seller.email, subject, dedupKey });
   908→   908→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "inapp", recipient: seller.email, subject, dedupKey });
   909→   909→      await logActivity({
   910→   910→        actorType: "system",
   911→   911→        action: toStatus === "paid" ? "payout.provider_confirmed" : "payout.provider_failed",
   912→   912→        entityType: "payout",
   913→   913→        entityId: payout.id,
   914→   914→        meta: { providerRef: args.providerRef, referenceNo: payout.referenceNo, amountMinor: payout.amountMinor },
   915→   915→      });
   916→   916→    } catch (e) {
   917→   917→      console.error(`[post-commit] payout ${payout.referenceNo} fan-out failed (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   918→   918→    }
   919→   919→  }
   920→   920→
   921→   921→  const freshPayout = await db.payout.findUnique({ where: { id: payout.id } });
   922→   922→  const payoutStatus = freshPayout?.status ?? payout.status;
   923→   923→
   924→   924→  if (!outcome.claimed) {
   925→   925→    // Lost the race (e.g. an admin transitioned the payout concurrently) —
   926→   926→    // answer ok with payoutConflict so the provider's retry converges; never
   927→   927→    // throw raw 5xx at a PSP. Decision recorded (processed=true).
   928→   928→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   929→   929→    return {
   930→   930→      ok: true,
   931→   931→      duplicate: true,
   932→   932→      payoutConflict: true,
   933→   933→      eventType: args.eventType,
   934→   934→      orderNumber: "",
   935→   935→      orderStatus: payoutStatus,
   936→   936→      intentStatus: "",
   937→   937→      payoutStatus,
   938→   938→    };
   939→   939→  }
   940→   940→
   941→   941→  await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   942→   942→  return {
   943→   943→    ok: true,
   944→   944→    duplicate: false,
   945→   945→    eventType: args.eventType,
   946→   946→    orderNumber: "",
   947→   947→    orderStatus: payoutStatus,
   948→   948→    intentStatus: "",
   949→   949→    payoutStatus,
   950→   950→  };
   951→   951→}
   952→   952→
   953→   953→export function webhookPayloadJson(body: unknown): string {
   954→   954→  return toJson(body);
   955→   955→}
