    68→    68→export function statusTone(status?: string | null): Tone {

    72→    72→  if (RED_STATUSES.has(s)) return "red";

    74→    74→  return "gray";

   133→   133→export function fmtDT(iso?: string | Date | null): string {
   134→   134→  if (!iso) return "—";
   135→   135→  const d = typeof iso === "string" ? new Date(iso) : iso;
   136→   136→  if (Number.isNaN(d.getTime())) return "—";
   137→   137→  return d.toLocaleString("en-MY", {
   138→   138→    day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit", hour12: false,
   139→   139→  });
   140→   140→}
   141→   141→
   142→   142→export function fmtDate(iso?: string | Date | null): string {
   143→   143→  if (!iso) return "—";
   144→   144→  const d = typeof iso === "string" ? new Date(iso) : iso;
   145→   145→  if (Number.isNaN(d.getTime())) return "—";
   146→   146→  return d.toLocaleDateString("en-MY", { day: "2-digit", month: "short", year: "numeric" });
   147→   147→}
   148→   148→
   149→   149→export function timeAgo(iso?: string | Date | null): string {
   150→   150→  if (!iso) return "—";
   151→   151→  const d = typeof iso === "string" ? new Date(iso) : iso;
   152→   152→  if (Number.isNaN(d.getTime())) return "—";
   153→   153→  const secs = Math.floor((Date.now() - d.getTime()) / 1000);
   154→   154→  if (secs < 60) return "just now";
   155→   155→  if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   156→   156→  if (secs < 86400) return `${Math.floor(secs / 3600)}h ago`;
   157→   157→  if (secs < 86400 * 30) return `${Math.floor(secs / 86400)}d ago`;
   158→   158→  return fmtDate(d);
   159→   159→}

   641→   641→/* -------------------------------------------------------------- breakdown */
   642→   642→
   643→   643→const BREAKDOWN_MONEY_KEYS = new Set([
   644→   644→  "base", "percent_part", "fixed_fee_allocated", "discount_allocated", "shipping_allocated",
   645→   645→]);
   646→   646→
   647→   647→function breakdownValue(key: string, v: unknown): string | null {
   648→   648→  if (v === null || v === undefined) return null;
   649→   649→  if (key === "flags") {
   650→   650→    const arr = Array.isArray(v) ? v.map(String) : [];
   651→   651→    return arr.length ? arr.join(", ") : null;
   652→   652→  }
   653→   653→  if (BREAKDOWN_MONEY_KEYS.has(key)) return formatMinor(Number(v));
   654→   654→  if (typeof v === "number") return key === "percent" ? `${v}%` : String(v);
   655→   655→  if (typeof v === "string") return v;
   656→   656→  return JSON.stringify(v);
   657→   657→}

   659→   659→/** Renders a commission breakdownJson payload as key:value chips. */
   660→   660→export function BreakdownChips({ json, className }: { json?: string | null; className?: string }) {
   661→   661→  const data = parseJsonSafe<Record<string, unknown>>(json, {});
   662→   662→  const entries = Object.entries(data).filter(([k, v]) => breakdownValue(k, v) !== null);
   663→   663→  if (entries.length === 0) return <span className="text-xs text-muted-foreground">No breakdown snapshot</span>;
   664→   664→  return (
   665→   665→    <div className={cn("flex flex-wrap gap-1.5", className)}>
   666→   666→      {entries.map(([k, v]) => {
   667→   667→        const val = breakdownValue(k, v) as string;

   674→   674→              highlight ? TONE_CLASS.teal : "bg-muted/60 text-muted-foreground"

   676→   676→          >

   687→   687→
   688→   688→function flatObj(v: unknown): Record<string, unknown> {
   689→   689→  return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : {};
   690→   690→}
   691→   691→
   692→   692→function diffVal(key: string, v: unknown): string {
   693→   693→  if (v === null || v === undefined || v === "") return "—";
   694→   694→  if (typeof v === "object") return JSON.stringify(v);
   695→   695→  if (key.endsWith("Minor")) return formatMinor(Number(v));
   696→   696→  if (typeof v === "boolean") return v ? "true" : "false";
   697→   697→  return String(v);
   698→   698→}

   700→   700→/** Field-by-field before/after viewer for audit log rows. */
   701→   701→export function DiffViewer({ beforeJson, afterJson }: { beforeJson?: string | null; afterJson?: string | null }) {
   702→   702→  const before = flatObj(parseJsonSafe<unknown>(beforeJson, {}));
   703→   703→  const after = flatObj(parseJsonSafe<unknown>(afterJson, {}));
   704→   704→  const keys = Array.from(new Set([...Object.keys(before), ...Object.keys(after)])).sort();
   705→   705→  if (keys.length === 0) {
   706→   706→    return <p className="text-xs text-muted-foreground">No structured before/after snapshot recorded for this entry.</p>;
   707→   707→  }
   708→   708→  return (
   709→   709→    <div className="overflow-hidden rounded-lg border">
   710→   710→      <Table>
   711→   711→        <TableHeader>
   712→   712→          <TableRow>
   713→   713→            <TableHead className="w-40">Field</TableHead>
   714→   714→            <TableHead>Before</TableHead>
   715→   715→            <TableHead>After</TableHead>
   716→   716→          </TableRow>
   717→   717→        </TableHeader>
   718→   718→        <TableBody>
   719→   719→          {keys.map((k) => {
   720→   720→            const b = k in before ? diffVal(k, before[k]) : null;
   721→   721→            const a = k in after ? diffVal(k, after[k]) : null;
   722→   722→            const changed = b !== a;
   723→   723→            const added = b === null;
   724→   724→            const removed = a === null;
   725→   725→            return (
   726→   726→              <TableRow key={k} className={cn(changed && !removed && "bg-amber-50/60 dark:bg-amber-950/20")}>
   727→   727→                <TableCell className="font-mono text-[11px] font-medium">{k}</TableCell>
   728→   728→                <TableCell className={cn("text-xs tabular-nums", removed && "text-red-600 line-through dark:text-red-400")}>
   729→   729→                  {b ?? "—"}
   730→   730→                </TableCell>
   731→   731→                <TableCell className={cn("text-xs tabular-nums", added && "font-medium text-emerald-700 dark:text-emerald-400")}>
   732→   732→                  {a ?? "—"}
   733→   733→                </TableCell>
   734→   734→              </TableRow>
   735→   735→            );
   736→   736→          })}
   737→   737→        </TableBody>
   738→   738→      </Table>
   739→   739→    </div>
   740→   740→  );
   741→   741→}
