    98→/// Phase 1 (C4): TOTP MFA enrollment for administrators.
    99→/// `secretEnc` = AES-256-GCM (iv:tag:ciphertext, base64 segments) encrypted
   100→/// with MFA_ENCRYPTION_KEY. `recoveryHashes` = JSON array of sha256(code)
   101→/// hex digests - one-time-use recovery codes, never stored in plaintext.
   102→/// `confirmedAt` null = enrollment started but not yet verified by a valid
   103→/// TOTP code; login enforcement only applies once confirmed.
   104→/// Phase 2.1: `lastUsedTotpCounter` = the TOTP time-step counter of the most
   105→/// recently ACCEPTED code — CAS-guarded monotonic update rejects any code
   106→/// from an already-used time step (replay protection, RFC 6238 §5.2). Null
   107→/// until the first accepted code.
   108→model AdminMfaEnrollment {

   115→  createdAt           DateTime  @default(now())
   116→  updatedAt           DateTime  @updatedAt
   117→  admin               AdminUser @relation(fields: [adminId], references: [id], onDelete: Cascade)
   118→}

   177→// FINAL CLOSURE (guest-order OTP + shopper email verification) — ONE challenge
   178→// table serves both purposes (no column was added to ShopperUser/Order on
   179→// purpose: a missing column would break every existing query pre-migration,
   180→// while a missing TABLE is tolerated fail-closed via the P2021 guard, exactly
   181→// like AdminMfaEnrollment):
   182→//   purpose "guest-order"   → subjectId = Order.id. Successful verify DELETES
   183→//     the row (CAS) — the issued order-access grant is the artifact. The
   184→//     knowledge pair (email + NM- order number) can now only REQUEST this
   185→//     challenge; it no longer authorizes any order data.
   186→//   purpose "shopper-email" → subjectId = ShopperUser.id. Row PERSISTS with
   187→//     verifiedAt set — the durable proof of email ownership that gates
   188→//     guest-order claiming in claimGuestStateForShopper().
   189→// Codes are stored ONLY as HMAC-SHA256(EMAIL_OTP_SECRET, "otp:v1:" + purpose +
   190→// ":" + subjectId + ":" + utcDayBucket + ":" + code) — never plaintext, never
   191→// logged. `attempts` survives re-issuance while a row is unexpired (repeated
   192→// issuance cannot reset brute-force accounting); it resets only after expiry.
   193→// No FK on purpose/subjectId: the subject is polymorphic by design.
   194→model EmailOtpChallenge {
   195→  id         String    @id @default(cuid())
   196→  purpose    String
   197→  subjectId  String
   198→  codeHash   String
   199→  attempts   Int       @default(0)
   200→  expiresAt  DateTime
   201→  verifiedAt DateTime?
   202→  createdAt  DateTime  @default(now())
   203→  updatedAt  DateTime  @updatedAt
   204→
   205→  @@unique([purpose, subjectId])
   206→}

   848→// LATE-PAYMENT RECONCILIATION (mission: final operational gaps, Phase 2).
   849→// A payment.succeeded webhook that arrives AFTER the expiry sweep released the
   850→// group's stock is refused with 409 GROUP_EXPIRED (money never enters the
   851→// ledger). For a REAL PSP the buyer's funds are already captured at the
   852→// provider — this table is the durable, traceable REFUND OBLIGATION for those
   853→// funds: who must be refunded, how much (the webhook amount = captured funds),
   854→// and where the provider interaction stands. It is an obligation tracker, NOT
   855→// a ledger posting — ledger postings remain exclusively under claim-first
   856→// payment success. Duplicate webhooks converge on ONE row per (provider, ref)
   857→// via the unique key; retries can never duplicate the obligation.
   858→model PaymentReconciliation {
   859→  id                String    @id @default(cuid())
   860→  provider          String
   861→  // PaymentGroup.ref — the PSP checkout session the late payment belongs to.

   864→  // MUST send it; the mock rails identify by providerRef).
   865→  providerPaymentId String?
   866→  // Amount captured at the provider per the verified webhook (this is what
   867→  // must be RETURNED — not the group total when they diverge; divergence is
   868→  // recorded in resolutionNote for operator verification).
   869→  amountMinor       Int
   870→  currency          String    @default("MYR")
   871→  reason            String    @default("late_payment_after_expiry")
   872→  // requested   — obligation recorded, no successful provider attempt yet
   873→  // processing   — attempt in flight (durable intent written pre-call)
   874→  // succeeded    — refund confirmed FROM the provider's authoritative record
   875→  // failed       — provider REJECTED the refund (permanent; operator item)
   876→  // unknown      — provider outcome pending/timed out (re-checked next tick)
   877→  status            String    @default("requested")
   878→  providerRefundId  String?
   879→  attempts          Int       @default(0)
   880→  lastAttemptAt     DateTime?
   881→  lastError         String?
   882→  resolutionNote    String?
   883→  createdAt         DateTime  @default(now())
   884→  updatedAt         DateTime  @updatedAt

   887→  @@index([status])
   888→  @@index([updatedAt])
