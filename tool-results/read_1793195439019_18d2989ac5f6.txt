   694→// LATE-PAYMENT RECONCILIATION (mission: final operational gaps, Phase 2).
   695→// A payment.succeeded webhook that arrives AFTER the expiry sweep released the
   696→// group's stock is refused with 409 GROUP_EXPIRED (money never enters the
   697→// ledger). For a REAL PSP the buyer's funds are already captured at the
   698→// provider — this table is the durable, traceable REFUND OBLIGATION for those
   699→// funds: who must be refunded, how much (the webhook amount = captured funds),
   700→// and where the provider interaction stands. It is an obligation tracker, NOT
   701→// a ledger posting — ledger postings remain exclusively under claim-first
   702→// payment success. Duplicate webhooks converge on ONE row per (provider, ref)
   703→// via the unique key; retries can never duplicate the obligation.
   704→model PaymentReconciliation {
   705→  id                String    @id @default(cuid())
   706→  provider          String
   707→  // PaymentGroup.ref — the PSP checkout session the late payment belongs to.
   708→  providerRef       String
   709→  // PSP's own payment object id when the event carries one (real adapters
   710→  // MUST send it; the mock rails identify by providerRef).
   711→  providerPaymentId String?
   712→  // Amount captured at the provider per the verified webhook (this is what
   713→  // must be RETURNED — not the group total when they diverge; divergence is
   714→  // recorded in resolutionNote for operator verification).
   715→  amountMinor       Int
   716→  currency          String    @default("MYR")
   717→  reason            String    @default("late_payment_after_expiry")
   718→  // requested   — obligation recorded, no successful provider attempt yet
   719→  // processing   — attempt in flight (durable intent written pre-call)
   720→  // succeeded    — refund confirmed FROM the provider's authoritative record
   721→  // failed       — provider REJECTED the refund (permanent; operator item)
   722→  // unknown      — provider outcome pending/timed out (re-checked next tick)
   723→  status            String    @default("requested")
   724→  providerRefundId  String?
   725→  attempts          Int       @default(0)
   726→  lastAttemptAt     DateTime?
   727→  lastError         String?
   728→  resolutionNote    String?
   729→  createdAt         DateTime  @default(now())
   730→  updatedAt         DateTime  @updatedAt
   731→
   732→  @@unique([provider, providerRef])
   733→  @@index([status])
   734→  @@index([updatedAt])
   735→}

  1320→  showFreeShippingBar     Boolean  @default(true)
  1321→  freeShipThresholdMinor  Int      @default(10000)

  1323→  upsellEnabled           Boolean  @default(false)

  1325→  upsellProductIdsJson    String   @default("[]")   // backend-only storage
