   204→   204→/** grams → human weight: 800 → "800 g", 2400 → "2.4 kg", 2000 → "2 kg". */
   205→   205→function fmtWeight(grams: number): string {
   206→   206→  if (!Number.isFinite(grams) || grams <= 0) return "—";
   207→   207→  if (grams < 1000) return `${grams} g`;
   208→   208→  return `${(grams / 1000).toFixed(2).replace(/\.?0+$/, "")} kg`;
   209→   209→}

   211→   211→/** ISO date → "Fri, 20 Sep". */
   212→   212→function fmtDay(iso: string): string {
   213→   213→  try {
   214→   214→    return format(new Date(iso), "EEE, d MMM");
   215→   215→  } catch {
   216→   216→    return iso;
   217→   217→  }
   218→   218→}

   220→   220→function fmtEddWindow(fromISO: string, toISO: string): string {
   221→   221→  const from = fmtDay(fromISO);
   222→   222→  const to = fmtDay(toISO);
   223→   223→  return from === to ? from : `${from} – ${to}`;
   224→   224→}

   373→   373→  const { data: order, loading, error, refetch } = useApi<OrderDetail>(`/api/store/orders/${orderId}`);
   374→   374→  const [shipOpen, setShipOpen] = useState(false);

   376→   376→  const [labelBusy, setLabelBusy] = useState(false);
   377→   377→
   378→   378→  if (loading) return <RowsSkeleton rows={6} />;
   379→   379→  if (error || !order) {
   380→   380→    return (
   381→   381→      <div className="space-y-3">
   382→   382→        <Button variant="ghost" onClick={onBack} className="gap-1.5"><ArrowLeft className="h-4 w-4" /> Back to orders</Button>
   383→   383→        <Card><CardContent className="p-6 text-sm text-rose-600">Could not load order{error ? `: ${error}` : ""}</CardContent></Card>
   384→   384→      </div>
   385→   385→    );
   386→   386→  }
   387→   387→
   388→   388→  const commission = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   389→   389→  const net = order.totalMinor - commission;
   390→   390→  const canFulfill = order.status === "paid" || order.status === "processing";
   391→   391→  const canShip = (order.status === "paid" || order.status === "processing") && !order.shipmentStatus;
   392→   392→  const canRefund = ["paid", "processing", "shipped", "delivered", "completed"].includes(order.status);
   393→   393→  // Platform-booked shipments only (manual AWBs get no generated label).
   394→   394→  const hasLabel = !!order.shipmentBookedAt && !!order.trackingNumber;
   395→   395→
   396→   396→  const fulfill = async () => {
   397→   397→    const ok = await runMutation(() => apiSend("POST", `/api/store/orders/${order.id}/fulfill`), `Order ${order.orderNumber} marked fulfilled`);
   398→   398→    if (ok) refetch();
   399→   399→  };
   400→   400→
   401→   401→  const downloadLabel = async () => {
   402→   402→    setLabelBusy(true);
   403→   403→    try {
   404→   404→      // A plain <a href> wouldn't carry the Bearer header — fetch the PDF as a
   405→   405→      // blob and open it (same auth pattern as every other seller API call).
   406→   406→      const res = await fetch(`/api/store/orders/${order.id}/label`, { cache: "no-store", headers: authHeaders() });
   407→   407→      if (!res.ok) throw new Error("Label is not available for this shipment");
   408→   408→      const blob = await res.blob();
   409→   409→      const url = URL.createObjectURL(blob);
   410→   410→      const win = window.open(url, "_blank", "noopener,noreferrer");
   411→   411→      if (!win) {
   412→   412→        // Popup blocked (sandboxed iframes) — fall back to a download click.
   413→   413→        const a = document.createElement("a");
   414→   414→        a.href = url;
   415→   415→        a.download = `AWB-${order.trackingNumber ?? order.orderNumber}.pdf`;
   416→   416→        document.body.appendChild(a);
   417→   417→        a.click();
   418→   418→        a.remove();
   419→   419→      }
   420→   420→      window.setTimeout(() => URL.revokeObjectURL(url), 60_000);
   421→   421→    } catch (err) {
   422→   422→      toast.error(err instanceof Error ? err.message : "Could not fetch the shipping label");
   423→   423→    } finally {
   424→   424→      setLabelBusy(false);
   425→   425→    }
   426→   426→  };
   427→   427→
   428→   428→  return (
   429→   429→    <div className="space-y-4">
   430→   430→      {/* print-only packing slip */}
   431→   431→      <PackingSlip order={order} />
   432→   432→
   433→   433→      <div className="flex flex-wrap items-center gap-2 print:hidden">
   434→   434→        <Button variant="ghost" onClick={onBack} className="gap-1.5">
   435→   435→          <ArrowLeft className="h-4 w-4" /> Orders
   436→   436→        </Button>
   437→   437→        <h2 className="text-lg font-semibold">{order.orderNumber}</h2>
   438→   438→        <StatusChip status={order.status} />
   439→   439→        <span className="text-xs text-muted-foreground">
   440→   440→          {order.channel} · created {fmtDateTime(order.createdAt)}
   441→   441→        </span>
   442→   442→        <Button variant="outline" size="sm" className="ml-auto gap-1.5" onClick={() => window.print()}>
   443→   443→          <Printer className="h-3.5 w-3.5" /> Print packing slip
   444→   444→        </Button>
   445→   445→      </div>

   833→   833→                  : "Book shipment"}
   834→   834→            </Button>
   835→   835→          </div>
   836→   836→        ) : null}
   837→   837→      </DialogContent>
   838→   838→    </Dialog>
   839→   839→  );
   840→   840→}

   844→   844→function RefundDialog({ open, onOpenChange, order, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; order: OrderDetail; onDone: () => void }) {
   845→   845→  const refunded = order.refunds.filter((r) => r.status !== "rejected").reduce((s, r) => s + r.amountMinor, 0);
   846→   846→  const refundable = Math.max(0, order.totalMinor - refunded);
   847→   847→  const [amount, setAmount] = useState<number>(refundable);
   848→   848→  const [reason, setReason] = useState("");
   849→   849→  const [restock, setRestock] = useState(true);
   850→   850→  const [busy, setBusy] = useState(false);
   851→   851→
   852→   852→  useEffect(() => {
   853→   853→    if (open) {
   854→   854→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   855→   855→      setAmount(Math.max(0, order.totalMinor - order.refunds.filter((r) => r.status !== "rejected").reduce((s, r) => s + r.amountMinor, 0)));
   856→   856→      setReason("");
   857→   857→      setRestock(true);
   858→   858→    }
   859→   859→  }, [open, order]);
   860→   860→
   861→   861→  const submit = async () => {
   862→   862→    if (amount <= 0 || amount > refundable) {
   863→   863→      toast.error(`Amount must be between RM 0 and ${formatMinor(refundable)}`);
   864→   864→      return;
   865→   865→    }
   866→   866→    setBusy(true);
   867→   867→    const ok = await runMutation(
   868→   868→      () => apiSend("POST", `/api/store/orders/${order.id}/refund-request`, { amountMinor: amount, reason: reason || null, restock }),
   869→   869→      "Refund requested — an admin will review it"
   870→   870→    );
   871→   871→    setBusy(false);
   872→   872→    if (ok) {
   873→   873→      onOpenChange(false);
   874→   874→      onDone();
   875→   875→    }
   876→   876→  };
   877→   877→
   878→   878→  return (
   879→   879→    <Dialog open={open} onOpenChange={onOpenChange}>
   880→   880→      <DialogContent className="max-w-md">
   881→   881→        <DialogHeader>
   882→   882→          <DialogTitle>Request refund — {order.orderNumber}</DialogTitle>
   883→   883→        </DialogHeader>
   884→   884→        <div className="space-y-3.5">
   885→   885→          <div className="space-y-1.5">
   886→   886→            <Label>Amount (RM)</Label>
   887→   887→            <MoneyInput valueMinor={amount} onChangeMinor={setAmount} />
   888→   888→            <p className="text-xs text-muted-foreground">Refundable balance: {formatMinor(refundable)}</p>
   889→   889→          </div>
   890→   890→          <div className="space-y-1.5">
   891→   891→            <Label htmlFor="refund-reason">Reason</Label>
   892→   892→            <Textarea id="refund-reason" rows={3} value={reason} onChange={(e) => setReason(e.target.value)} placeholder="Damaged in transit, item out of stock…" />
   893→   893→          </div>
   894→   894→          <div className="flex items-center justify-between rounded-xl border p-3">
   895→   895→            <div>
   896→   896→              <p className="text-sm font-medium">Restock items</p>
   897→   897→              <p className="text-xs text-muted-foreground">Return the items to inventory when the refund is processed.</p>
   898→   898→            </div>
   899→   899→            <Switch checked={restock} onCheckedChange={setRestock} aria-label="Restock items" />
   900→   900→          </div>
   901→   901→          <p className="text-xs text-muted-foreground">Requests are reviewed by SHOPMESH admins; the ledger reversal happens on approval.</p>
   902→   902→          <Button onClick={submit} disabled={busy} className="w-full min-h-[44px] gap-2">
   903→   903→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Undo2 className="h-4 w-4" />} Submit request
   904→   904→          </Button>
   905→   905→        </div>
   906→   906→      </DialogContent>
   907→   907→    </Dialog>
   908→   908→  );
   909→   909→}
