   412→          <DialogFooter>

   501→          <div className="overflow-x-auto">
   502→            {payoutsReq.error ? (
   503→              <ErrorState message={payoutsReq.error} onRetry={() => void payoutsReq.refetch()} />
   504→            ) : payoutsReq.loading ? (
   505→              <TableSkeleton rows={4} cols={6} />
   506→            ) : payouts.length === 0 ? (
   507→              <EmptyState title="No payouts in this batch" />
   508→            ) : (
   509→              <Table>
   510→                <TableHeader>
   511→                  <TableRow>
   512→                    <TableHead className="w-8" />
   513→                    <TableHead>Reference</TableHead>
   514→                    <TableHead>Seller</TableHead>
   515→                    <TableHead className="text-right">Amount</TableHead>
   516→                    <TableHead>Status</TableHead>
   517→                    <TableHead className="hidden md:table-cell">Method</TableHead>
   518→                    <TableHead className="hidden text-right lg:table-cell">Updated</TableHead>
   519→                  </TableRow>
   520→                </TableHeader>
   521→                <TableBody>
   522→                  {payouts.map((p) => (
   523→                    <Fragment key={p.id}>
   524→                      <TableRow className="cursor-pointer" onClick={() => setExpanded(expanded === p.id ? null : p.id)}>
   525→                        <TableCell>{expanded === p.id ? <ChevronDown className="h-4 w-4" /> : <ChevronRight className="h-4 w-4" />}</TableCell>
   526→                        <TableCell className="font-mono text-xs font-semibold">{p.referenceNo}</TableCell>
   527→                        <TableCell>
   528→                          <p className="font-medium">{p.sellerName ?? "—"}</p>
   529→                          <p className="text-xs text-muted-foreground">{p.sellerEmail ?? ""}</p>
   530→                        </TableCell>
   531→                        <TableCell className="text-right"><Money minor={p.amountMinor} bold /></TableCell>
   532→                        <TableCell>
   533→                          <StatusChip status={p.status} />
   534→                          {p.failReason ? <p className="mt-1 max-w-[180px] truncate text-[11px] text-red-600 dark:text-red-400">{p.failReason}</p> : null}
   535→                        </TableCell>
   536→                        <TableCell className="hidden text-sm capitalize md:table-cell">{p.method?.replace(/_/g, " ") ?? "—"}</TableCell>
   537→                        <TableCell className="hidden text-right text-xs text-muted-foreground lg:table-cell">{timeAgo(p.paidAt ?? p.createdAt)}</TableCell>
   538→                      </TableRow>
   539→                      {expanded === p.id ? (
   540→                        <TableRow>
   541→                          <TableCell colSpan={7} className="bg-muted/30">
   542→                            <PayoutExpand
   543→                              payout={p}
   544→                              onMarkPaid={() => setPayTarget(p)}
   545→                              onMarkFailed={() => setFailTarget(p)}
   546→                            />
   547→                          </TableCell>
   548→                        </TableRow>
   549→                      ) : null}
   550→                    </Fragment>
   551→                  ))}
   552→                </TableBody>
   553→              </Table>
   554→            )}
   555→          </div>
   556→        </CardContent>
   557→      </Card>
   558→
   559→      {/* mark paid */}
   560→      <MarkPaidDialog
   561→        payout={payTarget}
   562→        onClose={() => setPayTarget(null)}
   563→        onDone={() => {
   564→          setPayTarget(null);
   565→          done();
   566→        }}
   567→      />
   568→
   569→      {/* mark failed */}
   570→      <ReasonDialog
   571→        open={failTarget !== null}
   572→        onOpenChange={(o) => !o && setFailTarget(null)}
   573→        title="Mark payout as failed"
   574→        description={
   575→          failTarget
   576→            ? `${failTarget.referenceNo} · ${formatMinor(failTarget.amountMinor)} — the debit is reversed in the ledger (T4f) and funds return to the seller wallet.`
   577→            : ""
   578→        }
   579→        placeholder="Reason (e.g. bank account closed)…"
   580→        submitLabel="Mark failed"
   581→        destructive
   582→        onSubmit={(reason) => {
   583→          if (!failTarget) return;
   584→          apiSend(`/api/admin/payouts/${failTarget.id}/fail`, "POST", { reason })
   585→            .then(() => {
   586→              toast.success("Payout marked failed", { description: "Ledger reversal posted; funds returned to wallet." });
   587→              setFailTarget(null);
   588→              done();
   589→            })
   590→            .catch((e: unknown) => toast.error("Failed to update payout", { description: e instanceof Error ? e.message : String(e) }));
   591→        }}
   592→      />
   593→    </div>
   594→  );
   595→}

   597→function PayoutExpand({
   598→  payout,
   599→  onMarkPaid,
   600→  onMarkFailed,
   601→}: {
   602→  payout: PayoutRow;
   603→  onMarkPaid: () => void;
   604→  onMarkFailed: () => void;
   605→}) {
   606→  const st = parseJsonSafe<Statement>(payout.statementJson, {});
   607→  const lines = st.lines ?? [];
   608→  // Reconciliation audit: statement lines must sum EXACTLY to the net payout.
   609→  // Ledger-truth statements (reconciled: true) always do; legacy demo snapshots
   610→  // with decorated numbers fail this check and are flagged honestly.
   611→  const linesSum = lines.reduce((s, l) => s + l.amountMinor, 0);
   612→  const reconciles = lines.length > 0 && linesSum === payout.amountMinor;
   613→  return (
   614→    <div className="grid grid-cols-1 gap-4 p-3 lg:grid-cols-3">
   615→      <div className="lg:col-span-2">
   616→        <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   617→          Settlement statement{st.sellerName ? ` — ${st.sellerName}` : ""}
   618→        </p>
   619→        {lines.length === 0 ? (
   620→          <p className="text-xs text-muted-foreground">No statement lines recorded for this payout.</p>
   621→        ) : (
   622→          <div className="overflow-hidden rounded-lg border bg-card">
   623→            <div className={cn("flex items-center gap-1.5 border-b px-3 py-2 text-[11px] font-semibold", reconciles ? "bg-emerald-50 text-emerald-800 dark:bg-emerald-950/40 dark:text-emerald-300" : "bg-amber-50 text-amber-800 dark:bg-amber-950/40 dark:text-amber-300")}>
   624→              {reconciles ? (
   625→                <>
   626→                  <ShieldCheck className="h-3.5 w-3.5" /> Ledger-reconciled — every figure is aggregated from ledger lines and Σ(lines) = net payout
   627→                </>
   628→              ) : (
   629→                <>
   630→                  <AlertTriangle className="h-3.5 w-3.5" /> Not reconciled — Σ(lines) {formatMinor(linesSum)} ≠ payout {formatMinor(payout.amountMinor)} (legacy demo snapshot)
   631→                </>
   632→              )}
   633→            </div>
   634→            <Table>
   635→              <TableBody>
   636→                {lines.map((l, i) => (
   637→                  <TableRow key={i}>
   638→                    <TableCell className="text-sm">{l.label}</TableCell>
   639→                    <TableCell className="w-32 text-right"><Money minor={l.amountMinor} /></TableCell>
   640→                  </TableRow>
   641→                ))}
   642→                <TableRow className="border-t-2">
   643→                  <TableCell className="text-sm font-semibold">Net payout</TableCell>
   644→                  <TableCell className="w-32 text-right"><Money minor={payout.amountMinor} bold /></TableCell>
   645→                </TableRow>
   646→              </TableBody>
   647→            </Table>
   648→          </div>
   649→        )}
   650→        {st.orderCount != null || st.periodTo ? (
   651→          <p className="mt-2 text-[11px] text-muted-foreground">
   652→            {st.orderCount != null ? `${st.orderCount} paid order${st.orderCount === 1 ? "" : "s"}` : ""}
   653→            {st.periodFrom && st.periodTo ? ` · ${fmtDT(st.periodFrom)} → ${fmtDT(st.periodTo)}` : ""}
   654→          </p>
   655→        ) : null}
   656→      </div>
   657→
   658→      <div className="space-y-3">
   659→        <div>
   660→          <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank</p>
   661→          {payout.bankAccount ? (
   662→            <div className="mt-1.5 space-y-1 rounded-lg border bg-card p-3 text-sm">
   663→              <p className="flex items-center gap-1.5 font-medium">
   664→                <Landmark className="h-3.5 w-3.5 text-teal-600 dark:text-teal-400" /> {payout.bankAccount.bankName}
   665→              </p>
   666→              <p className="text-muted-foreground">{payout.bankAccount.holderName}</p>
   667→              <p className="font-mono text-xs">{payout.bankAccount.accountNumberMasked}</p>
   668→            </div>
   669→          ) : (
   670→            <p className="mt-1.5 text-xs text-muted-foreground">No bank account linked</p>
   671→          )}
   672→        </div>
   673→        {payout.providerRef ? (
   674→          <div>
   675→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank reference</p>
   676→            <p className="mt-1 font-mono text-xs">{payout.providerRef}</p>
   677→          </div>
   678→        ) : null}
   679→        {payout.status === "processing" ? (
   680→          <div className="flex flex-col gap-2">
   681→            <Button size="sm" className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700" onClick={onMarkPaid}>
   682→              Mark paid…
   683→            </Button>
   684→            <Button size="sm" variant="outline" className="min-h-[44px] border-red-200 text-red-600 hover:bg-red-50 dark:border-red-900 dark:hover:bg-red-950/40" onClick={onMarkFailed}>
   685→              Mark failed…
   686→            </Button>
   687→          </div>
   688→        ) : (
   689→          <p className={cn("text-xs", payout.status === "paid" ? "text-emerald-700 dark:text-emerald-400" : "text-muted-foreground")}>
   690→            {payout.status === "paid" ? `Paid ${fmtDT(payout.paidAt)}` : "No actions available"}
   691→          </p>
   692→        )}
   693→      </div>
   694→    </div>
   695→  );
   696→}

   698→function MarkPaidDialog({
   699→  payout,
   700→  onClose,
   701→  onDone,
   702→}: {
   703→  payout: PayoutRow | null;
   704→  onClose: () => void;
   705→  onDone: () => void;
   706→}) {
   707→  const [refNo, setRefNo] = useState("");
   708→  const [saving, setSaving] = useState(false);
   709→
   710→  useEffect(() => {
   711→    if (payout) setRefNo("");
   712→  }, [payout]);
   713→
   714→  const submit = async () => {
   715→    if (!payout || !refNo.trim()) return;
   716→    setSaving(true);
   717→    try {
   718→      await apiSend(`/api/admin/payouts/${payout.id}/mark-paid`, "POST", { referenceNo: refNo.trim() });
   719→      toast.success("Payout marked paid", {
   720→        description: `${payout.referenceNo} · bank ref ${refNo.trim()} — ledger T4 posted.`,
   721→      });
   722→      onDone();
   723→    } catch (e) {
   724→      toast.error("Mark paid failed", { description: e instanceof Error ? e.message : String(e) });
   725→    } finally {
   726→      setSaving(false);
   727→    }
   728→  };
   729→
   730→  return (
   731→    <Dialog open={payout !== null} onOpenChange={(o) => !o && onClose()}>
   732→      <DialogContent className="sm:max-w-md">
   733→        <DialogHeader>
   734→          <DialogTitle>Mark payout as paid</DialogTitle>
   735→          <DialogDescription>
   736→            {payout ? `${payout.referenceNo} · ${formatMinor(payout.amountMinor)} to ${payout.sellerName ?? "seller"}.` : ""}
   737→          </DialogDescription>
   738→        </DialogHeader>
   739→        <div className="space-y-1.5">
   740→          <Label htmlFor="payout-ref">Bank reference number (required)</Label>
   741→          <Input
   742→            id="payout-ref"
   743→            value={refNo}
   744→            onChange={(e) => setRefNo(e.target.value)}
   745→            placeholder="e.g. BK-20261008-8817"
   746→            className="min-h-[44px]"
   747→            autoFocus
   748→          />
   749→          <p className="text-[11px] text-muted-foreground">Recorded on the payout and posted to the ledger as payout_confirmed (T4).</p>
   750→        </div>
   751→        <DialogFooter>
   752→          <Button variant="outline" onClick={onClose} disabled={saving} className="min-h-[44px] sm:min-h-0">Cancel</Button>
   753→          <Button onClick={() => void submit()} disabled={saving || !refNo.trim()} className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700 sm:min-h-0">
   754→            {saving ? "Posting…" : "Confirm paid"}
   755→          </Button>
   756→        </DialogFooter>
   757→      </DialogContent>
   758→    </Dialog>
   759→  );
   760→}
