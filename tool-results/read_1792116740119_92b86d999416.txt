   180→// ---------------------------------------------------------------------------
   181→// data hook — profile + wallet + purchase counts + like/viewed counters
   182→// ---------------------------------------------------------------------------
   183→
   184→function useMeData(enabled: boolean) {
   185→  const [profile, setProfile] = useState<ProfileT | null>(null);
   186→  const [wallet, setWallet] = useState<WalletT | null>(null);
   187→  const [counts, setCounts] = useState<BucketCounts | null>(null);
   188→  const [likeCount, setLikeCount] = useState(0);
   189→  const [viewedCount, setViewedCount] = useState(0);
   190→  const [loading, setLoading] = useState(enabled);
   191→
   192→  const refresh = useCallback(async () => {
   193→    if (!enabled) return;
   194→    const [p, w, o, lk, vw] = await Promise.allSettled([
   195→      fetchJson<ProfileT>("/api/me/profile"),
   196→      fetchJson<WalletT>("/api/me/wallet"),
   197→      fetchJson<{ bucketCounts: BucketCounts; orders: OrderRowT[] }>("/api/me/orders?size=1"),
   198→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=LIKE&size=1"),
   199→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=VIEWED&size=1"),
   200→    ]);
   201→    if (p.status === "fulfilled") setProfile(p.value);
   202→    if (w.status === "fulfilled") {
   203→      // DORMANT SCAFFOLD — MESH FINANCE: while MESH_FINANCE_ENABLED is false
   204→      // the wallet API answers a `disabled` envelope (wallet/checkIn null).
   205→      // Normalize it into the UI shape so the kept scaffold cards render
   206→      // inert zeroes instead of crashing; the cards stay hidden behind the flag.
   207→      const raw = w.value as Partial<WalletT> & { disabled?: boolean };
   208→      setWallet({
   209→        wallet: raw.wallet ?? { balanceMinor: 0, coinBalance: 0, spayLaterLimitMinor: 0, spayLaterUsedMinor: 0, spayLaterAvailableMinor: 0, checkInStreak: 0 },
   210→        vouchers: raw.vouchers ?? { available: 0, used: 0 },
   211→        checkIn: raw.checkIn ?? { canClaim: false, nextClaimAt: null },
   212→      });
   213→    }
   214→    if (o.status === "fulfilled") setCounts(o.value.bucketCounts);
   215→    if (lk.status === "fulfilled") setLikeCount(lk.value.meta?.total ?? 0);
   216→    if (vw.status === "fulfilled") setViewedCount(vw.value.meta?.total ?? 0);
   217→    setLoading(false);
   218→  }, [enabled]);
   219→
   220→  useEffect(() => {
   221→    // eslint-disable-next-line react-hooks/set-state-in-effect -- initial dashboard data load (async fetch inside refresh)
   222→    void refresh();
   223→  }, [refresh]);
   224→
   225→  // Address CRUD (in this hub's sub-view or during checkout) must reflect on
   226→  // the dashboard teaser instantly — "N saved addresses" never goes stale.
   227→  useEffect(() => {
   228→    if (!enabled) return;
   229→    const sync = () => void refresh();
   230→    window.addEventListener("shopmesh:addresses-changed", sync);
   231→    return () => window.removeEventListener("shopmesh:addresses-changed", sync);
   232→  }, [enabled, refresh]);
   233→
   234→  return { profile, wallet, counts, likeCount, viewedCount, loading, refresh };
   235→}

   241→export default function MeHubView({ ctx, initialTab }: { ctx: ShopperCtx; initialTab?: string }) {
   242→  const meState = useApp((s) => s.me);
   243→  const openShopperAuth = useApp((s) => s.openShopperAuth);

   246→  const [purchaseTab, setPurchaseTab] = useState<string>(initialTab && initialTab.startsWith("bucket:") ? initialTab.slice(7) : "all");

   248→
   249→  if (!shopper) {
   250→    return (

   252→        <EmptyState
   253→          icon={User}
   254→          title="Your dashboard awaits"

   256→          action={
   257→            <Button onClick={() => openShopperAuth()} className="min-h-[44px] bg-brand-gradient-deep hover:brightness-110">
   258→              Sign in / Create account
   259→            </Button>
   260→          }
   261→        />
   262→      </div>
   263→    );
   264→  }
   265→
   266→  if (sub) {
   267→    const back = () => setSub(null);
   268→    switch (sub) {
   269→      case "purchases":
   270→        return <PurchasesView ctx={ctx} onBack={back} initialTab={purchaseTab} />;
   271→      case "wallet":
   272→        // DORMANT SCAFFOLD — MESH FINANCE: WalletView is kept intact for the
   273→        // roadmap; until MESH_FINANCE_ENABLED flips, this entry renders a
   274→        // coming-soon view instead of any stored-value UI.
   275→        return MESH_FINANCE_ENABLED ? (
   276→          <WalletView onBack={back} wallet={data.wallet} onRefresh={data.refresh} />
   277→        ) : (
   278→          <FinanceComingSoonView onBack={back} />
   279→        );
   280→      case "vouchers":
   281→        return <VouchersView onBack={back} onOpenMarket={() => ctx.nav({ name: "market" })} />;
   282→      case "likes":
   283→        return <ActivityGridView key="likes" type="LIKE" title="My Likes" onBack={back} ctx={ctx} />;
   284→      case "viewed":
   285→        return <ActivityGridView key="viewed" type="VIEWED" title="Recently Viewed" onBack={back} ctx={ctx} />;
   286→      case "addresses":
   287→        return <AddressesView onBack={back} />;
   288→    }
   289→  }
   290→
   291→  return (
   292→    <Dashboard
   293→      ctx={ctx}
   294→      data={data}
   295→      onSub={(s) => setSub(s)}
   296→      onPurchaseTab={(tab) => {
   297→        setPurchaseTab(tab);
   298→        setSub("purchases");
   299→      }}
   300→    />
   301→  );
   302→}

   608→function Avatar({ profile, size = "md" }: { profile: ProfileT | null; size?: "sm" | "md" | "lg" }) {
   609→  const px = size === "lg" ? "h-16 w-16 text-xl" : size === "md" ? "h-10 w-10 text-sm" : "h-8 w-8 text-xs";
   610→  const gradKey = profile?.avatarUrl?.startsWith("grad:") ? profile.avatarUrl : null;
   611→  const gradClass =
   612→    gradKey === "grad:amber"
   613→      ? "from-amber-400 to-orange-500"
   614→      : gradKey === "grad:rose"
   615→        ? "from-rose-400 to-pink-600"
   616→        : gradKey === "grad:violet"
   617→          ? "from-violet-400 to-purple-600"
   618→          : gradKey === "grad:emerald"
   619→            ? "from-emerald-400 to-teal-600"
   620→            : gradKey === "grad:slate"
   621→              ? "from-slate-400 to-slate-600"
   622→              : "from-teal-500 to-teal-800";
   623→  if (profile?.avatarUrl && !gradKey && /^https?:\/\/|^\//.test(profile.avatarUrl)) {
   624→    return <img src={profile.avatarUrl} alt={`${profile.fullName} avatar`} className={cn(px, "shrink-0 rounded-full object-cover ring-2 ring-white/70")} />;
   625→  }
   626→  const initial = (profile?.fullName ?? "?").trim().slice(0, 1).toUpperCase();
   627→  return (
   628→    <span className={cn(px, "flex shrink-0 items-center justify-center rounded-full bg-gradient-to-br font-extrabold text-white ring-2 ring-white/70", gradClass)} aria-hidden="true">
   629→      {initial}
   630→    </span>
   631→  );
   632→}

   833→function CoinCard({ wallet, onRefresh }: { wallet: WalletT | null; onRefresh: () => Promise<void> }) {
   834→  const [busy, setBusy] = useState(false);
   835→  const canClaim = wallet?.checkIn?.canClaim ?? false;
   836→
   837→  const claim = async () => {
   838→    setBusy(true);
   839→    try {
   840→      const res = await postJson<{ awarded: number; coinBalance: number; checkInStreak: number }>("/api/me/coins/claim", {});
   841→      toast.success(`+${res.awarded} coins claimed!`, { description: `Streak ${res.checkInStreak} day${res.checkInStreak === 1 ? "" : "s"} · balance ${res.coinBalance} coins` });
   842→      await onRefresh();
   843→    } catch (e) {
   844→      toast.error(e instanceof Error ? e.message : "Could not check in");
   845→    } finally {
   846→      setBusy(false);
   847→    }
   848→  };
   849→
   850→  return (
   851→    <div className="rounded-2xl border bg-card p-4 shadow-soft">
   852→      <div className="flex items-center gap-2 pb-1.5">

   858→        <Button
   859→          size="sm"
   860→          disabled={!canClaim || busy}
   861→          onClick={() => void claim()}
   862→          className={cn(
   863→            "relative h-7 min-h-[28px] px-2.5 text-[11px] font-bold",
