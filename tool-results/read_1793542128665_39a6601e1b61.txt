     2→// M09 COMMISSION ENGINE — full spec implementation (PART 3 M09)
     3→// Resolution priority: fee holiday > [scoped CommissionOverride rows and
     4→// direct field overrides interleaved by scope: product > category ancestor >
     5→// seller] > global default. Money math in integer minor units.
     6→// Commission snapshots are stored per order_item and NEVER recomputed.
     7→//
     8→// F-08 remediation: the scoped CommissionOverride table (Admin→Finance
     9→// "Commission Overrides") is resolved HERE, interleaved into the M09 chain at
    10→// its scope position — an active row within its validity window beats the
    11→// direct field at the SAME scope, and falls back to the next rule in the
    12→// chain otherwise (exactly what the admin UI promises). percent /
    13→// fixedFeeMinor / holdDays are resolved field-wise: a row that only sets
    14→// holdDays contributes its holdDays while the rate keeps falling through.
    15→// Multiple rows matching the same scope resolve deterministically:
    16→// earliest createdAt wins (id as final tie-break).
    17→//
    18→// ----------------------------------------------------------------------------
    19→// PHASE 2 / A2 — BATCHED rate resolution (re-land of F2-C1, adapted to the
    20→// F-08 override semantics). The original per-item `resolveRate` issued
    21→// ≈5 + 2·D sequential queries PER LINE (settings + seller + overrides +
    22→// holidays + product + one query per ancestor level, twice). `loadRateContext`
    23→// loads EVERYTHING a whole cart (all seller groups) needs in a FIXED handful
    24→// of round-trips (~6, independent of item count), and `resolveRateFromContext`
    25→// resolves each line purely in memory with byte-identical semantics. The
    26→// sequential resolver below remains available as `resolveRate` (single-item
    27→// convenience, now itself built on the batched loader); correctness parity is
    28→// proven by scripts/test-commission-parity.ts against a verbatim copy of the
    29→// pre-batch sequential resolver.

   103→const NONE = "__no_target__"; // sentinel that can never match a real cuid — see scopedOverrideRows note

   105→/**
   106→ * Load the FULL rate-resolution context for a set of lines in a FIXED number of
   107→ * round-trips: settings + sellers + products + holidays + seed categories in
   108→ * one parallel batch, then (rare) extra category rounds chasing parents to
   109→ * closure, then the scoped override rows for the exact closure. Item-count
   110→ * independent: a 1-line and a 50-line cart cost the same ~6 queries.
   111→ *
   112→ * `now` is shared by every consumer of the returned context — one checkout,
   113→ * one clock (the pre-batch code could drift a few ms across groups; the math
   114→ * is date-window sensitive only at exact holiday/override boundaries, where a
   115→ * single snapshot is strictly MORE deterministic).
   116→ */
   117→export async function loadRateContext(input: {
   118→  sellerIds: string[];
   119→  productIds?: string[];
   120→  categoryIds?: string[];
   121→  now?: Date;
   122→}): Promise<RateContext> {
   123→  const now = input.now ?? new Date();
   124→  const sellerIds = [...new Set(input.sellerIds.filter((s): s is string => typeof s === "string" && s.length > 0))];
   125→  const productIds = [...new Set((input.productIds ?? []).filter((p): p is string => typeof p === "string" && p.length > 0))];
   126→  const seedCategoryIds = [...new Set((input.categoryIds ?? []).filter((c): c is string => typeof c === "string" && c.length > 0))];
   127→
   128→  // Round 1 — five independent reads in ONE parallel batch. NOTE: the product
   129→  // read also returns `categoryId`, which is merged into the category seeds
   130→  // below — a caller that passes productIds but forgets categoryIds still gets
   131→  // a correct closure (defense against the silent-wrong-results footgun).
   132→  const [settings, sellerRows, productRows, holidays, seedCategories] = await Promise.all([
   133→    getCommissionSettings(),
   134→    sellerIds.length
   135→      ? db.seller.findMany({
   136→          where: { id: { in: sellerIds } },
   137→          select: { id: true, commissionOverridePercent: true, holdDaysOverride: true },
   138→        })
   139→      : [],
   140→    productIds.length
   141→      ? db.product.findMany({
   142→          where: { id: { in: productIds } },
   143→          select: { id: true, commissionPercentOverride: true, categoryId: true },
   144→        })
   145→      : [],
   146→    db.feeHoliday.findMany({
   147→      where: { isActive: true, startsAt: { lte: now }, endsAt: { gte: now } },
   148→      orderBy: { startsAt: "asc" },
   149→      select: { id: true, scope: true, scopeIds: true },
   150→    }),
   151→    seedCategoryIds.length
   152→      ? db.category.findMany({
   153→          where: { id: { in: seedCategoryIds } },
   154→          select: { id: true, parentId: true, commissionPercent: true },
   155→        })
   156→      : [],
   157→  ]);
   158→
   159→  const categories = new Map<string, RateContextCategory | undefined>();
   160→  for (const c of seedCategories) categories.set(c.id, { parentId: c.parentId, commissionPercent: c.commissionPercent });
   161→  // Merge the products' own categoryIds into the seeds (see note above) and
   162→  // queue the closure chase round for the combined set.
   163→  const products = new Map<string, number | null | undefined>(
   164→    productRows.map((p) => [p.id, p.commissionPercentOverride] as const)
   165→  );
   166→  const productDerivedCategoryIds = productRows.map((p) => p.categoryId).filter((c): c is string => typeof c === "string" && c.length > 0);
   167→  if (productDerivedCategoryIds.length > 0) {
   168→    const notYetLoaded = productDerivedCategoryIds.filter((c) => !categories.has(c));
   169→    if (notYetLoaded.length > 0) {
   170→      const rows = await db.category.findMany({
   171→        where: { id: { in: [...new Set(notYetLoaded)] } },
   172→        select: { id: true, parentId: true, commissionPercent: true },
   173→      });
   174→      for (const r of rows) categories.set(r.id, { parentId: r.parentId, commissionPercent: r.commissionPercent });
   175→      for (const c of new Set(notYetLoaded)) if (!categories.has(c)) categories.set(c, undefined);
   176→    }
   177→  }
   178→
   179→  // Rounds 2..k — chase parents to closure. Each round loads only ids not yet
   180→  // in the Map; bounded by tree depth (typically 1 extra round, often 0).
   181→  for (;;) {
   182→    const missing = new Set<string>();
   183→    for (const c of categories.values()) {
   184→      if (c?.parentId && !categories.has(c.parentId)) missing.add(c.parentId);
   185→    }
   186→    if (missing.size === 0) break;
   187→    const rows = await db.category.findMany({
   188→      where: { id: { in: [...missing] } },
   189→      select: { id: true, parentId: true, commissionPercent: true },
   190→    });
   191→    for (const r of rows) categories.set(r.id, { parentId: r.parentId, commissionPercent: r.commissionPercent });
   192→    if (rows.length < missing.size) {
   193→      // A parent id references a deleted category — mark it explicitly so the
   194→      // loop terminates (chain walk treats absent as "no parent", matching
   195→      // the sequential resolver's findUnique→null behavior).
   196→      for (const m of missing) if (!categories.has(m)) categories.set(m, undefined);
   197→    }
   198→  }
   199→
   200→  // Round 3 — scoped override rows for the EXACT closure (no over-fetch:
   201→  // category-scope rows limited to reachable ancestors, product/seller rows
   202→  // limited to this cart). Same shape/order as the original per-item query.
   203→  const closureCategoryIds = [...categories.keys()];
   204→  const overrides = await db.commissionOverride.findMany({
   205→    where: {
   206→      isActive: true,
   207→      AND: [
   208→        { OR: [{ validFrom: null }, { validFrom: { lte: now } }] },
   209→        { OR: [{ validTo: null }, { validTo: { gte: now } }] },
   210→        {
   211→          OR: [
   212→            { scope: "product", productId: { in: productIds.length > 0 ? productIds : [NONE] } },
   213→            { scope: "category", categoryId: { in: closureCategoryIds.length > 0 ? closureCategoryIds : [NONE] } },
   214→            { scope: "seller", sellerId: { in: sellerIds.length > 0 ? sellerIds : [NONE] } },
   215→          ],
   216→        },
   217→      ],
   218→    },
   219→    orderBy: [{ createdAt: "asc" }, { id: "asc" }],
   220→    select: {
   221→      id: true, scope: true, productId: true, categoryId: true, sellerId: true,
   222→      percent: true, fixedFeeMinor: true, holdDays: true,
   223→    },
   224→  });
   225→
   226→  return {
   227→    settings,
   228→    sellers: new Map<string, RateContextSeller | undefined>(
   229→      sellerRows.map((s) => [s.id, { commissionOverridePercent: s.commissionOverridePercent, holdDaysOverride: s.holdDaysOverride }] as const)
   230→    ),
   231→    holidays: holidays.map((h) => ({ id: h.id, scope: h.scope, scopeIds: h.scopeIds ?? "[]" })),
   232→    products,
   233→    categories,
   234→    overrides,
   235→  };
   236→}
   237→
   238→/** In-memory scoped-override selection — EXACT mirror of the original
   239→ *  per-item `scopedOverrideRows` filter: product rows for THIS product, then
   240→ *  category rows in ancestor-chain order (nearest first), then seller rows —
   241→ *  all from the pre-fetched createdAt-asc list, so within-class precedence
   242→ *  (earliest createdAt, id tie-break) is preserved by the filters themselves. */
   243→function scopedOverrideRowsFromContext(
   244→  ctx: RateContext,
   245→  item: { sellerId: string; productId?: string | null },
   246→  chain: string[]
   247→): RateContextOverride[] {
   248→  const productRows = ctx.overrides.filter((r) => r.scope === "product" && r.productId != null && r.productId === item.productId);
   249→  const categoryRows = chain.flatMap((catId) => ctx.overrides.filter((r) => r.scope === "category" && r.categoryId === catId));
   250→  const sellerRows = ctx.overrides.filter((r) => r.scope === "seller" && r.sellerId === item.sellerId);
   251→  return [...productRows, ...categoryRows, ...sellerRows];
   252→}
   253→
   254→/** Ancestor chain from the preloaded closure — cycle-safe, identical walk to
   255→ *  the original `categoryAncestorChain` (absent row ⇒ chain stops). */
   256→function chainFromContext(ctx: RateContext, categoryId: string | null | undefined): string[] {
   257→  const chain: string[] = [];
   258→  let cur = categoryId ?? null;
   259→  const seen = new Set<string>();
   260→  while (cur && !seen.has(cur)) {
   261→    seen.add(cur);
   262→    chain.push(cur);
   263→    cur = ctx.categories.get(cur)?.parentId ?? null;
   264→  }
   265→  return chain;
   266→}

   268→/**
   269→ * Resolve the effective commission rate for an item FROM A PRELOADED CONTEXT —
   270→ * pure, synchronous, zero queries. Byte-identical semantics to the original
   271→ * sequential resolver (parity-gated by scripts/test-commission-parity.ts):
   272→ *   0. fee holiday (all / sellers / categories-direct) …
   273→ *   … through scoped override rows and direct field overrides interleaved by
   274→ *   scope: product > category ancestor > seller … global default.
   275→ */
   276→export function resolveRateFromContext(
   277→  ctx: RateContext,
   278→  item: {
   279→    sellerId: string;
   280→    categoryId?: string | null;
   281→    productId?: string | null;
   282→  }
   283→): RateResolution {
   284→  const settings = ctx.settings;
   285→  const chain = chainFromContext(ctx, item.categoryId);
   286→
   287→  const rows = scopedOverrideRowsFromContext(ctx, item, chain);
   288→  const holdRow = rows.find((r) => r.holdDays != null);
   289→  const fixedRow = rows.find((r) => r.fixedFeeMinor != null);
   290→  const seller = ctx.sellers.get(item.sellerId);
   291→  const fallback: RateResolution = {
   292→    percent: settings.defaultPercent,
   293→    source: "global",
   294→    holdDays: holdRow?.holdDays ?? seller?.holdDaysOverride ?? settings.payoutHoldDays,
   295→    fixedFeeMinor: fixedRow?.fixedFeeMinor ?? settings.fixedFeeMinor,
   296→  };
   297→
   298→  // 1. active fee holiday covering (seller, category, now)
   299→  for (const h of ctx.holidays) {
   300→    if (h.scope === "all") {
   301→      return { ...fallback, percent: 0, source: "fee_holiday", holidayId: h.id };
   302→    }
   303→    const ids = JSON.parse(h.scopeIds || "[]") as string[];
   304→    if (h.scope === "sellers" && ids.includes(item.sellerId)) {
   305→      return { ...fallback, percent: 0, source: "fee_holiday", holidayId: h.id };
   306→    }
   307→    if (h.scope === "categories" && item.categoryId && ids.includes(item.categoryId)) {
   308→      return { ...fallback, percent: 0, source: "fee_holiday", holidayId: h.id };
   309→    }
   310→  }
   311→
   312→  // 2. product-scope override row (F-08) — windowed, beats the direct field
   313→  const pRow = rows.find((r) => r.scope === "product" && r.percent != null);
   314→  if (pRow) {
   315→    return { ...fallback, percent: pRow.percent!, source: "override", sourceId: pRow.id };
   316→  }
   317→
   318→  // 3. product direct override
   319→  if (item.productId) {
   320→    const p = ctx.products.get(item.productId);
   321→    if (p != null) {
   322→      return { ...fallback, percent: p, source: "product", sourceId: item.productId };
   323→    }
   324→  }
   325→
   326→  // 4. category-scope override rows (F-08) — nearest ancestor first
   327→  const cRow = rows.find((r) => r.scope === "category" && r.percent != null);
   328→  if (cRow) {
   329→    return { ...fallback, percent: cRow.percent!, source: "override", sourceId: cRow.id };
   330→  }
   331→
   332→  // 5. nearest category ancestor with commission_percent (direct field)
   333→  for (const catId of chain) {
   334→    const cat = ctx.categories.get(catId);
   335→    if (cat?.commissionPercent != null) {
   336→      return { ...fallback, percent: cat.commissionPercent, source: "category", sourceId: catId };
   337→    }
   338→  }
   339→
   340→  // 6. seller-scope override row (F-08)
   341→  const sRow = rows.find((r) => r.scope === "seller" && r.percent != null);
   342→  if (sRow) {
   343→    return { ...fallback, percent: sRow.percent!, source: "override", sourceId: sRow.id };
   344→  }
   345→
   346→  // 7. seller direct override
   347→  if (seller?.commissionOverridePercent != null) {
   348→    return { ...fallback, percent: seller.commissionOverridePercent, source: "seller", sourceId: item.sellerId };
   349→  }
   350→
   351→  // 8. global default
   352→  return fallback;
   353→}

   420→/**
   421→ * INVARIANT (Option B mirror requirement — FIN-1, v2 with affiliate):
   422→ *   sum(item.sellerNetMinor) + sum(item.commissionMinor) + sum(item.affiliatePoolMinor)
   423→ *     === order.totalMinor
   424→ * for EVERY order, where totalMinor = subtotalMinor − discountMinor +
   425→ * shippingMinor + taxMinor. The ledger T1 (postOrderPaid) debits the PSP
   426→ * gateway totalMinor and credits exactly sellerNet + commission + affiliate
   427→ * pool, so any drift here would throw LEDGER_UNBALANCED at payment time. The
   428→ * customer-paid gross per item is (lineSubtotal − discount + shipping + tax
   429→ * allocations); seller net is that gross MINUS commission MINUS the
   430→ * affiliate pool — never gross-plus-shipping again (the pre-FIN-1 formula
   431→ * double-counted shipping when appliesToShipping=true).
   432→ */
   433→export function reconcileSplits(
   434→  items: { sellerNetMinor: number; commissionMinor: number; affiliatePoolMinor?: number; lineSubtotalMinor: number }[],
   435→  totalMinor: number
   436→): { items: { sellerNetMinor: number; commissionMinor: number; affiliatePoolMinor?: number; lineSubtotalMinor: number }[]; driftMinor: number } {
   437→  const sumNet = items.reduce((s, r) => s + r.sellerNetMinor, 0);
   438→  const sumComm = items.reduce((s, r) => s + r.commissionMinor, 0);
   439→  const sumAff = items.reduce((s, r) => s + (r.affiliatePoolMinor ?? 0), 0);
   440→  const drift = totalMinor - (sumNet + sumComm + sumAff);
   441→  if (drift === 0 || items.length === 0) return { items, driftMinor: 0 };
   442→  // Deterministic remainder handling: assign the whole drift to the LARGEST
   443→  // line (same tie-break family as allocateProportionally: value desc, then
   444→  // original order) so the split reconciles EXACTLY to the order total.
   445→  let maxIdx = 0;
   446→  for (let i = 1; i < items.length; i++) {
   447→    if (items[i].lineSubtotalMinor > items[maxIdx].lineSubtotalMinor) maxIdx = i;
   448→  }
   449→  const adjusted = items.map((r, i) => (i === maxIdx ? { ...r, sellerNetMinor: r.sellerNetMinor + drift } : r));
   450→  return { items: adjusted, driftMinor: drift };
   451→}

   453→/**
   454→ * Compute per-item commission snapshots for an order (M09 BASE + MATH).
   455→ * - orderDiscount allocated proportionally to line subtotals (remainder to highest-value line)
   456→ * - shipping allocated proportionally (or 0 if not applicable per line config)
   457→ * - commission = round_half_up(base * percent/100) + allocated fixed fee
   458→ * - sellerNet = (subtotal − discount + shipping + tax) − commission per item —
   459→ *   the Option B invariant sum(sellerNet)+sum(commission)===order total holds
   460→ *   in ALL paths, including appliesToShipping=true (see reconcileSplits doc).
   461→ * - min_commission_minor floored per ORDER (skipped on fee holidays, flagged)
   462→ *
   463→ * PHASE 2 / A2: `preloaded` lets a multi-group checkout load ONE RateContext
   464→ * for every seller group and share it across the per-group calls (the context
   465→ * loader is item-count independent — see loadRateContext). Without it, the
   466→ * context is loaded from this order's own items: same semantics, ~6 queries
   467→ * either way (versus (5+2·D) sequential queries PER LINE pre-batching).
   468→ */
   469→export async function computeOrderCommissions(
   470→  input: {
   471→    sellerId: string;
   472→    items: CommissionItemInput[];
   473→    orderDiscountMinor: number;
   474→    shippingMinorTotal: number;
   475→    /** Order-level tax (currently always 0 in checkout pricing) — included in
   476→     *  the sellerNet reconciliation so the invariant survives tax going live. */
   477→    orderTaxMinor?: number;
   478→    appliesShippingToItem?: boolean; // whether shipping counts toward commission base
   479→    currency?: string;
   480→    now?: Date;
   481→  },
   482→  preloaded?: RateContext
   483→): Promise<CommissionOrderResult> {
   484→  const now = input.now ?? new Date();
   485→  const ctx =
   486→    preloaded ??
   487→    (await loadRateContext({
   488→      sellerIds: [input.sellerId],
   489→      productIds: input.items.map((it) => it.productId ?? "").filter(Boolean),
   490→      categoryIds: input.items.map((it) => it.categoryId ?? "").filter(Boolean),
   491→      now,
   492→    }));
   493→  const settings = ctx.settings;
   494→
   495→  const subtotals = input.items.map((it) => it.unitPriceMinor * it.qty);
   496→  const discounts = allocateProportionally(Math.max(0, input.orderDiscountMinor), subtotals);
   497→  const shippingAlloc = allocateProportionally(Math.max(0, input.shippingMinorTotal), subtotals);
   498→  const taxAlloc = allocateProportionally(Math.max(0, input.orderTaxMinor ?? 0), subtotals);
   499→
   500→  // resolve rate per item — pure in-memory resolution over the shared context
   501→  const rates: RateResolution[] = input.items.map((it) =>
   502→    resolveRateFromContext(ctx, {
   503→      sellerId: input.sellerId,
   504→      categoryId: it.categoryId ?? null,
   505→      productId: it.productId ?? null,
   506→    })
   507→  );
   508→
   509→  const perOrderFixed = settings.fixedFeeMinor;
   510→  const fixedAlloc = allocateProportionally(perOrderFixed, subtotals);
   511→
   512→  const results: CommissionItemResult[] = input.items.map((it, i) => {
   513→    const lineSubtotal = subtotals[i];
   514→    const rate = rates[i];
   515→    const baseRuleList = settings.basePriceRule === "list_price";
   516→    let base = baseRuleList ? lineSubtotal : lineSubtotal - discounts[i];
   517→    if (settings.appliesToShipping) base += shippingAlloc[i];
   518→    const percentPart = commissionOf(base, rate.percent);
   519→    const commission = percentPart + fixedAlloc[i];
   520→    // AFFILIATE (Hub): seller-funded pool on the COMMISSIONABLE BASE — eligible
   521→    // merchandise value after seller-funded discounts, EXCLUDING shipping and
   522→    // tax (spec §8). Only lines with resolved server-side attribution and a
   523→    // positive effective rate generate a pool.
   524→    const aff = it.affiliate && it.affiliate.rateBps > 0 ? it.affiliate : null;
   525→    const affBase = Math.max(0, lineSubtotal - discounts[i]);
   526→    const affiliatePool = aff ? affiliatePoolMinor(affBase, aff.rateBps) : 0;
   527→    // Customer-paid gross for this line = subtotal − discount + shipping + tax
   528→    // (allocation parts sum EXACTLY to the order-level inputs). sellerNet is
   529→    // the residual after commission AND the affiliate pool — NOT base+shipping
   530→    // again, which counted shipping twice whenever the commission base
   531→    // included it (FIN-1 fix).
   532→    const grossMinor = lineSubtotal - discounts[i] + shippingAlloc[i] + taxAlloc[i];
   533→    const sellerNet = grossMinor - commission - affiliatePool;
   534→    return {
   535→      title: it.title,
   536→      variantLabel: it.variantLabel ?? null,
   537→      unitPriceMinor: it.unitPriceMinor,
   538→      qty: it.qty,
   539→      lineSubtotalMinor: lineSubtotal,
   540→      discountMinor: discounts[i],
   541→      shippingMinor: shippingAlloc[i],
   542→      baseMinor: base,
   543→      percent: rate.percent,
   544→      percentPartMinor: percentPart,
   545→      fixedFeeMinor: fixedAlloc[i],
   546→      commissionMinor: commission,
   547→      sellerNetMinor: sellerNet,
   548→      affiliatePoolMinor: affiliatePool,
   549→      breakdown: {
   550→        rule_source: rate.source,
   551→        source_id: rate.sourceId,
   552→        holiday_id: rate.holidayId,
   553→        percent: rate.percent,
   554→        base,
   555→        percent_part: percentPart,
   556→        fixed_fee_allocated: fixedAlloc[i],
   557→        discount_allocated: discounts[i],
   558→        shipping_allocated: shippingAlloc[i],
   559→        ...(aff
   560→          ? { affiliate_rate_bps: aff.rateBps, affiliate_base: affBase, affiliate_pool: affiliatePool }
   561→          : {}),
   562→        flags: [] as string[],
   563→      },
   564→    };
   565→  });
   566→
   567→  // min commission floor — per ORDER, skipped when any item ran on a fee holiday
   568→  const onHoliday = rates.some((r) => r.source === "fee_holiday");
   569→  const totalPercentPart = results.reduce((s, r) => s + r.percentPartMinor, 0);
   570→  if (!onHoliday && settings.minCommissionMinor > 0 && totalPercentPart < settings.minCommissionMinor && input.items.length > 0) {
   571→    const bump = settings.minCommissionMinor - totalPercentPart;
   572→    // remainder to highest-value line
   573→    let maxIdx = 0;
   574→    for (let i = 1; i < results.length; i++) {
   575→      if (results[i].lineSubtotalMinor > results[maxIdx].lineSubtotalMinor) maxIdx = i;
   576→    }
   577→    results[maxIdx].percentPartMinor += bump;
   578→    results[maxIdx].commissionMinor += bump;
   579→    results[maxIdx].sellerNetMinor -= bump;
   580→    results[maxIdx].breakdown.flags.push("min_commission_floor");
   581→  }
   582→
   583→  // FINAL RECONCILIATION (Option B invariant): sellerNet + commission must
   584→  // equal the order total (subtotal − discount + shipping + tax) EXACTLY.
   585→  // Drift is structurally zero with the math above (allocation parts sum
   586→  // exactly); reconcileSplits stays as the deterministic safety net — any
   587→  // remainder lands on the largest line, never silently lost.
   588→  const expectedTotal =
   589→    subtotals.reduce((s, v) => s + v, 0) -
   590→    Math.max(0, input.orderDiscountMinor) +
   591→    Math.max(0, input.shippingMinorTotal) +
   592→    Math.max(0, input.orderTaxMinor ?? 0);
   593→  const reconciled = reconcileSplits(
   594→    results.map((r) => ({ sellerNetMinor: r.sellerNetMinor, commissionMinor: r.commissionMinor, affiliatePoolMinor: r.affiliatePoolMinor, lineSubtotalMinor: r.lineSubtotalMinor })),
   595→    expectedTotal
   596→  );
   597→  for (let i = 0; i < results.length; i++) results[i].sellerNetMinor = reconciled.items[i].sellerNetMinor;
   598→
   599→  const holdDays = rates[0]?.holdDays ?? settings.payoutHoldDays;
   600→  const orderCommission = results.reduce((s, r) => s + r.commissionMinor, 0);
   601→  const orderSellerNet = results.reduce((s, r) => s + r.sellerNetMinor, 0);
   602→  const orderAffiliatePool = results.reduce((s, r) => s + r.affiliatePoolMinor, 0);
   603→
   604→  return { items: results, orderCommissionMinor: orderCommission, orderSellerNetMinor: orderSellerNet, orderAffiliatePoolMinor: orderAffiliatePool, holdDays, holdReleaseRule: settings.holdReleaseRule };
   605→}

   640→ *  compatibility; batch callers should prefer loadRateContext +
   641→ *  resolveRateFromContext). Same semantics as the pre-batch sequential
   642→ *  resolver — proven byte-identical by scripts/test-commission-parity.ts. */
   643→export async function resolveRate(ctx: {
   644→  sellerId: string;
   645→  categoryId?: string | null;
   646→  productId?: string | null;
   647→  now?: Date;
   648→}): Promise<RateResolution> {

   656→}
