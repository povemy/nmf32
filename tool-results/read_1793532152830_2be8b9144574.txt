    52→const IMG_EXT: Record<string, string> = { "image/jpeg": ".jpg", "image/png": ".png", "image/webp": ".webp", "image/gif": ".gif" };
    53→
    54→/** SSRF-guarded, size-limited copy of an external product image into public/uploads.
    55→ * B5/F-06: the raw `fetch(parsed, { redirect: "follow" })` is replaced by
    56→ * safeFetchExternalImage — manual redirects (max 3 hops) with FULL re-validation
    57→ * per hop, DNS pinning via a guarded undici Agent dispatcher, content-type
    58→ * whitelist + magic-byte sniffing, and the 5MB cap enforced before the body is
    59→ * read. assertPublicHost stays as a cheap legacy pre-check; safeFetch is the
    60→ * authoritative guard. */
    61→export async function copyExternalImage(storeId: string, url: string): Promise<string | null> {
    62→  try {
    63→    const parsed = new URL(url);
    64→    await assertPublicHost(parsed);
    65→    const res = await safeFetchExternalImage(url);
    66→    if (!res.ok) return null;
    67→    const ext = IMG_EXT[res.contentType];
    68→    if (!ext) return null;
    69→    const buf = res.buffer;
    70→    if (buf.length === 0 || buf.length > 5 * 1024 * 1024) return null;
    71→    const hash = sha256Hex(url).slice(0, 16);
    72→    const dir = path.join(process.cwd(), "public", "uploads", storeId);
    73→    await fs.promises.mkdir(dir, { recursive: true });
    74→    const file = path.join(dir, `${hash}${ext}`);
    75→    await fs.promises.writeFile(file, buf);
    76→    return `/uploads/${storeId}/${hash}${ext}`;
    77→  } catch {
    78→    return null;
    79→  }
    80→}

    88→async function ensureStoreForJob(job: { id: string; platform: string }, shop: NormalizedShop): Promise<{ id: string; slug: string }> {
    89→  // reuse store bound to a prior import from the same source shop
    90→  const prior = await db.store.findFirst({
    91→    where: { products: { some: { sourcePlatform: job.platform, sourceShopId: shop.sourceShopId } } },
    92→    select: { id: true, slug: true },
    93→  });
    94→  if (prior) return prior;
    95→
    96→  const label = platformLabel(job.platform);
    97→  const slug = await uniqueSlug(shop.name);
    98→  const store = await db.store.create({
    99→    data: {
   100→      sellerId: null,
   101→      name: shop.name.slice(0, 80),
   102→      slug,
   103→      tagline: `Imported from ${label}`,
   104→      description: shop.name ? `${shop.name} — catalogue imported from ${label}.` : null,
   105→      logoUrl: shop.logoUrl ?? null,
   106→      status: "generating",
   107→      lifecycleSource: "import",
   108→    },
   109→    select: { id: true, slug: true },
   110→  });
   111→  await transitionStore({ storeId: store.id, to: "pending_verification", actorType: "system", note: "import generated" }).catch(() => undefined);
   112→  // home page draft so preview + publish render real sections
   113→  await db.cmsPage.create({
   114→    data: {
   115→      storeId: store.id,
   116→      slug: "home",
   117→      title: `${shop.name} — Home`,
   118→      pageType: "home",
   119→      draftJson: JSON.stringify({ sections: defaultHomeSections(shop.name, `Imported from ${label}`), version: 1 }),
   120→    },
   121→  }).catch(() => {
   122→    // home page might exist for reused slug store — ignore
   123→  });
   124→  return store;
   125→}

   131→async function upsertProduct(job: { id: string; platform: string; isFixture: boolean }, storeId: string, storeName: string, sourceShopId: string, np: NormalizedProduct): Promise<{ productId: string; warnings: string[] }> {
   132→  const warnings = [...np.warnings];
   133→  const existing = await db.product.findUnique({
   134→    where: { storeId_sourcePlatform_sourceProductId: { storeId, sourcePlatform: job.platform, sourceProductId: np.sourceProductId } },
   135→    include: { images: true, variants: true },
   136→  });
   137→  const overrides = parseJson<Overrides>(existing?.overridesJson, {});
   138→  const description = sanitiseDescriptionHtml(np.descriptionHtml);
   139→  const minPrice = np.variants.length > 0 ? Math.min(...np.variants.map((v) => v.priceMinor)) : 0;
   140→
   141→  let productId: string;
   142→  if (!existing) {
   143→    const slug = await slugifyProductTitle(storeId, np.title);
   144→    const categoryId = await mapCategory(np.categoryHint);
   145→    // C1.4 (Phase 2.2) — IMPORTED PRODUCTS ENTER MODERATION. Previously the
   146→    // create branch assigned `active` merely because the import succeeded /
   147→    // the source item was available, bypassing the keyword flagging and
   148→    // review applied to manually created products — a "Replica Gun" title
   149→    // imported straight to public. Now EVERY imported product is born
   150→    // `pending_review` (admin moderation queue approves to `active`, which
   151→    // also sets publishedAt); unavailable source items stay `archived`.
   152→    // Keyword flags run over the same text fields as the manual path and are
   153→    // persisted to flagsJson for the moderation queue.
   154→    const descriptionText = htmlToText(description);
   155→    const flags = matchFlags(`${np.title} ${descriptionText}`);
   156→    const created = await db.product.create({
   157→      data: {
   158→        storeId,
   159→        categoryId,
   160→        title: np.title.slice(0, 160),
   161→        slug,
   162→        description,
   163→        status: np.isAvailable ? "pending_review" : "archived",
   164→        flagsJson: JSON.stringify(flags),
   165→        priceMinor: minPrice,
   166→        sourcePlatform: job.platform,
   167→        sourceShopId,
   168→        sourceProductId: np.sourceProductId,
   169→        sourceUrl: np.sourceUrl ?? null,
   170→        sourceBrand: np.brand ?? null,
   171→        sourceStatus: np.sourceStatus,
   172→        isFixture: job.isFixture,
   173→        lastSyncedAt: new Date(),
   174→        overridesJson: "{}",
   175→        ...seoFields(np, storeName),
   176→        images: { create: np.images.map((im) => ({ url: im.url, sortOrder: im.sortOrder, alt: im.alt ?? null })) },
   177→        variants: {
   178→          create: np.variants.map((v) => ({
   179→            name: v.options.map((o) => `${o.name}: ${o.value}`).join(" / ") || "Default",
   180→            optionName: v.options[0]?.name ?? null,
   181→            optionValue: v.options[0]?.value ?? null,
   182→            sourceModelId: v.sourceModelId,
   183→            imageUrl: v.imageUrl ?? null,
   184→            sku: v.sourceSku ?? null,
   185→            priceMinor: v.priceMinor,
   186→            compareAtPriceMinor: v.compareAtPriceMinor ?? null,
   187→            stock: v.stock ?? 0,
   188→            isActive: v.isAvailable,
   189→          })),
   190→        },
   191→      },
   192→      select: { id: true },
   193→    });
   194→    productId = created.id;
   195→  } else {
   196→    productId = existing.id;
   197→    // C1.4 (Phase 2.2) — re-import no longer resurrects archived products to
   198→    // `active` and never overwrites an admin verdict (banned / rejected /
   199→    // pending_review survive). Available at source keeps the CURRENT status
   200→    // (a seller deliberately re-submits an archived product via the product
   201→    // routes → pending_review → admin approval); unavailable archives it.
   202→    // Keyword flags re-run on the incoming content: a newly-flagged previously
   203→    // active product is demoted to pending_review with the flags recorded.
   204→    const descriptionText = htmlToText(description);
   205→    const flags = matchFlags(`${np.title.slice(0, 160)} ${descriptionText}`);
   206→    const incomingStatus = np.isAvailable ? existing.status : "archived";
   207→    const demoteForFlags = flags.length > 0 && existing.status === "active";
   208→    await db.product.update({
   209→      where: { id: productId },
   210→      data: {
   211→        // seller-locked fields survive re-import
   212→        ...(overrides.title ? {} : { title: np.title.slice(0, 160) }),
   213→        ...(overrides.description ? {} : { description }),
   214→        priceMinor: minPrice,
   215→        sourceStatus: np.sourceStatus,
   216→        sourceBrand: np.brand ?? existing.sourceBrand ?? null,
   217→        sourceUrl: np.sourceUrl ?? existing.sourceUrl,
   218→        lastSyncedAt: new Date(),
   219→        // disappeared at source → archive (never delete: order history preserved)
   220→        status: demoteForFlags ? "pending_review" : incomingStatus,
   221→        ...(flags.length > 0 || existing.flagsJson !== "[]" ? { flagsJson: JSON.stringify(flags) } : {}),
   222→        ...(overrides.description ? {} : seoFields(np, storeName)),
   223→      },
   224→      select: { id: true },
   225→    });
   226→    if (!overrides.images) {
   227→      await db.productImage.deleteMany({ where: { productId } });
   228→      if (np.images.length > 0) {
   229→        await db.productImage.createMany({ data: np.images.map((im) => ({ productId, url: im.url, sortOrder: im.sortOrder, alt: im.alt ?? null })) });
   230→      }
   231→    }
   232→    if (!overrides.variants) {
   233→      await db.productVariant.deleteMany({ where: { productId } });
   234→      await db.productVariant.createMany({
   235→        data: np.variants.map((v) => ({
   236→          productId,
   237→          name: v.options.map((o) => `${o.name}: ${o.value}`).join(" / ") || "Default",
   238→          optionName: v.options[0]?.name ?? null,
   239→          optionValue: v.options[0]?.value ?? null,
   240→          sourceModelId: v.sourceModelId,
   241→          imageUrl: v.imageUrl ?? null,
   242→          sku: v.sourceSku ?? null,
   243→          priceMinor: v.priceMinor,
   244→          compareAtPriceMinor: v.compareAtPriceMinor ?? null,
   245→          stock: v.stock ?? 0,
   246→          isActive: v.isAvailable,
   247→        })),
   248→      });
   249→    }
   250→  }
   251→
   252→  // image processing: copy external images into local storage (permitted strategy),
   253→  // fall back to source URL reference with a warning on failure.
   254→  let imagesCopied = 0;
   255→  const rows = await db.productImage.findMany({ where: { productId }, orderBy: { sortOrder: "asc" } });
   256→  for (let i = 0; i < rows.length; i++) {
   257→    const row = rows[i];
   258→    if (row.url.startsWith("/")) continue; // already local (fixture/mockup)
   259→    const local = await copyExternalImage(storeId, row.url);
   260→    if (local) {
   261→      await db.productImage.update({ where: { id: row.id }, data: { url: local } });
   262→      imagesCopied++;
   263→    } else {
   264→      warnings.push(`Image ${i + 1} kept as source reference (copy failed)`);
   265→    }
   266→  }
   267→  if (imagesCopied > 0) await bumpProgress(job.id, { imagesCopied });
   268→  return { productId, warnings };
   269→}

   280→export async function createImportJob(input: { inputUrl: string; allowFixture?: boolean }): Promise<{ jobId: string; trackingToken: string; status: string }> {
   281→  const { connector, resolved } = await resolveForConnector(input.inputUrl);
   282→  const auth = await connector.authStatus();
   283→  const isFixture = resolved.platform === "fixture";
   284→  if (isFixture && input.allowFixture === false) {
   285→    throw new ApiError(400, "FIXTURE_NOT_ALLOWED", "Fixture imports are disabled.");
   286→  }
   287→  const trackingToken = randomToken(24);
   288→  const job = await db.importJob.create({
   289→    data: {
   290→      trackingToken,
   291→      platform: resolved.platform,
   292→      inputUrl: resolved.normalizedUrl,
   293→      urlType: resolved.urlType,
   294→      sourceShopId: resolved.shopId ?? null,
   295→      status: "queued",
   296→      isFixture,
   297→      summaryJson: JSON.stringify({
   298→        platform: resolved.platform,
   299→        platformLabel: platformLabel(resolved.platform),
   300→        isFixture,
   301→      } satisfies JobSummary),
   302→    },
   303→    select: { id: true },
   304→  });
   305→  if (!isFixture && (auth.needsAuthorisation || !auth.envConfigured)) {
   306→    await db.importJob.update({
   307→      where: { id: job.id },
   308→      data: {
   309→        status: "awaiting_authorisation",
   310→        phaseDetail: auth.envConfigured
   311→          ? "Seller authorisation required before the catalogue can be imported."
   312→          : "Marketplace app credentials are not configured on the server.",
   313→        summaryJson: JSON.stringify({
   314→          platform: resolved.platform,
   315→          platformLabel: platformLabel(resolved.platform),
   316→          isFixture,
   317→          authorisationRequired: true,
   318→          beginUrl: auth.beginUrl ?? null,
   319→          missingEnv: auth.missing,
   320→        } satisfies JobSummary),
   321→      },
   322→    });
   323→    await logActivity({ actorType: "shopper", action: "import.awaiting_authorisation", entityType: "import_job", entityId: job.id, meta: { platform: resolved.platform } }).catch(() => undefined);
   324→  }
   325→  return { jobId: job.id, trackingToken, status: isFixture ? "queued" : "awaiting_authorisation" };
   326→}

   330→export async function runJob(jobId: string): Promise<{ status: string }> {
   331→  const job = await db.importJob.findUnique({ where: { id: jobId } });
   332→  if (!job) throw new ApiError(404, "JOB_NOT_FOUND", "Import job not found.");
   333→  if (job.status === "completed" || job.status === "completed_with_warnings") return { status: job.status };
   334→  if (job.status === "awaiting_authorisation") return { status: job.status };
   335→
   336→  const connector = getConnector(job.platform as Platform);
   337→  const warnings: string[] = [];
   338→  try {
   339→    // connection resolution: fixture creates a pseudo-connection keyed by shop slug
   340→    let connectionId: string | null = job.connectionId;
   341→    if (!connectionId) {
   342→      if (job.isFixture) {
   343→        const conn = await db.marketplaceConnection.upsert({
   344→          where: { platform_shopId: { platform: "fixture", shopId: job.sourceShopId ?? "demo-store" } },
   345→          create: {
   346→            platform: "fixture",
   347→            shopId: job.sourceShopId ?? "demo-store",
   348→            shopName: job.sourceShopId ?? "Demo Store",
   349→            status: "connected",
   350→            envConfigured: true,
   351→          },
   352→          update: { status: "connected" },
   353→          select: { id: true },
   354→        });
   355→        connectionId = conn.id;
   356→      } else {
   357→        const conn = await db.marketplaceConnection.findFirst({ where: { platform: job.platform, status: "connected" }, orderBy: { createdAt: "desc" }, select: { id: true } });
   358→        if (!conn) throw new ApiError(409, "NO_CONNECTION", `No authorised ${platformLabel(job.platform)} connection. Connect the marketplace first.`);
   359→        connectionId = conn.id;
   360→      }
   361→      await db.importJob.update({ where: { id: jobId }, data: { connectionId } });
   362→    }
   363→
   364→    if (!job.isFixture) await setPhase(jobId, "connecting", "Connecting to marketplace…");
   365→    await setPhase(jobId, "fetching_shop", "Retrieving shop information…");
   366→    const shop = await connector.getShop(connectionId);
   367→    summaryCache.set(jobId, JSON.stringify({ shop } satisfies JobSummary));
   368→
   369→    const store = await ensureStoreForJob(job, shop);
   370→    await db.importJob.update({ where: { id: jobId }, data: { storeId: store.id } });
   371→    await transitionStore({ storeId: store.id, to: "pending_claim", actorType: "system", note: `import job ${jobId}` }).catch(() => undefined);
   372→
   373→    // ---- discover products
   374→    await setPhase(jobId, "importing_products", "Listing marketplace products…");
   375→    const discovered: { sourceProductId: string }[] = [];
   376→    if (job.urlType === "product" && job.sourceProductId) {
   377→      discovered.push({ sourceProductId: job.sourceProductId });
   378→    } else {
   379→      let cursor: string | null = null;
   380→      for (let page = 0; page < 25; page++) {
   381→        const pageRes = await connector.listProductsPage(connectionId, cursor, 50);
   382→        discovered.push(...pageRes.products);
   383→        await bumpProgress(jobId, { discovered: discovered.length });
   384→        if (!pageRes.nextCursor) break;
   385→        cursor = pageRes.nextCursor;
   386→      }
   387→    }
   388→    if (discovered.length === 0) throw new ApiError(422, "EMPTY_CATALOGUE", "No products were found for this shop.");
   389→
   390→    // ---- per-product import (partial failures never abort the job)
   391→    let succeeded = 0;
   392→    let failed = 0;
   393→    let skipped = 0;
   394→    const storeRow = await db.store.findUnique({ where: { id: store.id }, select: { name: true } });
   395→    for (const d of discovered) {
   396→      const item = await db.importJobItem.upsert({
   397→        where: { jobId_sourcePlatform_sourceProductId: { jobId, sourcePlatform: job.platform, sourceProductId: d.sourceProductId } },
   398→        create: { jobId, sourcePlatform: job.platform, sourceShopId: shop.sourceShopId, sourceProductId: d.sourceProductId, status: "pending" },
   399→        update: {},
   400→        select: { id: true, status: true, productId: true },
   401→      });
   402→      if (item.status === "ok") {
   403→        skipped++;
   404→        await bumpProgress(jobId, { processed: succeeded + failed + skipped, skipped });
   405→        continue;
   406→      }
   407→      try {
   408→        const np: NormalizedProduct = await connector.getProduct(connectionId, d.sourceProductId);
   409→        await db.importJobItem.update({
   410→          where: { id: item.id },
   411→          data: { title: np.title, sourceShopId: shop.sourceShopId, status: "importing", warningsJson: "[]" },
   412→        });
   413→        const { productId, warnings: pw } = await upsertProduct(job, store.id, storeRow?.name ?? shop.name, shop.sourceShopId, np);
   414→        for (const w of pw) warnings.push(`${np.title}: ${w}`);
   415→        await db.importJobItem.update({
   416→          where: { id: item.id },
   417→          data: { productId, status: "ok", warningsJson: JSON.stringify(pw), processedAt: new Date(), title: np.title },
   418→        });
   419→        succeeded++;
   420→      } catch (err) {
   421→        failed++;
   422→        const msg = err instanceof Error ? err.message : String(err);
   423→        warnings.push(`${d.sourceProductId}: ${msg}`);
   424→        await db.importJobItem.update({
   425→          where: { id: item.id },
   426→          data: { status: "failed", error: msg.slice(0, 500), processedAt: new Date() },
   427→        });
   428→      }
   429→      await bumpProgress(jobId, { processed: succeeded + failed + skipped, succeeded, failed, skipped });
   430→    }
   431→
   432→    // ---- variants summary phase (variants are created inline; phase kept for progress fidelity)
   433→    await setPhase(jobId, "importing_variants", "Importing product variants…");
   434→    const variantCount = await db.productVariant.count({ where: { product: { storeId: store.id, sourcePlatform: job.platform } } });
   435→    await setPhase(jobId, "processing_images", "Processing product images…");
   436→
   437→    // ---- SEO
   438→    await setPhase(jobId, "generating_seo", "Generating SEO metadata…");
   439→    const importedProducts = await db.product.findMany({
   440→      where: { storeId: store.id, sourcePlatform: job.platform },
   441→      select: { id: true, title: true, description: true, seoTitle: true, images: { orderBy: { sortOrder: "asc" }, take: 1 } },
   442→    });
   443→    for (const p of importedProducts) {
   444→      if (!p.seoTitle) {
   445→        const { seoTitle, seoDescription } = deriveProductSeo(p.title, p.description, storeRow?.name ?? shop.name);
   446→        await db.product.update({ where: { id: p.id }, data: { seoTitle, seoDescription } });
   447→      }
   448→      const ogImage = p.images[0]?.url ?? null;
   449→      await db.seoMeta.upsert({
   450→        where: { entityType_entityId: { entityType: "product", entityId: p.id } },
   451→        create: {
   452→          entityType: "product", entityId: p.id,
   453→          seoTitle: `${p.title} | ${storeRow?.name ?? shop.name}`.slice(0, 60),
   454→          seoDescription: (htmlToText(p.description) || p.title).slice(0, 155),
   455→          ogImage,
   456→          structuredDataJson: JSON.stringify({ "@type": "Product", name: p.title }),
   457→          warningsJson: "[]",
   458→        },
   459→        update: { ogImage },
   460→      });
   461→    }
   462→    const storeSeo = await db.seoMeta.upsert({
   463→      where: { entityType_entityId: { entityType: "store", entityId: store.id } },
   464→      create: {
   465→        entityType: "store", entityId: store.id,
   466→        seoTitle: `${storeRow?.name ?? shop.name} — Official Store`.slice(0, 60),
   467→        seoDescription: `Shop ${storeRow?.name ?? shop.name} on NOMOFEE. Imported from ${platformLabel(job.platform)}.`.slice(0, 155),
   468→        ogImage: null,
   469→        structuredDataJson: JSON.stringify({ "@type": "Store", name: storeRow?.name ?? shop.name }),
   470→        warningsJson: "[]",
   471→      },
   472→      update: {},
   473→    });
   474→
   475→    // ---- storefront
   476→    await setPhase(jobId, "generating_storefront", "Generating storefront…");
   477→    const home = await db.cmsPage.findFirst({ where: { storeId: store.id, slug: "home" } });
   478→    if (home && !home.publishedJson) {
   479→      await db.cmsPage.update({ where: { id: home.id }, data: { publishedJson: home.draftJson, publishedVersion: 1 } });
   480→    }
   481→
   482→    // ---- claim token: while the store is unclaimed, each completed run issues a
   483→    // fresh claim link (previous links revoked — the owner may have lost theirs).
   484→    // Once claimed, the claim phase is skipped entirely.
   485→    await setPhase(jobId, "awaiting_claim", "Preparing secure claim link…");
   486→    const storeFresh = await db.store.findUnique({ where: { id: store.id }, select: { sellerId: true, status: true } });
   487→    let claimToken: string | null = null;
   488→    let claimExpiresAt: Date | null = null;
   489→    let claimMethod: string | null = null;
   490→    if (!storeFresh?.sellerId) {
   491→      await db.storeClaim.updateMany({ where: { storeId: store.id, status: "active" }, data: { status: "revoked" } });
   492→      claimToken = randomToken(24);
   493→      claimExpiresAt = new Date(Date.now() + 14 * 86400 * 1000);
   494→      claimMethod = job.isFixture ? "claim_code" : "pending";
   495→      await db.storeClaim.create({
   496→        data: {
   497→          storeId: store.id,
   498→          jobId: job.id,
   499→          tokenHash: sha256Hex(claimToken),
   500→          status: "active",
   501→          verificationMethod: claimMethod,
   502→          expiresAt: claimExpiresAt,
   503→        },
   504→      });
   505→    }
   506→
   507→    const failedItems = await db.importJobItem.count({ where: { jobId, status: "failed" } });
   508→    const summary: JobSummary = {
   509→      shop: { name: shop.name, sourceShopId: shop.sourceShopId, region: shop.region ?? null },
   510→      platform: job.platform,
   511→      platformLabel: platformLabel(job.platform),
   512→      isFixture: job.isFixture,
   513→      storeId: store.id,
   514→      storeSlug: store.slug,
   515→      productsOk: succeeded,
   516→      productsFailed: failed,
   517→      warnings: warnings.slice(0, 20),
   518→      claimMethod: claimMethod ?? undefined,
   519→    };
   520→    if (claimToken && claimExpiresAt) {
   521→      summary.claimToken = claimToken;
   522→      summary.claimExpiresAt = claimExpiresAt.toISOString();
   523→    }
   524→    await db.importJob.update({
   525→      where: { id: jobId },
   526→      data: {
   527→        status: failedItems > 0 ? "completed_with_warnings" : "completed",
   528→        phaseDetail: failedItems > 0 ? `${failedItems} product(s) could not be imported.` : null,
   529→        summaryJson: JSON.stringify(summary),
   530→        storeCreated: true,
   531→        startedAt: job.startedAt ?? new Date(),
   532→        finishedAt: new Date(),
   533→      },
   534→    });
   535→    await logActivity({ actorType: "system", action: "import.completed", entityType: "import_job", entityId: jobId, meta: { platform: job.platform, succeeded, failed } }).catch(() => undefined);
   536→    void storeSeo;
   537→    return { status: failedItems > 0 ? "completed_with_warnings" : "completed" };
   538→  } catch (err) {
   539→    const msg = err instanceof Error ? err.message : String(err);
   540→    await db.importJob.update({
   541→      where: { id: jobId },
   542→      data: { status: "failed", errorJson: JSON.stringify({ code: err instanceof ApiError ? err.code : "IMPORT_FAILED", message: msg.slice(0, 500) }), finishedAt: new Date() },
   543→    });
   544→    return { status: "failed" };
   545→  }
   546→}

   548→/** Resume failed/pending items of a non-terminal job. */
   549→export async function resumeJob(jobId: string): Promise<{ status: string }> {
   550→  const job = await db.importJob.findUnique({ where: { id: jobId } });
   551→  if (!job) throw new ApiError(404, "JOB_NOT_FOUND", "Import job not found.");
   552→  if (job.status === "failed" || job.status === "completed_with_warnings") {
   553→    await db.importJob.update({ where: { id: jobId }, data: { status: "queued", errorJson: "{}" } });
   554→  }
   555→  return runJob(jobId);
   556→}

   558→/** Verify a raw claim token against the store claim row (single-use validation happens at bind time). */
   559→export async function validateClaimToken(jobId: string, rawToken: string): Promise<{ storeId: string; storeSlug: string; storeName: string; method: string }> {
   560→  const job = await db.importJob.findUnique({ where: { id: jobId }, select: { storeId: true, isFixture: true } });
   561→  const tokenHash = sha256Hex(rawToken);
   562→  const claim = await db.storeClaim.findFirst({
   563→    where: { tokenHash, status: "active", expiresAt: { gt: new Date() } },
   564→    include: { store: { select: { id: true, slug: true, name: true } } },
   565→  });
   566→  if (!claim || !job || claim.storeId !== job.storeId) {
   567→    throw new ApiError(403, "INVALID_CLAIM_TOKEN", "This claim link is invalid, already used, or expired.");
   568→  }
   569→  return { storeId: claim.store.id, storeSlug: claim.store.slug, storeName: claim.store.name, method: claim.verificationMethod };
   570→}

   573→export function hashClaimToken(rawToken: string): string {
   574→  return sha256Hex(rawToken);
