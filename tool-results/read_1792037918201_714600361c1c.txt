   133→   133→export function fmtDT(iso?: string | Date | null): string {
   134→   134→  if (!iso) return "—";
   135→   135→  const d = typeof iso === "string" ? new Date(iso) : iso;
   136→   136→  if (Number.isNaN(d.getTime())) return "—";
   137→   137→  return d.toLocaleString("en-MY", {
   138→   138→    day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit", hour12: false,
   139→   139→  });
   140→   140→}
   141→   141→
   142→   142→export function fmtDate(iso?: string | Date | null): string {
   143→   143→  if (!iso) return "—";
   144→   144→  const d = typeof iso === "string" ? new Date(iso) : iso;
   145→   145→  if (Number.isNaN(d.getTime())) return "—";
   146→   146→  return d.toLocaleDateString("en-MY", { day: "2-digit", month: "short", year: "numeric" });
   147→   147→}
   148→   148→
   149→   149→export function timeAgo(iso?: string | Date | null): string {
   150→   150→  if (!iso) return "—";
   151→   151→  const d = typeof iso === "string" ? new Date(iso) : iso;
   152→   152→  if (Number.isNaN(d.getTime())) return "—";
   153→   153→  const secs = Math.floor((Date.now() - d.getTime()) / 1000);
   154→   154→  if (secs < 60) return "just now";
   155→   155→  if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   156→   156→  if (secs < 86400) return `${Math.floor(secs / 3600)}h ago`;
   157→   157→  if (secs < 86400 * 30) return `${Math.floor(secs / 86400)}d ago`;
   158→   158→  return fmtDate(d);
   159→   159→}

   636→   636→/* -------------------------------------------------------------- breakdown */
   637→   637→
   638→   638→const BREAKDOWN_MONEY_KEYS = new Set([
   639→   639→  "base", "percent_part", "fixed_fee_allocated", "discount_allocated", "shipping_allocated",
   640→   640→]);
   641→   641→
   642→   642→function breakdownValue(key: string, v: unknown): string | null {
   643→   643→  if (v === null || v === undefined) return null;
   644→   644→  if (key === "flags") {
   645→   645→    const arr = Array.isArray(v) ? v.map(String) : [];
   646→   646→    return arr.length ? arr.join(", ") : null;
   647→   647→  }
   648→   648→  if (BREAKDOWN_MONEY_KEYS.has(key)) return formatMinor(Number(v));
   649→   649→  if (typeof v === "number") return key === "percent" ? `${v}%` : String(v);
   650→   650→  if (typeof v === "string") return v;
   651→   651→  return JSON.stringify(v);
   652→   652→}

   654→   654→/** Renders a commission breakdownJson payload as key:value chips. */
   655→   655→export function BreakdownChips({ json, className }: { json?: string | null; className?: string }) {
   656→   656→  const data = parseJsonSafe<Record<string, unknown>>(json, {});
   657→   657→  const entries = Object.entries(data).filter(([k, v]) => breakdownValue(k, v) !== null);
   658→   658→  if (entries.length === 0) return <span className="text-xs text-muted-foreground">No breakdown snapshot</span>;
   659→   659→  return (
   660→   660→    <div className={cn("flex flex-wrap gap-1.5", className)}>
   661→   661→      {entries.map(([k, v]) => {
   662→   662→        const val = breakdownValue(k, v) as string;
   663→   663→        const highlight = k === "rule_source";
   664→   664→        return (

   666→   666→            key={k}
   667→   667→            className={cn(
   668→   668→              "inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[11px]",
   669→   669→              highlight ? TONE_CLASS.teal : "bg-muted/60 text-muted-foreground"
   670→   670→            )}
   671→   671→          >
   672→   672→            <span className="font-medium capitalize">{k.replace(/_/g, " ")}</span>
   673→   673→            <span className="tabular-nums">{val}</span>

   682→   682→
   683→   683→function flatObj(v: unknown): Record<string, unknown> {
   684→   684→  return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : {};
   685→   685→}
   686→   686→
   687→   687→function diffVal(key: string, v: unknown): string {
   688→   688→  if (v === null || v === undefined || v === "") return "—";
   689→   689→  if (typeof v === "object") return JSON.stringify(v);
   690→   690→  if (key.endsWith("Minor")) return formatMinor(Number(v));
   691→   691→  if (typeof v === "boolean") return v ? "true" : "false";
   692→   692→  return String(v);
   693→   693→}

   695→   695→/** Field-by-field before/after viewer for audit log rows. */
   696→   696→export function DiffViewer({ beforeJson, afterJson }: { beforeJson?: string | null; afterJson?: string | null }) {
   697→   697→  const before = flatObj(parseJsonSafe<unknown>(beforeJson, {}));
   698→   698→  const after = flatObj(parseJsonSafe<unknown>(afterJson, {}));
   699→   699→  const keys = Array.from(new Set([...Object.keys(before), ...Object.keys(after)])).sort();
   700→   700→  if (keys.length === 0) {
   701→   701→    return <p className="text-xs text-muted-foreground">No structured before/after snapshot recorded for this entry.</p>;
   702→   702→  }
   703→   703→  return (
   704→   704→    <div className="overflow-hidden rounded-lg border">
   705→   705→      <Table>
   706→   706→        <TableHeader>
   707→   707→          <TableRow>
   708→   708→            <TableHead className="w-40">Field</TableHead>
   709→   709→            <TableHead>Before</TableHead>
   710→   710→            <TableHead>After</TableHead>
   711→   711→          </TableRow>
   712→   712→        </TableHeader>
   713→   713→        <TableBody>
   714→   714→          {keys.map((k) => {
   715→   715→            const b = k in before ? diffVal(k, before[k]) : null;
   716→   716→            const a = k in after ? diffVal(k, after[k]) : null;
   717→   717→            const changed = b !== a;
   718→   718→            const added = b === null;
   719→   719→            const removed = a === null;
   720→   720→            return (
   721→   721→              <TableRow key={k} className={cn(changed && !removed && "bg-amber-50/60 dark:bg-amber-950/20")}>
   722→   722→                <TableCell className="font-mono text-[11px] font-medium">{k}</TableCell>
   723→   723→                <TableCell className={cn("text-xs tabular-nums", removed && "text-red-600 line-through dark:text-red-400")}>
   724→   724→                  {b ?? "—"}
   725→   725→                </TableCell>
   726→   726→                <TableCell className={cn("text-xs tabular-nums", added && "font-medium text-emerald-700 dark:text-emerald-400")}>
   727→   727→                  {a ?? "—"}
   728→   728→                </TableCell>
   729→   729→              </TableRow>
   730→   730→            );
   731→   731→          })}
   732→   732→        </TableBody>
   733→   733→      </Table>
   734→   734→    </div>
   735→   735→  );
   736→   736→}
