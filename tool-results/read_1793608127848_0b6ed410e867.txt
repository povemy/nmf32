    43→// F1 [Measured]: under concurrent same-seller confirmations, the unconditional
    44→// `ledgerAccount.upsert` on the SHARED SELLER/GATEWAY/COMMISSION account rows
    45→// was itself a hot-row write inside every Serializable tx (write-write SSI
    46→// conflicts + extra roundtrips). Accounts are effectively immutable once
    47→// created, so: read-first + per-process memo. The common case becomes a
    48→// read (no write, no conflict domain growth) and 1 roundtrip instead of an
    49→// upsert; first-creation still upserts (P2002-safe).
    50→const ensuredAccounts = new Set<string>();
    51→
    52→/** Same upsert, runnable on an explicit transaction client (FIN-1 atomicity). */
    53→async function ensureAccountTx(code: string, walletId: string | null | undefined, tx: Prisma.TransactionClient) {
    54→  if (ensuredAccounts.has(code)) return;
    55→  const meta = ACCOUNT_META[code] ?? {
    56→    name: code.startsWith("GATEWAY:") ? `Gateway ${code.slice(8)}` : code.startsWith("SELLER:") ? "Seller Payable" : code,
    57→    type: code.startsWith("SELLER:") ? "liability" : code.startsWith("GATEWAY:") ? "asset" : "asset",
    58→  };
    59→  const existing = await tx.ledgerAccount.findUnique({ where: { code }, select: { code: true } });
    60→  if (existing) {
    61→    ensuredAccounts.add(code);
    62→    return;
    63→  }
    64→  try {
    65→    await tx.ledgerAccount.upsert({
    66→      where: { code },
    67→      create: { code, name: meta.name, type: meta.type, walletId: walletId ?? null },
    68→      update: {},
    69→    });
    70→    ensuredAccounts.add(code);
    71→  } catch (e) {
    72→    // Raced creation from another tx — the row exists now; treat as ensured.
    73→    if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
    74→      ensuredAccounts.add(code);
    75→      return;
    76→    }
    77→    throw e;
    78→  }
    79→}

    94→// ---------------------------------------------------------------------------
    95→// FIN-1 POSTING CORE — atomic + idempotent.
    96→//
    97→// ATOMICITY: every posting runs inside a single prisma.$transaction (lines,
    98→// account upserts, wallet-cache refresh all commit or roll back together).
    99→// Recipes called with an outer transaction client run INSIDE it — never
   100→// nested — so a webhook/payout flow can group ledger + business writes.
   101→//
   102→// IDEMPOTENCY (dedupKey semantics — read before changing):
   103→//   • dedupKey is an OPTIONAL caller/recipe-supplied unique helper key stored
   104→//     on LedgerTransaction. When null/undefined the row is written with NULL —
   105→//     Postgres unique indexes treat NULLs as distinct, so non-deduped postings
   106→//     keep the legacy always-insert behavior (e.g. repeatable admin
   107→//     adjustments).
   108→//   • Recipes pass EXPLICIT keys only where a posting must happen EXACTLY
   109→//     once per business event (T1 per order, T2 per settlement ref, T3/T4/T4f
   110→//     per payout, T5 per refund row). Generic auto-keys are deliberately NOT
   111→//     derived here: a (txnType,refType,refId) default would falsely dedup
   112→//     legitimately-repeatable postings such as multi partial refunds per
   113→//     order (refType=order for every refund) — that is why postRefund takes
   114→//     the per-Refund-row refundId for its key.
   115→//   • On a P2002 unique violation the EXISTING transaction is fetched and
   116→//     returned with idempotentReplay:true (see wasReplay) instead of throwing
   117→//     — duplicate webhook deliveries / double-clicked admins converge on the
   118→//     original posting and never create a second one.
   119→// ---------------------------------------------------------------------------
   120→
   121→export type PostedTransaction = Prisma.LedgerTransactionGetPayload<{ include: { lines: true } }> & {
   122→  /** True when the dedupKey already existed and the ORIGINAL posting was returned. */
   123→  idempotentReplay?: boolean;
   124→};
   125→
   126→/** True when a posted result is actually the original txn replayed (dedup hit). */
   127→export function wasReplay(t: PostedTransaction): boolean {
   128→  return t.idempotentReplay === true;
   129→}

   131→/** Post a balanced double-entry transaction. Throws if unbalanced. */
   132→export async function postTransaction(input: {
   133→  txnType: string;
   134→  refType?: string | null;
   135→  refId?: string | null;
   136→  description?: string | null;
   137→  currency?: string;
   138→  lines: LedgerLineInput[];
   139→  occurredAt?: Date;
   140→  /** Optional unique idempotency key (see block comment above). NULL = no dedup. */
   141→  dedupKey?: string | null;
   142→  /** Existing transaction client — when omitted the posting wraps itself in db.$transaction. */
   143→  tx?: Prisma.TransactionClient;
   144→}): Promise<PostedTransaction> {
   145→  const run = async (tx: Prisma.TransactionClient): Promise<PostedTransaction> => {
   146→    const debit = input.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   147→    const credit = input.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   148→    if (debit !== credit || debit === 0) {
   149→      throw new Error(`LEDGER_UNBALANCED debit=${debit} credit=${credit} txn=${input.txnType}`);
   150→    }
   151→    for (const l of input.lines) await ensureAccountTx(l.accountCode, l.walletId ?? null, tx);
   152→    // With a dedupKey we may need to RECOVER from a unique violation inside
   153→    // this (possibly caller-provided) transaction. Postgres aborts the whole
   154→    // tx on a failed statement (25P02 — even a SELECT is refused afterwards),
   155→    // so the create is fenced with a SAVEPOINT: on P2002 we rewind to it and
   156→    // the original row becomes readable again. Without this, the
   157→    // replay-instead-of-throw contract below is unreachable inside a real tx.
   158→    const sp = input.dedupKey
   159→      ? `fin1_sp_${Date.now().toString(36)}${Math.floor(Math.random() * 1e9).toString(36)}`
   160→      : null;
   161→    if (sp) await tx.$executeRawUnsafe(`SAVEPOINT ${sp}`);
   162→    try {
   163→      const created = await tx.ledgerTransaction.create({
   164→        data: {
   165→          txnType: input.txnType,
   166→          refType: input.refType ?? null,
   167→          refId: input.refId ?? null,
   168→          description: input.description ?? null,
   169→          occurredAt: input.occurredAt ?? new Date(),
   170→          dedupKey: input.dedupKey ?? null,
   171→          lines: {
   172→            create: input.lines.map((l) => ({
   173→              accountCode: l.accountCode,
   174→              direction: l.direction,
   175→              amountMinor: l.amountMinor,
   176→              walletId: l.walletId ?? null,
   177→              availableAt: l.availableAt ?? null,
   178→              memo: l.memo ?? null,
   179→              // Denormalized ref (FIN-1 helper columns) so ref-scoped line
   180→              // queries don't need a join; mirrors the parent txn.
   181→              refType: input.refType ?? null,
   182→              refId: input.refId ?? null,
   183→            })),
   184→          },
   185→        },
   186→        include: { lines: true },
   187→      });
   188→      if (sp) await tx.$executeRawUnsafe(`RELEASE SAVEPOINT ${sp}`);
   189→      return created;
   190→    } catch (e) {
   191→      // Idempotent replay: the dedupKey already exists → return the ORIGINAL
   192→      // transaction instead of throwing (never double-post, never 500).
   193→      if (input.dedupKey && e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   194→        if (sp) await tx.$executeRawUnsafe(`ROLLBACK TO SAVEPOINT ${sp}`);
   195→        const existing = await tx.ledgerTransaction.findUnique({
   196→          where: { dedupKey: input.dedupKey },
   197→          include: { lines: true },
   198→        });
   199→        if (existing) return { ...existing, idempotentReplay: true };
   200→      }
   201→      throw e;
   202→    }
   203→  };
   204→  return input.tx ? run(input.tx) : db.$transaction(run);
   205→}

   209→export type WalletTruth = {
   210→  pendingMinor: number;
   211→  availableMinor: number;
   212→  lifetimeEarnedMinor: number;
   213→  lifetimeCommissionMinor: number;
   214→};
   215→
   216→/** Compute wallet balances from ledger lines (the source of truth). */
   217→export async function walletTruth(walletId: string, tx?: Prisma.TransactionClient): Promise<WalletTruth> {
   218→  const payable = ACCOUNTS.sellerPayable(walletId);
   219→  const client = tx ?? db;
   220→  const lines = await client.ledgerLine.findMany({
   221→    where: { accountCode: payable },
   222→    select: { direction: true, amountMinor: true, availableAt: true },
   223→  });
   224→  const now = new Date();
   225→  let totalCredits = 0, maturedCredits = 0, totalDebits = 0, lifetimeCommission = 0;
   226→  for (const l of lines) {
   227→    if (l.direction === "credit") {
   228→      totalCredits += l.amountMinor;
   229→      if (!l.availableAt || l.availableAt <= now) maturedCredits += l.amountMinor;
   230→    } else {
   231→      totalDebits += l.amountMinor;
   232→    }
   233→  }
   234→  // lifetime commission from revenue account lines tagged with this wallet? simpler: from txns T1 memo — use commission revenue lines with walletId
   235→  const commissionLines = await client.ledgerLine.findMany({
   236→    where: { accountCode: ACCOUNTS.revenueCommission(), walletId, direction: "credit" },
   237→    select: { amountMinor: true },
   238→  });
   239→  lifetimeCommission = commissionLines.reduce((s, l) => s + l.amountMinor, 0);
   240→  return {
   241→    pendingMinor: totalCredits - maturedCredits,
   242→    availableMinor: maturedCredits - totalDebits,
   243→    lifetimeEarnedMinor: totalCredits - lifetimeCommission,
   244→    lifetimeCommissionMinor: lifetimeCommission,
   245→  };
   246→}

   248→/**
   249→ * Refresh cached wallet columns from ledger truth (nightly invariant in prod;
   250→ * on-write here). Pass the recipe's `tx` so the refresh reads uncommitted
   251→ * lines and commits atomically with the posting itself.
   252→ *
   253→ * CONTENTION NOTE (measured): inside a Serializable tx this UPDATE on the
   254→ * wallet row is the hot-row epicenter of same-seller payment-confirmation
   255→ * storms (performance/REPORT.md §5.4 — 0/15 succeeded). Hot paths (webhook
   256→ * confirmation) now pass no `tx` and run this AFTER the money tx commits —
   257→ * the cache is a derived display value and payout authorization recomputes
   258→ * truth independently, so a failed/late post-commit refresh is safe (the
   259→ * invariant classifier reports time-maturity staleness as benign).
   260→ */
   261→export async function refreshWalletCache(walletId: string, tx?: Prisma.TransactionClient) {
   262→  const truth = await walletTruth(walletId, tx);
   263→  const client = tx ?? db;
   264→  await client.wallet.update({
   265→    where: { id: walletId },
   266→    data: {
   267→      pendingMinor: truth.pendingMinor,
   268→      availableMinor: truth.availableMinor,
   269→      lifetimeEarnedMinor: truth.lifetimeEarnedMinor,
   270→      lifetimeCommissionMinor: truth.lifetimeCommissionMinor,
   271→    },
   272→  });
   273→  return truth;
   274→}
   275→
   276→/**
   277→ * Best-effort cache refresh for display surfaces: never throws — on failure
   278→ * returns the last-known cache and the error, so a wallet PAGE never 500s
   279→ * because a concurrent payment tx briefly holds the wallet row. Money
   280→ * decisions never use this helper (they recompute truth transactionally).
   281→ */
   282→export async function safeRefreshWalletCache(walletId: string) {
   283→  try {
   284→    const truth = await refreshWalletCache(walletId);
   285→    return { refreshed: true as const, truth, error: null };
   286→  } catch (e) {
   287→    const cached = await db.wallet.findUnique({ where: { id: walletId } });
   288→    return {
   289→      refreshed: false as const,
   290→      truth: null,
   291→      error: e instanceof Error ? e.message : String(e),
   292→      cached,
   293→    };
   294→  }
   295→}

   299→/**
   300→ * T1 — order paid: DR GATEWAY:{p} total | CR SELLER:PAYABLE (available_at=paid+hold) | CR COMMISSION
   301→ * [+ CR AFFILIATE_LIABILITY pool when the order carries affiliate attribution].
   302→ * Dedup: exactly ONE T1 per order (`order_paid:order:{orderId}`) — a group
   303→ * webhook pays each order once; duplicate deliveries replay harmlessly
   304→ * (P2002 → original txn returned with idempotentReplay:true).
   305→ *
   306→ * AFFILIATE 4th line: `affiliateMinor` is the seller-funded commission pool
   307→ * Σ(OrderItem.affiliatePoolMinor) — already subtracted from sellerNetMinor
   308→ * by computeOrderCommissions, so credits sum to the total exactly:
   309→ *   sellerNet + commission + affiliate === total  (Option B invariant v2).
   310→ * Line omitted when affiliateMinor = 0 (legacy rows stay byte-compatible).
   311→ */
   312→export async function postOrderPaid(input: {
   313→  orderId: string;
   314→  orderNumber: string;
   315→  walletId: string;
   316→  provider: string;
   317→  totalMinor: number;
   318→  sellerNetMinor: number;
   319→  commissionMinor: number;
   320→  /** Seller-funded affiliate pool for this order (0 ⇒ no affiliate line). */
   321→  affiliateMinor?: number;
   322→  holdUntil: Date;
   323→  currency?: string;
   324→  occurredAt?: Date;
   325→  tx?: Prisma.TransactionClient;
   326→  /** Set false on hot paths (payment confirmation) that refresh the cache
   327→   *  AFTER the money tx commits instead of inside it (see refreshWalletCache
   328→   *  contention note). Default true preserves every other caller. */
   329→  refreshCache?: boolean;
   330→}) {
   331→  const lines: LedgerLineInput[] = [
   332→    { accountCode: ACCOUNTS.gateway(input.provider), direction: "debit", amountMinor: input.totalMinor, memo: `Order ${input.orderNumber}` },
   333→    { accountCode: ACCOUNTS.sellerPayable(input.walletId), direction: "credit", amountMinor: input.sellerNetMinor, walletId: input.walletId, availableAt: input.holdUntil, memo: `Seller net for ${input.orderNumber}` },
   334→    { accountCode: ACCOUNTS.revenueCommission(), direction: "credit", amountMinor: input.commissionMinor, walletId: input.walletId, memo: `Commission for ${input.orderNumber}` },
   335→  ];
   336→  if (input.affiliateMinor && input.affiliateMinor > 0) {
   337→    lines.push({ accountCode: ACCOUNTS.affiliateLiability(), direction: "credit", amountMinor: input.affiliateMinor, memo: `Affiliate pool for ${input.orderNumber}` });
   338→  }
   339→  const t = await postTransaction({
   340→    txnType: "order_paid",
   341→    refType: "order",
   342→    refId: input.orderId,
   343→    description: `Order ${input.orderNumber} paid via ${input.provider}`,
   344→    dedupKey: `order_paid:order:${input.orderId}`,
   345→    lines,
   346→    occurredAt: input.occurredAt,
   347→    tx: input.tx,
   348→  });
   349→  if (input.refreshCache !== false) await refreshWalletCache(input.walletId, input.tx);
   350→  return t;
   351→}

   445→// ------------------------------------------------------- AFFILIATE PAYOUTS
   446→// Manual controlled workflow (mirrors the seller payout recipes, but the
   447→// liability side is PLATFORM:AFFILIATE_LIABILITY, never a seller wallet):
   448→//   T-A3 approve  : DR AFFILIATE_LIABILITY | CR PAYOUT_CLEARING
   449→//   T-A4 mark-paid: DR PAYOUT_CLEARING     | CR BANK_MAIN
   450→//   T-A4f fail    : DR PAYOUT_CLEARING     | CR AFFILIATE_LIABILITY (release)
   451→// Idempotency: exactly one posting per payout row per stage (dedupKeys).

   501→/**
   502→ * T5 — refund: DR SELLER:PAYABLE (seller share) + DR COMMISSION (reversal)
   503→ * [+ DR AFFILIATE_LIABILITY (affiliate reversal)] | CR GATEWAY:{p}
   504→ * Dedup: exactly ONE T5 per Refund ROW (`refund:refund:{refundId}`) — an order
   505→ * may legally receive MULTIPLE partial refunds, so the key is per refund row,
   506→ * NOT per order (a (txnType,order) key would false-dedup the 2nd partial).
   507→ *
   508→ * AFFILIATE: `affiliateReversalMinor` mirrors commissionReversalMinor —
   509→ * derived proportionally from the frozen OrderItem.affiliatePoolMinor
   510→ * snapshots, clamped to the remaining reversible pool. The seller share
   511→ * shrinks by the same amount (the pool was carved from seller proceeds at
   512→ * checkout, so a refund restores it symmetrically).
   513→ */
   514→export async function postRefund(input: {
   515→  orderId: string;
   516→  orderNumber: string;
   517→  walletId: string;
   518→  provider: string;
   519→  amountMinor: number;
   520→  commissionReversalMinor: number;
   521→  /** Affiliate pool reversal for this refund (0 ⇒ no affiliate line). */
   522→  affiliateReversalMinor?: number;
   523→  /** C1.2 (Phase 2.2): the portion of the affiliate reversal whose liability
   524→   *  already moved into PAYOUT_CLEARING (an APPROVED in-flight affiliate
   525→   *  payout claimed the entries; T-A3 posted). That portion debits
   526→   *  PAYOUT_CLEARING instead of AFFILIATE_LIABILITY so neither account goes
   527→   *  negative and the clearing account nets to zero across T-A3 / this line /
   528→   *  T-A4(-f) at the adjusted payout amount. Must be ≤ affiliateReversalMinor. */
   529→  affiliateClearingMinor?: number;
   530→  /** The Refund row driving this posting — required for its dedup key. */
   531→  refundId: string;
   532→  tx?: Prisma.TransactionClient;
   533→}) {
   534→  const affiliateReversal = Math.max(0, input.affiliateReversalMinor ?? 0);
   535→  const affiliateClearing = Math.min(Math.max(0, input.affiliateClearingMinor ?? 0), affiliateReversal);
   536→  const affiliateLiability = affiliateReversal - affiliateClearing;
   537→  const sellerShare = input.amountMinor - input.commissionReversalMinor - affiliateReversal;
   538→  const lines: LedgerLineInput[] = [
   539→    { accountCode: ACCOUNTS.sellerPayable(input.walletId), direction: "debit", amountMinor: sellerShare, walletId: input.walletId, memo: `Refund seller share ${input.orderNumber}` },
   540→    { accountCode: ACCOUNTS.revenueCommission(), direction: "debit", amountMinor: input.commissionReversalMinor, walletId: input.walletId, memo: `Commission reversal ${input.orderNumber}` },
   541→  ];
   542→  if (affiliateLiability > 0) {
   543→    lines.push({ accountCode: ACCOUNTS.affiliateLiability(), direction: "debit", amountMinor: affiliateLiability, memo: `Affiliate reversal ${input.orderNumber}` });
   544→  }
   545→  if (affiliateClearing > 0) {
   546→    // C1.2: liability for these entries was already parked in payout clearing
   547→    // by T-A3 — reversing it out of AFFILIATE_LIABILITY would drive that
   548→    // account negative; take it from the clearing account instead.
   549→    lines.push({ accountCode: ACCOUNTS.payoutClearing(), direction: "debit", amountMinor: affiliateClearing, memo: `Affiliate reversal (in-flight payout) ${input.orderNumber}` });
   550→  }
   551→  lines.push({ accountCode: ACCOUNTS.gateway(input.provider), direction: "credit", amountMinor: input.amountMinor, memo: `Refund ${input.orderNumber}` });
   552→  const t = await postTransaction({
   553→    txnType: "refund",
   554→    refType: "order",
   555→    refId: input.orderId,
   556→    description: `Refund for ${input.orderNumber}`,
   557→    dedupKey: `refund:refund:${input.refundId}`,
   558→    lines,
   559→    tx: input.tx,
   560→  });
   561→  await refreshWalletCache(input.walletId, input.tx);
   562→  return t;
   563→}

   599→// ------------------------------------------------------------- INVARIANTS
   600→
   601→export type WalletCacheDrift = {
   602→  walletId: string;
   603→  cachedAvailable: number;
   604→  cachedPending: number;
   605→  truthAvailable: number;
   606→  truthPending: number;
   607→  /** available cache − truth, pending cache − truth (signed). */
   608→  availableDelta: number;
   609→  pendingDelta: number;
   610→  /** Sum of holds whose availableAt matured after the cache's last write —
   611→   *  the full explanation when classification is TIME_MATURITY. */
   612→  maturedSinceLastWrite: number;
   613→  lastCacheWriteAt: Date;
   614→};
   615→
   616→export type LedgerInvariantResult = {
   617→  ok: boolean;
   618→  /** HARD problems only: unbalanced transactions + hard cache drift. */
   619→  problems: string[];
   620→  /** CACHE_VALUE_DRIFT / unexplained staleness — totals differ or the
   621→   *  staleness is NOT explained by holds maturing after the last write.
   622→   *  Always investigated; never auto-dismissed. */
   623→  hardDrift: WalletCacheDrift[];
   624→  /** Benign, self-healing staleness: totals equal AND the cache equals
   625→   *  ledger truth recomputed as-of its own last write — holds matured by
   626→   *  clock afterwards. Money correctness is NOT affected (payout
   627→   *  authorization recomputes truth transactionally). */
   628→  timeMaturityStale: WalletCacheDrift[];
   629→  txnCount: number;
   630→  walletCount: number;
   631→};
   632→
   633→/**
   634→ * Ledger invariants, classification-aware (mission: wallet-ledger drift).
   635→ *
   636→ * Wallet caches are event-driven derived values; hold credits mature by the
   637→ * CLOCK (availableAt <= now) with no triggering write, so a cache frozen
   638→ * since before a maturity is EXPECTED to disagree with truth-now while its
   639→ * TOTAL stays correct. The invariant therefore separates:
   640→ *   • unbalanced transactions            → hard, money correctness
   641→ *   • totals-differ / unexplained drift  → hard (CACHE_VALUE_DRIFT)
   642→ *   • explained time-maturity staleness  → benign, reported for visibility
   643→ * (Health checks must distinguish these — a stale display cache is not a
   644→ * lost-money event, and a nightly full-ledger scan on every poll is itself a
   645→ * hazard; see /api/health for the TTL-wrapped consumption.)
   646→ */
   647→export async function runLedgerInvariants(): Promise<LedgerInvariantResult> {
   648→  const problems: string[] = [];
   649→  const txns = await db.ledgerTransaction.findMany({ include: { lines: true } });
   650→  for (const t of txns) {
   651→    const debit = t.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   652→    const credit = t.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   653→    if (debit !== credit) problems.push(`txn ${t.id} (${t.txnType}) unbalanced: D${debit} C${credit}`);
   654→  }
   655→
   656→  const wallets = await db.wallet.findMany();
   657→  const payableCodes = wallets.map((w) => ACCOUNTS.sellerPayable(w.id));
   658→  const lines = payableCodes.length
   659→    ? await db.ledgerLine.findMany({
   660→        where: { accountCode: { in: payableCodes } },
   661→        select: { accountCode: true, direction: true, amountMinor: true, availableAt: true },
   662→      })
   663→    : [];
   664→  const byWallet = new Map<string, { direction: string; amountMinor: number; availableAt: Date | null }[]>();
   665→  for (const l of lines) {
   666→    // accountCode embeds the walletId: SELLER:{walletId}:PAYABLE
   667→    const wid = l.accountCode.slice("SELLER:".length, -":PAYABLE".length);
   668→    const arr = byWallet.get(wid) ?? [];
   669→    arr.push({ direction: l.direction, amountMinor: l.amountMinor, availableAt: l.availableAt });
   670→    byWallet.set(wid, arr);
   671→  }
   672→
   673→  const hardDrift: WalletCacheDrift[] = [];
   674→  const timeMaturityStale: WalletCacheDrift[] = [];
   675→  const now = new Date();
   676→  for (const w of wallets) {
   677→    const wLines = byWallet.get(w.id) ?? [];
   678→    let totalCredits = 0,
   679→      maturedNow = 0,
   680→      totalDebits = 0,
   681→      maturedAtWrite = 0,
   682→      maturedSinceWrite = 0;
   683→    const writeAt = w.updatedAt;
   684→    for (const l of wLines) {
   685→      if (l.direction === "credit") {
   686→        totalCredits += l.amountMinor;
   687→        const matured = !l.availableAt || l.availableAt <= now;
   688→        if (matured) maturedNow += l.amountMinor;
   689→        if (!l.availableAt || l.availableAt <= writeAt) maturedAtWrite += l.amountMinor;
   690→        else if (l.availableAt <= now) maturedSinceWrite += l.amountMinor;
   691→      } else {
   692→        totalDebits += l.amountMinor;
   693→      }
   694→    }
   695→    // Truth as-of NOW and as-of the cache's last write. Debits apply in both
   696→    // views (they are event-driven, not clock-driven).
   697→    const truthNowAvail = maturedNow - totalDebits;
   698→    const truthNowPend = totalCredits - maturedNow;
   699→    const truthAtWriteAvail = maturedAtWrite - totalDebits;
   700→    const truthAtWritePend = totalCredits - maturedAtWrite;
   701→
   702→    if (truthNowAvail === w.availableMinor && truthNowPend === w.pendingMinor) continue; // MATCH
   703→
   704→    const base: WalletCacheDrift = {
   705→      walletId: w.id,
   706→      cachedAvailable: w.availableMinor,
   707→      cachedPending: w.pendingMinor,
   708→      truthAvailable: truthNowAvail,
   709→      truthPending: truthNowPend,
   710→      availableDelta: w.availableMinor - truthNowAvail,
   711→      pendingDelta: w.pendingMinor - truthNowPend,
   712→      maturedSinceLastWrite: maturedSinceWrite,
   713→      lastCacheWriteAt: writeAt,
   714→    };
   715→    const totalsEqual = w.availableMinor + w.pendingMinor === truthNowAvail + truthNowPend;
   716→    if (totalsEqual && truthAtWriteAvail === w.availableMinor && truthAtWritePend === w.pendingMinor) {
   717→      timeMaturityStale.push(base); // benign, clock-driven, self-heals on next event/read
   718→    } else {
   719→      hardDrift.push(base);
   720→      problems.push(
   721→        `wallet ${w.id} cache drift (${totalsEqual ? "UNEXPLAINED staleness" : "CACHE_VALUE_DRIFT — totals differ"}): available cache=${w.availableMinor} truth=${truthNowAvail}, pending cache=${w.pendingMinor} truth=${truthNowPend}`
   722→      );
   723→    }
   724→  }
   725→  return { ok: problems.length === 0, problems, hardDrift, timeMaturityStale, txnCount: txns.length, walletCount: wallets.length };
   726→}

   757→// ------------------------------------------------------------- LEDGER-TRUTH STATEMENTS
   758→// INTEGRITY RULE (user directive): every number printed on a payout statement
   759→// must be DERIVED from ledger lines for the payout period — never decorated.
   760→// The lines always reconcile to the payout amount via an explicit
   761→// "Retained in wallet" balancing line, so Σ(lines) ≡ net payout.
   762→
   763→export type LedgerStatement = {
   764→  /** Parsed shape of the JSON string returned by buildLedgerStatementFromLedger
   765→   *  (Payout.statementJson column stores the serialized form). */
   766→  sellerName: string;
   767→  periodFrom: string;
   768→  periodTo: string;
   769→  orderCount: number;
   770→  reconciled: true;
   771→  generatedAt: string;
   772→  lines: { label: string; amountMinor: number }[];
   773→};
   774→
   775→/** Aggregate a wallet's ledger activity for [periodFrom, periodTo] and build a
   776→ *  payout statement whose lines sum EXACTLY to `payoutAmountMinor`.
   777→ *  Components (all from ledger truth):
   778→ *    • sales        — payable credits from order_paid (seller net) + matching
   779→ *                     commission-revenue credits (gross sales)
   780→ *    • commission   — commission-revenue credits from order_paid in the period
   781→ *    • refunds      — payable debits from refund txns + commission reversals
   782→ *    • adjustments  — net admin_adjustment movement on the payable account
   783→ *    • retained     — residual balancing line so Σ(lines) ≡ payout (holds, caps,
   784→ *                     prior-balance carry)                                   */
   785→export async function buildLedgerStatementFromLedger(params: {
   786→  walletId: string;
   787→  sellerName: string;
   788→  payoutAmountMinor: number;
   789→  orderCount: number;
   790→  periodFrom: Date;
   791→  periodTo: Date;
   792→}): Promise<string> {
   793→  const { walletId, sellerName, payoutAmountMinor, orderCount, periodFrom, periodTo } = params;
   794→  const payable = ACCOUNTS.sellerPayable(walletId);
   795→  const commissionAccount = ACCOUNTS.revenueCommission();
   796→
   797→  // Wallet-scope lines in the period, joined to their txn type. Period is
   798→  // evaluated on txn.occurredAt (business time) — rows are often backdated
   799→  // (seed) or posted later (payout T3 lands after the statement snapshot).
   800→  const lines = await db.ledgerLine.findMany({
   801→    where: {
   802→      OR: [{ accountCode: payable }, { accountCode: commissionAccount, walletId }],
   803→      txn: { occurredAt: { gte: periodFrom, lte: periodTo } },
   804→    },
   805→    select: { accountCode: true, direction: true, amountMinor: true, txn: { select: { txnType: true } } },
   806→  });
   807→
   808→  let salesNet = 0; // payable credits from order_paid
   809→  let commission = 0; // commission-revenue credits from order_paid
   810→  let refundSellerShare = 0; // payable debits from refund txns
   811→  let refundCommissionReversal = 0; // commission debits from refund txns
   812→  let adjustments = 0; // admin_adjustment net (credits − debits on payable)
   813→
   814→  for (const l of lines) {
   815→    const type = l.txn.txnType;
   816→    if (l.accountCode === payable) {
   817→      if (l.direction === "credit") {
   818→        if (type === "order_paid") salesNet += l.amountMinor;
   819→        else if (type === "admin_adjustment") adjustments += l.amountMinor;
   820→      } else {
   821→        if (type === "refund") refundSellerShare += l.amountMinor;
   822→        else if (type === "admin_adjustment") adjustments -= l.amountMinor;
   823→      }
   824→    } else if (l.accountCode === commissionAccount) {
   825→      if (l.direction === "credit") {
   826→        if (type === "order_paid") commission += l.amountMinor;
   827→      } else if (type === "refund") {
   828→        refundCommissionReversal += l.amountMinor;
   829→      }
   830→    }
   831→  }
   832→
   833→  const sales = salesNet + commission; // gross sales for the period
   834→  const refunds = refundSellerShare + refundCommissionReversal;
   835→  const netMovement = sales - commission - refunds + adjustments;
   836→  // Balancing line so Σ(statement lines) ≡ net payout: when the period's net
   837→  // movement exceeds the payout (pending holds, min-payout cap, prior carry),
   838→  // the difference is "retained"; when a payout draws on prior balance, it's
   839→  // a positive carry line. Either way the statement always reconciles.
   840→  const retained = payoutAmountMinor - netMovement;
   841→
   842→  const linesOut: { label: string; amountMinor: number }[] = [];
   843→  if (sales !== 0) linesOut.push({ label: "Sales (settled to available)", amountMinor: sales });
   844→  if (commission !== 0) linesOut.push({ label: "Platform commission", amountMinor: -commission });
   845→  if (refunds !== 0) linesOut.push({ label: "Refunds", amountMinor: -refunds });
   846→  if (adjustments !== 0) linesOut.push({ label: "Adjustments", amountMinor: adjustments });
   847→  if (retained !== 0) linesOut.push({ label: "Retained in wallet (hold/cap — not paid out)", amountMinor: retained });
   848→
   849→  return toJson({
   850→    sellerName,
   851→    periodFrom: periodFrom.toISOString(),
   852→    periodTo: periodTo.toISOString(),
   853→    orderCount,
   854→    reconciled: true,
   855→    generatedAt: new Date().toISOString(),
   856→    lines: linesOut,
   857→  });
   858→}
   859→
