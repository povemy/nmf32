    78→/** Total unread seller messages — powers the "My messages" header badge.
    79→ *  Polls lightly + refreshes when the messages view is opened/left. */
    80→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    81→  const [unread, setUnread] = useState(0);
    82→  useEffect(() => {
    83→    if (!shopperSignedIn) {
    84→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    85→      setUnread(0);
    86→      return;
    87→    }
    88→    let alive = true;
    89→    const load = () =>
    90→      fetchJson<{ total: number }>("/api/chat/unread")
    91→        .then((d) => alive && setUnread(d.total ?? 0))
    92→        .catch(() => {});
    93→    load();
    94→    const t = setInterval(load, 15000);
    95→    return () => {
    96→      alive = false;
    97→      clearInterval(t);
    98→    };
    99→  }, [shopperSignedIn]);
   100→  // instant re-sync when a thread is read inside the messages view (the view
   101→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   102→  useEffect(() => {
   103→    if (!shopperSignedIn) return;
   104→    const sync = () =>
   105→      fetchJson<{ total: number }>("/api/chat/unread")
   106→        .then((d) => setUnread(d.total ?? 0))
   107→        .catch(() => {});
   108→    window.addEventListener("shopmesh:chat-sync", sync);
   109→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   110→  }, [shopperSignedIn]);
   111→  // opening (or leaving) the messages view should immediately re-sync the badge
   112→  useEffect(() => {
   113→    if (!shopperSignedIn) return;
   114→    const t = setTimeout(
   115→      () =>
   116→        fetchJson<{ total: number }>("/api/chat/unread")
   117→          .then((d) => setUnread(d.total ?? 0))
   118→          .catch(() => {}),
   119→      viewName === "messages" ? 600 : 0
   120→    );
   121→    return () => clearTimeout(t);
   122→  }, [shopperSignedIn, viewName]);
   123→  return unread;
   124→}

   126→export default function ShopperShell() {
   127→  const setSurface = useApp((s) => s.setSurface);
   128→  const activeStoreSlug = useApp((s) => s.activeStoreSlug);
   129→  const setActiveStoreSlug = useApp((s) => s.setActiveStoreSlug);
   130→
   131→  const [view, setView] = useState<ShopperView>({ name: "market" });
   132→  const [ready, setReady] = useState(false);
   133→  const [cartOpen, setCartOpen] = useState(false);
   134→  const [headerSearch, setHeaderSearch] = useState("");
   135→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   136→  // intentionally NOT part of ShopperView — a synthetic render override.
   137→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   138→  const restored = useRef(false);
   139→
   140→  // ---- mount: activeStoreSlug override, else restore last view --------------
   141→  useEffect(() => {
   142→    if (restored.current) return;
   143→    restored.current = true;
   144→    // claim deep-link wins over everything
   145→    try {
   146→      const params = new URLSearchParams(window.location.search);
   147→      const cj = params.get("claim_job");
   148→      const ct = params.get("claim_token");
   149→      if (cj && ct) {
   150→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   151→        setClaimLink({ jobId: cj, token: ct });
   152→        setReady(true);
   153→        return;
   154→      }
   155→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   156→      const sf = params.get("sf");
   157→      if (sf) {
   158→        const sp = params.get("sp");
   159→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   160→        setReady(true);
   161→        return;
   162→      }
   163→    } catch {
   164→      /* ignore */
   165→    }
   166→    if (activeStoreSlug) {
   167→      setView({ name: "storefront", slug: activeStoreSlug });
   168→      setActiveStoreSlug(null);
   169→    } else {
   170→      try {
   171→        const raw = window.localStorage.getItem(VIEW_KEY);
   172→        if (raw) {
   173→          const v = JSON.parse(raw) as ShopperView;
   174→          if (v && typeof v.name === "string") setView(v);
   175→        }
   176→      } catch {
   177→        /* ignore */
   178→      }
   179→    }
   180→    setReady(true);
   181→  }, [activeStoreSlug, setActiveStoreSlug]);
   182→
   183→  // ---- nav helper: persist + scroll to top ----------------------------------
   184→  const nav = useCallback((v: ShopperView) => {
   185→    setView(v);
   186→    try {
   187→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   188→    } catch {
   189→      /* ignore */
   190→    }
   191→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   192→  }, []);
   193→
   194→  // refresh storefront/store-page when slug changes externally
   195→  useEffect(() => {
   196→    if (!ready) return;
   197→    if (activeStoreSlug) {
   198→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   199→      setView({ name: "storefront", slug: activeStoreSlug });
   200→      setActiveStoreSlug(null);
   201→    }
   202→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   203→
   204→  const ctx = useMemo<ShopperCtx>(() => ({ nav, openCart: () => setCartOpen(true) }), [nav]);
   205→
   206→  const showChrome = !isStoreChrome(view);
   207→
   208→  return (
   209→    <div className="flex-1 flex flex-col min-h-screen">
   210→      <Toaster position="top-center" richColors />
   211→      <ShopperAuthDialog />
   212→
   213→      {showChrome ? (
   214→        <MarketHeader
   215→          view={view}
   216→          nav={nav}
   217→          openCart={() => setCartOpen(true)}
   218→          search={headerSearch}
   219→          setSearch={setHeaderSearch}
   220→        />
   221→      ) : null}
   222→
   223→      <AnimatePresence mode="wait">
   224→        <motion.main
   225→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : ""}`}
   226→          initial={{ opacity: 0, y: 8 }}
   227→          animate={{ opacity: 1, y: 0 }}
   228→          exit={{ opacity: 0, y: -8 }}
   229→          transition={{ duration: 0.18, ease: "easeOut" }}
   230→          className="flex-1 flex flex-col"
   231→        >
   232→          {claimLink ? null : (
   233→            <>
   234→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   235→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   236→              {view.name === "category" ? (
   237→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   238→              ) : null}
   239→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   240→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   241→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   242→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   243→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   244→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   245→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   246→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   247→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   248→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   249→            </>
   250→          )}
   251→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}
   252→        </motion.main>
   253→      </AnimatePresence>
   254→
   255→      {showChrome ? (
   256→        <ShopperFooter
   257→          onSeller={() => setSurface("seller")}
   258→          onLookup={() => nav({ name: "lookup" })}
   259→          onMarket={() => nav({ name: "market" })}
   260→          onGenerate={() => nav({ name: "generate" })}
   261→        />
   262→      ) : null}
   263→
   264→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   265→    </div>
   266→  );
   267→}

   718→function timeAgoShort(iso: string): string {
   719→  const diff = Date.now() - new Date(iso).getTime();
   720→  const m = Math.floor(diff / 60000);
   721→  if (m < 1) return "now";
   722→  if (m < 60) return `${m}m`;
   723→  const h = Math.floor(m / 60);
   724→  if (h < 24) return `${h}h`;
   725→  const d = Math.floor(h / 24);
   726→  if (d < 7) return `${d}d`;
   727→  return new Date(iso).toLocaleDateString(undefined, { day: "numeric", month: "short" });
   728→}

   730→function MessagesView({ ctx, initialThreadId }: { ctx: ShopperCtx; initialThreadId?: string }) {
   731→  const me = useApp((s) => s.me);
   732→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   733→  const shopper = me.loaded ? me.shopper : null;
   734→
   735→  const [threads, setThreads] = useState<ThreadRow[] | null>(null);
   736→  const [activeId, setActiveId] = useState<string | null>(initialThreadId ?? null);
   737→  const [messages, setMessages] = useState<ChatMsg[] | null>(null);
   738→  const [activeThread, setActiveThread] = useState<{ storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null } | null>(null);
   739→  const [context, setContext] = useState<ThreadContextCard | null>(null);
   740→  const [draft, setDraft] = useState("");
   741→  const [sending, setSending] = useState(false);
   742→  const socketRef = useRef<Socket | null>(null);
   743→  const scrollRef = useRef<HTMLDivElement>(null);
   744→
   745→  const loadThreads = useCallback(async () => {
   746→    try {
   747→      const rows = await fetchJson<ThreadRow[]>("/api/chat/threads");
   748→      setThreads(rows);
   749→      return rows;
   750→    } catch {
   751→      setThreads([]);
   752→      return [];
   753→    }
   754→  }, []);
   755→
   756→  const loadMessages = useCallback(async (threadId: string) => {
   757→    try {
   758→      const d = await fetchJson<{
   759→        thread: { storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null };
   760→        context: ThreadContextCard | null;
   761→        messages: ChatMsg[];
   762→      }>(`/api/chat/threads/${threadId}`);
   763→      setMessages(d.messages);
   764→      setActiveThread(d.thread);
   765→      setContext(d.context ?? null);
   766→    } catch {
   767→      setMessages([]);
   768→      setContext(null);
   769→    }
   770→  }, []);
   771→
   772→  // initial load
   773→  useEffect(() => {
   774→    if (!shopper) return;
   775→    let alive = true;
   776→    loadThreads().then(() => {
   777→      if (!alive) return;
   778→      // Only auto-open when resuming an intent (contact-seller deep link).
   779→      // Plain entry shows the list first so unread badges stay visible —
   780→      // auto-reading the newest thread would instantly consume the badge.
   781→      if (initialThreadId) setActiveId(initialThreadId);
   782→    });
   783→    return () => {
   784→      alive = false;
   785→    };
   786→  }, [shopper, initialThreadId, loadThreads]);
   787→
   788→  // load conversation when thread changes
   789→  useEffect(() => {
   790→    if (!activeId) return;
   791→    setMessages(null);
   792→    // the detail GET marks the thread read server-side — refresh the list so
   793→    // the unread pill on the row (and the header badge) drops immediately
   794→    loadMessages(activeId).then(() => {
   795→      loadThreads();
   796→      window.dispatchEvent(new Event("shopmesh:chat-sync"));
   797→    });
   798→  }, [activeId, loadMessages, loadThreads]);
   799→
   800→  // live updates via the socket.io chat hub
   801→  useEffect(() => {
   802→    if (!shopper) return;
   803→    const socket = io("/?XTransformPort=3003", {
   804→      transports: ["websocket", "polling"],
   805→      reconnection: true,
   806→      reconnectionAttempts: 5,
   807→      timeout: 10000,
   808→    });
   809→    socketRef.current = socket;
   810→    socket.on("connect", () => {
   811→      // join all thread rooms so the sidebar stays fresh too
   812→      const rooms = (threads ?? []).map((t) => `thread:${t.id}`);
   813→      socket.emit("join", { rooms });
   814→    });
   815→    socket.on("chat:message", (msg: ChatMsg & { threadId?: string }) => {
   816→      if (msg.threadId && msg.threadId === activeId) {
   817→        setMessages((prev) => (prev && !prev.some((m) => m.id === msg.id) ? [...prev, msg] : prev));
   818→        if (msg.senderType === "seller") {
   819→          // message arrived while the shopper is looking at this conversation —
   820→          // clear the badge straight away instead of waiting for the poller
   821→          void postJson(`/api/chat/threads/${activeId}/read`, {}).catch(() => {});
   822→        }
   823→      }
   824→      loadThreads();
   825→    });
   826→    return () => {
   827→      socket.disconnect();
   828→      socketRef.current = null;
   829→    };
   830→  }, [shopper, threads, activeId, loadThreads]);
   831→
   832→  // auto-scroll to latest
   833→  useEffect(() => {
   834→    const el = scrollRef.current;
   835→    if (el) el.scrollTop = el.scrollHeight;
   836→  }, [messages]);
