   318→/** Total unread shopper messages — powers the "Messages" nav badge. */
   319→function useSellerChatUnread(active: SellerNav["view"]) {
   320→  const [unread, setUnread] = useState(0);
   321→  useEffect(() => {
   322→    let alive = true;
   323→    const load = () =>
   324→      apiGet<{ total: number }>("/api/store/chat/unread")
   325→        .then((d) => alive && setUnread(d.total ?? 0))
   326→        .catch(() => {});
   327→    load();
   328→    const t = setInterval(load, 15000);
   329→    return () => {
   330→      alive = false;
   331→      clearInterval(t);
   332→    };
   333→  }, [active]);
   334→  // instant re-sync when a thread is read inside the messages view (the view
   335→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   336→  useEffect(() => {
   337→    const sync = () =>
   338→      apiGet<{ total: number }>("/api/store/chat/unread")
   339→        .then((d) => setUnread(d.total ?? 0))
   340→        .catch(() => {});
   341→    window.addEventListener("nomofee:chat-sync", sync);
   342→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   343→  }, []);
   344→  return unread;
   345→}

   487→function AuthScreen({ onAuthed }: { onAuthed: () => Promise<void> }) {
   488→  const setSurface = useApp((s) => s.setSurface);
   489→  const [loginEmail, setLoginEmail] = useState("");
   490→  const [loginPassword, setLoginPassword] = useState("");

   492→  const [reg, setReg] = useState({ fullName: "", displayName: "", email: "", password: "", phone: "", storeName: "" });
   493→
   494→  const goMarket = () => setSurface("shopper");
   495→
   496→  const login = async () => {
   497→    if (!loginEmail || !loginPassword) {
   498→      toast.error("Enter your email and password");
   499→      return;
   500→    }
   501→    setBusy(true);
   502→    const ok = await runMutation(async () => {
   503→      const body = await apiSend<{ token?: string }>("POST", "/api/auth/login", {
   504→        email: loginEmail,
   505→        password: loginPassword,
   506→        portal: "seller",
