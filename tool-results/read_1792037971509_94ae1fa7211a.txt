   191→/** Chip tone per shipment status — same Tailwind tokens as StatusChip. */
   192→function shipmentChipClass(status: string): string {
   193→  if (status === "DELIVERED") return "bg-emerald-100 text-emerald-800 border-emerald-200";
   194→  if (status === "EXCEPTION" || status === "RETURNED") return "bg-rose-100 text-rose-800 border-rose-200";
   195→  return "bg-muted text-muted-foreground border-border";
   196→}
   197→
   198→function shipmentBadgeTone(status: string): string {
   199→  if (status === "DELIVERED") return "bg-emerald-100 text-emerald-800 border-emerald-200";
   200→  if (status === "EXCEPTION" || status === "RETURNED") return "bg-rose-100 text-rose-800 border-rose-200";
   201→  return "bg-amber-100 text-amber-800 border-amber-200";
   202→}

   204→/** grams → human weight: 800 → "800 g", 2400 → "2.4 kg", 2000 → "2 kg". */
   205→function fmtWeight(grams: number): string {
   206→  if (!Number.isFinite(grams) || grams <= 0) return "—";
   207→  if (grams < 1000) return `${grams} g`;
   208→  return `${(grams / 1000).toFixed(2).replace(/\.?0+$/, "")} kg`;
   209→}
   210→
   211→/** ISO date → "Fri, 20 Sep". */
   212→function fmtDay(iso: string): string {
   213→  try {
   214→    return format(new Date(iso), "EEE, d MMM");
   215→  } catch {
   216→    return iso;
   217→  }
   218→}
   219→
   220→function fmtEddWindow(fromISO: string, toISO: string): string {
   221→  const from = fmtDay(fromISO);
   222→  const to = fmtDay(toISO);
   223→  return from === to ? from : `${from} – ${to}`;
   224→}

   236→export default function SellerOrders({ focusOrderId, onNavigate }: { focusOrderId?: string; onNavigate: (n: SellerNav) => void }) {
   237→  const [tab, setTab] = useState("to_fulfill");
   238→  const [search, setSearch] = useState("");
   239→  const [page, setPage] = useState(1);
   240→  const [rows, setRows] = useState<OrderRow[]>([]);
   241→  const [total, setTotal] = useState(0);
   242→  const [loading, setLoading] = useState(true);
   243→  const [selectedId, setSelectedId] = useState<string | null>(null);
   244→  const [tick, setTick] = useState(0);
   245→
   246→  useEffect(() => {
   247→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   248→    if (focusOrderId) setSelectedId(focusOrderId);
   249→  }, [focusOrderId]);
   250→
   251→  useEffect(() => {
   252→    let alive = true;
   253→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   254→    setLoading(true);
   255→    apiGetPaged<OrderRow[]>(`/api/store/orders?status=${tab}&search=${encodeURIComponent(search)}&page=${page}&size=12`)
   256→      .then(({ rows: r, meta }) => {
   257→        if (!alive) return;
   258→        setRows(r);
   259→        setTotal(meta?.total ?? r.length);
   260→      })
   261→      .catch(() => alive && setRows([]))
   262→      .finally(() => alive && setLoading(false));
   263→    return () => {
   264→      alive = false;
   265→    };
   266→  }, [tab, search, page, tick]);
   267→
   268→  if (selectedId) {
   269→    return (
   270→      <OrderDetailView
   271→        orderId={selectedId}
   272→        onBack={() => {
   273→          setSelectedId(null);
   274→          setTick((t) => t + 1);
   275→          onNavigate({ view: "orders" });
   276→        }}
   277→      />
   278→    );
   279→  }
   280→
   281→  const totalPages = Math.max(1, Math.ceil(total / 12));
   282→
   283→  return (
   284→    <div className="space-y-4">
   285→      <Tabs value={tab} onValueChange={(v) => { setTab(v); setPage(1); }}>
   286→        <TabsList className="flex w-full flex-wrap justify-start gap-1 sm:w-auto">
   287→          {TABS.map((t) => (
   288→            <TabsTrigger key={t.key} value={t.key} className="min-h-[36px]">
   289→              {t.label}
   290→            </TabsTrigger>
   291→          ))}
   292→        </TabsList>
   293→      </Tabs>
   294→
   295→      <div className="relative">
   296→        <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   297→        <Input
   298→          placeholder="Search order number, customer, email…"
   299→          className="pl-8 sm:max-w-sm"
   300→          value={search}
   301→          onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   302→        />
   303→      </div>
   304→
   305→      {loading ? (
   306→        <RowsSkeleton rows={5} />
   307→      ) : rows.length === 0 ? (
   308→        <EmptyState icon={<ClipboardList className="h-8 w-8" />} title="No orders here" hint="Orders appear as shoppers check out on your storefront." />
   309→      ) : (
   310→        <div className="overflow-x-auto rounded-xl border">
   311→          <Table>
   312→            <TableHeader>
   313→              <TableRow>
   314→                <TableHead>Order</TableHead>
   315→                <TableHead>Customer</TableHead>
   316→                <TableHead>Status</TableHead>
   317→                <TableHead className="text-right">Items</TableHead>
   318→                <TableHead className="text-right">Total</TableHead>
   319→                <TableHead className="text-right">Paid</TableHead>
   320→              </TableRow>
   321→            </TableHeader>
   322→            <TableBody>
   323→              {rows.map((o) => (
   324→                <TableRow key={o.id} className="cursor-pointer" onClick={() => setSelectedId(o.id)}>
   325→                  <TableCell>
   326→                    <p className="font-medium">{o.orderNumber}</p>
   327→                    <p className="text-xs text-muted-foreground">{ageOf(o.createdAt)}</p>
   328→                  </TableCell>
   329→                  <TableCell>
   330→                    <p className="max-w-40 truncate">{o.customerName}</p>
   331→                    <p className="max-w-40 truncate text-xs text-muted-foreground">{o.customerEmail}</p>
   332→                  </TableCell>
   333→                  <TableCell>
   334→                    <div className="flex flex-col items-start gap-1">
   335→                      <StatusChip status={o.status} />
   336→                      {o.shipmentStatus ? (
   337→                        <Badge
   338→                          variant="outline"
   339→                          className={cn("gap-1 border px-1.5 text-[10px] font-medium", shipmentBadgeTone(o.shipmentStatus))}
   340→                        >
   341→                          <Truck className="h-3 w-3" aria-hidden />
   342→                          {[o.shippingCourierName ?? o.trackingProvider ?? "Shipment", shipmentShortLabel(o.shipmentStatus)].join(" · ")}
   343→                        </Badge>
   344→                      ) : null}
   345→                    </div>
   346→                  </TableCell>
   347→                  <TableCell className="text-right tabular-nums">{o.itemCount}</TableCell>
   348→                  <TableCell className="text-right tabular-nums">{formatMinor(o.totalMinor)}</TableCell>
   349→                  <TableCell className="text-right text-xs text-muted-foreground">{o.paidAt ? fmtDateTime(o.paidAt) : "—"}</TableCell>
   350→                </TableRow>
   351→              ))}
   352→            </TableBody>
   353→          </Table>
   354→        </div>
   355→      )}
   356→
   357→      {rows.length > 0 ? (
   358→        <div className="flex items-center justify-between text-sm text-muted-foreground">
   359→          <span>{total} order{total === 1 ? "" : "s"} · page {page} of {totalPages}</span>
   360→          <div className="flex gap-2">
   361→            <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage((p) => p - 1)}>Previous</Button>
   362→            <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage((p) => p + 1)}>Next</Button>
   363→          </div>
   364→        </div>
   365→      ) : null}
   366→    </div>
   367→  );
   368→}

   372→function OrderDetailView({ orderId, onBack }: { orderId: string; onBack: () => void }) {
   373→  const { data: order, loading, error, refetch } = useApi<OrderDetail>(`/api/store/orders/${orderId}`);
   374→  const [shipOpen, setShipOpen] = useState(false);
   375→  const [refundOpen, setRefundOpen] = useState(false);
   376→  const [labelBusy, setLabelBusy] = useState(false);
   377→
   378→  if (loading) return <RowsSkeleton rows={6} />;
   379→  if (error || !order) {
   380→    return (
   381→      <div className="space-y-3">
   382→        <Button variant="ghost" onClick={onBack} className="gap-1.5"><ArrowLeft className="h-4 w-4" /> Back to orders</Button>
   383→        <Card><CardContent className="p-6 text-sm text-rose-600">Could not load order{error ? `: ${error}` : ""}</CardContent></Card>
   384→      </div>
   385→    );
   386→  }
   387→
   388→  const commission = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   389→  const net = order.totalMinor - commission;
   390→  const canFulfill = order.status === "paid" || order.status === "processing";
   391→  const canShip = (order.status === "paid" || order.status === "processing") && !order.shipmentStatus;
   392→  const canRefund = ["paid", "processing", "shipped", "delivered", "completed"].includes(order.status);
   393→  // Platform-booked shipments only (manual AWBs get no generated label).
   394→  const hasLabel = !!order.shipmentBookedAt && !!order.trackingNumber;
   395→
   396→  const fulfill = async () => {
   397→    const ok = await runMutation(() => apiSend("POST", `/api/store/orders/${order.id}/fulfill`), `Order ${order.orderNumber} marked fulfilled`);
   398→    if (ok) refetch();
   399→  };
   400→
   401→  const downloadLabel = async () => {
   402→    setLabelBusy(true);
   403→    try {
   404→      // A plain <a href> wouldn't carry the Bearer header — fetch the PDF as a
   405→      // blob and open it (same auth pattern as every other seller API call).
   406→      const res = await fetch(`/api/store/orders/${order.id}/label`, { cache: "no-store", headers: authHeaders() });
   407→      if (!res.ok) throw new Error("Label is not available for this shipment");
   408→      const blob = await res.blob();
   409→      const url = URL.createObjectURL(blob);
   410→      const win = window.open(url, "_blank", "noopener,noreferrer");
   411→      if (!win) {
   412→        // Popup blocked (sandboxed iframes) — fall back to a download click.
   413→        const a = document.createElement("a");
   414→        a.href = url;
   415→        a.download = `AWB-${order.trackingNumber ?? order.orderNumber}.pdf`;
   416→        document.body.appendChild(a);
   417→        a.click();
   418→        a.remove();
   419→      }
   420→      window.setTimeout(() => URL.revokeObjectURL(url), 60_000);
   421→    } catch (err) {
   422→      toast.error(err instanceof Error ? err.message : "Could not fetch the shipping label");
   423→    } finally {
   424→      setLabelBusy(false);
   425→    }
   426→  };
   427→
   428→  return (
   429→    <div className="space-y-4">
   430→      {/* print-only packing slip */}
   431→      <PackingSlip order={order} />
   432→
   433→      <div className="flex flex-wrap items-center gap-2 print:hidden">
   434→        <Button variant="ghost" onClick={onBack} className="gap-1.5">
   435→          <ArrowLeft className="h-4 w-4" /> Orders
   436→        </Button>
   437→        <h2 className="text-lg font-semibold">{order.orderNumber}</h2>
   438→        <StatusChip status={order.status} />
   439→        <span className="text-xs text-muted-foreground">
   440→          {order.channel} · created {fmtDateTime(order.createdAt)}
   441→        </span>
   442→        <Button variant="outline" size="sm" className="ml-auto gap-1.5" onClick={() => window.print()}>
   443→          <Printer className="h-3.5 w-3.5" /> Print packing slip
   444→        </Button>
   445→      </div>
   446→
   447→      {/* actions */}
   448→      <div className="flex flex-wrap gap-2 print:hidden">
   449→        {canFulfill ? (
   450→          <Button onClick={fulfill} className="gap-2 min-h-[44px]">
   451→            <PackageCheck className="h-4 w-4" /> Mark fulfilled
   452→          </Button>
   453→        ) : null}
   454→        {canShip ? (
   455→          <Button variant="outline" onClick={() => setShipOpen(true)} className="gap-2 min-h-[44px]">
   456→            <Truck className="h-4 w-4" /> Ship order…
   457→          </Button>
   458→        ) : null}
   459→        {canRefund ? (
   460→          <Button variant="outline" onClick={() => setRefundOpen(true)} className="gap-2 min-h-[44px] text-rose-600">
   461→            <Undo2 className="h-4 w-4" /> Request refund
   462→          </Button>
   463→        ) : null}
   464→      </div>
   465→
   466→      {/* shipment panel — replaces the Ship action once a shipment is booked.
   467→          Seller order detail doesn't include shipment events (see orders/[id] GET),
   468→          so status is shown as a chip; the buyer timeline lives on /me/orders. */}
   469→      {order.shipmentStatus ? (
   470→        <div className="flex flex-col gap-3 rounded-xl border bg-muted/40 p-4 sm:flex-row sm:items-center print:hidden">
   471→          <div className="flex min-w-0 flex-1 items-start gap-3">
   472→            <span className="mt-0.5 flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-primary/10 text-primary">
   473→              <Truck className="h-4 w-4" aria-hidden />
   474→            </span>
   475→            <div className="min-w-0">
   476→              <div className="flex flex-wrap items-center gap-2">
   477→                <p className="text-sm font-medium">{order.shippingCourierName ?? order.trackingProvider ?? "Shipment"}</p>
   478→                <Badge variant="outline" className={cn("border font-medium", shipmentChipClass(order.shipmentStatus))}>
   479→                  {shipmentLabel(order.shipmentStatus)}
   480→                </Badge>
   481→              </div>
   482→              <p className="mt-0.5 truncate text-xs text-muted-foreground">
   483→                AWB <span className="font-mono text-foreground">{order.trackingNumber}</span>
   484→              </p>
   485→            </div>
   486→          </div>
   487→          {hasLabel ? (
   488→            <Button variant="outline" size="sm" onClick={downloadLabel} disabled={labelBusy} className="gap-1.5 self-start sm:self-auto">
   489→              {labelBusy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Download className="h-3.5 w-3.5" />} Download label
   490→            </Button>
   491→          ) : null}
   492→        </div>
   493→      ) : order.trackingNumber ? (
   494→        <span className="flex items-center gap-1.5 rounded-lg border bg-muted/40 px-3 text-xs text-muted-foreground">
   495→          <Truck className="h-3.5 w-3.5" /> {order.trackingProvider}: {order.trackingNumber}
   496→        </span>
   497→      ) : null}
   498→
   499→      <div className="grid gap-4 lg:grid-cols-3">
   500→        {/* items */}
   501→        <Card className="lg:col-span-2">
   502→          <CardHeader><CardTitle className="text-sm">Items</CardTitle></CardHeader>
   503→          <CardContent className="overflow-x-auto">
   504→            <Table>
   505→              <TableHeader>
   506→                <TableRow>
   507→                  <TableHead>Item</TableHead>
   508→                  <TableHead className="text-right">Qty</TableHead>
   509→                  <TableHead className="text-right">Unit</TableHead>
   510→                  <TableHead className="text-right">Line total</TableHead>
   511→                  <TableHead className="text-right">Commission</TableHead>
   512→                </TableRow>
   513→              </TableHeader>
   514→              <TableBody>
   515→                {order.items.map((i) => (
   516→                  <TableRow key={i.id}>

   638→/**
   639→ * Task 10-b — ShipOrderDialog.
   640→ * Primary flow: POST /shipping/quote → pick a rate → POST /ship {mode:"platform", rateId}
   641→ * books through the platform's aggregator account (AWB + label + auto tracking).
   642→ * Legacy fallback ("I'll use my own courier"): POST /ship {provider, trackingNumber}.
   643→ */
   644→function ShipOrderDialog({ open, onOpenChange, order, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; order: OrderDetail; onDone: () => void }) {
   645→  const [phase, setPhase] = useState<"loading" | "ready" | "error">("loading");
   646→  const [quote, setQuote] = useState<ShippingQuote | null>(null);
   647→  const [quoteError, setQuoteError] = useState<string | null>(null);
   648→  const [rateId, setRateId] = useState<string>("");
   649→  const [manualOpen, setManualOpen] = useState(false);
   650→  const [provider, setProvider] = useState(order.trackingProvider ?? "");
   651→  const [tracking, setTracking] = useState("");
   652→  const [busy, setBusy] = useState(false);
   653→
   654→  const loadQuote = useCallback(async () => {
   655→    setPhase("loading");
   656→    setQuoteError(null);
   657→    try {
   658→      const q = await apiSend<ShippingQuote>("POST", `/api/store/orders/${order.id}/shipping/quote`);
   659→      setQuote(q);
   660→      // Rates arrive sorted cheapest-first — pre-select the cheapest.
   661→      setRateId(q.rates[0]?.rateId ?? "");
   662→      setPhase("ready");
   663→    } catch (err) {
   664→      setQuote(null);
   665→      const msg = err instanceof Error ? err.message : "Could not fetch shipping rates";
   666→      setQuoteError(msg);
   667→      toast.error(msg);
   668→      setPhase("error");
   669→    }
   670→  }, [order.id]);
   671→
   672→  useEffect(() => {
   673→    if (!open) return;
   674→    setProvider(order.trackingProvider ?? "");
   675→    setTracking("");
   676→    setManualOpen(false);
   677→    void loadQuote();
   678→  }, [open, order.id, order.trackingProvider, loadQuote]);
   679→
   680→  const selectedRate = quote?.rates.find((r) => r.rateId === rateId) ?? null;
   681→  const manualMode = manualOpen;
   682→
   683→  const confirm = async () => {
   684→    if (manualMode) {
   685→      if (!provider.trim() || !tracking.trim()) {
   686→        toast.error("Courier and tracking number are required");
   687→        return;
   688→      }
   689→    } else if (!rateId) {
   690→      toast.error("Select a shipping rate first");
   691→      return;
   692→    }
   693→
   694→    setBusy(true);
   695→    try {
   696→      const res = await apiSend<ShipResult>(
   697→        "POST",
   698→        `/api/store/orders/${order.id}/ship`,
   699→        manualMode
   700→          ? { provider: provider.trim(), trackingNumber: tracking.trim() }
   701→          : { mode: "platform", rateId }
   702→      );
   703→      const courier = res.shippingCourierName ?? res.trackingProvider ?? "courier";
   704→      toast.success(
   705→        manualMode
   706→          ? `${order.orderNumber} shipped — ${provider.trim()} · AWB ${res.trackingNumber}`
   707→          : `Shipment booked — ${courier} · AWB ${res.trackingNumber}`
   708→      );
   709→      onOpenChange(false);
   710→      onDone();
   711→    } catch (err) {
   712→      toast.error(err instanceof Error ? err.message : "Could not book the shipment");
   713→    } finally {
   714→      setBusy(false);
   715→    }
   716→  };
   717→
   718→  return (
   719→    <Dialog open={open} onOpenChange={onOpenChange}>
   720→      <DialogContent className="max-w-md">
   721→        <DialogHeader>
   722→          <DialogTitle>Ship {order.orderNumber}</DialogTitle>
   723→        </DialogHeader>
   724→
   725→        {phase === "loading" ? (
   726→          <div className="space-y-2" role="status" aria-label="Loading shipping rates">
   727→            <Skeleton className="h-9 w-full" />
   728→            <Skeleton className="h-[68px] w-full" />
   729→            <Skeleton className="h-[68px] w-full" />
   730→            <Skeleton className="h-[68px] w-full" />
   731→            <Skeleton className="h-[68px] w-full" />
   732→          </div>
   733→        ) : phase === "error" ? (
   734→          <div className="space-y-3">
   735→            <Alert>
   736→              <AlertDescription className="text-xs">{quoteError ?? "Could not fetch shipping rates"}</AlertDescription>
   737→            </Alert>
   738→            <Button variant="outline" onClick={() => void loadQuote()} disabled={busy} className="w-full min-h-[44px] gap-2">
   739→              <RefreshCw className="h-4 w-4" /> Try again
   740→            </Button>
   741→          </div>
   742→        ) : quote && quote.alreadyBooked ? (
   743→          <div className="space-y-3">
   744→            <Alert>
   745→              <AlertDescription className="text-xs">
   746→                A shipment is already booked for this order. Close this dialog to see the shipment details.
   747→              </AlertDescription>
   748→            </Alert>
   749→          </div>
   750→        ) : quote ? (
   751→          <div className="space-y-3.5">
   752→            {/* parcel summary */}
   753→            <div className="flex flex-wrap items-center gap-x-2 gap-y-1 rounded-lg border bg-muted/40 px-3 py-2 text-xs text-muted-foreground">
   754→              <MapPin className="h-3.5 w-3.5 shrink-0" aria-hidden />
   755→              <span>To {[quote.destState, quote.destPostcode].filter(Boolean).join(" ") || "—"}</span>
   756→              <span aria-hidden>·</span>
   757→              <Weight className="h-3.5 w-3.5 shrink-0" aria-hidden />
   758→              <span>{fmtWeight(quote.weightGrams)} parcel</span>

   760→
   761→            {/* header + explainer */}
   762→            <div className="space-y-1">
   763→              <p className="text-sm font-medium">Shipping rates</p>

   765→                Platform booking — label + auto tracking included. Shipping rides the platform&apos;s courier account; nothing is charged to your wallet.
   766→              </p>
   767→            </div>
   768→
   769→            {/* rates — cheapest first, pre-selected */}
   770→            <RadioGroup value={rateId} onValueChange={setRateId} className="max-h-72 gap-2 overflow-y-auto pr-1">
   771→              {quote.rates.map((r, i) => (
   772→                <label

   780→                  <RadioGroupItem id={`rate-${r.rateId}`} value={r.rateId} />

   787→                          Cheapest
   788→                        </Badge>
   789→                      ) : null}
   790→                    </span>

   844→function RefundDialog({ open, onOpenChange, order, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; order: OrderDetail; onDone: () => void }) {
   845→  const refunded = order.refunds.filter((r) => r.status !== "rejected").reduce((s, r) => s + r.amountMinor, 0);
   846→  const refundable = Math.max(0, order.totalMinor - refunded);
   847→  const [amount, setAmount] = useState<number>(refundable);
   848→  const [reason, setReason] = useState("");
   849→  const [restock, setRestock] = useState(true);
   850→  const [busy, setBusy] = useState(false);
   851→
   852→  useEffect(() => {
   853→    if (open) {
   854→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   855→      setAmount(Math.max(0, order.totalMinor - order.refunds.filter((r) => r.status !== "rejected").reduce((s, r) => s + r.amountMinor, 0)));
   856→      setReason("");
   857→      setRestock(true);
   858→    }
   859→  }, [open, order]);
   860→
   861→  const submit = async () => {
   862→    if (amount <= 0 || amount > refundable) {
   863→      toast.error(`Amount must be between RM 0 and ${formatMinor(refundable)}`);
   864→      return;
   865→    }
   866→    setBusy(true);
   867→    const ok = await runMutation(
   868→      () => apiSend("POST", `/api/store/orders/${order.id}/refund-request`, { amountMinor: amount, reason: reason || null, restock }),
   869→      "Refund requested — an admin will review it"
   870→    );
   871→    setBusy(false);
   872→    if (ok) {
   873→      onOpenChange(false);
   874→      onDone();
   875→    }
   876→  };
   877→
   878→  return (
   879→    <Dialog open={open} onOpenChange={onOpenChange}>
   880→      <DialogContent className="max-w-md">
   881→        <DialogHeader>
   882→          <DialogTitle>Request refund — {order.orderNumber}</DialogTitle>
   883→        </DialogHeader>
   884→        <div className="space-y-3.5">
   885→          <div className="space-y-1.5">
   886→            <Label>Amount (RM)</Label>
   887→            <MoneyInput valueMinor={amount} onChangeMinor={setAmount} />
   888→            <p className="text-xs text-muted-foreground">Refundable balance: {formatMinor(refundable)}</p>
   889→          </div>
   890→          <div className="space-y-1.5">
   891→            <Label htmlFor="refund-reason">Reason</Label>
   892→            <Textarea id="refund-reason" rows={3} value={reason} onChange={(e) => setReason(e.target.value)} placeholder="Damaged in transit, item out of stock…" />
   893→          </div>
   894→          <div className="flex items-center justify-between rounded-xl border p-3">
   895→            <div>
   896→              <p className="text-sm font-medium">Restock items</p>
   897→              <p className="text-xs text-muted-foreground">Return the items to inventory when the refund is processed.</p>
   898→            </div>
   899→            <Switch checked={restock} onCheckedChange={setRestock} aria-label="Restock items" />
   900→          </div>
   901→          <p className="text-xs text-muted-foreground">Requests are reviewed by SHOPMESH admins; the ledger reversal happens on approval.</p>
   902→          <Button onClick={submit} disabled={busy} className="w-full min-h-[44px] gap-2">
   903→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Undo2 className="h-4 w-4" />} Submit request
   904→          </Button>
   905→        </div>
   906→      </DialogContent>
   907→    </Dialog>
   908→  );
   909→}
