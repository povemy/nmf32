    81→/** Total unread seller messages — powers the "My messages" header badge.
    82→ *  Polls lightly + refreshes when the messages view is opened/left. */
    83→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    84→  const [unread, setUnread] = useState(0);
    85→  useEffect(() => {
    86→    if (!shopperSignedIn) {
    87→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    88→      setUnread(0);
    89→      return;
    90→    }
    91→    let alive = true;
    92→    const load = () =>
    93→      fetchJson<{ total: number }>("/api/chat/unread")
    94→        .then((d) => alive && setUnread(d.total ?? 0))
    95→        .catch(() => {});
    96→    load();
    97→    const t = setInterval(load, 15000);
    98→    return () => {
    99→      alive = false;
   100→      clearInterval(t);
   101→    };
   102→  }, [shopperSignedIn]);
   103→  // instant re-sync when a thread is read inside the messages view (the view
   104→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   105→  useEffect(() => {
   106→    if (!shopperSignedIn) return;
   107→    const sync = () =>
   108→      fetchJson<{ total: number }>("/api/chat/unread")
   109→        .then((d) => setUnread(d.total ?? 0))
   110→        .catch(() => {});
   111→    window.addEventListener("nomofee:chat-sync", sync);
   112→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   113→  }, [shopperSignedIn]);
   114→  // opening (or leaving) the messages view should immediately re-sync the badge
   115→  useEffect(() => {
   116→    if (!shopperSignedIn) return;
   117→    const t = setTimeout(
   118→      () =>
   119→        fetchJson<{ total: number }>("/api/chat/unread")
   120→          .then((d) => setUnread(d.total ?? 0))
   121→          .catch(() => {}),
   122→      viewName === "messages" ? 600 : 0
   123→    );
   124→    return () => clearTimeout(t);
   125→  }, [shopperSignedIn, viewName]);
   126→  return unread;
   127→}

   129→export default function ShopperShell() {

   135→  const [ready, setReady] = useState(false);
   136→  const [cartOpen, setCartOpen] = useState(false);
   137→  const [headerSearch, setHeaderSearch] = useState("");
   138→  // Back-navigation history stack (user request: back button on every page).
   139→  // viewRef mirrors `view` so nav() can push the PREVIOUS view without reading
   140→  // state inside the setState updater (StrictMode-safe).

   142→  const viewRef = useRef<ShopperView>({ name: "market" });
   143→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   144→  // intentionally NOT part of ShopperView — a synthetic render override.
   145→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   146→  const restored = useRef(false);
   147→
   148→  // ---- mount: activeStoreSlug override, else restore last view --------------
   149→  useEffect(() => {
   150→    if (restored.current) return;
   151→    restored.current = true;
   152→    // claim deep-link wins over everything
   153→    try {
   154→      const params = new URLSearchParams(window.location.search);
   155→      const cj = params.get("claim_job");
   156→      const ct = params.get("claim_token");
   157→      if (cj && ct) {
   158→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   159→        setClaimLink({ jobId: cj, token: ct });
   160→        setReady(true);
   161→        return;
   162→      }
   163→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   164→      const sf = params.get("sf");
   165→      if (sf) {
   166→        const sp = params.get("sp");
   167→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   168→        setReady(true);
   169→        return;
   170→      }
   171→    } catch {
   172→      /* ignore */
   173→    }
   174→    if (activeStoreSlug) {
   175→      setView({ name: "storefront", slug: activeStoreSlug });
   176→      setActiveStoreSlug(null);
   177→    } else {
   178→      try {
   179→        const raw = window.localStorage.getItem(VIEW_KEY);
   180→        if (raw) {
   181→          const v = JSON.parse(raw) as ShopperView;
   182→          if (v && typeof v.name === "string") setView(v);
   183→        }
   184→      } catch {
   185→        /* ignore */
   186→      }
   187→    }
   188→    setReady(true);
   189→  }, [activeStoreSlug, setActiveStoreSlug]);
   190→
   191→  // ---- nav helper: persist + scroll to top ----------------------------------
   192→  useEffect(() => {
   193→    viewRef.current = view;
   194→  }, [view]);
   195→
   196→  const nav = useCallback((v: ShopperView) => {
   197→    if (viewRef.current.name !== v.name || JSON.stringify(viewRef.current) !== JSON.stringify(v)) {
   198→      viewStack.current.push(viewRef.current);
   199→      if (viewStack.current.length > 25) viewStack.current.shift();
   200→    }
   201→    setView(v);
   202→    try {
   203→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   204→    } catch {
   205→      /* ignore */
   206→    }
   207→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   208→  }, []);
   209→
   210→  const back = useCallback(() => {
   211→    const prev = viewStack.current.pop() ?? { name: "market" as const };
   212→    setView(prev);
   213→    try {
   214→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(prev));
   215→    } catch {
   216→      /* ignore */
   217→    }
   218→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   219→  }, []);
   220→
   221→  // refresh storefront/store-page when slug changes externally
   222→  useEffect(() => {
   223→    if (!ready) return;
   224→    if (activeStoreSlug) {
   225→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   226→      setView({ name: "storefront", slug: activeStoreSlug });
   227→      setActiveStoreSlug(null);
   228→    }
   229→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   230→
   231→  const ctx = useMemo<ShopperCtx>(() => ({ nav, back, openCart: () => setCartOpen(true) }), [nav, back]);
   232→
   233→  const showChrome = !isStoreChrome(view);
   234→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   235→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   236→
   237→  return (
   238→    <div className="flex-1 flex flex-col min-h-screen">
   239→      <Toaster position="top-center" richColors />
   240→      <ShopperAuthDialog />
   241→
   242→      {showChrome ? (
   243→        <MarketHeader
   244→          view={view}
   245→          nav={nav}
   246→          openCart={() => setCartOpen(true)}
   247→          search={headerSearch}
   248→          setSearch={setHeaderSearch}
   249→        />
   250→      ) : null}
   251→
   252→      <AnimatePresence mode="wait">
   253→        <motion.main
   254→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : view.name === "me" ? view.tab ?? "" : ""}`}
   255→          initial={{ opacity: 0, y: 8 }}
   256→          animate={{ opacity: 1, y: 0 }}

   258→          transition={{ duration: 0.18, ease: "easeOut" }}
   259→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   260→        >
   261→          {claimLink ? null : (
   262→            <>
   263→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   264→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   265→              {view.name === "category" ? (
   266→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   267→              ) : null}
   268→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   269→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   270→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   271→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   272→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   273→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   274→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   275→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   276→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   277→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   278→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}
   279→            </>
   280→          )}
   281→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}

   285→      {showBottomNav ? (
   286→        <BottomNav
   287→          view={view}
   288→          nav={nav}
   289→          openCart={() => setCartOpen(true)}

   292→
   293→      {showChrome ? (

   297→          onMarket={() => nav({ name: "market" })}
   298→          onGenerate={() => nav({ name: "generate" })}
   299→        />
   300→      ) : null}
   301→
   302→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   303→    </div>
   304→  );
   305→}

   311→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   312→  const me = useApp((s) => s.me);
   313→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   314→  const signedIn = Boolean(me.loaded && me.shopper);
   315→  const { count } = useCart();
   316→  const unreadChats = useChatUnread(signedIn, view.name);
   317→
   318→  const items = [
   319→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   320→    {
   321→      key: "orders",
   322→      label: "Orders",
   323→      icon: Package,
   324→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   325→      badge: 0,
   326→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   327→    },
   328→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   329→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   330→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   331→  ];
   332→
   333→  return (
   334→    <nav
   335→      aria-label="Bottom navigation"
   336→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   337→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   338→    >
   339→      <div className="mx-auto grid max-w-lg grid-cols-5">
   340→        {items.map((it) => {
   341→          const Icon = it.icon;
   342→          return (
   343→            <button
   344→              key={it.key}
   345→              type="button"
   346→              onClick={it.onClick}
   347→              aria-label={it.label}
   348→              aria-current={it.active ? "page" : undefined}
   349→              className={cn(
   350→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   351→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   352→              )}
   353→            >
   354→              <span className="relative">
   355→                <Icon className="h-5 w-5" aria-hidden="true" />
   356→                {it.badge > 0 ? (
   357→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   358→                    {it.badge > 9 ? "9+" : it.badge}
   359→                  </span>
   360→                ) : null}
   361→              </span>
   362→              {it.label}
   363→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   364→            </button>
   365→          );
   366→        })}
   367→      </div>
   368→    </nav>
   369→  );
   370→}
