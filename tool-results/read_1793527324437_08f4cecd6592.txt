    43→    43→// F1 [Measured]: under concurrent same-seller confirmations, the unconditional
    44→    44→// `ledgerAccount.upsert` on the SHARED SELLER/GATEWAY/COMMISSION account rows
    45→    45→// was itself a hot-row write inside every Serializable tx (write-write SSI
    46→    46→// conflicts + extra roundtrips). Accounts are effectively immutable once
    47→    47→// created, so: read-first + per-process memo. The common case becomes a
    48→    48→// read (no write, no conflict domain growth) and 1 roundtrip instead of an
    49→    49→// upsert; first-creation still upserts (P2002-safe).
    50→    50→const ensuredAccounts = new Set<string>();

    52→    52→/** Same upsert, runnable on an explicit transaction client (FIN-1 atomicity). */
    53→    53→async function ensureAccountTx(code: string, walletId: string | null | undefined, tx: Prisma.TransactionClient) {
    54→    54→  if (ensuredAccounts.has(code)) return;
    55→    55→  const meta = ACCOUNT_META[code] ?? {
    56→    56→    name: code.startsWith("GATEWAY:") ? `Gateway ${code.slice(8)}` : code.startsWith("SELLER:") ? "Seller Payable" : code,
    57→    57→    type: code.startsWith("SELLER:") ? "liability" : code.startsWith("GATEWAY:") ? "asset" : "asset",
    58→    58→  };
    59→    59→  const existing = await tx.ledgerAccount.findUnique({ where: { code }, select: { code: true } });
    60→    60→  if (existing) {
    61→    61→    ensuredAccounts.add(code);
    62→    62→    return;
    63→    63→  }
    64→    64→  try {
    65→    65→    await tx.ledgerAccount.upsert({
    66→    66→      where: { code },
    67→    67→      create: { code, name: meta.name, type: meta.type, walletId: walletId ?? null },
    68→    68→      update: {},
    69→    69→    });
    70→    70→    ensuredAccounts.add(code);
    71→    71→  } catch (e) {
    72→    72→    // Raced creation from another tx — the row exists now; treat as ensured.
    73→    73→    if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
    74→    74→      ensuredAccounts.add(code);
    75→    75→      return;
    76→    76→    }
    77→    77→    throw e;
    78→    78→  }
    79→    79→}

    94→    94→// ---------------------------------------------------------------------------
    95→    95→// FIN-1 POSTING CORE — atomic + idempotent.
    96→    96→//
    97→    97→// ATOMICITY: every posting runs inside a single prisma.$transaction (lines,
    98→    98→// account upserts, wallet-cache refresh all commit or roll back together).
    99→    99→// Recipes called with an outer transaction client run INSIDE it — never
   100→   100→// nested — so a webhook/payout flow can group ledger + business writes.
   101→   101→//
   102→   102→// IDEMPOTENCY (dedupKey semantics — read before changing):
   103→   103→//   • dedupKey is an OPTIONAL caller/recipe-supplied unique helper key stored
   104→   104→//     on LedgerTransaction. When null/undefined the row is written with NULL —
   105→   105→//     Postgres unique indexes treat NULLs as distinct, so non-deduped postings
   106→   106→//     keep the legacy always-insert behavior (e.g. repeatable admin
   107→   107→//     adjustments).
   108→   108→//   • Recipes pass EXPLICIT keys only where a posting must happen EXACTLY
   109→   109→//     once per business event (T1 per order, T2 per settlement ref, T3/T4/T4f
   110→   110→//     per payout, T5 per refund row). Generic auto-keys are deliberately NOT
   111→   111→//     derived here: a (txnType,refType,refId) default would falsely dedup
   112→   112→//     legitimately-repeatable postings such as multi partial refunds per
   113→   113→//     order (refType=order for every refund) — that is why postRefund takes
   114→   114→//     the per-Refund-row refundId for its key.
   115→   115→//   • On a P2002 unique violation the EXISTING transaction is fetched and
   116→   116→//     returned with idempotentReplay:true (see wasReplay) instead of throwing
   117→   117→//     — duplicate webhook deliveries / double-clicked admins converge on the
   118→   118→//     original posting and never create a second one.
   119→   119→// ---------------------------------------------------------------------------
   120→   120→
   121→   121→export type PostedTransaction = Prisma.LedgerTransactionGetPayload<{ include: { lines: true } }> & {
   122→   122→  /** True when the dedupKey already existed and the ORIGINAL posting was returned. */
   123→   123→  idempotentReplay?: boolean;
   124→   124→};
   125→   125→
   126→   126→/** True when a posted result is actually the original txn replayed (dedup hit). */
   127→   127→export function wasReplay(t: PostedTransaction): boolean {
   128→   128→  return t.idempotentReplay === true;
   129→   129→}

   131→   131→/** Post a balanced double-entry transaction. Throws if unbalanced. */
   132→   132→export async function postTransaction(input: {
   133→   133→  txnType: string;
   134→   134→  refType?: string | null;
   135→   135→  refId?: string | null;
   136→   136→  description?: string | null;
   137→   137→  currency?: string;
   138→   138→  lines: LedgerLineInput[];
   139→   139→  occurredAt?: Date;
   140→   140→  /** Optional unique idempotency key (see block comment above). NULL = no dedup. */
   141→   141→  dedupKey?: string | null;
   142→   142→  /** Existing transaction client — when omitted the posting wraps itself in db.$transaction. */
   143→   143→  tx?: Prisma.TransactionClient;
   144→   144→}): Promise<PostedTransaction> {
   145→   145→  const run = async (tx: Prisma.TransactionClient): Promise<PostedTransaction> => {
   146→   146→    const debit = input.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   147→   147→    const credit = input.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   148→   148→    if (debit !== credit || debit === 0) {
   149→   149→      throw new Error(`LEDGER_UNBALANCED debit=${debit} credit=${credit} txn=${input.txnType}`);
   150→   150→    }
   151→   151→    for (const l of input.lines) await ensureAccountTx(l.accountCode, l.walletId ?? null, tx);
   152→   152→    // With a dedupKey we may need to RECOVER from a unique violation inside
   153→   153→    // this (possibly caller-provided) transaction. Postgres aborts the whole
   154→   154→    // tx on a failed statement (25P02 — even a SELECT is refused afterwards),
   155→   155→    // so the create is fenced with a SAVEPOINT: on P2002 we rewind to it and
   156→   156→    // the original row becomes readable again. Without this, the
   157→   157→    // replay-instead-of-throw contract below is unreachable inside a real tx.
   158→   158→    const sp = input.dedupKey
   159→   159→      ? `fin1_sp_${Date.now().toString(36)}${Math.floor(Math.random() * 1e9).toString(36)}`
   160→   160→      : null;
   161→   161→    if (sp) await tx.$executeRawUnsafe(`SAVEPOINT ${sp}`);
   162→   162→    try {
   163→   163→      const created = await tx.ledgerTransaction.create({
   164→   164→        data: {
   165→   165→          txnType: input.txnType,
   166→   166→          refType: input.refType ?? null,
   167→   167→          refId: input.refId ?? null,
   168→   168→          description: input.description ?? null,
   169→   169→          occurredAt: input.occurredAt ?? new Date(),
   170→   170→          dedupKey: input.dedupKey ?? null,
   171→   171→          lines: {
   172→   172→            create: input.lines.map((l) => ({
   173→   173→              accountCode: l.accountCode,
   174→   174→              direction: l.direction,
   175→   175→              amountMinor: l.amountMinor,
   176→   176→              walletId: l.walletId ?? null,
   177→   177→              availableAt: l.availableAt ?? null,
   178→   178→              memo: l.memo ?? null,
   179→   179→              // Denormalized ref (FIN-1 helper columns) so ref-scoped line
   180→   180→              // queries don't need a join; mirrors the parent txn.
   181→   181→              refType: input.refType ?? null,
   182→   182→              refId: input.refId ?? null,
   183→   183→            })),
   184→   184→          },
   185→   185→        },
   186→   186→        include: { lines: true },
   187→   187→      });
   188→   188→      if (sp) await tx.$executeRawUnsafe(`RELEASE SAVEPOINT ${sp}`);
   189→   189→      return created;
   190→   190→    } catch (e) {
   191→   191→      // Idempotent replay: the dedupKey already exists → return the ORIGINAL
   192→   192→      // transaction instead of throwing (never double-post, never 500).
   193→   193→      if (input.dedupKey && e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   194→   194→        if (sp) await tx.$executeRawUnsafe(`ROLLBACK TO SAVEPOINT ${sp}`);
   195→   195→        const existing = await tx.ledgerTransaction.findUnique({
   196→   196→          where: { dedupKey: input.dedupKey },
   197→   197→          include: { lines: true },
   198→   198→        });
   199→   199→        if (existing) return { ...existing, idempotentReplay: true };
   200→   200→      }
   201→   201→      throw e;
   202→   202→    }
   203→   203→  };
   204→   204→  return input.tx ? run(input.tx) : db.$transaction(run);
   205→   205→}

   209→   209→export type WalletTruth = {
   210→   210→  pendingMinor: number;
   211→   211→  availableMinor: number;
   212→   212→  lifetimeEarnedMinor: number;
   213→   213→  lifetimeCommissionMinor: number;

   216→   216→/** Compute wallet balances from ledger lines (the source of truth). */
   217→   217→export async function walletTruth(walletId: string, tx?: Prisma.TransactionClient): Promise<WalletTruth> {
   218→   218→  const payable = ACCOUNTS.sellerPayable(walletId);
   219→   219→  const client = tx ?? db;
   220→   220→  const lines = await client.ledgerLine.findMany({
   221→   221→    where: { accountCode: payable },
   222→   222→    select: { direction: true, amountMinor: true, availableAt: true },
   223→   223→  });
   224→   224→  const now = new Date();
   225→   225→  let totalCredits = 0, maturedCredits = 0, totalDebits = 0, lifetimeCommission = 0;
   226→   226→  for (const l of lines) {
   227→   227→    if (l.direction === "credit") {
   228→   228→      totalCredits += l.amountMinor;
   229→   229→      if (!l.availableAt || l.availableAt <= now) maturedCredits += l.amountMinor;
   230→   230→    } else {
   231→   231→      totalDebits += l.amountMinor;
   232→   232→    }
   233→   233→  }
   234→   234→  // lifetime commission from revenue account lines tagged with this wallet? simpler: from txns T1 memo — use commission revenue lines with walletId
   235→   235→  const commissionLines = await client.ledgerLine.findMany({
   236→   236→    where: { accountCode: ACCOUNTS.revenueCommission(), walletId, direction: "credit" },
   237→   237→    select: { amountMinor: true },
   238→   238→  });
   239→   239→  lifetimeCommission = commissionLines.reduce((s, l) => s + l.amountMinor, 0);
   240→   240→  return {
   241→   241→    pendingMinor: totalCredits - maturedCredits,
   242→   242→    availableMinor: maturedCredits - totalDebits,
   243→   243→    lifetimeEarnedMinor: totalCredits - lifetimeCommission,
   244→   244→    lifetimeCommissionMinor: lifetimeCommission,
   245→   245→  };
   246→   246→}

   248→   248→/**
   249→   249→ * Refresh cached wallet columns from ledger truth (nightly invariant in prod;
   250→   250→ * on-write here). Pass the recipe's `tx` so the refresh reads uncommitted
   251→   251→ * lines and commits atomically with the posting itself.
   252→   252→ *
   253→   253→ * CONTENTION NOTE (measured): inside a Serializable tx this UPDATE on the
   254→   254→ * wallet row is the hot-row epicenter of same-seller payment-confirmation
   255→   255→ * storms (performance/REPORT.md §5.4 — 0/15 succeeded). Hot paths (webhook
   256→   256→ * confirmation) now pass no `tx` and run this AFTER the money tx commits —
   257→   257→ * the cache is a derived display value and payout authorization recomputes
   258→   258→ * truth independently, so a failed/late post-commit refresh is safe (the
   259→   259→ * invariant classifier reports time-maturity staleness as benign).
   260→   260→ */
   261→   261→export async function refreshWalletCache(walletId: string, tx?: Prisma.TransactionClient) {
   262→   262→  const truth = await walletTruth(walletId, tx);
   263→   263→  const client = tx ?? db;
   264→   264→  await client.wallet.update({
   265→   265→    where: { id: walletId },
   266→   266→    data: {
   267→   267→      pendingMinor: truth.pendingMinor,
   268→   268→      availableMinor: truth.availableMinor,
   269→   269→      lifetimeEarnedMinor: truth.lifetimeEarnedMinor,
   270→   270→      lifetimeCommissionMinor: truth.lifetimeCommissionMinor,
   271→   271→    },
   272→   272→  });
   273→   273→  return truth;
   274→   274→}
   275→   275→
   276→   276→/**
   277→   277→ * Best-effort cache refresh for display surfaces: never throws — on failure
   278→   278→ * returns the last-known cache and the error, so a wallet PAGE never 500s
   279→   279→ * because a concurrent payment tx briefly holds the wallet row. Money
   280→   280→ * decisions never use this helper (they recompute truth transactionally).
   281→   281→ */
   282→   282→export async function safeRefreshWalletCache(walletId: string) {
   283→   283→  try {
   284→   284→    const truth = await refreshWalletCache(walletId);
   285→   285→    return { refreshed: true as const, truth, error: null };
   286→   286→  } catch (e) {
   287→   287→    const cached = await db.wallet.findUnique({ where: { id: walletId } });
   288→   288→    return {
   289→   289→      refreshed: false as const,
   290→   290→      truth: null,
   291→   291→      error: e instanceof Error ? e.message : String(e),
   292→   292→      cached,
   293→   293→    };
   294→   294→  }
   295→   295→}

   329→   329→  refreshCache?: boolean;

   392→   392→export async function postPayoutInitiated(input: { walletId: string; payoutId: string; referenceNo: string; amountMinor: number; tx?: Prisma.TransactionClient; refreshCache?: boolean }) {
   393→   393→  const t = await postTransaction({

   584→   584→// ------------------------------------------------------------- INVARIANTS
   585→   585→
   586→   586→export type WalletCacheDrift = {
   587→   587→  walletId: string;
   588→   588→  cachedAvailable: number;
   589→   589→  cachedPending: number;
   590→   590→  truthAvailable: number;
   591→   591→  truthPending: number;
   592→   592→  /** available cache − truth, pending cache − truth (signed). */
   593→   593→  availableDelta: number;
   594→   594→  pendingDelta: number;
   595→   595→  /** Sum of holds whose availableAt matured after the cache's last write —
   596→   596→   *  the full explanation when classification is TIME_MATURITY. */
   597→   597→  maturedSinceLastWrite: number;
   598→   598→  lastCacheWriteAt: Date;
   599→   599→};
   600→   600→
   601→   601→export type LedgerInvariantResult = {
   602→   602→  ok: boolean;
   603→   603→  /** HARD problems only: unbalanced transactions + hard cache drift. */
   604→   604→  problems: string[];
   605→   605→  /** CACHE_VALUE_DRIFT / unexplained staleness — totals differ or the
   606→   606→   *  staleness is NOT explained by holds maturing after the last write.
   607→   607→   *  Always investigated; never auto-dismissed. */
   608→   608→  hardDrift: WalletCacheDrift[];
   609→   609→  /** Benign, self-healing staleness: totals equal AND the cache equals
   610→   610→   *  ledger truth recomputed as-of its own last write — holds matured by
   611→   611→   *  clock afterwards. Money correctness is NOT affected (payout
   612→   612→   *  authorization recomputes truth transactionally). */
   613→   613→  timeMaturityStale: WalletCacheDrift[];
   614→   614→  txnCount: number;
   615→   615→  walletCount: number;
   616→   616→};
   617→   617→
   618→   618→/**
   619→   619→ * Ledger invariants, classification-aware (mission: wallet-ledger drift).
   620→   620→ *
   621→   621→ * Wallet caches are event-driven derived values; hold credits mature by the
   622→   622→ * CLOCK (availableAt <= now) with no triggering write, so a cache frozen
   623→   623→ * since before a maturity is EXPECTED to disagree with truth-now while its
   624→   624→ * TOTAL stays correct. The invariant therefore separates:
   625→   625→ *   • unbalanced transactions            → hard, money correctness
   626→   626→ *   • totals-differ / unexplained drift  → hard (CACHE_VALUE_DRIFT)
   627→   627→ *   • explained time-maturity staleness  → benign, reported for visibility
   628→   628→ * (Health checks must distinguish these — a stale display cache is not a
   629→   629→ * lost-money event, and a nightly full-ledger scan on every poll is itself a
   630→   630→ * hazard; see /api/health for the TTL-wrapped consumption.)
   631→   631→ */
   632→   632→export async function runLedgerInvariants(): Promise<LedgerInvariantResult> {
   633→   633→  const problems: string[] = [];
   634→   634→  const txns = await db.ledgerTransaction.findMany({ include: { lines: true } });
   635→   635→  for (const t of txns) {
   636→   636→    const debit = t.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   637→   637→    const credit = t.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   638→   638→    if (debit !== credit) problems.push(`txn ${t.id} (${t.txnType}) unbalanced: D${debit} C${credit}`);
   639→   639→  }
   640→   640→
   641→   641→  const wallets = await db.wallet.findMany();
   642→   642→  const payableCodes = wallets.map((w) => ACCOUNTS.sellerPayable(w.id));
   643→   643→  const lines = payableCodes.length
   644→   644→    ? await db.ledgerLine.findMany({
   645→   645→        where: { accountCode: { in: payableCodes } },
   646→   646→        select: { accountCode: true, direction: true, amountMinor: true, availableAt: true },
   647→   647→      })
   648→   648→    : [];
   649→   649→  const byWallet = new Map<string, { direction: string; amountMinor: number; availableAt: Date | null }[]>();
   650→   650→  for (const l of lines) {
   651→   651→    // accountCode embeds the walletId: SELLER:{walletId}:PAYABLE
   652→   652→    const wid = l.accountCode.slice("SELLER:".length, -":PAYABLE".length);
   653→   653→    const arr = byWallet.get(wid) ?? [];
   654→   654→    arr.push({ direction: l.direction, amountMinor: l.amountMinor, availableAt: l.availableAt });
   655→   655→    byWallet.set(wid, arr);
   656→   656→  }
   657→   657→
   658→   658→  const hardDrift: WalletCacheDrift[] = [];
   659→   659→  const timeMaturityStale: WalletCacheDrift[] = [];
   660→   660→  const now = new Date();
   661→   661→  for (const w of wallets) {
   662→   662→    const wLines = byWallet.get(w.id) ?? [];
   663→   663→    let totalCredits = 0,
   664→   664→      maturedNow = 0,
   665→   665→      totalDebits = 0,
   666→   666→      maturedAtWrite = 0,
   667→   667→      maturedSinceWrite = 0;
   668→   668→    const writeAt = w.updatedAt;
   669→   669→    for (const l of wLines) {
   670→   670→      if (l.direction === "credit") {
   671→   671→        totalCredits += l.amountMinor;
   672→   672→        const matured = !l.availableAt || l.availableAt <= now;
   673→   673→        if (matured) maturedNow += l.amountMinor;
   674→   674→        if (!l.availableAt || l.availableAt <= writeAt) maturedAtWrite += l.amountMinor;
   675→   675→        else if (l.availableAt <= now) maturedSinceWrite += l.amountMinor;
   676→   676→      } else {
   677→   677→        totalDebits += l.amountMinor;
   678→   678→      }
   679→   679→    }
   680→   680→    // Truth as-of NOW and as-of the cache's last write. Debits apply in both
   681→   681→    // views (they are event-driven, not clock-driven).
   682→   682→    const truthNowAvail = maturedNow - totalDebits;
   683→   683→    const truthNowPend = totalCredits - maturedNow;
   684→   684→    const truthAtWriteAvail = maturedAtWrite - totalDebits;
   685→   685→    const truthAtWritePend = totalCredits - maturedAtWrite;
   686→   686→
   687→   687→    if (truthNowAvail === w.availableMinor && truthNowPend === w.pendingMinor) continue; // MATCH
   688→   688→
   689→   689→    const base: WalletCacheDrift = {
   690→   690→      walletId: w.id,
   691→   691→      cachedAvailable: w.availableMinor,
   692→   692→      cachedPending: w.pendingMinor,
   693→   693→      truthAvailable: truthNowAvail,
   694→   694→      truthPending: truthNowPend,
   695→   695→      availableDelta: w.availableMinor - truthNowAvail,
   696→   696→      pendingDelta: w.pendingMinor - truthNowPend,
   697→   697→      maturedSinceLastWrite: maturedSinceWrite,
   698→   698→      lastCacheWriteAt: writeAt,
   699→   699→    };
   700→   700→    const totalsEqual = w.availableMinor + w.pendingMinor === truthNowAvail + truthNowPend;
   701→   701→    if (totalsEqual && truthAtWriteAvail === w.availableMinor && truthAtWritePend === w.pendingMinor) {
   702→   702→      timeMaturityStale.push(base); // benign, clock-driven, self-heals on next event/read
   703→   703→    } else {
   704→   704→      hardDrift.push(base);
   705→   705→      problems.push(
   706→   706→        `wallet ${w.id} cache drift (${totalsEqual ? "UNEXPLAINED staleness" : "CACHE_VALUE_DRIFT — totals differ"}): available cache=${w.availableMinor} truth=${truthNowAvail}, pending cache=${w.pendingMinor} truth=${truthNowPend}`
   707→   707→      );
   708→   708→    }
   709→   709→  }
   710→   710→  return { ok: problems.length === 0, problems, hardDrift, timeMaturityStale, txnCount: txns.length, walletCount: wallets.length };
   711→   711→}

   742→   742→// ------------------------------------------------------------- LEDGER-TRUTH STATEMENTS
   743→   743→// INTEGRITY RULE (user directive): every number printed on a payout statement
   744→   744→// must be DERIVED from ledger lines for the payout period — never decorated.
   745→   745→// The lines always reconcile to the payout amount via an explicit
   746→   746→// "Retained in wallet" balancing line, so Σ(lines) ≡ net payout.
   747→   747→
   748→   748→export type LedgerStatement = {
   749→   749→  /** Parsed shape of the JSON string returned by buildLedgerStatementFromLedger
   750→   750→   *  (Payout.statementJson column stores the serialized form). */
   751→   751→  sellerName: string;
   752→   752→  periodFrom: string;
   753→   753→  periodTo: string;
   754→   754→  orderCount: number;
   755→   755→  reconciled: true;
   756→   756→  generatedAt: string;
   757→   757→  lines: { label: string; amountMinor: number }[];
   758→   758→};
   759→   759→
   760→   760→/** Aggregate a wallet's ledger activity for [periodFrom, periodTo] and build a
   761→   761→ *  payout statement whose lines sum EXACTLY to `payoutAmountMinor`.
   762→   762→ *  Components (all from ledger truth):
   763→   763→ *    • sales        — payable credits from order_paid (seller net) + matching
   764→   764→ *                     commission-revenue credits (gross sales)
   765→   765→ *    • commission   — commission-revenue credits from order_paid in the period
   766→   766→ *    • refunds      — payable debits from refund txns + commission reversals
   767→   767→ *    • adjustments  — net admin_adjustment movement on the payable account
   768→   768→ *    • retained     — residual balancing line so Σ(lines) ≡ payout (holds, caps,
   769→   769→ *                     prior-balance carry)                                   */
   770→   770→export async function buildLedgerStatementFromLedger(params: {
   771→   771→  walletId: string;
   772→   772→  sellerName: string;
   773→   773→  payoutAmountMinor: number;
   774→   774→  orderCount: number;
   775→   775→  periodFrom: Date;
   776→   776→  periodTo: Date;
   777→   777→}): Promise<string> {
   778→   778→  const { walletId, sellerName, payoutAmountMinor, orderCount, periodFrom, periodTo } = params;
   779→   779→  const payable = ACCOUNTS.sellerPayable(walletId);
   780→   780→  const commissionAccount = ACCOUNTS.revenueCommission();
   781→   781→
   782→   782→  // Wallet-scope lines in the period, joined to their txn type. Period is
   783→   783→  // evaluated on txn.occurredAt (business time) — rows are often backdated
   784→   784→  // (seed) or posted later (payout T3 lands after the statement snapshot).
   785→   785→  const lines = await db.ledgerLine.findMany({
   786→   786→    where: {
   787→   787→      OR: [{ accountCode: payable }, { accountCode: commissionAccount, walletId }],
   788→   788→      txn: { occurredAt: { gte: periodFrom, lte: periodTo } },
   789→   789→    },
   790→   790→    select: { accountCode: true, direction: true, amountMinor: true, txn: { select: { txnType: true } } },
   791→   791→  });
   792→   792→
   793→   793→  let salesNet = 0; // payable credits from order_paid
   794→   794→  let commission = 0; // commission-revenue credits from order_paid
   795→   795→  let refundSellerShare = 0; // payable debits from refund txns
   796→   796→  let refundCommissionReversal = 0; // commission debits from refund txns
   797→   797→  let adjustments = 0; // admin_adjustment net (credits − debits on payable)
   798→   798→
   799→   799→  for (const l of lines) {
   800→   800→    const type = l.txn.txnType;
   801→   801→    if (l.accountCode === payable) {
   802→   802→      if (l.direction === "credit") {
   803→   803→        if (type === "order_paid") salesNet += l.amountMinor;
   804→   804→        else if (type === "admin_adjustment") adjustments += l.amountMinor;
   805→   805→      } else {
   806→   806→        if (type === "refund") refundSellerShare += l.amountMinor;
   807→   807→        else if (type === "admin_adjustment") adjustments -= l.amountMinor;
   808→   808→      }
   809→   809→    } else if (l.accountCode === commissionAccount) {
   810→   810→      if (l.direction === "credit") {
   811→   811→        if (type === "order_paid") commission += l.amountMinor;
   812→   812→      } else if (type === "refund") {
   813→   813→        refundCommissionReversal += l.amountMinor;
   814→   814→      }
   815→   815→    }
   816→   816→  }
   817→   817→
   818→   818→  const sales = salesNet + commission; // gross sales for the period
   819→   819→  const refunds = refundSellerShare + refundCommissionReversal;
   820→   820→  const netMovement = sales - commission - refunds + adjustments;
   821→   821→  // Balancing line so Σ(statement lines) ≡ net payout: when the period's net
   822→   822→  // movement exceeds the payout (pending holds, min-payout cap, prior carry),
   823→   823→  // the difference is "retained"; when a payout draws on prior balance, it's
   824→   824→  // a positive carry line. Either way the statement always reconciles.
   825→   825→  const retained = payoutAmountMinor - netMovement;
   826→   826→
   827→   827→  const linesOut: { label: string; amountMinor: number }[] = [];
   828→   828→  if (sales !== 0) linesOut.push({ label: "Sales (settled to available)", amountMinor: sales });
   829→   829→  if (commission !== 0) linesOut.push({ label: "Platform commission", amountMinor: -commission });
   830→   830→  if (refunds !== 0) linesOut.push({ label: "Refunds", amountMinor: -refunds });
   831→   831→  if (adjustments !== 0) linesOut.push({ label: "Adjustments", amountMinor: adjustments });
   832→   832→  if (retained !== 0) linesOut.push({ label: "Retained in wallet (hold/cap — not paid out)", amountMinor: retained });
   833→   833→
   834→   834→  return toJson({
   835→   835→    sellerName,
   836→   836→    periodFrom: periodFrom.toISOString(),
   837→   837→    periodTo: periodTo.toISOString(),
   838→   838→    orderCount,
   839→   839→    reconciled: true,
   840→   840→    generatedAt: new Date().toISOString(),
   841→   841→    lines: linesOut,
   842→   842→  });
   843→   843→}
