    88→    },
    89→  });
    90→  for (let i = 0; i < p.imgSeeds.length; i++) {
    91→    await db.productImage.create({
    92→      data: { productId: product.id, url: img(p.imgSeeds[i], 900, 900, p.title), sortOrder: i },
    93→    });
    94→  }
    95→  const variants = p.variants ?? [{ name: "Default", price: p.price, stock: p.stock ?? 40 }];
    96→  for (const v of variants) {
    97→    await db.productVariant.create({
    98→      data: {

   108→    });
   109→  }
   110→  return product;

   253→async function seedOrder(opts: {
   254→  storeId: string; storeName: string; walletId: string;
   255→  customer: { name: string; email: string; phone?: string };
   256→  items: SeedOrderItem[];
   257→  status: string; provider: string; daysBack: number;
   258→  discountMinor?: number; couponCode?: string; shippingMinor?: number;
   259→  withLedger?: boolean; withSettlement?: boolean;
   260→}) {
   261→  const createdAt = daysAgo(opts.daysBack, 9 + (opts.daysBack % 6));
   262→  const settings = await db.commissionSetting.findUnique({ where: { id: 1 } });
   263→  const holdDays = settings?.payoutHoldDays ?? 7;
   264→  const subtotal = opts.items.reduce((s, i) => s + i.unitPrice * i.qty, 0);
   265→  const discount = opts.discountMinor ?? 0;
   266→  const shipping = opts.shippingMinor ?? (subtotal >= 10000 ? 0 : 800);
   267→  const total = subtotal - discount + shipping;
   268→
   269→  const storeForCommission = await db.store.findUnique({ where: { id: opts.storeId } });
   270→  if (!storeForCommission?.sellerId) throw new Error(`seed order: store ${opts.storeId} or its seller missing`);
   271→
   272→  const computed = await computeOrderCommissions({
   273→    sellerId: storeForCommission.sellerId,
   274→    items: opts.items.map((i) => ({
   275→      title: i.title, unitPriceMinor: i.unitPrice, qty: i.qty,
   276→      productId: i.product.id, categoryId: i.product.categoryId, variantLabel: i.variantLabel ?? null,
   277→    })),
   278→    orderDiscountMinor: discount,
   279→    shippingMinorTotal: shipping,
   280→    now: createdAt,
   281→  });
   282→
   283→  const paid = !["payment_pending", "cancelled"].includes(opts.status);
   284→  const paidAt = paid ? createdAt : null;
   285→  const shippedAt = ["shipped", "delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 86400000) : null;
   286→  const deliveredAt = ["delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 3 * 86400000) : null;
   287→  const completedAt = opts.status === "completed" ? new Date(createdAt.getTime() + 9 * 86400000) : null;
   288→
   289→  const order = await db.order.create({
   290→    data: {
   291→      storeId: opts.storeId,
   292→      orderNumber: nextOrderNumber(),
   293→      customerName: opts.customer.name,
   294→      customerEmail: opts.customer.email,
   295→      customerPhone: opts.customer.phone ?? "60125550011",
   296→      status: opts.status,
   297→      channel: "storefront",
   298→      subtotalMinor: subtotal,
   299→      discountMinor: discount,
   300→      shippingMinor: shipping,
   301→      totalMinor: total,
   302→      couponCode: opts.couponCode ?? null,
   303→      shippingAddressJson: toJson({
   304→        line1: "42 Jalan Merdeka", city: "Kuala Lumpur", state: "Selangor", postcode: "43650", country: "MY",
   305→      }),
   306→      paymentProvider: opts.provider,
   307→      paidAt,
   308→      fulfilledAt: ["processing", "shipped", "delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 43200000) : null,
   309→      shippedAt,
   310→      deliveredAt,
   311→      completedAt,
   312→      createdAt,
   313→      updatedAt: createdAt,
   314→    },
   315→  });
   316→
   317→  for (let i = 0; i < opts.items.length; i++) {
   318→    const item = opts.items[i];
   319→    const c = computed.items[i];
   320→    await db.orderItem.create({
   321→      data: {
   322→        orderId: order.id,
   323→        productId: item.product.id,
   324→        variantId: item.variantId,
   325→        title: item.title,
   326→        variantLabel: item.variantLabel ?? null,
   327→        unitPriceMinor: item.unitPrice,
   328→        qty: item.qty,
   329→        lineSubtotalMinor: item.unitPrice * item.qty,
   330→        discountMinor: c.discountMinor,
   331→        shippingMinor: c.shippingMinor,
   332→        commissionPercent: c.percent,
   333→        commissionMinor: c.commissionMinor,
   334→        fixedFeeMinor: c.fixedFeeMinor,
   335→        sellerNetMinor: c.sellerNetMinor,
   336→        breakdownJson: toJson(c.breakdown),
   337→      },
   338→    });
   339→    await db.product.update({ where: { id: item.product.id }, data: { salesCount: { increment: item.qty } } });
   340→    await db.productVariant.update({ where: { id: item.variantId }, data: { stock: { decrement: item.qty } } });
   341→  }
   342→
   343→  const historyTo: string[] = [];
   344→  if (paid) historyTo.push("paid");
   345→  if (["processing", "shipped", "delivered", "completed"].includes(opts.status)) historyTo.push("processing");
   346→  if (shippedAt) historyTo.push("shipped");
   347→  if (deliveredAt) historyTo.push("delivered");
   348→  if (completedAt) historyTo.push("completed");
   349→  let prev = "payment_pending";
   350→  for (const st of historyTo) {
   351→    await db.orderStatusHistory.create({
   352→      data: { orderId: order.id, fromStatus: prev, toStatus: st, actor: st === "paid" ? "system" : "seller", createdAt: paidAt ?? createdAt },
   353→    });
   354→    prev = st;
   355→  }
   356→
   357→  if (opts.withLedger && paid) {
   358→    await db.paymentIntent.create({
   359→      data: {
   360→        orderId: order.id, provider: opts.provider,
   361→        providerRef: `ref_${order.orderNumber.toLowerCase()}`,
   362→        amountMinor: total, status: "succeeded", expiresAt: daysAhead(1), createdAt, updatedAt: createdAt,
   363→      },
   364→    });
   365→    await postOrderPaid({
   366→      orderId: order.id, orderNumber: order.orderNumber, walletId: opts.walletId,
   367→      provider: opts.provider, totalMinor: total,
   368→      sellerNetMinor: computed.orderSellerNetMinor, commissionMinor: computed.orderCommissionMinor,
   369→      holdUntil: new Date((paidAt as Date).getTime() + holdDays * 86400000),
   370→      occurredAt: paidAt as Date,
   371→    });
   372→    if (opts.withSettlement && completedAt) {
   373→      await postGatewaySettlement({
   374→        provider: opts.provider, amountMinor: total,
   375→        ref: `stl_${order.orderNumber.toLowerCase()}`, occurredAt: new Date((paidAt as Date).getTime() + 172800000),
   376→      });
   377→    }
   378→  } else if (opts.status === "payment_pending") {
   379→    await db.paymentIntent.create({
   380→      data: {
   381→        orderId: order.id, provider: opts.provider,
   382→        providerRef: `ref_${order.orderNumber.toLowerCase()}`,
   383→        amountMinor: total, status: "requires_payment", expiresAt: new Date(createdAt.getTime() + 1800000),
   384→        redirectUrl: `/pay/${`ref_${order.orderNumber.toLowerCase()}`}`,
   385→      },
   386→    });
   387→  }
   388→  return order;
   389→}
