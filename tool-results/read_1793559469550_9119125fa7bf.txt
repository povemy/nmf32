     3→     3→// docs/affiliate/*.md are the authoritative design docs.
     4→     4→//
     5→     5→// COMMERCIAL MODEL (fixed, spec §1/§8):
     6→     6→//   • Seller-funded commission pool per eligible OrderItem: P = B × R (bps,
     7→     7→//     integer sen). The pool is carved OUT of seller proceeds at checkout
     8→     8→//     (sellerNet already subtracts it — see computeOrderCommissions).
     9→     9→//   • Split of the POOL: S = floor(P × 10 / 100) to the direct affiliate's
    10→    10→//     ONE-LEVEL sponsor; D = P − S to the direct affiliate. No sponsor ⇒
    11→    11→//     D = P, S = 0. D + S = P ALWAYS (never exceeds the pool).
    12→    12→//   • Attribution travels UP exactly one generation. No recruitment bonuses,
    13→    13→//     no click rewards, no registration rewards — liabilities only ever
    14→    14→//     originate from genuine eligible paid sales.
    15→    15→//
    16→    16→// INTEGRITY CONTRACTS:
    17→    17→//   • Pool snapshots freeze on OrderItem at checkout (commission contract).
    18→    18→//   • AffiliateCommission rows post EXACTLY ONCE per order item at payment
    19→    19→//     confirmation (unique orderItemId; duplicate webhooks are no-ops).
    20→    20→//   • The 90/10 SPLIT is computed at POSTING time from the then-current
    21→    21→//     sponsor-override flag + sponsor eligibility, then never rewritten —
    22→    22→//     so the override can be disabled without corrupting existing records.
    23→    23→//   • Every money mutation rides the immutable ledger (lib/ledger.ts).

    86→    86→function clampInt(v: unknown, min: number, max: number, fallback: number): number {
    87→    87→  const n = typeof v === "number" ? v : Number(v);
    88→    88→  if (!Number.isFinite(n)) return fallback;
    89→    89→  return Math.min(max, Math.max(min, Math.round(n)));
    90→    90→}

   112→   112→/** Effective seller affiliate rate (bps) for a product, honouring every gate.
   113→   113→ *  0 ⇒ not enrolled (no commission). Product override ?? store default,
   114→   114→ *  clamped to the platform max; programme master switch must be on. */
   115→   115→export function effectiveAffiliateRateBps(
   116→   116→  s: AffiliateSettings,
   117→   117→  store: { affiliateOptIn: boolean; affiliateDefaultRateBps: number },
   118→   118→  product: { affiliateEnabled: boolean; affiliateRateBps: number | null }
   119→   119→): number {
   120→   120→  if (!s.enabled || !store.affiliateOptIn || !product.affiliateEnabled) return 0;
   121→   121→  const rate = product.affiliateRateBps ?? store.affiliateDefaultRateBps;
   122→   122→  return Math.max(0, Math.min(rate, s.maxRateBps));
   123→   123→}

   128→   128→/** Pool P = floor(B × R / 10^4) in integer sen.
   129→   129→ *  POST-IMPLEMENTATION AUDIT (P4): the ONE canonical rounding policy for the
   130→   130→ *  entire affiliate money path is FLOOR (commission-definitions.md §9 is
   131→   131→ *  normative: "Where this document conflicts with prose elsewhere, this
   132→   132→ *  document governs"). Previously half-up (platform percent convention) —
   133→   133→ *  a one-sen drift vs the normative doc. Same floor policy is used at the
   134→   134→ *  checkout snapshot, payment posting, refund reversal, reconciliation and
   135→   135→ *  every report. The platform commission's own percent part keeps its
   136→   136→ *  historical half-up convention (separate ledger line, unchanged). */
   137→   137→export function affiliatePoolMinor(baseMinor: number, rateBps: number): number {
   138→   138→  if (baseMinor <= 0 || rateBps <= 0) return 0;
   139→   139→  return Math.floor((baseMinor * rateBps) / 10000);
   140→   140→}
   141→   141→
   142→   142→/** 90/10 split of the POOL: S = floor(P × 10 / 100); D = P − S (D + S = P). */
   143→   143→export function splitAffiliatePool(poolMinor: number): { directMinor: number; sponsorMinor: number } {
   144→   144→  if (poolMinor <= 0) return { directMinor: 0, sponsorMinor: 0 };
   145→   145→  const sponsorMinor = Math.floor((poolMinor * 10) / 100);
   146→   146→  return { directMinor: poolMinor - sponsorMinor, sponsorMinor };
   147→   147→}

   204→   204→/** Read the FIRST-TOUCH referral cookie and validate it against the touch
   205→   205→ *  table (unexpired, unconsumed). Returns the sponsor profile or null. */
   206→   206→export async function resolveReferralCookie(): Promise<{ affiliateId: string; displayName: string; code: string } | null> {
   207→   207→  try {
   208→   208→    const store = await cookies();
   209→   209→    const raw = store.get(REFERRAL_COOKIE)?.value;
   210→   210→    if (!raw) return null;
   211→   211→    const parsed = parseJson<{ code?: string }>(raw, {});
   212→   212→    if (!parsed?.code) return null;
   213→   213→    const touch = await db.affiliateReferralTouch.findFirst({
   214→   214→      where: { code: parsed.code, consumedAt: null, expiresAt: { gt: new Date() } },
   215→   215→      orderBy: { touchedAt: "desc" },
   216→   216→      select: { affiliateId: true, code: true },
   217→   217→    });
   218→   218→    const profile = touch
   219→   219→      ? await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId }, select: { id: true, displayName: true, code: true, status: true } })
   220→   220→      : null;
   221→   221→    if (!profile || profile.status !== "active") return null;
   222→   222→    return { affiliateId: profile.id, displayName: profile.displayName, code: profile.code };
   223→   223→  } catch {
   224→   224→    return null;
   225→   225→  }
   226→   226→}

   251→   251→/** Activate the current shopper as an affiliate. Idempotent (already-active →
   252→   252→ *  error with the existing profile). Sponsor precedence (spec §5):
   253→   253→ *  1. Valid first-touch referral cookie (30-day window, unconsumed);
   254→   254→ *  2. Else a one-time manual code entry — confirmed explicitly;
   255→   255→ *  3. Else no sponsor (recorded permanently as none).
   256→   256→ *  The relationship locks forever after activation: no click, seller or admin
   257→   257→ *  tool can casually change it (admin corrections are separately audited). */
   258→   258→export async function activateAffiliate(input: ActivationInput) {
   259→   259→  const settings = await getAffiliateSettings();
   260→   260→  if (!settings.enabled) {
   261→   261→    throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   262→   262→  }
   263→   263→
   264→   264→  const existing = await db.affiliateProfile.findUnique({ where: { userId: input.userId } });
   265→   265→  if (existing) {
   266→   266→    throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   267→   267→  }
   268→   268→
   269→   269→  // ---- Sponsor resolution (server-side only) -----------------------------
   270→   270→  let sponsor: { id: string; displayName: string } | null = null;
   271→   271→  let sponsorSource: string | null = null;
   272→   272→
   273→   273→  const touch = await resolveReferralCookie();
   274→   274→  if (touch) {
   275→   275→    const profile = await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId } });
   276→   276→    if (profile && profile.status === "active" && profile.userId !== input.userId) {
   277→   277→      sponsor = { id: profile.id, displayName: profile.displayName };
   278→   278→      sponsorSource = "referral_touch";
   279→   279→    }
   280→   280→  }
   281→   281→  if (!sponsor && input.sponsorCode) {
   282→   282→    const candidate = await findAffiliateByCode(input.sponsorCode);
   283→   283→    if (!candidate) throw new ApiError(400, "INVALID_SPONSOR_CODE", "That referral code was not found");
   284→   284→    if (candidate.userId === input.userId) {
   285→   285→      throw new ApiError(400, "SELF_REFERRAL", "You cannot refer yourself");
   286→   286→    }
   287→   287→    if (candidate.status !== "active") {
   288→   288→      throw new ApiError(400, "SPONSOR_INACTIVE", "That sponsor is not currently eligible");
   289→   289→    }
   290→   290→    if (!input.sponsorConfirmed) {
   291→   291→      throw new ApiError(400, "SPONSOR_CONFIRMATION_REQUIRED", "Confirm the sponsor relationship before activating");
   292→   292→    }
   293→   293→    sponsor = { id: candidate.id, displayName: candidate.displayName };
   294→   294→    sponsorSource = "code_entry";
   295→   295→  }
   296→   296→
   297→   297→  const code = randomToken(5); // 10-hex-char referral code, unpredictable
   298→   298→  const slug = await uniqueSlug(slugifyAffiliate(input.slug || input.displayName || input.fullName));
   299→   299→
   300→   300→  const profile = await db.$transaction(async (tx) => {
   301→   301→    // Re-check inside the tx (concurrent double-submit).
   302→   302→    const dupe = await tx.affiliateProfile.findUnique({ where: { userId: input.userId } });
   303→   303→    if (dupe) throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   304→   304→
   305→   305→    const created = await tx.affiliateProfile.create({
   306→   306→      data: {
   307→   307→        userId: input.userId,
   308→   308→        code,
   309→   309→        slug,
   310→   310→        displayName: (input.displayName || input.fullName).slice(0, 60),
   311→   311→        bio: input.bio?.slice(0, 500) ?? null,
   312→   312→        status: "active",
   313→   313→        termsVersion: AFFILIATE_TERMS_VERSION,
   314→   314→        termsAcceptedAt: new Date(),
   315→   315→        activatedAt: new Date(),
   316→   316→        sponsorId: sponsor?.id ?? null,
   317→   317→        sponsorAssignedAt: sponsor ? new Date() : null,
   318→   318→        sponsorSource,
   319→   319→      },
   320→   320→    });
   321→   321→
   322→   322→    // Consume the referral touch exactly once (relationship now recorded).
   323→   323→    if (sponsor && sponsorSource === "referral_touch" && touch) {
   324→   324→      await tx.affiliateReferralTouch.updateMany({
   325→   325→        where: { code: touch.code, consumedAt: null },
   326→   326→        data: { consumedAt: new Date(), consumedUserId: input.userId },
   327→   327→      });
   328→   328→    }
   329→   329→    return created;
   330→   330→  });
   331→   331→
   332→   332→  return { profile, sponsor, sponsorSource };
   333→   333→}

   335→   335→/** Admin-only sponsor correction — audited, with documented reason. Cycle
   336→   336→ *  guard: with one-level relationships a 2-cycle (A↔B) is the only possible
   337→   337→ *  loop; rejecting `newSponsor.sponsorId === affiliateId` kills it. */
   338→   338→export async function correctSponsor(input: {
   339→   339→  affiliateId: string;
   340→   340→  newSponsorId: string | null;
   341→   341→  reason: string;
   342→   342→  adminId: string;
   343→   343→}) {
   344→   344→  if (!input.reason || input.reason.trim().length < 10) {
   345→   345→    throw new ApiError(400, "VALIDATION", "A documented reason (≥10 chars) is required for sponsor corrections");
   346→   346→  }
   347→   347→  const affiliate = await db.affiliateProfile.findUnique({ where: { id: input.affiliateId } });
   348→   348→  if (!affiliate) throw new ApiError(404, "NOT_FOUND", "Affiliate not found");
   349→   349→  if (input.newSponsorId === affiliate.id) {
   350→   350→    throw new ApiError(400, "SELF_REFERRAL", "An affiliate cannot sponsor themselves");
   351→   351→  }
   352→   352→  if (input.newSponsorId) {
   353→   353→    const sponsor = await db.affiliateProfile.findUnique({ where: { id: input.newSponsorId } });
   354→   354→    if (!sponsor) throw new ApiError(404, "NOT_FOUND", "New sponsor not found");
   355→   355→    if (sponsor.sponsorId === affiliate.id) {
   356→   356→      throw new ApiError(400, "SPONSOR_CYCLE", "This correction would create a sponsor cycle");
   357→   357→    }
   358→   358→  }
   359→   359→  return db.$transaction(async (tx) => {
   360→   360→    const updated = await tx.affiliateProfile.update({
   361→   361→      where: { id: input.affiliateId },
   362→   362→      data: {
   363→   363→        sponsorId: input.newSponsorId,
   364→   364→        sponsorAssignedAt: input.newSponsorId ? new Date() : null,
   365→   365→        sponsorSource: "admin_correction",
   366→   366→      },
   367→   367→    });
   368→   368→    await tx.affiliateSponsorAudit.create({
   369→   369→      data: {
   370→   370→        affiliateId: input.affiliateId,
   371→   371→        fromSponsorId: affiliate.sponsorId,
   372→   372→        toSponsorId: input.newSponsorId,
   373→   373→        reason: input.reason.trim(),
   374→   374→        actorType: "admin",
   375→   375→        actorId: input.adminId,
   376→   376→      },
   377→   377→    });
   378→   378→    return updated;
   379→   379→  });
   380→   380→}

   392→   392→export async function postAffiliateCommissionsForOrder(

   406→   406→    ? new Map((await tx.affiliateProfile.findMany({ where: { id: { in: sponsorIds } }, select: { id: true, status: true } })).map((p) => [p.id, p.status]))
   407→   407→    : new Map<string, string>();

   411→   411→  const links = linkIds.length
   412→   412→    ? new Map((await tx.affiliateLink.findMany({ where: { id: { in: linkIds } }, select: { id: true, type: true, channel: true } })).map((l) => [l.id, l]))
   413→   413→    : new Map<string, { type: string; channel: string }>();
   414→   414→

   416→   416→  for (const item of attributed) {
   417→   417→    const existing = await tx.affiliateCommission.findUnique({ where: { orderItemId: item.id }, select: { id: true } });
   418→   418→    if (existing) continue; // idempotent replay
   419→   419→
   420→   420→    const pool = item.affiliatePoolMinor;
   421→   421→    let sponsorMinor = 0;
   422→   422→    let sponsorHeld = false;
   423→   423→    // POST-IMPLEMENTATION AUDIT (P3 — deterministic policy freeze): the
   424→   424→    // sponsor-override decision comes from the CHECKOUT-TIME snapshot
   425→   425→    // (OrderItem.affiliateSponsorEnabled), never from the live setting. A
   426→   426→    // settings change between checkout and payment confirmation must not
   427→   427→    // retroactively alter an existing order's compensation arrangement:
   428→   428→    //   • override ON at checkout → this order keeps the 90/10 split even if
   429→   429→    //     an admin later disables sponsor compensation for NEW orders;
   430→   430→    //   • override OFF at checkout → the direct promoter receives 100% (D = P)
   431→   431→    //     for this order, even if the override is later enabled.
   432→   432→    // NULL (legacy rows created before the snapshot column existed) falls
   433→   433→    // back to the current setting — the historical behaviour those orders
   434→   434→    // were created under. The sponsor's CURRENT status still decides only
   435→   435→    // whether the 10% share posts as pending or held (never who the
   436→   436→    // beneficiary is), matching commission-definitions.md §4.
   437→   437→    const sponsorOverrideOn = item.affiliateSponsorEnabled ?? settings.sponsorEnabled;
   438→   438→    if (item.affiliateSponsorId && sponsorOverrideOn) {
   439→   439→      const status = sponsors.get(item.affiliateSponsorId) ?? "missing";
   440→   440→      if (status === "active") {
   441→   441→        sponsorMinor = Math.floor((pool * 10) / 100);
   442→   442→      } else {
   443→   443→        // Sponsor under review / gone: HOLD the 10% separately — the direct
   444→   444→        // promoter's 90% must NOT absorb it and must NOT be penalised.
   445→   445→        sponsorMinor = Math.floor((pool * 10) / 100);
   446→   446→        sponsorHeld = true;
   447→   447→      }
   448→   448→    }
   449→   449→    const directMinor = pool - sponsorMinor; // D + S = P always
   450→   450→    const bd = parseJson<{ base?: number }>(item.breakdownJson, {});
   451→   451→
   452→   452→    const commission = await tx.affiliateCommission.create({
   453→   453→      data: {
   454→   454→        orderItemId: item.id,
   455→   455→        orderId: order.id,
   456→   456→        orderNumber: order.orderNumber,
   457→   457→        storeId: order.storeId,
   458→   458→        productId: item.productId ?? null,
   459→   459→        affiliateId: item.affiliateDirectId!,
   460→   460→        sponsorId: item.affiliateSponsorId ?? null,
   461→   461→        linkId: item.affiliateLinkId ?? null,
   462→   462→        linkType: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.type ?? null) : null,
   463→   463→        channel: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.channel ?? null) : null,
   464→   464→        commissionableBaseMinor: Math.max(0, bd.base ?? 0),
   465→   465→        rateBps: item.affiliateRateBps ?? 0,
   466→   466→        poolMinor: pool,
   467→   467→        status: "pending",
   468→   468→        breakdownJson: item.breakdownJson,
   469→   469→      },
   470→   470→    });
   471→   471→
   472→   472→    await tx.affiliateCommissionEntry.create({
   473→   473→      data: {
   474→   474→        commissionId: commission.id,
   475→   475→        beneficiaryAffiliateId: item.affiliateDirectId!,
   476→   476→        role: "direct",
   477→   477→        amountMinor: directMinor,
   478→   478→        status: "pending",
   479→   479→      },
   480→   480→    });
   481→   481→    if (item.affiliateSponsorId && sponsorMinor > 0) {
   482→   482→      await tx.affiliateCommissionEntry.create({
   483→   483→        data: {
   484→   484→          commissionId: commission.id,
   485→   485→          beneficiaryAffiliateId: item.affiliateSponsorId,
   486→   486→          role: "sponsor",
   487→   487→          amountMinor: sponsorMinor,
   488→   488→          status: sponsorHeld ? "held" : "pending",
   489→   489→          holdReason: sponsorHeld ? "sponsor_under_review" : null,
   490→   490→        },
   491→   491→      });
   492→   492→    }
   493→   493→    posted++;
   494→   494→  }
   495→   495→  return posted;
   496→   496→}

   502→   502→/** Approve an order's pending entries (authoritative delivered/completed). */
   503→   503→export async function approveAffiliateCommissionsForOrder(orderId: string, tx?: Tx): Promise<number> {
   504→   504→  const client = tx ?? db;
   505→   505→  const commissions = await client.affiliateCommission.findMany({
   506→   506→    where: { orderId, status: "pending" },
   507→   507→    select: { id: true },
   508→   508→  });
   509→   509→  if (commissions.length === 0) return 0;
   510→   510→  const settings = await getAffiliateSettings();
   511→   511→  const now = new Date();
   512→   512→  const availableAt = new Date(now.getTime() + settings.reviewDays * 86400000);
   513→   513→  let approved = 0;
   514→   514→  for (const c of commissions) {
   515→   515→    const res = await client.affiliateCommissionEntry.updateMany({
   516→   516→      where: { commissionId: c.id, status: "pending" },
   517→   517→      data: { status: "approved", availableAt },
   518→   518→    });
   519→   519→    if (res.count > 0) {
   520→   520→      await client.affiliateCommission.update({ where: { id: c.id }, data: { status: "approved", approvedAt: now } });
   521→   521→      approved += res.count;
   522→   522→    }
   523→   523→  }
   524→   524→  return approved;

   527→   527→/** Batch sweep (cron): approve commissions on delivered/completed orders +
   528→   528→ *  mature approved entries whose review window elapsed. Bounded batches. */
   529→   529→export async function sweepAffiliateLifecycle(): Promise<{ approved: number; matured: number }> {
   530→   530→  const settings = await getAffiliateSettings();
   531→   531→  if (!settings.enabled) return { approved: 0, matured: 0 };
   532→   532→
   533→   533→  // 1) Orders delivered/completed with pending commissions → approve.
   534→   534→  const pendingCommissions = await db.affiliateCommission.findMany({
   535→   535→    where: { status: "pending" },
   536→   536→    select: { orderId: true },
   537→   537→    distinct: ["orderId"],
   538→   538→    take: 50,
   539→   539→  });
   540→   540→  let approved = 0;
   541→   541→  if (pendingCommissions.length > 0) {
   542→   542→    const orderIds = pendingCommissions.map((c) => c.orderId);
   543→   543→    const eligible = await db.order.findMany({
   544→   544→      where: { id: { in: orderIds }, status: { in: ["delivered", "completed"] } },
   545→   545→      select: { id: true },
   546→   546→    });
   547→   547→    for (const o of eligible) {
   548→   548→      approved += await approveAffiliateCommissionsForOrder(o.id);
   549→   549→    }
   550→   550→  }
   551→   551→
   552→   552→  // 2) Matured approved entries → available (clock transition, CAS-guarded).
   553→   553→  const now = new Date();
   554→   554→  const matured = await db.affiliateCommissionEntry.updateMany({
   555→   555→    where: { status: "approved", availableAt: { lte: now } },
   556→   556→    data: { status: "available" },
   557→   557→  });
   558→   558→  if (matured.count > 0) {
   559→   559→    await db.affiliateCommission.updateMany({
   560→   560→      where: { status: "approved", entries: { some: { status: "available" } } },
   561→   561→      data: { status: "available" },
   562→   562→    });
   563→   563→  }
   564→   564→  return { approved, matured: matured.count };
   565→   565→}
   566→   566→

   568→   568→// Balances (affiliate-side truth = AffiliateCommissionEntry rows; the ledger
   569→   569→// mirrors totals on PLATFORM:AFFILIATE_LIABILITY for double-entry integrity).
   570→   570→// ---------------------------------------------------------------------------
   571→   571→export type AffiliateBalances = {
   572→   572→  pendingMinor: number;
   573→   573→  approvedMinor: number; // approved but review window not elapsed
   574→   574→  availableMinor: number; // matured, not yet claimed by a payout
   575→   575→  payoutInFlightMinor: number; // claimed by requested/processing payouts
   576→   576→  paidMinor: number;
   577→   577→  reversedMinor: number;
   578→   578→  heldMinor: number;
   579→   579→  recoveryMinor: number; // reversals booked against already-paid entries
   580→   580→  requestableMinor: number; // available − recovery, floored at 0
   581→   581→};

   583→   583→export async function affiliateBalances(affiliateId: string, tx?: Prisma.TransactionClient): Promise<AffiliateBalances> {
   584→   584→  // C1.2 (Phase 2.2): optional tx client — callers inside a transaction now
   585→   585→  // read balance truth from the SAME snapshot (requestAffiliatePayout
   586→   586→  // previously read balances on the GLOBAL client, outside its transaction).
   587→   587→  const client = tx ?? db;
   588→   588→  const entries = await client.affiliateCommissionEntry.findMany({
   589→   589→    where: { beneficiaryAffiliateId: affiliateId },
   590→   590→    select: { status: true, amountMinor: true, reversedMinor: true, availableAt: true, payoutId: true },
   591→   591→  });
   592→   592→  const now = Date.now();
   593→   593→  const b: AffiliateBalances = {
   594→   594→    pendingMinor: 0, approvedMinor: 0, availableMinor: 0, payoutInFlightMinor: 0,
   595→   595→    paidMinor: 0, reversedMinor: 0, heldMinor: 0, recoveryMinor: 0, requestableMinor: 0,
   596→   596→  };
   597→   597→  for (const e of entries) {
   598→   598→    const matured = (e.availableAt?.getTime() ?? 0) <= now;
   599→   599→    switch (e.status) {
   600→   600→      case "pending":
   601→   601→      case "approved":
   602→   602→        if (e.status === "pending" || !matured) {
   603→   603→          if (e.status === "pending") b.pendingMinor += e.amountMinor - e.reversedMinor;
   604→   604→          else b.approvedMinor += e.amountMinor - e.reversedMinor;
   605→   605→          break;
   606→   606→        }
   607→   607→        // fallthrough: matured approved
   608→   608→      case "available":
   609→   609→        if (e.payoutId) b.payoutInFlightMinor += e.amountMinor - e.reversedMinor;
   610→   610→        else b.availableMinor += e.amountMinor - e.reversedMinor;
   611→   611→        break;
   612→   612→      case "paid":
   613→   613→        b.paidMinor += e.amountMinor;
   614→   614→        b.recoveryMinor += e.reversedMinor; // late refund after payout
   615→   615→        break;
   616→   616→      case "reversed":
   617→   617→        b.reversedMinor += e.amountMinor;
   618→   618→        break;
   619→   619→      case "held":
   620→   620→        b.heldMinor += e.amountMinor - e.reversedMinor;
   621→   621→        break;
   622→   622→    }
   623→   623→  }
   624→   624→  b.requestableMinor = Math.max(0, b.availableMinor - b.recoveryMinor);
   625→   625→  return b;
   626→   626→}
   627→   627→
   628→   628→// ---------------------------------------------------------------------------
   629→   629→// Refund reversals — proportional against the ORIGINAL immutable allocations
   630→   630→// (spec §11). Shipping-only refunds book no product-commission reversal (the
   631→   631→// pool base excluded shipping by construction). Late refunds after payout
   632→   632→// become recovery entries: reversedMinor accumulates on the PAID entry and
   633→   633→// offsets future earnings — the original payout history is never rewritten.
   634→   634→// ---------------------------------------------------------------------------
   635→   635→function fullyReversedFn(newReversed: number, amountMinor: number): boolean {
   636→   636→  return newReversed >= amountMinor;
   637→   637→}
   638→   638→export async function reverseAffiliateForRefund(
   639→   639→  tx: Tx,
   640→   640→  input: {
   641→   641→    refundId: string;
   642→   642→    orderId: string;
   643→   643→    refundAmountMinor: number;
   644→   644→    /** Fresh order items (with affiliate snapshots) under the lock. */
   645→   645→    items: { id: string; lineSubtotalMinor: number; refundedMinor: number; breakdownJson: string; affiliatePoolMinor: number }[];
   646→   646→    /** Σ affiliateReversalMinor of prior PROCESSED refunds on this order. */
   647→   647→    priorAffiliateReversalMinor: number;
   648→   648→  }
   649→   649→): Promise<{ affiliateReversalMinor: number; affiliateClearingMinor: number; rule: string }> {
   650→   650→  const attributed = input.items.filter((i) => i.affiliatePoolMinor > 0);
   651→   651→  if (attributed.length === 0) return { affiliateReversalMinor: 0, affiliateClearingMinor: 0, rule: "no_affiliate" };
   652→   652→
   653→   653→  // ---- C1.2 (Phase 2.2) pre-pass: load commissions + entries, then LOCK the
   654→   654→  // open AffiliatePayout rows BEFORE any entry write. Lock order is
   655→   655→  // payout → entries on every flow that touches both (this function and the
   656→   656→  // admin approve / mark-paid / fail claims, which take the payout row lock
   657→   657→  // first via their conditional updateMany) — no AB-BA deadlock window.
   658→   658→  const commissions = (
   659→   659→    await Promise.all(
   660→   660→      attributed.map((it) => tx.affiliateCommission.findUnique({ where: { orderItemId: it.id }, include: { entries: true } })),
   661→   661→    )
   662→   662→  ).filter((c): c is NonNullable<typeof c> => c !== null);
   663→   663→  const openPayoutIds = [
   664→   664→    ...new Set(commissions.flatMap((c) => c.entries.map((e) => e.payoutId).filter((p): p is string => p !== null))),
   665→   665→  ];
   666→   666→  const openPayouts = openPayoutIds.length
   667→   667→    ? await tx.affiliatePayout.findMany({
   668→   668→        where: { id: { in: openPayoutIds }, status: { in: ["requested", "processing"] } },
   669→   669→        select: { id: true, status: true, amountMinor: true },
   670→   670→      })
   671→   671→    : [];
   672→   672→  // Deterministic lock order (sorted ids) — multiple payouts can be affected
   673→   673→  // when one order's commissions span several payout requests.
   674→   674→  for (const p of [...openPayouts].sort((a, b) => a.id.localeCompare(b.id))) {
   675→   675→    await tx.$queryRaw`SELECT id FROM "AffiliatePayout" WHERE id = ${p.id} FOR UPDATE`;
   676→   676→  }
   677→   677→  const payoutById = new Map(openPayouts.map((p) => [p.id, p]));
   678→   678→
   679→   679→  // Customer-paid gross per item (breakdown-proportional, mirroring
   680→   680→  // deriveCommissionReversal's allocation family).
   681→   681→  const grossArr = attributed.map((it) => {
   682→   682→    const bd = parseJson<{ discount_allocated?: number; shipping_allocated?: number }>(it.breakdownJson, {});
   683→   683→    return Math.max(0, it.lineSubtotalMinor - (bd.discount_allocated ?? 0) + (bd.shipping_allocated ?? 0));
   684→   684→  });
   685→   685→  const allocs: number[] = [];
   686→   686→  {
   687→   687→    // floor + remainder-to-largest (same as allocateProportionally)
   688→   688→    const totalGross = grossArr.reduce((s, v) => s + v, 0);
   689→   689→    if (totalGross <= 0) {
   690→   690→      attributed.forEach(() => allocs.push(0));
   691→   691→    } else {
   692→   692→      let allocated = 0;
   693→   693→      for (let i = 0; i < attributed.length; i++) {
   694→   694→        const a = Math.floor((input.refundAmountMinor * grossArr[i]) / totalGross);
   695→   695→        allocs.push(a);
   696→   696→        allocated += a;
   697→   697→      }
   698→   698→      let rem = input.refundAmountMinor - allocated;
   699→   699→      while (rem > 0) {
   700→   700→        let maxIdx = 0;
   701→   701→        for (let i = 1; i < grossArr.length; i++) if (grossArr[i] > grossArr[maxIdx]) maxIdx = i;
   702→   702→        allocs[maxIdx] += 1;
   703→   703→        rem -= 1;
   704→   704→      }
   705→   705→    }
   706→   706→  }
   707→   707→
   708→   708→  // Per-item affiliate reversal, clamped to the remaining reversible pool.
   709→   709→  const orderPool = attributed.reduce((s, it) => s + it.affiliatePoolMinor, 0);
   710→   710→  const remainingReversible = Math.max(0, orderPool - input.priorAffiliateReversalMinor);
   711→   711→  let total = 0;
   712→   712→  const perItem: { item: (typeof attributed)[number]; reversal: number }[] = [];
   713→   713→  for (let i = 0; i < attributed.length; i++) {
   714→   714→    if (grossArr[i] <= 0 || allocs[i] <= 0) {
   715→   715→      perItem.push({ item: attributed[i], reversal: 0 });
   716→   716→      continue;
   717→   717→    }
   718→   718→    const r = Math.min(Math.floor((allocs[i] * attributed[i].affiliatePoolMinor) / grossArr[i]), attributed[i].affiliatePoolMinor);
   719→   719→    perItem.push({ item: attributed[i], reversal: r });
   720→   720→    total += r;
   721→   721→  }
   722→   722→  total = Math.min(Math.max(0, total), input.refundAmountMinor, remainingReversible);
   723→   723→
   724→   724→  // C1.2: cumulative per-payout adjustments for THIS refund.
   725→   725→  //   requested  → liability-side: T5 keeps debiting AFFILIATE_LIABILITY and
   726→   726→  //                the payout's frozen amount shrinks (no ledger posted for a
   727→   727→  //                requested payout yet — fully consistent).
   728→   728→  //   processing → clearing-side: T-A3 already moved the frozen amount from
   729→   729→  //                AFFILIATE_LIABILITY into PAYOUT_CLEARING, so the reversal
   730→   730→  //                must debit PAYOUT_CLEARING (returned as
   731→   731→  //                affiliateClearingMinor for the T5 split) and the payout
   732→   732→  //                amount shrinks so T-A4 pays the reduced figure — the
   733→   733→  //                clearing account nets to exactly zero across T-A3, T5's
   734→   734→  //                split line and T-A4/T-A4f at the adjusted amount.
   735→   735→  const adjByPayout = new Map<string, { liability: number; clearing: number }>();
   736→   736→  let clearingTotal = 0;
   737→   737→
   738→   738→  // Apply to the commission + entries (proportional split within each pool).
   739→   739→  if (total > 0) {
   740→   740→    // Scale per-item reversals down proportionally if the clamp reduced total.
   741→   741→    const rawTotal = perItem.reduce((s, p) => s + p.reversal, 0);
   742→   742→    const scale = rawTotal > 0 ? total / rawTotal : 0;
   743→   743→    for (const p of perItem) {
   744→   744→      const itemReversal = p.reversal === 0 ? 0 : Math.min(p.reversal, Math.floor(p.reversal * scale));
   745→   745→      if (itemReversal <= 0) continue;
   746→   746→      const commission = commissions.find((c) => c.orderItemId === p.item.id);
   747→   747→      if (!commission) continue; // pre-affiliate legacy rows: nothing to reverse
   748→   748→      const direct = commission.entries.find((e) => e.role === "direct");
   749→   749→      const sponsor = commission.entries.find((e) => e.role === "sponsor");
   750→   750→      // Split proportionally to the frozen allocation shares (P4: floor policy
   751→   751→      // — same rounding as the pool computation). The SPONSOR share is the
   752→   752→      // floor-proportional part and the DIRECT share takes the residual
   753→   753→      // (commission-definitions.md §7 step 3: "direct absorbs the rounding
   754→   754→      // residual"), mirroring the forward split where D = P − S. Invariant:
   755→   755→      // Σ(dRev+sRev) === itemReversal whenever caps allow, and never more.
   756→   756→      const sponsorShare = sponsor ? Math.min(sponsor.amountMinor - sponsor.reversedMinor, Math.floor((itemReversal * sponsor.amountMinor) / commission.poolMinor)) : 0;
   757→   757→      const directShare = direct ? Math.min(direct.amountMinor - direct.reversedMinor, itemReversal - sponsorShare) : 0;
   758→   758→      for (const [entry, share] of [[direct, directShare], [sponsor, sponsorShare]] as const) {
   759→   759→        if (!entry || share <= 0) continue;
   760→   760→        const newReversed = Math.min(entry.amountMinor, entry.reversedMinor + share);
   761→   761→        const actualApplied = newReversed - entry.reversedMinor;
   762→   762→        await tx.affiliateCommissionEntry.update({
   763→   763→          where: { id: entry.id },
   764→   764→          data: {
   765→   765→            reversedMinor: newReversed,
   766→   766→            // PAID entries stay paid — the reversal is a recovery entry that
   767→   767→            // offsets future earnings (spec §11 late refunds); pending/
   768→   768→            // approved/available entries flip to reversed.
   769→   769→            ...(entry.status !== "paid" && fullyReversedFn(newReversed, entry.amountMinor) ? { status: "reversed" as const } : {}),
   770→   770→          },
   771→   771→        });
   772→   772→        // ---- C1.2 payout coordination: shrink the frozen amount of the open
   773→   773→        // payout that claimed this entry, by exactly the share applied. A
   774→   774→        // refund and a payout can no longer both consume the same commission
   775→   775→        // liability.
   776→   776→        if (actualApplied > 0 && entry.payoutId) {
   777→   777→          const payout = payoutById.get(entry.payoutId);
   778→   778→          if (payout && (payout.status === "requested" || payout.status === "processing")) {
   779→   779→            const adj = adjByPayout.get(payout.id) ?? { liability: 0, clearing: 0 };
   780→   780→            if (payout.status === "processing") { adj.clearing += actualApplied; clearingTotal += actualApplied; }
   781→   781→            else adj.liability += actualApplied;
   782→   782→            adjByPayout.set(payout.id, adj);
   783→   783→          }
   784→   784→        }
   785→   785→      }
   786→   786→      const anyLive = commission.entries.some((e) => e.status !== "reversed");
   787→   787→      if (!anyLive) {
   788→   788→        await tx.affiliateCommission.update({
   789→   789→          where: { id: commission.id },
   790→   790→          data: { status: "reversed", reversedAt: new Date() },
   791→   791→        });
   792→   792→      }
   793→   793→    }
   794→   794→  }
   795→   795→
   796→   796→  // ---- C1.2: apply the per-payout amount reductions (the payout rows are
   797→   797→  // already locked). A REQUESTED payout whose frozen amount is fully consumed
   798→   798→  // is auto-released (rejected + entries unclaimed) — no ledger was posted
   799→   799→  // for it, so this is clean; the affiliate simply re-requests later.
   800→   800→  for (const [payoutId, adj] of adjByPayout) {
   801→   801→    const payout = payoutById.get(payoutId);
   802→   802→    if (!payout) continue;
   803→   803→    const reduce = adj.liability + adj.clearing;
   804→   804→    if (reduce <= 0) continue;
   805→   805→    const newAmount = Math.max(0, payout.amountMinor - reduce);
   806→   806→    if (payout.status === "requested" && newAmount <= 0) {
   807→   807→      await tx.affiliatePayout.update({
   808→   808→        where: { id: payoutId, status: "requested" },
   809→   809→        data: {
   810→   810→          status: "rejected",
   811→   811→          amountMinor: 0,
   812→   812→          failReason: "Auto-released: commissions fully reversed by refunds before approval",
   813→   813→          processedAt: new Date(),
   814→   814→        },
   815→   815→      });
   816→   816→      // Release the still-live claimed entries back to the available pool.
   817→   817→      await tx.affiliateCommissionEntry.updateMany({
   818→   818→        where: { payoutId, status: { in: ["available", "approved"] } },
   819→   819→        data: { payoutId: null },
   820→   820→      });
   821→   821→    } else if (newAmount !== payout.amountMinor) {
   822→   822→      await tx.affiliatePayout.update({
   823→   823→        where: { id: payoutId, status: { in: ["requested", "processing"] } },
   824→   824→        data: { amountMinor: newAmount },
   825→   825→      });
   826→   826→    }
   827→   827→  }
   828→   828→
   829→   829→  return { affiliateReversalMinor: total, affiliateClearingMinor: clearingTotal, rule: "breakdown_snapshot_proportional" };
   830→   830→}
   831→   831→export async function requestAffiliatePayout(affiliateId: string): Promise<{ payoutId: string; amountMinor: number }> {
   832→   832→  const settings = await getAffiliateSettings();
   833→   833→  if (!settings.enabled) throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   834→   834→
   835→   835→  // C1.2 (Phase 2.2) — brought to the SAME standard as the seller payout
   836→   836→  // system (store/payouts route): SERIALIZABLE isolation + bounded P2034
   837→   837→  // retry of the whole transaction, balance truth read via the transaction
   838→   838→  // client, and a CONDITIONAL entry claim that re-asserts `payoutId: null`.
   839→   839→  //
   840→   840→  // The previous implementation ran at READ COMMITTED with a check-then-act
   841→   841→  // one-open-request guard and a NON-conditional `updateMany({ id IN ... })`
   842→   842→  // claim: two concurrent requests both read the same `payoutId: null`
   843→   843→  // entries, both froze the FULL requestable amount into two AffiliatePayout
   844→   844→  // rows, and the second write overwrote the first's claim (last-write-wins)
   845→   845→  // — a double-payment with per-row dedupKeys that the ledger could NOT
   846→   846→  // catch. Serializable + conditional claims + the partial unique index
   847→   847→  // (migration 20261105000000) close every layer of that window.
   848→   848→  return withSerializableTxRetry(async (tx) => {
   849→   849→    // One open request per affiliate — re-checked INSIDE the Serializable
   850→   850→    // transaction (a concurrent duplicate submit now aborts one of the two
   851→   851→    // transactions instead of both passing the check).
   852→   852→    const open = await tx.affiliatePayout.findFirst({
   853→   853→      where: { affiliateId, status: { in: ["requested", "processing"] } },
   854→   854→      select: { id: true },
   855→   855→    });
   856→   856→    if (open) throw new ApiError(409, "PAYOUT_OPEN", "You already have a payout request in progress");
   857→   857→
   858→   858→    // Balance truth from THIS transaction's snapshot (was the global client).
   859→   859→    const balances = await affiliateBalances(affiliateId, tx);
   860→   860→    if (balances.requestableMinor < settings.minPayoutMinor) {
   861→   861→      throw new ApiError(400, "BELOW_MIN_PAYOUT", `Minimum payout is RM ${(settings.minPayoutMinor / 100).toFixed(2)}`);
   862→   862→    }
   863→   863→
   864→   864→    const amountMinor = balances.requestableMinor;
   865→   865→    const now = new Date();
   866→   866→    // Claim every matured, unclaimed entry; bind to the payout row we create.
   867→   867→    const toClaim = await tx.affiliateCommissionEntry.findMany({
   868→   868→      where: {
   869→   869→        beneficiaryAffiliateId: affiliateId,
   870→   870→        status: { in: ["approved", "available"] },
   871→   871→        availableAt: { lte: now },
   872→   872→        payoutId: null,
   873→   873→      },
   874→   874→      select: { id: true, amountMinor: true, reversedMinor: true, commissionId: true },
   875→   875→    });
   876→   876→    const payout = await tx.affiliatePayout.create({
   877→   877→      data: { affiliateId, amountMinor, status: "requested", method: "manual_bank" },
   878→   878→    });
   879→   879→    if (toClaim.length > 0) {
   880→   880→      // CONDITIONAL claim — `payoutId: null` re-asserted in the WHERE so a
   881→   881→      // concurrent claimant's binding can never be silently overwritten.
   882→   882→      // (Under Serializable any concurrent writer of these rows aborts this
   883→   883→      // transaction first; the conditional guard is belt-and-braces against
   884→   884→      // non-transactional writers and keeps the count assertion honest.)
   885→   885→      const claim = await tx.affiliateCommissionEntry.updateMany({
   886→   886→        where: { id: { in: toClaim.map((e) => e.id) }, payoutId: null },
   887→   887→        data: { payoutId: payout.id },
   888→   888→      });
   889→   889→      if (claim.count !== toClaim.length) {
   890→   890→        throw new ApiError(409, "PAYOUT_CLAIM_CONFLICT", "Commission entries changed concurrently — please retry the payout request");
   891→   891→      }
   892→   892→    }
   893→   893→    // Ledger-derived statement (schema contract: built from entries at
   894→   894→    // request time, never hand-edited) — order references + balance snapshot.
   895→   895→    const orderRefs = toClaim.length
   896→   896→      ? await tx.affiliateCommission.findMany({
   897→   897→          where: { id: { in: [...new Set(toClaim.map((e) => e.commissionId))] } },
   898→   898→          select: { id: true, orderNumber: true, productId: true, storeId: true },
   899→   899→        })
   900→   900→      : [];
   901→   901→    const refMap = new Map(orderRefs.map((c) => [c.id, c]));
   902→   902→    await tx.affiliatePayout.update({
   903→   903→      where: { id: payout.id },
   904→   904→      data: {
   905→   905→        statementJson: toJson({
   906→   906→          builtAt: now.toISOString(),
   907→   907→          balances: balances,
   908→   908→          entries: toClaim.map((e) => ({
   909→   909→            entryId: e.id,
   910→   910→            commissionId: e.commissionId,
   911→   911→            orderNumber: refMap.get(e.commissionId)?.orderNumber ?? null,
   912→   912→            amountMinor: e.amountMinor,
   913→   913→            reversedMinor: e.reversedMinor,
   914→   914→          })),
   915→   915→        }),
   916→   916→      },
   917→   917→    });
   918→   918→    return { payoutId: payout.id, amountMinor };
   919→   919→  });
   920→   920→}
