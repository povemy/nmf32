   340→// FIN-3-c — seller in-app notification bell. Feeds from the NotificationLog
   341→// rows (channel "inapp") the finance flows already write for the seller email:
   342→// order.paid (FIN-2-b) + payout requested/approved/paid/failed/rejected
   343→// (FIN-2-a/FIN-3-b). The shopper bell is a different subsystem (Notification
   344→// model via /api/me/notifications) and is intentionally untouched.
   345→
   346→type SellerNotificationRow = { id: string; subject: string; createdAt: string; event: string };
   347→
   348→type SellerNotificationsPayload = { notifications: SellerNotificationRow[]; recentCount: number };
   349→
   350→/** House scrollbar style (same treatment as the shopper bell panel). */
   351→const SELLER_BELL_SCROLL =
   352→  "max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent";
   353→
   354→/** 24h recency = client-side "new" heuristic: NotificationLog has no read /
   355→ *  seen column (adding fake read-tracking was out of scope), so "within the
   356→ *  last 24h" is the honest approximation of "unread". */
   357→const RECENT_WINDOW_MS = 24 * 60 * 60 * 1000;
   358→
   359→function SellerNotificationBell() {
   360→  const [open, setOpen] = useState(false);
   361→  const [data, setData] = useState<SellerNotificationsPayload | null>(null);
   362→  const [loading, setLoading] = useState(false);
   363→  const [errored, setErrored] = useState(false);
   364→
   365→  const load = useCallback(async () => {
   366→    setLoading(true);
   367→    try {
   368→      const d = await apiGet<SellerNotificationsPayload>("/api/store/notifications");
   369→      setData(d);
   370→      setErrored(false);
   371→    } catch (e) {
   372→      // Silent by design — a failed bell must never break the seller shell:
   373→      // keep the last data (no badge flicker), warn for debugging only.
   374→      setErrored(true);
   375→      console.warn("[seller-bell] notifications fetch failed", e instanceof Error ? e.message : e);
   376→    } finally {
   377→      setLoading(false);
   378→    }
   379→  }, []);
   380→
   381→  // Badge freshness without aggressive polling: fetch once on mount (dot only
   382→  // needs recentCount) and refetch whenever the popover opens.
   383→  useEffect(() => {
   384→    void load();
   385→  }, [load]);
   386→
   387→  const onOpenChange = (v: boolean) => {
   388→    setOpen(v);
   389→    if (v) void load();
   390→  };
   391→
   392→  const recent = data?.recentCount ?? 0;
   393→  const rows = data?.notifications ?? null;
   394→
   395→  return (
   396→    <Popover open={open} onOpenChange={onOpenChange}>
   397→      <PopoverTrigger asChild>
   398→        <Button
   399→          variant="ghost"
   400→          size="icon"
   401→          className="relative rounded-full border border-slate-200 bg-white text-slate-500 shadow-sm hover:border-teal-400 hover:bg-white hover:text-teal-700"
   402→          aria-label="Seller notifications"
   403→          aria-haspopup="dialog"
   404→          aria-expanded={open}
   405→        >
   406→          <Bell className="h-4 w-4" aria-hidden="true" />
   407→          {recent > 0 ? (
   408→            <span
   409→              className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-amber-500 ring-2 ring-white"
   410→              aria-hidden="true"
   411→            />
   412→          ) : null}
   413→        </Button>
   414→      </PopoverTrigger>
   415→      <PopoverContent align="end" className="w-80 rounded-2xl p-0 sm:w-96" role="dialog" aria-label="Seller notifications panel">
   416→        <div className="flex items-center gap-2 border-b border-slate-200/80 px-4 py-3">
   417→          <h2 className="text-sm font-bold text-slate-900">Notifications</h2>
   418→          {recent > 0 ? (
   419→            <span className="rounded-full bg-amber-50 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 tabular-nums">
   420→              {recent} new
   421→            </span>
   422→          ) : null}
   423→        </div>
   424→
   425→        {loading && !rows ? (
   426→          <div className="space-y-3 p-4" aria-label="Loading notifications">
   427→            {Array.from({ length: 3 }).map((_, i) => (
   428→              <div key={i} className="flex items-center gap-3">
   429→                <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   430→                <div className="flex-1 space-y-1.5">
   431→                  <Skeleton className="h-3.5 w-3/4" />
   432→                  <Skeleton className="h-3 w-1/3" />
   433→                </div>
   434→              </div>
   435→            ))}
   436→          </div>
   437→        ) : !rows || rows.length === 0 ? (
   438→          <div className="flex flex-col items-center gap-2 px-6 py-10 text-center" aria-label="No notifications">
   439→            <span className="rounded-full bg-muted p-4">
   440→              <Bell className="h-6 w-6 text-muted-foreground" aria-hidden="true" />
   441→            </span>
   442→            <p className="text-sm font-semibold">{errored && !data ? "Notifications unavailable" : "No notifications yet"}</p>
   443→            <p className="max-w-[16rem] text-xs text-muted-foreground">
   444→              {errored && !data
   445→                ? "Couldn't load right now — it will retry next time you open the bell."
   446→                : "Payout and order updates will appear here."}
   447→            </p>
   448→          </div>
   449→        ) : (
   450→          <ul className={cn("divide-y divide-slate-100", SELLER_BELL_SCROLL)} aria-label="Notification list">
   451→            {rows.map((n) => {
   452→              const isNew = Date.now() - new Date(n.createdAt).getTime() < RECENT_WINDOW_MS;
   453→              return (
   454→                <li key={n.id} className="flex items-start gap-3 px-4 py-3">
   455→                  <div className="min-w-0 flex-1">
   456→                    <p className="truncate text-[13px] font-medium" title={n.subject}>
   457→                      {n.subject}
   458→                    </p>
   459→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   460→                      <span title={new Date(n.createdAt).toLocaleString("en-MY")}>{ageOf(n.createdAt)}</span>
   461→                      <span aria-hidden="true"> · </span>
   462→                      <span className="lowercase">{n.event}</span>
   463→                    </p>
   464→                  </div>
   465→                  {isNew ? (
   466→                    <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-amber-500" role="img" aria-label="Recent" />
   467→                  ) : null}
   468→                </li>
   469→              );
   470→            })}
   471→          </ul>
   472→        )}
   473→      </PopoverContent>
   474→    </Popover>
   475→  );
   476→}

   480→/** Total unread shopper messages — powers the "Messages" nav badge. */
   481→function useSellerChatUnread(active: SellerNav["view"]) {
   482→  const [unread, setUnread] = useState(0);
   483→  useEffect(() => {
   484→    let alive = true;
   485→    const load = () =>
   486→      apiGet<{ total: number }>("/api/store/chat/unread")
   487→        .then((d) => alive && setUnread(d.total ?? 0))
   488→        .catch(() => {});
   489→    load();
   490→    const t = setInterval(load, 15000);
   491→    return () => {
   492→      alive = false;
   493→      clearInterval(t);
   494→    };
   495→  }, [active]);
   496→  // instant re-sync when a thread is read inside the messages view (the view
   497→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   498→  useEffect(() => {
   499→    const sync = () =>
   500→      apiGet<{ total: number }>("/api/store/chat/unread")
   501→        .then((d) => setUnread(d.total ?? 0))
   502→        .catch(() => {});

   504→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   505→  }, []);
   506→  return unread;
   507→}
