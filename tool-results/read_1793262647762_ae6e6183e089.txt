   339→/** Prev-period delta: ▲/▼ % (emerald/rose), "New" badge when prev=0 & value>0,
   340→ *  "—" when there is no prior data. goodDown flips the polarity (refunds). */
   341→function Delta({ value, prev, goodDown = false }: { value: number | null; prev: number | null | undefined; goodDown?: boolean }) {
   342→  if (prev == null || value == null) return <span className="text-xs text-muted-foreground">—</span>;
   343→  if (prev === 0) {
   344→    return value > 0 ? (
   345→      <Badge variant="outline" className="border-emerald-200 bg-emerald-100 text-emerald-800">
   346→        New
   347→      </Badge>
   348→    ) : (
   349→      <span className="text-xs text-muted-foreground">—</span>
   350→    );
   351→  }
   352→  const pct = ((value - prev) / prev) * 100;
   353→  if (pct === 0) return <span className="text-xs text-muted-foreground">0% vs prev</span>;
   354→  const up = pct > 0;
   355→  const good = goodDown ? !up : up;
   356→  return (
   357→    <span className={cn("text-xs font-medium tabular-nums", good ? "text-emerald-600" : "text-rose-600")}>
   358→      {up ? "▲" : "▼"} {Math.abs(Math.round(pct * 10) / 10)}%
   359→      <span className="ml-1 font-normal text-muted-foreground">vs prev</span>
   360→    </span>
   361→  );
   362→}

   436→/** Paged fetch hook (apiGetPaged keeps the top-level pagination meta). */
   437→function usePagedApi<T>(url: string | null) {
   438→  const [data, setData] = useState<T | null>(null);
   439→  const [meta, setMeta] = useState<{ page: number; size: number; total: number } | null>(null);
   440→  const [loading, setLoading] = useState(true);
   441→  const [error, setError] = useState<string | null>(null);
   442→  const [tick, setTick] = useState(0);
   443→
   444→  useEffect(() => {
   445→    if (!url) return;
   446→    let alive = true;
   447→    // eslint-disable-next-line react-hooks/set-state-in-effect -- data-fetch reset
   448→    setLoading(true);
   449→    apiGetPaged<T>(url)
   450→      .then((r) => {
   451→        if (alive) {
   452→          setData(r.rows);
   453→          setMeta(r.meta);
   454→          setError(null);
   455→        }
   456→      })
   457→      .catch((e: unknown) => alive && setError(e instanceof Error ? e.message : "Failed to load"))
   458→      .finally(() => alive && setLoading(false));
   459→    return () => {
   460→      alive = false;
   461→    };
   462→  }, [url, tick]);
   463→
   464→  const refetch = useCallback(() => setTick((t) => t + 1), []);
   465→  return { data, meta, loading, error, refetch };
   466→}
