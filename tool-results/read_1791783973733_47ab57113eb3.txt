   100→  });
   101→  const [name, setName] = useState(() => saved.name ?? "");

   106→  const [stateSel, setStateSel] = useState(() => saved.state ?? "Selangor");
   107→  const [postcode, setPostcode] = useState(() => saved.postcode ?? "");
   108→  const [note, setNote] = useState("");
   109→  const [shippingRateId, setShippingRateId] = useState("standard");
   110→  const [provider, setProvider] = useState("fpx_toyyibpay");
   111→  const [couponInput, setCouponInput] = useState("");
   112→  const [couponCode, setCouponCode] = useState("");

   137→        setQuote(null);

   139→        setQuoting(false);

   144→
   145→  // auto-quote on items / shipping change (keeps applied coupon)

   149→

   151→    const code = couponInput.trim().toUpperCase();
   152→    if (!code) return;
   153→    setCouponErr(null);
   154→    setQuoting(true);
   155→    try {
   156→      const q = await postJson<QuoteT>("/api/checkout/quote", {
   157→        storeSlug: cart?.storeSlug,
   158→        items: payloadItems,
   159→        couponCode: code,
   160→        shippingRateId,
   161→      });
   162→      if (q.coupon?.valid) {
   163→        setCouponCode(code);
   164→        setQuote(q);

   166→          description: q.coupon.freeShip ? "Free shipping unlocked" : `You save ${formatMinor(q.coupon.discountMinor ?? 0)}`,

   168→      } else {
   169→        setCouponCode("");
   170→        setCouponErr(q.coupon?.reason ?? "Invalid coupon");
   171→      }
   172→    } catch (e) {
   173→      setCouponErr(e instanceof Error ? e.message : "Could not validate coupon");
   174→    } finally {
   175→      setQuoting(false);
   176→    }
   177→  }

   182→    line1.trim().length > 0 &&
   183→    city.trim().length > 0 &&
   184→    postcode.trim().length > 0;
   185→
   186→  async function placeOrder() {
   187→    setTouched(true);
   188→    if (!formValid || items.length === 0) return;

   191→      const res = await postJson<{ orderNumber: string; orderPaymentRef: string; totalMinor: number }>(

   199→          provider,

   206→      toast.success(`Order ${res.orderNumber} created`, { description: "Redirecting to secure payment…" });
   207→      ctx.nav({ name: "pay", ref: res.orderPaymentRef });
   208→    } catch (e) {
   209→      toast.error(e instanceof Error ? e.message : "Checkout failed — please try again");
   210→    } finally {
   211→      setSubmitting(false);
   212→    }
   213→  }
   214→
   215→  if (cartLoading) {
   216→    return (
   217→      <div className="mx-auto max-w-6xl px-4 py-8 flex-1">
   218→        <Skeleton className="h-8 w-48 mb-6" />
   219→        <div className="grid gap-6 lg:grid-cols-3">
   220→          <div className="lg:col-span-2 space-y-4">
   221→            <Skeleton className="h-64 rounded-xl" />
   222→            <Skeleton className="h-40 rounded-xl" />
   223→          </div>
   224→          <Skeleton className="h-80 rounded-xl" />
   225→        </div>
   226→      </div>

   228→  }
   229→
   230→  if (items.length === 0) {

   234→          icon={ShoppingBag}
   235→          title="Your cart is empty"
   236→          hint="Add something lovely from the marketplace or a store first."
   237→          action={
   238→            <Button onClick={() => ctx.nav({ name: "market" })} className="bg-teal-700 hover:bg-teal-800">
   239→              Start shopping
   240→            </Button>
   241→          }
   242→        />
   243→      </div>
   244→    );
   245→  }
   246→
   247→  const total = quote?.totalMinor ?? items.reduce((s, i) => s + i.lineTotalMinor, 0);
   248→  const freeShipProgress = Math.min(100, Math.round((items.reduce((s, i) => s + i.lineTotalMinor, 0) / 10000) * 100));
   249→
   250→  const field = (label: string, node: React.ReactNode, error?: boolean) => (
   251→    <div className="space-y-1.5">
   252→      <Label className="text-xs font-semibold text-muted-foreground">{label}</Label>
   253→      {node}
   254→      {error ? <p className="text-xs text-rose-600">Required</p> : null}
   255→    </div>
   256→  );
   257→
   258→  return (
   259→    <div className="mx-auto max-w-6xl px-4 py-8 flex-1">
   260→      <h1 className="text-2xl font-extrabold tracking-tight">Checkout</h1>
   261→      <p className="text-sm text-muted-foreground">
   262→        Items from <span className="font-semibold text-teal-700">{cart?.storeName}</span> · one store per order
   263→      </p>
   264→
   265→      <div className="mt-6 grid gap-6 lg:grid-cols-5">
   266→        {/* left column */}
   267→        <div className="lg:col-span-3 space-y-6">
   268→          {/* contact */}
   269→          <section className="rounded-xl border bg-card p-5 space-y-4" aria-labelledby="checkout-contact">
   270→            <h2 id="checkout-contact" className="flex items-center gap-2 text-base font-bold">
   271→              <User className="h-4 w-4 text-teal-700" /> Contact
   272→            </h2>
   273→            <div className="grid gap-4 sm:grid-cols-2">
   274→              {field(

   276→                <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Aina Binti Ahmad" className="min-h-[44px]" aria-required />
   277→              )}
   278→              {field(

   281→              )}
   282→              {field("Phone", <Input value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="012-345 6789" className="min-h-[44px]" />)}

   326→                    key={s.id}
   327→                    htmlFor={`ship-${s.id}`}

   329→                      "flex cursor-pointer items-center gap-3 rounded-lg border p-3.5 transition-colors min-h-[44px]",

   331→                    )}
   332→                  >

   334→                    <div className="flex-1">

   344→              <div className="flex items-center gap-2 rounded-lg bg-emerald-50 border border-emerald-200 px-3 py-2 text-xs font-medium text-emerald-800">

   346→              </div>
   347→            ) : (
   348→              <div className="pt-1">
   349→                <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   350→                  <div className="h-full rounded-full bg-teal-600 transition-all" style={{ width: `${freeShipProgress}%` }} />
   351→                </div>
   352→                <p className="pt-1.5 text-xs text-muted-foreground">

   355→              </div>
   356→            )}

   358→
   359→          {/* payment method */}

   362→              <CreditCard className="h-4 w-4 text-teal-700" /> Payment method

   367→                  key={p.id}

   370→                    "flex cursor-pointer items-center gap-3 rounded-lg border p-3.5 transition-colors min-h-[44px]",

   374→                  <RadioGroupItem value={p.id} id={`pay-${p.id}`} />
   375→                  <p.icon className="h-5 w-5 text-teal-700" />

   388→            <ul className="space-y-3 max-h-64 overflow-y-auto pr-1">
   389→              {items.map((i) => (
   390→                <li key={i.id} className="flex gap-3">
   391→                  <img src={i.imageUrl} alt={i.title} loading="lazy" className="h-14 w-14 rounded-lg border object-cover" />
   392→                  <div className="min-w-0 flex-1">
   393→                    <p className="text-sm font-medium leading-tight line-clamp-2">{i.title}</p>
   394→                    {i.variantLabel ? <p className="text-xs text-muted-foreground">{i.variantLabel}</p> : null}
   395→                    <p className="text-xs text-muted-foreground">Qty {i.qty}</p>
   396→                  </div>
   397→                  <span className="text-sm font-semibold">{formatMinor(i.lineTotalMinor)}</span>
   398→                </li>
   399→              ))}
   400→            </ul>
   401→
   402→            {/* coupon */}

   406→                  <Tag className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />

   408→                    value={couponInput}

   410→                    placeholder="Coupon code"
   411→                    className="pl-8 min-h-[44px]"
   412→                    aria-label="Coupon code"
   413→                    onKeyDown={(e) => e.key === "Enter" && applyCoupon()}
   414→                  />
   415→                </div>
   416→                <Button variant="outline" onClick={applyCoupon} disabled={quoting || !couponInput.trim()} className="min-h-[44px] border-teal-700/40 text-teal-800 hover:bg-teal-50">
   417→                  {quoting ? <Loader2 className="h-4 w-4 animate-spin" /> : <TicketPercent className="h-4 w-4" />} Apply
   418→                </Button>
   419→              </div>
   420→              {couponErr ? <p className="text-xs font-medium text-rose-600">{couponErr}</p> : null}
   421→              {quote?.coupon?.valid ? (
   422→                <p className="text-xs font-medium text-emerald-700">
   423→                  {quote.coupon.code} applied {quote.coupon.freeShip ? "— free shipping" : `— ${formatMinor(quote.coupon.discountMinor ?? 0)} off`}
   424→                </p>
   425→              ) : null}
   426→            </div>
   427→
   428→            <Separator />
   429→
   430→            <dl className="space-y-2 text-sm">
   431→              <div className="flex justify-between">
   432→                <dt className="text-muted-foreground">Subtotal</dt>
   433→                <dd className="font-semibold">{formatMinor(quote?.subtotalMinor ?? 0)}</dd>
   434→              </div>
   435→              {quote && quote.discountMinor > 0 ? (
   436→                <div className="flex justify-between text-emerald-700">
   437→                  <dt>Discount</dt>
   438→                  <dd className="font-semibold">-{formatMinor(quote.discountMinor)}</dd>
   439→                </div>
   440→              ) : null}
   441→              <div className="flex justify-between">
   442→                <dt className="text-muted-foreground">Shipping</dt>

   445→              <Separator />
   446→              <div className="flex justify-between text-base">
   447→                <dt className="font-bold">Total</dt>

   449→              </div>
   450→            </dl>
   451→
   452→            {/* note */}
   453→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} placeholder="Note to seller (optional)" rows={2} className="min-h-[44px] text-sm" />
   454→
   455→            {touched && !formValid ? (
   456→              <p className="rounded-lg bg-rose-50 border border-rose-200 px-3 py-2 text-xs font-medium text-rose-700">
   457→                Please fill in the required fields (name, email, address, city, postcode).
   458→              </p>
   459→            ) : null}
   460→
   461→            <Button
   462→              size="lg"
   463→              onClick={placeOrder}
   464→              disabled={submitting || quoting}
   465→              className="w-full min-h-[48px] bg-teal-700 text-white hover:bg-teal-800"
   466→            >
   467→              {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   468→              {submitting ? "Creating order…" : `Pay ${formatMinor(total)}`}
   469→            </Button>
   470→            <p className="text-center text-[11px] text-muted-foreground">Sandbox demo — no real money moves.</p>
   471→          </div>
   472→        </aside>
   473→      </div>
   474→    </div>
   475→  );
   476→}

   502→      setFailed(false);
   503→      if (p.status === "succeeded") {
   504→        setPhase((ph) => (ph === "awaiting" ? "success" : ph));

   506→        setPhase((ph) => (ph === "awaiting" ? "failure" : ph));
   507→      }
   508→    } catch {
   509→      setFailed(true);
   510→    }
   511→  }, [paymentRef]);
   512→
   513→  // initial load + 3s polling for external status changes
   514→  useEffect(() => {

   517→    return () => clearInterval(t);
   518→  }, [load]);
   519→

   521→  useEffect(() => {
   522→    if (phase !== "awaiting") return;

   684→function OrderCard({ o, compact = false }: { o: LookupResultT; compact?: boolean }) {
   685→  const [copied, setCopied] = useState(false);
   686→  const paid = !["payment_pending", "cancelled"].includes(o.status);
   687→
   688→  function copy() {
   689→    navigator.clipboard?.writeText(o.orderNumber).then(
   690→      () => {
   691→        setCopied(true);
   692→        toast.success("Order number copied");
   693→        setTimeout(() => setCopied(false), 1500);
   694→      },
   695→      () => toast.error("Could not copy")
   696→    );
   697→  }
   698→
   699→  return (
   700→    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }} className={cn("rounded-2xl border bg-card p-5 sm:p-6 space-y-5", compact && "shadow-sm")}>
   701→      <div className="flex items-start gap-4">
   702→        <motion.div
   703→          initial={{ scale: 0, rotate: -20 }}
   704→          animate={{ scale: 1, rotate: 0 }}
   705→          transition={{ type: "spring", stiffness: 260, damping: 14 }}
   706→          className={cn("flex h-12 w-12 shrink-0 items-center justify-center rounded-full", paid ? "bg-emerald-100" : "bg-amber-100")}

   708→          {paid ? <CheckCircle2 className="h-7 w-7 text-emerald-600" /> : <PackageSearch className="h-7 w-7 text-amber-600" />}

   710→        <div className="min-w-0 flex-1">
   711→          <p className="text-xs uppercase tracking-widest text-muted-foreground">Order</p>
   712→          <div className="flex items-center gap-2 flex-wrap">
   713→            <p className="text-xl font-extrabold font-mono tracking-tight">{o.orderNumber}</p>
   714→            <Button variant="ghost" size="icon" className="h-8 w-8" onClick={copy} aria-label="Copy order number">

   732→
   733→      <div className="grid gap-6 sm:grid-cols-2">

   735→          <h3 className="pb-3 text-sm font-bold uppercase tracking-wide text-muted-foreground">Items</h3>

   737→            {o.items.map((it, i) => (

   831→export function LookupView({ ctx }: { ctx: ShopperCtx }) {
   832→  const [email, setEmail] = useState(() => (typeof window !== "undefined" ? window.localStorage.getItem(EMAIL_KEY) ?? "" : ""));
   833→  const [orderNumber, setOrderNumber] = useState("");
   834→  const [result, setResult] = useState<LookupResultT | null>(null);
   835→  const [error, setError] = useState<string | null>(null);
   836→  const [busy, setBusy] = useState(false);
   837→
   838→  async function search() {
   839→    if (!email.trim() || !orderNumber.trim()) {
   840→      setError("Both email and order number are required.");
   841→      return;
   842→    }
   843→    setBusy(true);
   844→    setError(null);
   845→    setResult(null);
   846→    try {
   847→      const data = await fetchJson<LookupResultT>(
   848→        `/api/orders/lookup?email=${encodeURIComponent(email.trim())}&orderNumber=${encodeURIComponent(orderNumber.trim())}`
   849→      );
   850→      setResult(data);
   851→      window.localStorage.setItem(EMAIL_KEY, email.trim().toLowerCase());
   852→    } catch (e) {
   853→      setError(e instanceof Error ? e.message : "Lookup failed");
   854→    } finally {
   855→      setBusy(false);
   856→    }
   857→  }
   858→
   859→  return (
   860→    <div className="mx-auto max-w-3xl px-4 py-8 flex-1">
   861→      <div className="rounded-2xl border bg-card p-5 sm:p-6 space-y-4">
