   311→   311→/** Total unread shopper messages — powers the "Messages" nav badge. */
   312→   312→function useSellerChatUnread(active: SellerNav["view"]) {
   313→   313→  const [unread, setUnread] = useState(0);
   314→   314→  useEffect(() => {
   315→   315→    let alive = true;
   316→   316→    const load = () =>
   317→   317→      apiGet<{ total: number }>("/api/store/chat/unread")
   318→   318→        .then((d) => alive && setUnread(d.total ?? 0))
   319→   319→        .catch(() => {});
   320→   320→    load();
   321→   321→    const t = setInterval(load, 15000);
   322→   322→    return () => {
   323→   323→      alive = false;
   324→   324→      clearInterval(t);
   325→   325→    };
   326→   326→  }, [active]);
   327→   327→  // instant re-sync when a thread is read inside the messages view (the view
   328→   328→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   329→   329→  useEffect(() => {
   330→   330→    const sync = () =>
   331→   331→      apiGet<{ total: number }>("/api/store/chat/unread")
   332→   332→        .then((d) => setUnread(d.total ?? 0))
   333→   333→        .catch(() => {});
   334→   334→    window.addEventListener("nomofee:chat-sync", sync);
   335→   335→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   336→   336→  }, []);
   337→   337→  return unread;
   338→   338→}
