   330→// FIN-3-c — seller in-app notification bell. Feeds from the NotificationLog
   331→// rows (channel "inapp") the finance flows already write for the seller email:
   332→// order.paid (FIN-2-b) + payout requested/approved/paid/failed/rejected
   333→// (FIN-2-a/FIN-3-b). The shopper bell is a different subsystem (Notification
   334→// model via /api/me/notifications) and is intentionally untouched.
   335→
   336→type SellerNotificationRow = { id: string; subject: string; createdAt: string; event: string };
   337→
   338→type SellerNotificationsPayload = { notifications: SellerNotificationRow[]; recentCount: number };
   339→
   340→/** House scrollbar style (same treatment as the shopper bell panel). */
   341→const SELLER_BELL_SCROLL =
   342→  "max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent";
   343→
   344→/** 24h recency = client-side "new" heuristic: NotificationLog has no read /
   345→ *  seen column (adding fake read-tracking was out of scope), so "within the
   346→ *  last 24h" is the honest approximation of "unread". */
   347→const RECENT_WINDOW_MS = 24 * 60 * 60 * 1000;
   348→
   349→function SellerNotificationBell() {
   350→  const [open, setOpen] = useState(false);
   351→  const [data, setData] = useState<SellerNotificationsPayload | null>(null);
   352→  const [loading, setLoading] = useState(false);
   353→  const [errored, setErrored] = useState(false);
   354→
   355→  const load = useCallback(async () => {
   356→    setLoading(true);
   357→    try {
   358→      const d = await apiGet<SellerNotificationsPayload>("/api/store/notifications");
   359→      setData(d);
   360→      setErrored(false);
   361→    } catch (e) {
   362→      // Silent by design — a failed bell must never break the seller shell:
   363→      // keep the last data (no badge flicker), warn for debugging only.
   364→      setErrored(true);
   365→      console.warn("[seller-bell] notifications fetch failed", e instanceof Error ? e.message : e);
   366→    } finally {
   367→      setLoading(false);
   368→    }
   369→  }, []);
   370→
   371→  // Badge freshness without aggressive polling: fetch once on mount (dot only
   372→  // needs recentCount) and refetch whenever the popover opens.
   373→  useEffect(() => {
   374→    void load();
   375→  }, [load]);
   376→
   377→  const onOpenChange = (v: boolean) => {
   378→    setOpen(v);
   379→    if (v) void load();
   380→  };
   381→
   382→  const recent = data?.recentCount ?? 0;
   383→  const rows = data?.notifications ?? null;
   384→
   385→  return (
   386→    <Popover open={open} onOpenChange={onOpenChange}>
   387→      <PopoverTrigger asChild>
   388→        <Button
   389→          variant="ghost"
   390→          size="icon"
   391→          className="relative text-muted-foreground"
   392→          aria-label="Seller notifications"
   393→          aria-haspopup="dialog"
   394→          aria-expanded={open}
   395→        >
   396→          <Bell className="h-4 w-4" aria-hidden="true" />
   397→          {recent > 0 ? (
   398→            <span
   399→              className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-amber-500 ring-2 ring-background"
   400→              aria-hidden="true"
   401→            />
   402→          ) : null}
   403→        </Button>
   404→      </PopoverTrigger>
   405→      <PopoverContent align="end" className="w-80 rounded-2xl p-0 sm:w-96" role="dialog" aria-label="Seller notifications panel">
   406→        <div className="flex items-center gap-2 border-b px-4 py-3">
   407→          <h2 className="text-sm font-semibold">Notifications</h2>
   408→          {recent > 0 ? (
   409→            <span className="rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 tabular-nums">
   410→              {recent} new
   411→            </span>
   412→          ) : null}
   413→        </div>
   414→
   415→        {loading && !rows ? (
   416→          <div className="space-y-3 p-4" aria-label="Loading notifications">
   417→            {Array.from({ length: 3 }).map((_, i) => (
   418→              <div key={i} className="flex items-center gap-3">
   419→                <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   420→                <div className="flex-1 space-y-1.5">
   421→                  <Skeleton className="h-3.5 w-3/4" />
   422→                  <Skeleton className="h-3 w-1/3" />
   423→                </div>
   424→              </div>
   425→            ))}
   426→          </div>
   427→        ) : !rows || rows.length === 0 ? (
   428→          <div className="flex flex-col items-center gap-2 px-6 py-10 text-center" aria-label="No notifications">
   429→            <span className="rounded-full bg-muted p-4">
   430→              <Bell className="h-6 w-6 text-muted-foreground" aria-hidden="true" />
   431→            </span>
   432→            <p className="text-sm font-semibold">{errored && !data ? "Notifications unavailable" : "No notifications yet"}</p>
   433→            <p className="max-w-[16rem] text-xs text-muted-foreground">
   434→              {errored && !data
   435→                ? "Couldn't load right now — it will retry next time you open the bell."
   436→                : "Payout and order updates will appear here."}
   437→            </p>
   438→          </div>
   439→        ) : (
   440→          <ul className={cn("divide-y", SELLER_BELL_SCROLL)} aria-label="Notification list">
   441→            {rows.map((n) => {
   442→              const isNew = Date.now() - new Date(n.createdAt).getTime() < RECENT_WINDOW_MS;
   443→              return (
   444→                <li key={n.id} className="flex items-start gap-3 px-4 py-3">
   445→                  <div className="min-w-0 flex-1">
   446→                    <p className="truncate text-[13px] font-medium" title={n.subject}>
   447→                      {n.subject}
   448→                    </p>
   449→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   450→                      <span title={new Date(n.createdAt).toLocaleString("en-MY")}>{ageOf(n.createdAt)}</span>
   451→                      <span aria-hidden="true"> · </span>
   452→                      <span className="lowercase">{n.event}</span>
   453→                    </p>
   454→                  </div>
   455→                  {isNew ? (
   456→                    <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-amber-500" role="img" aria-label="Recent" />
   457→                  ) : null}
   458→                </li>
   459→              );
   460→            })}
   461→          </ul>
   462→        )}
   463→      </PopoverContent>
   464→    </Popover>
   465→  );
   466→}

   470→/** Total unread shopper messages — powers the "Messages" nav badge. */
   471→function useSellerChatUnread(active: SellerNav["view"]) {
   472→  const [unread, setUnread] = useState(0);
   473→  useEffect(() => {
   474→    let alive = true;
   475→    const load = () =>
   476→      apiGet<{ total: number }>("/api/store/chat/unread")
   477→        .then((d) => alive && setUnread(d.total ?? 0))
   478→        .catch(() => {});
   479→    load();
   480→    const t = setInterval(load, 15000);
   481→    return () => {
   482→      alive = false;
   483→      clearInterval(t);
   484→    };
   485→  }, [active]);
   486→  // instant re-sync when a thread is read inside the messages view (the view
   487→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   488→  useEffect(() => {
   489→    const sync = () =>
   490→      apiGet<{ total: number }>("/api/store/chat/unread")
   491→        .then((d) => setUnread(d.total ?? 0))
   492→        .catch(() => {});
   493→    window.addEventListener("nomofee:chat-sync", sync);
   494→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   495→  }, []);
   496→  return unread;
   497→}
