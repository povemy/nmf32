    94→  }
    95→  const variants = p.variants ?? [{ name: "Default", price: p.price, stock: p.stock ?? 40 }];

   241→type SeedOrderItem = { product: { title: string; id: string; categoryId: string | null }; variantId: string; title: string; variantLabel?: string; unitPrice: number; qty: number };

   243→async function seedOrder(opts: {
   244→  storeId: string; storeName: string; walletId: string;
   245→  customer: { name: string; email: string; phone?: string };
   246→  items: SeedOrderItem[];
   247→  status: string; provider: string; daysBack: number;
   248→  discountMinor?: number; couponCode?: string; shippingMinor?: number;
   249→  withLedger?: boolean; withSettlement?: boolean;
   250→}) {
   251→  const createdAt = daysAgo(opts.daysBack, 9 + (opts.daysBack % 6));
   252→  const settings = await db.commissionSetting.findUnique({ where: { id: 1 } });
   253→  const holdDays = settings?.payoutHoldDays ?? 7;
   254→  const subtotal = opts.items.reduce((s, i) => s + i.unitPrice * i.qty, 0);
   255→  const discount = opts.discountMinor ?? 0;
   256→  const shipping = opts.shippingMinor ?? (subtotal >= 10000 ? 0 : 800);
   257→  const total = subtotal - discount + shipping;
   258→
   259→  const storeForCommission = await db.store.findUnique({ where: { id: opts.storeId } });
   260→  if (!storeForCommission?.sellerId) throw new Error(`seed order: store ${opts.storeId} or its seller missing`);
   261→
   262→  const computed = await computeOrderCommissions({
   263→    sellerId: storeForCommission.sellerId,
   264→    items: opts.items.map((i) => ({
   265→      title: i.title, unitPriceMinor: i.unitPrice, qty: i.qty,
   266→      productId: i.product.id, categoryId: i.product.categoryId, variantLabel: i.variantLabel ?? null,
   267→    })),
   268→    orderDiscountMinor: discount,
   269→    shippingMinorTotal: shipping,
   270→    now: createdAt,
   271→  });
   272→
   273→  const paid = !["payment_pending", "cancelled"].includes(opts.status);
   274→  const paidAt = paid ? createdAt : null;
   275→  const shippedAt = ["shipped", "delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 86400000) : null;
   276→  const deliveredAt = ["delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 3 * 86400000) : null;
   277→  const completedAt = opts.status === "completed" ? new Date(createdAt.getTime() + 9 * 86400000) : null;
   278→
   279→  const order = await db.order.create({
   280→    data: {
   281→      storeId: opts.storeId,
   282→      orderNumber: nextOrderNumber(),
   283→      customerName: opts.customer.name,
   284→      customerEmail: opts.customer.email,
   285→      customerPhone: opts.customer.phone ?? "60125550011",
   286→      status: opts.status,
   287→      channel: "storefront",
   288→      subtotalMinor: subtotal,
   289→      discountMinor: discount,
   290→      shippingMinor: shipping,
   291→      totalMinor: total,
   292→      couponCode: opts.couponCode ?? null,
   293→      shippingAddressJson: toJson({
   294→        line1: "42 Jalan Merdeka", city: "Kuala Lumpur", state: "Selangor", postcode: "43650", country: "MY",
   295→      }),
   296→      paymentProvider: opts.provider,
   297→      paidAt,
   298→      fulfilledAt: ["processing", "shipped", "delivered", "completed"].includes(opts.status) ? new Date(createdAt.getTime() + 43200000) : null,
   299→      shippedAt,
   300→      deliveredAt,
   301→      completedAt,
   302→      createdAt,
   303→      updatedAt: createdAt,
   304→    },
   305→  });
   306→
   307→  for (let i = 0; i < opts.items.length; i++) {
   308→    const item = opts.items[i];
   309→    const c = computed.items[i];
   310→    await db.orderItem.create({
   311→      data: {
   312→        orderId: order.id,
   313→        productId: item.product.id,
   314→        variantId: item.variantId,
   315→        title: item.title,
   316→        variantLabel: item.variantLabel ?? null,
   317→        unitPriceMinor: item.unitPrice,
   318→        qty: item.qty,
   319→        lineSubtotalMinor: item.unitPrice * item.qty,
   320→        discountMinor: c.discountMinor,
   321→        shippingMinor: c.shippingMinor,
   322→        commissionPercent: c.percent,
   323→        commissionMinor: c.commissionMinor,
   324→        fixedFeeMinor: c.fixedFeeMinor,
   325→        sellerNetMinor: c.sellerNetMinor,
   326→        breakdownJson: toJson(c.breakdown),
   327→      },
   328→    });
   329→    await db.product.update({ where: { id: item.product.id }, data: { salesCount: { increment: item.qty } } });
   330→    await db.productVariant.update({ where: { id: item.variantId }, data: { stock: { decrement: item.qty } } });
   331→  }
   332→
   333→  const historyTo: string[] = [];
   334→  if (paid) historyTo.push("paid");
   335→  if (["processing", "shipped", "delivered", "completed"].includes(opts.status)) historyTo.push("processing");
   336→  if (shippedAt) historyTo.push("shipped");
   337→  if (deliveredAt) historyTo.push("delivered");
   338→  if (completedAt) historyTo.push("completed");
   339→  let prev = "payment_pending";
   340→  for (const st of historyTo) {
   341→    await db.orderStatusHistory.create({
   342→      data: { orderId: order.id, fromStatus: prev, toStatus: st, actor: st === "paid" ? "system" : "seller", createdAt: paidAt ?? createdAt },
   343→    });
   344→    prev = st;
   345→  }
   346→
   347→  if (opts.withLedger && paid) {
   348→    await db.paymentIntent.create({
   349→      data: {
   350→        orderId: order.id, provider: opts.provider,
   351→        providerRef: `ref_${order.orderNumber.toLowerCase()}`,
   352→        amountMinor: total, status: "succeeded", expiresAt: daysAhead(1), createdAt, updatedAt: createdAt,
   353→      },
   354→    });
   355→    await postOrderPaid({
   356→      orderId: order.id, orderNumber: order.orderNumber, walletId: opts.walletId,
   357→      provider: opts.provider, totalMinor: total,
   358→      sellerNetMinor: computed.orderSellerNetMinor, commissionMinor: computed.orderCommissionMinor,
   359→      holdUntil: new Date((paidAt as Date).getTime() + holdDays * 86400000),
   360→      occurredAt: paidAt as Date,
   361→    });
   362→    if (opts.withSettlement && completedAt) {
   363→      await postGatewaySettlement({
   364→        provider: opts.provider, amountMinor: total,
   365→        ref: `stl_${order.orderNumber.toLowerCase()}`, occurredAt: new Date((paidAt as Date).getTime() + 172800000),
   366→      });
   367→    }
   368→  } else if (opts.status === "payment_pending") {
   369→    await db.paymentIntent.create({
   370→      data: {
   371→        orderId: order.id, provider: opts.provider,
   372→        providerRef: `ref_${order.orderNumber.toLowerCase()}`,
   373→        amountMinor: total, status: "requires_payment", expiresAt: new Date(createdAt.getTime() + 1800000),
   374→        redirectUrl: `/pay/${`ref_${order.orderNumber.toLowerCase()}`}`,
   375→      },
   376→    });
   377→  }
   378→  return order;
   379→}
