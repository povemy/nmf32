   204→   204→export class ApiFail extends Error {
   205→   205→  status: number;
   206→   206→  code: string;
   207→   207→  constructor(status: number, code: string, message: string) {
   208→   208→    super(message);
   209→   209→    this.status = status;
   210→   210→    this.code = code;
   211→   211→  }
   212→   212→}

   214→   214→export async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   215→   215→  const res = await fetch(url, {
   216→   216→    cache: "no-store",
   217→   217→    ...init,
   218→   218→    // Device identity headers (cart + shopper token) — keeps guest state working
   219→   219→    // where cookies are unavailable (cross-site iframe preview panels).
   220→   220→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   221→   221→  });
   222→   222→  let body: unknown = null;
   223→   223→  try {
   224→   224→    body = await res.json();
   225→   225→  } catch {
   226→   226→    /* non-json */
   227→   227→  }
   228→   228→  const b = body as { data?: T; error?: { code?: string; message?: string } } | null;
   229→   229→  if (!res.ok || !b || b.error) {
   230→   230→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   231→   231→  }
   232→   232→  return (b.data ?? ({} as T)) as T;
   233→   233→}
   234→   234→
   235→   235→/** Like fetchJson but also returns the paged meta envelope. */
   236→   236→export async function fetchEnvelope<T>(
   237→   237→  url: string,
   238→   238→  init?: RequestInit
   239→   239→): Promise<{ data: T; meta?: { page: number; size: number; total: number } }> {
   240→   240→  const res = await fetch(url, {
   241→   241→    cache: "no-store",
   242→   242→    ...init,
   243→   243→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   244→   244→  });
   245→   245→  let body: unknown = null;
   246→   246→  try {
   247→   247→    body = await res.json();
   248→   248→  } catch {
   249→   249→    /* non-json */
   250→   250→  }
   251→   251→  const b = body as { data?: T; meta?: { page: number; size: number; total: number }; error?: { code?: string; message?: string } } | null;
   252→   252→  if (!res.ok || !b || b.error) {
   253→   253→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   254→   254→  }
   255→   255→  return { data: (b.data ?? ({} as T)) as T, meta: b.meta };
   256→   256→}

   296→   296→export function trackOrderRef(orderNumber: string, email: string) {
   297→   297→  if (typeof window === "undefined") return;
   298→   298→  try {
   299→   299→    const raw = JSON.parse(window.localStorage.getItem(ORDER_REFS_KEY) ?? "[]") as OrderRefT[];
   300→   300→    const clean = raw.filter((r) => r.orderNumber !== orderNumber);
   301→   301→    clean.unshift({ orderNumber, email: email.toLowerCase(), at: Date.now() });
   302→   302→    window.localStorage.setItem(ORDER_REFS_KEY, JSON.stringify(clean.slice(0, 25)));
   303→   303→  } catch {
   304→   304→    /* corrupted list — ignore */
   305→   305→  }
   306→   306→}

   318→   318→/** Fetch the account cart and live-refresh on sm-cart-updated events + on any
   319→   319→ *  sign-in / sign-out change (carts are account-scoped — guests have none). */
   320→   320→export function useCart() {
   321→   321→  const me = useApp((s) => s.me);
   322→   322→  const authed = Boolean(me.loaded && me.shopper);
   323→   323→  const [cart, setCart] = useState<CartT | null>(null);
   324→   324→  const [loading, setLoading] = useState(true);
   325→   325→
   326→   326→  const refresh = useCallback(async () => {
   327→   327→    try {
   328→   328→      const data = await fetchJson<{ cart: CartT }>("/api/cart");
   329→   329→      setCart(data.cart ?? { id: null, storeId: null, storeSlug: null, storeName: null, items: [] });
   330→   330→    } catch {
   331→   331→      setCart({ id: null, storeId: null, storeSlug: null, storeName: null, items: [] });
   332→   332→    } finally {
   333→   333→      setLoading(false);
   334→   334→    }
   335→   335→  }, []);
   336→   336→
   337→   337→  useEffect(() => {
   338→   338→    setLoading(true);
   339→   339→    void refresh();
   340→   340→    const onEvt = () => refresh();
   341→   341→    window.addEventListener(CART_EVENT, onEvt);
   342→   342→    return () => window.removeEventListener(CART_EVENT, onEvt);
   343→   343→  }, [refresh, authed]);
   344→   344→
   345→   345→  const count = cart?.items.reduce((s, i) => s + i.qty, 0) ?? 0;
   346→   346→  const subtotal = cart?.items.reduce((s, i) => s + i.lineTotalMinor, 0) ?? 0;
   347→   347→  return { cart, count, subtotal, loading, refresh };
   348→   348→}

   354→   354→export function Stars({ rating, count, className }: { rating: number; count?: number; className?: string }) {
   355→   355→  const full = Math.floor(rating);
   356→   356→  const half = rating - full >= 0.25 && rating - full < 0.75;
   357→   357→  const rounded = rating - full >= 0.75 ? full + 1 : full;
   358→   358→  return (
   359→   359→    <span className={cn("inline-flex items-center gap-0.5", className)} aria-label={`Rated ${rating.toFixed(1)} out of 5`}>
   360→   360→      {[0, 1, 2, 3, 4].map((i) =>
   361→   361→        i < full || (i === full && rounded > full && !half) ? (
   362→   362→          <Star key={i} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   363→   363→        ) : i === full && half ? (
   364→   364→          <span key={i} className="relative inline-flex">
   365→   365→            <Star className="h-3.5 w-3.5 text-muted-foreground/40" />
   366→   366→            <StarHalf className="h-3.5 w-3.5 absolute inset-0 fill-amber-400 text-amber-400" />
   367→   367→          </span>
   368→   368→        ) : (
   369→   369→          <Star key={i} className="h-3.5 w-3.5 text-muted-foreground/40" />
   370→   370→        )
   371→   371→      )}
   372→   372→      <span className="text-xs text-muted-foreground ml-1 tabular-nums">
   373→   373→        {rating > 0 ? rating.toFixed(1) : "New"}
   374→   374→        {typeof count === "number" && count > 0 ? ` (${count})` : ""}
   375→   375→      </span>
   376→   376→    </span>
   377→   377→  );
   378→   378→}

   702→   702→export function OrderFlowTimeline({ history, current }: { history: { toStatus: string; createdAt: string }[]; current: string }) {
   703→   703→  const special = ["cancelled", "refunded", "partially_refunded"].includes(current);
   704→   704→  const reachedAt = new Map<string, string>();
   705→   705→  for (const h of history) reachedAt.set(h.toStatus, h.createdAt);
   706→   706→  const flow = special ? [] : FLOW;
   707→   707→  const currentIdx = flow.indexOf(current === "completed" ? "delivered" : current);
   708→   708→
   709→   709→  return (
   710→   710→    <ol className="space-y-0" aria-label="Order status timeline">
   711→   711→      {flow.map((step, i) => {
   712→   712→        const done = currentIdx >= i && currentIdx !== -1;
   713→   713→        const ts = reachedAt.get(step);
   714→   714→        const isLast = i === flow.length - 1;
   715→   715→        return (
   716→   716→          <li key={step} className="relative flex gap-3 pb-5 last:pb-0">
   717→   717→            {!isLast ? <span className={cn("absolute left-[9px] top-5 bottom-0 w-px", done ? "bg-teal-600" : "bg-border")} /> : null}
   718→   718→            <span
   719→   719→              className={cn(
   720→   720→                "relative z-10 mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-full border-2",
   721→   721→                done ? "border-teal-600 bg-teal-600" : "border-muted-foreground/30 bg-background"
   722→   722→              )}
   723→   723→            >
   724→   724→              {done ? (
   725→   725→                <svg viewBox="0 0 10 8" className="h-2.5 w-2.5 fill-none stroke-white" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden>
   726→   726→                  <path d="M1 4l2.5 2.5L9 1" />

   729→   729→            </span>
   730→   730→            <div className="min-w-0">
   731→   731→              <p className={cn("text-sm font-medium leading-5", done ? "text-foreground" : "text-muted-foreground")}>
   732→   732→                {statusLabel(step)}
   733→   733→              </p>
   734→   734→              {ts ? (
   735→   735→                <p className="text-xs text-muted-foreground">{new Date(ts).toLocaleString("en-MY", { dateStyle: "medium", timeStyle: "short" })}</p>
   736→   736→              ) : null}

   738→   738→          </li>
   739→   739→        );
   740→   740→      })}
   741→   741→      {special ? (
   742→   742→        <li className="flex items-center gap-2">
   743→   743→          <StatusBadge status={current} />

   745→   745→        </li>
   746→   746→      ) : null}
   747→   747→    </ol>
   748→   748→  );
   749→   749→}

   766→   766→export function discountPercent(price: number, compareAt: number | null | undefined): number | null {
