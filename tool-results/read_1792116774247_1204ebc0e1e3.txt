   112→function makeIdemKey(): string {
   113→  try {
   114→    return crypto.randomUUID();
   115→  } catch {
   116→    return `idem_${Date.now()}_${Math.random().toString(36).slice(2)}`;
   117→  }
   118→}
   119→
   120→function loadDraft(): DraftT {
   121→  const savedContact: Record<string, string> = (() => {
   122→    if (typeof window === "undefined") return {};
   123→    try {
   124→      return JSON.parse(window.localStorage.getItem(CONTACT_KEY) ?? "{}") as Record<string, string>;
   125→    } catch {
   126→      return {};
   127→    }
   128→  })();
   129→  const base: DraftT = {
   130→    phase: 1,
   131→    name: savedContact.name ?? "",
   132→    email: savedContact.email ?? "",
   133→    phone: savedContact.phone ?? "",
   134→    addressId: null,
   135→    shippingRateId: "standard",
   136→    provider: "fpx_toyyibpay",
   137→    couponCode: "",
   138→    note: "",
   139→    agreed: false,
   140→    idemKey: makeIdemKey(),
   141→  };
   142→  if (typeof window === "undefined") return base;
   143→  try {
   144→    const raw = JSON.parse(window.sessionStorage.getItem(DRAFT_KEY) ?? "{}") as Partial<DraftT>;
   145→    return {
   146→      ...base,
   147→      ...raw,
   148→      phase: raw.phase === 2 || raw.phase === 3 ? raw.phase : 1,
   149→      idemKey: raw.idemKey || base.idemKey, // stable across retries → idempotent resubmits
   150→    };
   151→  } catch {
   152→    return base;
   153→  }
   154→}
   155→
   156→function persistDraft(d: DraftT) {
   157→  if (typeof window !== "undefined") window.sessionStorage.setItem(DRAFT_KEY, JSON.stringify(d));
   158→}
   159→
   160→function clearDraft() {
   161→  if (typeof window !== "undefined") window.sessionStorage.removeItem(DRAFT_KEY);
   162→}

   228→const STEPS = [
   229→  { n: 1 as const, label: "Information", icon: User },
   230→  { n: 2 as const, label: "Delivery", icon: Truck },
   231→  { n: 3 as const, label: "Payment", icon: CreditCard },

   233→
   234→function Stepper({ phase, onJump }: { phase: 1 | 2 | 3; onJump: (n: 1 | 2) => void }) {
   235→  return (
   236→    <ol className="flex min-w-0 items-center gap-2 overflow-hidden sm:gap-3" aria-label="Checkout progress">
   237→      {STEPS.map((s, i) => {
   238→        const done = phase > s.n;
   239→        const current = phase === s.n;
   240→        return (
   241→          <li key={s.n} className="flex min-w-0 flex-1 items-center gap-2 last:flex-none sm:gap-3">
   242→            <button
   243→              type="button"
   244→              disabled={!done}
   245→              onClick={() => done && onJump(s.n)}
   246→              className="flex min-w-0 items-center gap-2 disabled:cursor-default"
   247→              aria-current={current ? "step" : undefined}
   248→              aria-label={`Step ${s.n}: ${s.label}${done ? " (completed — click to edit)" : ""}`}
   249→            >
   250→              <span
   251→                className={cn(
   252→                  "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors",
   253→                  done && "border-transparent bg-brand-gradient text-white shadow-sm",
   254→                  current && "border-teal-600 bg-teal-50 text-teal-700",
   255→                  !done && !current && "border-muted-foreground/25 bg-card text-muted-foreground"
   256→                )}
   257→              >
   258→                {done ? <Check className="h-4 w-4" aria-hidden="true" /> : s.n}
   259→              </span>
   260→              <span
   261→                className={cn(
   262→                  "truncate text-xs font-bold sm:text-sm",
   263→                  current ? "text-teal-800" : done ? "text-foreground" : "text-muted-foreground"
   264→                )}
   265→              >
   266→                {s.label}
   267→              </span>
   268→            </button>
   269→            {i < STEPS.length - 1 ? (
   270→              <span className={cn("h-0.5 min-w-4 flex-1 rounded-full", phase > s.n ? "bg-brand-gradient" : "bg-border")} aria-hidden="true" />
   271→            ) : null}
   272→          </li>
   273→        );
   274→      })}
   275→    </ol>
   276→  );
   277→}

   329→function validateForm(form: typeof emptyForm): Record<string, string | null> {
   330→  return {
   331→    name: form.name.trim().length >= 2 ? null : "Recipient name is required",
   332→    phone: !form.phone.trim() || PHONE_RE.test(form.phone.trim()) ? null : "Enter a valid phone number",
   333→    line1: form.line1.trim() ? null : "Street address is required",
   334→    city: form.city.trim() ? null : "City is required",
   335→    state: form.state ? null : "State is required",
   336→    postcode: POSTCODE_RE.test(form.postcode.trim()) ? null : "5-digit postcode required",
   337→  };

   344→export function CheckoutView({ ctx }: { ctx: ShopperCtx }) {
   345→  const { cart, loading: cartLoading, refresh: refreshCart } = useCart();
   346→  const items = cart?.items ?? [];
   347→
   348→  const me = useApp((s) => s.me);
   349→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   350→  const shopper = me.loaded ? me.shopper : null;
   351→
   352→  const [draft, setDraft] = useState<DraftT>(loadDraft);
   353→  useEffect(() => persistDraft(draft), [draft]);
   354→
   355→  // Contact details come from the ACCOUNT (carts/checkout are account-scoped);
   356→  // only fill blanks so the shopper can still override per-order (gifts etc.).
   357→  useEffect(() => {
   358→    if (!shopper) return;
   359→    setDraft((d) => ({
   360→      ...d,
   361→      name: d.name.trim() || shopper.fullName,
   362→      email: d.email.trim() || shopper.email,
   363→      phone: d.phone.trim() || shopper.phone || "",
   364→    }));
   365→  }, [shopper]);
   366→
   367→  const [addresses, setAddresses] = useState<AddressT[] | null>(null);
   368→  const [addressesLoading, setAddressesLoading] = useState(true);
   369→  const [formMode, setFormMode] = useState<null | { kind: "new" } | { kind: "edit"; address: AddressT }>(null);
   370→  const [form, setForm] = useState(emptyForm);
   371→  const [formErrors, setFormErrors] = useState<Record<string, string | null>>({});
   372→  const [formBusy, setFormBusy] = useState(false);
   373→  const [deleteArm, setDeleteArm] = useState<string | null>(null);
   374→
   375→  const [quote, setQuote] = useState<QuoteT | null>(null);
   376→  const [quoting, setQuoting] = useState(false);
   377→  const [couponInput, setCouponInput] = useState("");
   378→  const [couponErr, setCouponErr] = useState<string | null>(null);
   379→  const [tried, setTried] = useState(false);
   380→  const [submitting, setSubmitting] = useState(false);
   381→
   382→  const selectedAddr = useMemo(
   383→    () => addresses?.find((a) => a.id === draft.addressId) ?? null,
   384→    [addresses, draft.addressId]
   385→  );
   386→
   387→  // load the device address book once
   388→  useEffect(() => {
   389→    let alive = true;
   390→    fetchJson<{ addresses: AddressT[] }>("/api/checkout/addresses")
   391→      .then((d) => {
   392→        if (!alive) return;
   393→        setAddresses(d.addresses);
   394→        setDraft((dft) => {
   395→          if (dft.addressId && d.addresses.some((a) => a.id === dft.addressId)) return dft;
   396→          const def = d.addresses.find((a) => a.isDefault) ?? d.addresses[0];
   397→          return def ? { ...dft, addressId: def.id } : dft;
   398→        });
   399→      })
   400→      .catch(() => alive && setAddresses([]))
   401→      .finally(() => alive && setAddressesLoading(false));
   402→    return () => {
   403→      alive = false;
   404→    };
   405→  }, []);
   406→
   407→  const payloadItems = useMemo(() => items.map((i) => ({ variantId: i.variantId, qty: i.qty })), [items]);
   408→  const destState = selectedAddr?.state ?? null;
   409→
   410→  const refreshQuote = useCallback(
   411→    async (code: string, rateId: string, state: string | null) => {
   412→      if (payloadItems.length === 0) return;
   413→      setQuoting(true);
   414→      try {
   415→        // OPTION A: no storeSlug — the server groups cart items per seller itself.
   416→        const q = await postJson<QuoteT>("/api/checkout/quote", {
   417→          items: payloadItems,
   418→          couponCode: code || undefined,
   419→          shippingRateId: rateId,
   420→          state: state ?? undefined,
   421→        });
   422→        setQuote(q);
   423→      } catch {
   424→        setQuote(null);
   425→      } finally {
   426→        setQuoting(false);
   427→      }
   428→    },
   429→    [payloadItems]
   430→  );
   431→
   432→  // auto-quote on cart / shipping / destination / coupon change
   433→  useEffect(() => {
   434→    refreshQuote(draft.couponCode, draft.shippingRateId, destState);
   435→  }, [refreshQuote, draft.couponCode, draft.shippingRateId, destState]);
   436→
   437→  // delivery option became invalid for this destination → fall back to Standard (server rule)
   438→  useEffect(() => {
   439→    if (!quote?.shippingOptions) return;
   440→    const sel = quote.shippingOptions.find((o) => o.id === draft.shippingRateId);
   441→    if (sel && !sel.available) {
   442→      setDraft((d) => ({ ...d, shippingRateId: "standard" }));
   443→      toast.error(sel.reason ?? "Selected delivery is unavailable — switched to Standard");
   444→    }
   445→  }, [quote, draft.shippingRateId]);
   446→
   447→  const stockIssues = useMemo(
   448→    () => (quote?.items ?? []).filter((i) => i.qty > i.stock),
   449→    [quote]
   450→  );
   451→
   452→  // ------------------------------------------------------------- coupon
   453→  async function applyCoupon() {
   454→    const code = couponInput.trim().toUpperCase();
   455→    if (!code) return;
   456→    setCouponErr(null);
   457→    setQuoting(true);
   458→    try {
   459→      const q = await postJson<QuoteT>("/api/checkout/quote", {
   460→        items: payloadItems,
   461→        couponCode: code,
   462→        shippingRateId: draft.shippingRateId,
   463→        state: destState ?? undefined,
   464→      });
   465→      if (q.coupon?.valid) {
   466→        setDraft((d) => ({ ...d, couponCode: code }));
   467→        setQuote(q);
   468→        toast.success(`Coupon ${code} applied`, {
   469→          description: q.coupon.freeShip ? "Free shipping unlocked" : `You save ${formatMinor(q.coupon.discountMinor ?? 0)}`,
   470→        });
   471→      } else {
   472→        setCouponErr(q.coupon?.reason ?? "Invalid coupon");
   473→      }
   474→    } catch (e) {
   475→      setCouponErr(e instanceof Error ? e.message : "Could not validate coupon");
   476→    } finally {
   477→      setQuoting(false);
   478→    }
   479→  }
   480→
   481→  function removeCoupon() {
   482→    setDraft((d) => ({ ...d, couponCode: "" }));
   483→    setCouponInput("");
   484→    setCouponErr(null);
   485→  }
   486→
   487→  // ------------------------------------------------------------- address book ops
   488→  async function saveAddress() {
   489→    const errs = validateForm(form);
   490→    setFormErrors(errs);
   491→    if (Object.values(errs).some(Boolean)) return;
   492→    setFormBusy(true);
   493→    try {
   494→      const payload = { ...form, name: form.name.trim(), line1: form.line1.trim(), city: form.city.trim(), postcode: form.postcode.trim() };
   495→      const res =
   496→        formMode?.kind === "edit"
   497→          ? await patchJson<{ address: AddressT; addresses: AddressT[] }>(`/api/checkout/addresses/${formMode.address.id}`, payload)
   498→          : await postJson<{ address: AddressT; addresses: AddressT[] }>("/api/checkout/addresses", payload);
   499→      setAddresses(res.addresses);
   500→      setDraft((d) => ({ ...d, addressId: res.address.id }));
   501→      setFormMode(null);
   502→      setForm(emptyForm);
   503→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   504→      toast.success(formMode?.kind === "edit" ? "Address updated" : "Address saved", {
   505→        description: `${res.address.line1}, ${res.address.city} ${res.address.postcode}`,
   506→      });
   507→    } catch (e) {
   508→      toast.error(e instanceof Error ? e.message : "Could not save address");
   509→    } finally {
   510→      setFormBusy(false);
   511→    }
   512→  }
   513→
   514→  async function removeAddress(id: string) {
   515→    try {
   516→      const res = await deleteJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`);
   517→      setAddresses(res.addresses);
   518→      setDeleteArm(null);
   519→      setDraft((d) => (d.addressId === id ? { ...d, addressId: res.addresses.find((a) => a.isDefault)?.id ?? res.addresses[0]?.id ?? null } : d));
   520→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   521→      toast.success("Address removed");
   522→    } catch (e) {
   523→      toast.error(e instanceof Error ? e.message : "Could not remove address");
   524→    }
   525→  }
   526→
   527→  async function makeDefault(id: string) {
   528→    try {
   529→      const res = await patchJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`, { isDefault: true });
   530→      setAddresses(res.addresses);
   531→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   532→      toast.success("Default address updated");
   533→    } catch (e) {
   534→      toast.error(e instanceof Error ? e.message : "Could not update address");
   535→    }
   536→  }
