    64→    64→// ---------------------------------------------------------------------------
    65→    65→// FIN-1 POSTING CORE — atomic + idempotent.
    66→    66→//
    67→    67→// ATOMICITY: every posting runs inside a single prisma.$transaction (lines,
    68→    68→// account upserts, wallet-cache refresh all commit or roll back together).
    69→    69→// Recipes called with an outer transaction client run INSIDE it — never
    70→    70→// nested — so a webhook/payout flow can group ledger + business writes.
    71→    71→//
    72→    72→// IDEMPOTENCY (dedupKey semantics — read before changing):
    73→    73→//   • dedupKey is an OPTIONAL caller/recipe-supplied unique helper key stored
    74→    74→//     on LedgerTransaction. When null/undefined the row is written with NULL —
    75→    75→//     Postgres unique indexes treat NULLs as distinct, so non-deduped postings
    76→    76→//     keep the legacy always-insert behavior (e.g. repeatable admin
    77→    77→//     adjustments).
    78→    78→//   • Recipes pass EXPLICIT keys only where a posting must happen EXACTLY
    79→    79→//     once per business event (T1 per order, T2 per settlement ref, T3/T4/T4f
    80→    80→//     per payout, T5 per refund row). Generic auto-keys are deliberately NOT
    81→    81→//     derived here: a (txnType,refType,refId) default would falsely dedup
    82→    82→//     legitimately-repeatable postings such as multi partial refunds per
    83→    83→//     order (refType=order for every refund) — that is why postRefund takes
    84→    84→//     the per-Refund-row refundId for its key.
    85→    85→//   • On a P2002 unique violation the EXISTING transaction is fetched and
    86→    86→//     returned with idempotentReplay:true (see wasReplay) instead of throwing
    87→    87→//     — duplicate webhook deliveries / double-clicked admins converge on the
    88→    88→//     original posting and never create a second one.
    89→    89→// ---------------------------------------------------------------------------
    90→    90→
    91→    91→export type PostedTransaction = Prisma.LedgerTransactionGetPayload<{ include: { lines: true } }> & {
    92→    92→  /** True when the dedupKey already existed and the ORIGINAL posting was returned. */
    93→    93→  idempotentReplay?: boolean;
    94→    94→};
    95→    95→
    96→    96→/** True when a posted result is actually the original txn replayed (dedup hit). */
    97→    97→export function wasReplay(t: PostedTransaction): boolean {
    98→    98→  return t.idempotentReplay === true;
    99→    99→}

   101→   101→/** Post a balanced double-entry transaction. Throws if unbalanced. */
   102→   102→export async function postTransaction(input: {
   103→   103→  txnType: string;
   104→   104→  refType?: string | null;
   105→   105→  refId?: string | null;
   106→   106→  description?: string | null;
   107→   107→  currency?: string;
   108→   108→  lines: LedgerLineInput[];
   109→   109→  occurredAt?: Date;
   110→   110→  /** Optional unique idempotency key (see block comment above). NULL = no dedup. */
   111→   111→  dedupKey?: string | null;
   112→   112→  /** Existing transaction client — when omitted the posting wraps itself in db.$transaction. */
   113→   113→  tx?: Prisma.TransactionClient;
   114→   114→}): Promise<PostedTransaction> {
   115→   115→  const run = async (tx: Prisma.TransactionClient): Promise<PostedTransaction> => {
   116→   116→    const debit = input.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   117→   117→    const credit = input.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   118→   118→    if (debit !== credit || debit === 0) {
   119→   119→      throw new Error(`LEDGER_UNBALANCED debit=${debit} credit=${credit} txn=${input.txnType}`);
   120→   120→    }
   121→   121→    for (const l of input.lines) await ensureAccountTx(l.accountCode, l.walletId ?? null, tx);
   122→   122→    // With a dedupKey we may need to RECOVER from a unique violation inside
   123→   123→    // this (possibly caller-provided) transaction. Postgres aborts the whole
   124→   124→    // tx on a failed statement (25P02 — even a SELECT is refused afterwards),
   125→   125→    // so the create is fenced with a SAVEPOINT: on P2002 we rewind to it and
   126→   126→    // the original row becomes readable again. Without this, the
   127→   127→    // replay-instead-of-throw contract below is unreachable inside a real tx.
   128→   128→    const sp = input.dedupKey
   129→   129→      ? `fin1_sp_${Date.now().toString(36)}${Math.floor(Math.random() * 1e9).toString(36)}`
   130→   130→      : null;
   131→   131→    if (sp) await tx.$executeRawUnsafe(`SAVEPOINT ${sp}`);
   132→   132→    try {
   133→   133→      const created = await tx.ledgerTransaction.create({
   134→   134→        data: {
   135→   135→          txnType: input.txnType,
   136→   136→          refType: input.refType ?? null,
   137→   137→          refId: input.refId ?? null,
   138→   138→          description: input.description ?? null,
   139→   139→          occurredAt: input.occurredAt ?? new Date(),
   140→   140→          dedupKey: input.dedupKey ?? null,
   141→   141→          lines: {
   142→   142→            create: input.lines.map((l) => ({
   143→   143→              accountCode: l.accountCode,
   144→   144→              direction: l.direction,
   145→   145→              amountMinor: l.amountMinor,
   146→   146→              walletId: l.walletId ?? null,
   147→   147→              availableAt: l.availableAt ?? null,
   148→   148→              memo: l.memo ?? null,
   149→   149→              // Denormalized ref (FIN-1 helper columns) so ref-scoped line
   150→   150→              // queries don't need a join; mirrors the parent txn.
   151→   151→              refType: input.refType ?? null,
   152→   152→              refId: input.refId ?? null,
   153→   153→            })),
   154→   154→          },
   155→   155→        },
   156→   156→        include: { lines: true },
   157→   157→      });
   158→   158→      if (sp) await tx.$executeRawUnsafe(`RELEASE SAVEPOINT ${sp}`);
   159→   159→      return created;
   160→   160→    } catch (e) {
   161→   161→      // Idempotent replay: the dedupKey already exists → return the ORIGINAL
   162→   162→      // transaction instead of throwing (never double-post, never 500).
   163→   163→      if (input.dedupKey && e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   164→   164→        if (sp) await tx.$executeRawUnsafe(`ROLLBACK TO SAVEPOINT ${sp}`);
   165→   165→        const existing = await tx.ledgerTransaction.findUnique({
   166→   166→          where: { dedupKey: input.dedupKey },
   167→   167→          include: { lines: true },
   168→   168→        });
   169→   169→        if (existing) return { ...existing, idempotentReplay: true };
   170→   170→      }
   171→   171→      throw e;
   172→   172→    }
   173→   173→  };
   174→   174→  return input.tx ? run(input.tx) : db.$transaction(run);
   175→   175→}

   179→   179→export type WalletTruth = {
   180→   180→  pendingMinor: number;
   181→   181→  availableMinor: number;
   182→   182→  lifetimeEarnedMinor: number;
   183→   183→  lifetimeCommissionMinor: number;
   184→   184→};
   185→   185→
   186→   186→/** Compute wallet balances from ledger lines (the source of truth). */
   187→   187→export async function walletTruth(walletId: string, tx?: Prisma.TransactionClient): Promise<WalletTruth> {
   188→   188→  const payable = ACCOUNTS.sellerPayable(walletId);
   189→   189→  const client = tx ?? db;
   190→   190→  const lines = await client.ledgerLine.findMany({
   191→   191→    where: { accountCode: payable },
   192→   192→    select: { direction: true, amountMinor: true, availableAt: true },
   193→   193→  });
   194→   194→  const now = new Date();
   195→   195→  let totalCredits = 0, maturedCredits = 0, totalDebits = 0, lifetimeCommission = 0;
   196→   196→  for (const l of lines) {
   197→   197→    if (l.direction === "credit") {
   198→   198→      totalCredits += l.amountMinor;
   199→   199→      if (!l.availableAt || l.availableAt <= now) maturedCredits += l.amountMinor;
   200→   200→    } else {
   201→   201→      totalDebits += l.amountMinor;
   202→   202→    }
   203→   203→  }
   204→   204→  // lifetime commission from revenue account lines tagged with this wallet? simpler: from txns T1 memo — use commission revenue lines with walletId
   205→   205→  const commissionLines = await client.ledgerLine.findMany({
   206→   206→    where: { accountCode: ACCOUNTS.revenueCommission(), walletId, direction: "credit" },
   207→   207→    select: { amountMinor: true },
   208→   208→  });
   209→   209→  lifetimeCommission = commissionLines.reduce((s, l) => s + l.amountMinor, 0);
   210→   210→  return {
   211→   211→    pendingMinor: totalCredits - maturedCredits,
   212→   212→    availableMinor: maturedCredits - totalDebits,
   213→   213→    lifetimeEarnedMinor: totalCredits - lifetimeCommission,
   214→   214→    lifetimeCommissionMinor: lifetimeCommission,
   215→   215→  };
   216→   216→}

   232→   232→      lifetimeCommissionMinor: truth.lifetimeCommissionMinor,
   233→   233→    },
   234→   234→  });
   235→   235→  return truth;

   369→   369→/**
   370→   370→ * T5 — refund: DR SELLER:PAYABLE (seller share) + DR COMMISSION (reversal) | CR GATEWAY:{p}
   371→   371→ * Dedup: exactly ONE T5 per Refund ROW (`refund:refund:{refundId}`) — an order
   372→   372→ * may legally receive MULTIPLE partial refunds, so the key is per refund row,
   373→   373→ * NOT per order (a (txnType,order) key would false-dedup the 2nd partial).
   374→   374→ */
   375→   375→export async function postRefund(input: {
   376→   376→  orderId: string;
   377→   377→  orderNumber: string;
   378→   378→  walletId: string;
   379→   379→  provider: string;
   380→   380→  amountMinor: number;
   381→   381→  commissionReversalMinor: number;
   382→   382→  /** The Refund row driving this posting — required for its dedup key. */
   383→   383→  refundId: string;
   384→   384→  tx?: Prisma.TransactionClient;
   385→   385→}) {
   386→   386→  const sellerShare = input.amountMinor - input.commissionReversalMinor;
   387→   387→  const t = await postTransaction({
   388→   388→    txnType: "refund",
   389→   389→    refType: "order",
   390→   390→    refId: input.orderId,
   391→   391→    description: `Refund for ${input.orderNumber}`,
   392→   392→    dedupKey: `refund:refund:${input.refundId}`,
   393→   393→    lines: [
   394→   394→      { accountCode: ACCOUNTS.sellerPayable(input.walletId), direction: "debit", amountMinor: sellerShare, walletId: input.walletId, memo: `Refund seller share ${input.orderNumber}` },
   395→   395→      { accountCode: ACCOUNTS.revenueCommission(), direction: "debit", amountMinor: input.commissionReversalMinor, walletId: input.walletId, memo: `Commission reversal ${input.orderNumber}` },
   396→   396→      { accountCode: ACCOUNTS.gateway(input.provider), direction: "credit", amountMinor: input.amountMinor, memo: `Refund ${input.orderNumber}` },
   397→   397→    ],
   398→   398→    tx: input.tx,
   399→   399→  });
   400→   400→  await refreshWalletCache(input.walletId, input.tx);
   401→   401→  return t;
   402→   402→}

   440→   440→/** Nightly invariant (ICCO): every txn balanced; wallet cache == ledger truth. */
   441→   441→export async function runLedgerInvariants() {
   442→   442→  const problems: string[] = [];
   443→   443→  const txns = await db.ledgerTransaction.findMany({ include: { lines: true } });
   444→   444→  for (const t of txns) {
   445→   445→    const debit = t.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   446→   446→    const credit = t.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   447→   447→    if (debit !== credit) problems.push(`txn ${t.id} (${t.txnType}) unbalanced: D${debit} C${credit}`);
   448→   448→  }
   449→   449→  const wallets = await db.wallet.findMany();
   450→   450→  for (const w of wallets) {
   451→   451→    const truth = await walletTruth(w.id);
   452→   452→    if (truth.availableMinor !== w.availableMinor || truth.pendingMinor !== w.pendingMinor) {
   453→   453→      problems.push(`wallet ${w.id} cache drift: available cache=${w.availableMinor} truth=${truth.availableMinor}, pending cache=${w.pendingMinor} truth=${truth.pendingMinor}`);
   454→   454→    }
   455→   455→  }
   456→   456→  // clearing + gateway accounts must net to zero-ish (settlements outstand-ing)
   457→   457→  return { ok: problems.length === 0, problems, txnCount: txns.length, walletCount: wallets.length };
   458→   458→}

   489→   489→// ------------------------------------------------------------- LEDGER-TRUTH STATEMENTS
   490→   490→// INTEGRITY RULE (user directive): every number printed on a payout statement
   491→   491→// must be DERIVED from ledger lines for the payout period — never decorated.
   492→   492→// The lines always reconcile to the payout amount via an explicit
   493→   493→// "Retained in wallet" balancing line, so Σ(lines) ≡ net payout.
   494→   494→
   495→   495→export type LedgerStatement = {
   496→   496→  /** Parsed shape of the JSON string returned by buildLedgerStatementFromLedger
   497→   497→   *  (Payout.statementJson column stores the serialized form). */
   498→   498→  sellerName: string;
   499→   499→  periodFrom: string;
   500→   500→  periodTo: string;
   501→   501→  orderCount: number;
   502→   502→  reconciled: true;
   503→   503→  generatedAt: string;
   504→   504→  lines: { label: string; amountMinor: number }[];
   505→   505→};
   506→   506→
   507→   507→/** Aggregate a wallet's ledger activity for [periodFrom, periodTo] and build a
   508→   508→ *  payout statement whose lines sum EXACTLY to `payoutAmountMinor`.
   509→   509→ *  Components (all from ledger truth):
   510→   510→ *    • sales        — payable credits from order_paid (seller net) + matching
   511→   511→ *                     commission-revenue credits (gross sales)
   512→   512→ *    • commission   — commission-revenue credits from order_paid in the period
   513→   513→ *    • refunds      — payable debits from refund txns + commission reversals
   514→   514→ *    • adjustments  — net admin_adjustment movement on the payable account
   515→   515→ *    • retained     — residual balancing line so Σ(lines) ≡ payout (holds, caps,
   516→   516→ *                     prior-balance carry)                                   */
   517→   517→export async function buildLedgerStatementFromLedger(params: {
   518→   518→  walletId: string;
   519→   519→  sellerName: string;
   520→   520→  payoutAmountMinor: number;
   521→   521→  orderCount: number;
   522→   522→  periodFrom: Date;
   523→   523→  periodTo: Date;
   524→   524→}): Promise<string> {
   525→   525→  const { walletId, sellerName, payoutAmountMinor, orderCount, periodFrom, periodTo } = params;
   526→   526→  const payable = ACCOUNTS.sellerPayable(walletId);
   527→   527→  const commissionAccount = ACCOUNTS.revenueCommission();
   528→   528→
   529→   529→  // Wallet-scope lines in the period, joined to their txn type. Period is
   530→   530→  // evaluated on txn.occurredAt (business time) — rows are often backdated
   531→   531→  // (seed) or posted later (payout T3 lands after the statement snapshot).
   532→   532→  const lines = await db.ledgerLine.findMany({
   533→   533→    where: {
   534→   534→      OR: [{ accountCode: payable }, { accountCode: commissionAccount, walletId }],
   535→   535→      txn: { occurredAt: { gte: periodFrom, lte: periodTo } },
   536→   536→    },
   537→   537→    select: { accountCode: true, direction: true, amountMinor: true, txn: { select: { txnType: true } } },
   538→   538→  });
   539→   539→
   540→   540→  let salesNet = 0; // payable credits from order_paid
   541→   541→  let commission = 0; // commission-revenue credits from order_paid
   542→   542→  let refundSellerShare = 0; // payable debits from refund txns
   543→   543→  let refundCommissionReversal = 0; // commission debits from refund txns
   544→   544→  let adjustments = 0; // admin_adjustment net (credits − debits on payable)
   545→   545→
   546→   546→  for (const l of lines) {
   547→   547→    const type = l.txn.txnType;
   548→   548→    if (l.accountCode === payable) {
   549→   549→      if (l.direction === "credit") {
   550→   550→        if (type === "order_paid") salesNet += l.amountMinor;
   551→   551→        else if (type === "admin_adjustment") adjustments += l.amountMinor;
   552→   552→      } else {
   553→   553→        if (type === "refund") refundSellerShare += l.amountMinor;
   554→   554→        else if (type === "admin_adjustment") adjustments -= l.amountMinor;
   555→   555→      }
   556→   556→    } else if (l.accountCode === commissionAccount) {
   557→   557→      if (l.direction === "credit") {
   558→   558→        if (type === "order_paid") commission += l.amountMinor;
   559→   559→      } else if (type === "refund") {
   560→   560→        refundCommissionReversal += l.amountMinor;
   561→   561→      }
   562→   562→    }
   563→   563→  }
   564→   564→
   565→   565→  const sales = salesNet + commission; // gross sales for the period
   566→   566→  const refunds = refundSellerShare + refundCommissionReversal;
   567→   567→  const netMovement = sales - commission - refunds + adjustments;
   568→   568→  // Balancing line so Σ(statement lines) ≡ net payout: when the period's net
   569→   569→  // movement exceeds the payout (pending holds, min-payout cap, prior carry),
   570→   570→  // the difference is "retained"; when a payout draws on prior balance, it's
   571→   571→  // a positive carry line. Either way the statement always reconciles.
   572→   572→  const retained = payoutAmountMinor - netMovement;
   573→   573→
   574→   574→  const linesOut: { label: string; amountMinor: number }[] = [];
   575→   575→  if (sales !== 0) linesOut.push({ label: "Sales (settled to available)", amountMinor: sales });
   576→   576→  if (commission !== 0) linesOut.push({ label: "Platform commission", amountMinor: -commission });
   577→   577→  if (refunds !== 0) linesOut.push({ label: "Refunds", amountMinor: -refunds });
   578→   578→  if (adjustments !== 0) linesOut.push({ label: "Adjustments", amountMinor: adjustments });
   579→   579→  if (retained !== 0) linesOut.push({ label: "Retained in wallet (hold/cap — not paid out)", amountMinor: retained });
   580→   580→
   581→   581→  return toJson({
   582→   582→    sellerName,
   583→   583→    periodFrom: periodFrom.toISOString(),
   584→   584→    periodTo: periodTo.toISOString(),
   585→   585→    orderCount,
   586→   586→    reconciled: true,
   587→   587→    generatedAt: new Date().toISOString(),
   588→   588→    lines: linesOut,
   589→   589→  });
   590→   590→}
