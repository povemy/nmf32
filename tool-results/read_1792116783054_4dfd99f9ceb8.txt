   287→/** Total unread shopper messages — powers the "Messages" nav badge. */
   288→function useSellerChatUnread(active: SellerNav["view"]) {
   289→  const [unread, setUnread] = useState(0);
   290→  useEffect(() => {
   291→    let alive = true;
   292→    const load = () =>
   293→      apiGet<{ total: number }>("/api/store/chat/unread")
   294→        .then((d) => alive && setUnread(d.total ?? 0))
   295→        .catch(() => {});
   296→    load();
   297→    const t = setInterval(load, 15000);
   298→    return () => {
   299→      alive = false;
   300→      clearInterval(t);
   301→    };
   302→  }, [active]);
   303→  // instant re-sync when a thread is read inside the messages view (the view
   304→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   305→  useEffect(() => {
   306→    const sync = () =>
   307→      apiGet<{ total: number }>("/api/store/chat/unread")
   308→        .then((d) => setUnread(d.total ?? 0))
   309→        .catch(() => {});
   310→    window.addEventListener("shopmesh:chat-sync", sync);
   311→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   312→  }, []);
   313→  return unread;
   314→}

   456→function AuthScreen({ onAuthed }: { onAuthed: () => Promise<void> }) {
   457→  const setSurface = useApp((s) => s.setSurface);
   458→  const [loginEmail, setLoginEmail] = useState("");
   459→  const [loginPassword, setLoginPassword] = useState("");
   460→  const [busy, setBusy] = useState(false);
   461→  const [reg, setReg] = useState({ fullName: "", displayName: "", email: "", password: "", phone: "", storeName: "" });
   462→
   463→  const goMarket = () => setSurface("shopper");
   464→
   465→  const login = async () => {
   466→    if (!loginEmail || !loginPassword) {
   467→      toast.error("Enter your email and password");
   468→      return;
   469→    }
   470→    setBusy(true);
   471→    const ok = await runMutation(async () => {
   472→      const body = await apiSend<{ token?: string }>("POST", "/api/auth/login", {
   473→        email: loginEmail,
   474→        password: loginPassword,
   475→        portal: "seller",
   476→      });
   477→      setSessionToken(body?.token); // localStorage bridge for cookie-less contexts
   478→    }, "Welcome back!");
   479→    setBusy(false);
   480→    if (ok) await onAuthed();
   481→  };
   482→
   483→  const register = async () => {
   484→    if (!reg.fullName.trim() || !reg.email.includes("@") || reg.password.length < 8) {
   485→      toast.error("Name, valid email and a password of at least 8 characters are required");
   486→      return;
   487→    }
   488→    setBusy(true);
   489→    const ok = await runMutation(async () => {
   490→      const body = await apiSend<{ token?: string }>("POST", "/api/auth/register", reg);
   491→      setSessionToken(body?.token); // localStorage bridge for cookie-less contexts
   492→    }, "Store created — let's set it up!");
   493→    setBusy(false);
   494→    if (ok) await onAuthed();
   495→  };
   496→
   497→  return (
   498→    <div className="flex-1 flex flex-col min-h-0">
   499→      {/* top chrome — the portal is reachable from the buyer hub (“Start Selling” /
   500→          “Open store”), so there is always a visible way back to the marketplace */}
   501→      <header className="border-b bg-background/95 backdrop-blur px-4 py-2.5">
   502→        <div className="mx-auto flex w-full max-w-6xl items-center gap-3">
   503→          <button
   504→            type="button"
   505→            onClick={goMarket}
   506→            className="flex min-h-[44px] items-center gap-2.5"
   507→            aria-label="SHOPMESH — back to the marketplace"
   508→          >
   509→            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-primary text-primary-foreground">
   510→              <ShoppingBag className="h-4 w-4" />
   511→            </span>
