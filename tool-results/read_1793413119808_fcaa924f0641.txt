    50→type Json = { data?: Record<string, unknown> | unknown[]; error?: { code?: string; message?: string }; [k: string]: unknown };

    52→async function api(
    53→  method: "GET" | "POST",
    54→  path: string,
    55→  opts: { token?: string; cookie?: string; body?: unknown; idemKey?: string } = {}
    56→): Promise<{ status: number; json: Json }> {
    57→  const headers: Record<string, string> = { "Content-Type": "application/json" };
    58→  if (opts.token) headers.Authorization = `Bearer ${opts.token}`;
    59→  if (opts.cookie) headers.cookie = opts.cookie;
    60→  if (opts.idemKey) headers["Idempotency-Key"] = opts.idemKey;
    61→  try {
    62→    const res = await fetch(`${BASE}${path}`, {
    63→      method,
    64→      headers,
    65→      body: opts.body === undefined ? undefined : JSON.stringify(opts.body),
    66→    });
    67→    let json: Json = {};
    68→    try {
    69→      json = (await res.json()) as Json;
    70→    } catch {
    71→      /* empty body */
    72→    }
    73→    return { status: res.status, json };
    74→  } catch (e) {
    75→    return { status: 0, json: { error: { code: "FETCH_ERROR", message: String(e) } } };
    76→  }
    77→}
    78→
    79→const errCode = (j: Json) => (j.error as { code?: string } | undefined)?.code ?? "";
    80→const errMsg = (j: Json) => (j.error as { message?: string } | undefined)?.message ?? "";

   128→function signedWebhookBody(providerRef: string, eventType: string, amountMinor: number) {
   129→  const ts = Date.now();
   130→  const sig = createHmac("sha256", WEBHOOK_SECRET).update(`${providerRef}.${eventType}.${amountMinor}.${ts}`).digest("hex");
   131→  return { providerRef, eventType, amountMinor, ts, sig };
   132→}

   137→async function sectionA(): Promise<{ ref: string } | null> {
   138→  section("A — Payment webhook idempotency");
   139→  try {
   140→    const cookie = await loginShopper();
   141→    const variant = await db.productVariant.findFirst({
   142→      where: { product: { store: { slug: "bytebazaar" }, status: "active" }, stock: { gt: 0 } },
   143→      select: { id: true },
   144→    });
   145→    if (!variant) throw new Error("no buyable bytebazaar variant");
   146→    const ref = await createCheckout(cookie, variant.id);
   147→    console.log(`  checkout ref=${ref}`);
   148→
   149→    const t0 = await txnCount(); // BEFORE the first delivery
   150→    const l0 = await lineCount();
   151→
   152→    const p1 = await simulate(ref);
   153→    check("A1 first /simulate 200", p1.status === 200, `status=${p1.status}`);
   154→
   155→    const group = await db.paymentGroup.findUnique({
   156→      where: { ref },
   157→      include: { orders: { select: { orderNumber: true, status: true, totalMinor: true } } },
   158→    });
   159→    check("A2 order(s) paid after 1st delivery", !!group && group.orders.every((o) => o.status === "paid"), group?.orders.map((o) => `${o.orderNumber}:${o.status}`).join(","));
   160→    const amountMinor = group?.totalMinor ?? 0;
   161→
   162→    // Baseline AFTER the first (legitimate) delivery — the exactly-once test is
   163→    // that everything after it moves NOTHING.
   164→    const tBase = await txnCount();
   165→    const lBase = await lineCount();
   166→    const firstMove = tBase === t0 + 1 && lBase === l0 + 3; // single T1 = 1 txn / 3 lines
   167→    check("A2b first delivery posted exactly one T1 (1 txn / 3 lines)", firstMove, `txns ${t0}→${tBase}, lines ${l0}→${lBase}`);
   168→
   169→    const p2 = await simulate(ref);
   170→    check("A3 second /simulate 200 (replay-safe, response shape preserved)", p2.status === 200, `status=${p2.status}`);
   171→    // NOTE: /simulate intentionally does not surface the processor's duplicate
   172→    // flag (its response shape feeds the checkout UI) — the duplicate:true
   173→    // assertion is made on the raw webhook route below, which returns it.
   174→
   175→    const t1 = await txnCount();
   176→    const l1 = await lineCount();
   177→    check("A4 ledger unchanged after duplicate /simulate", t1 === tBase && l1 === lBase, `txns ${tBase}→${t1}, lines ${lBase}→${l1}`);
   178→
   179→    // Exact duplicate webhook DELIVERY (same signed event, resent) → duplicate:true
   180→    const replay = await api("POST", "/api/webhooks/payments/mock", {
   181→      body: signedWebhookBody(ref, "payment.succeeded", amountMinor),
   182→    });
   183→    const dup = (replay.json.data as { duplicate?: boolean } | undefined)?.duplicate;
   184→    check("A5 duplicate webhook delivery → duplicate=true", replay.status === 200 && dup === true, `status=${replay.status} duplicate=${String(dup)}`);
   185→
   186→    const t2 = await txnCount();
   187→    const l2 = await lineCount();
   188→    check("A6 ledger unchanged after duplicate webhook delivery", t2 === tBase && l2 === lBase, `txns ${tBase}→${t2}, lines ${lBase}→${l2}`);
   189→    return { ref };
   190→  } catch (e) {
   191→    check("A section completed", false, String(e));
   192→    return null;
   193→  }
   194→}

   213→      const delta = target - wallet.availableMinor;

   215→        token: adminToken,
   216→        body: {
   217→          amountMinor: delta,

   220→        },

   223→        "B0 wallet top-up via audited admin adjust (T6)",

   225→        `credited ${delta} minor → available ${wallet.availableMinor}→${((adj.json.data as { after?: { availableMinor?: number } } | undefined)?.after?.availableMinor ?? "?")}`
   226→      );

   247→
   248→    const sim1 = await api("POST", "/api/webhooks/payments/payout-simulate", { token: adminToken, body: { referenceNo: payout.referenceNo } });

   252→    const t4 = await dedupCount(`payout_confirmed:payout:${payout.id}`);

   257→      "B6 payout.paid notifications exactly 2 (email+inapp)",

   259→      `rows=${notifs.length} channels=[${notifs.map((n) => n.channel).join(",")}]`
   260→    );
   261→
   262→    const sim2 = await api("POST", "/api/webhooks/payments/payout-simulate", { token: adminToken, body: { referenceNo: payout.referenceNo } });
   263→    const sim2Data = (sim2.json.data ?? {}) as { payoutStatus?: string; payoutConflict?: boolean; duplicate?: boolean };
   264→    const replayNoOp = sim2.status === 200 && (sim2Data.payoutConflict === true || sim2Data.duplicate === true);
   265→    check("B7 payout-simulate replay → no-op (payoutConflict/duplicate)", replayNoOp, `status=${sim2.status} payoutConflict=${String(sim2Data.payoutConflict)} duplicate=${String(sim2Data.duplicate)}`);
   266→
   267→    const t4b = await dedupCount(`payout_confirmed:payout:${payout.id}`);
   268→    const notifsB = await db.notificationLog.count({ where: { event: "payout.paid", dedupKey: `payout.paid:${payout.id}` } });
   269→    check("B8 replay moved nothing (T4 + notifications unchanged)", t4b === 1 && notifsB === 2, `T4=${t4b}, notifications=${notifsB}`);
   270→
   271→    return { payoutId: payout.id, referenceNo: payout.referenceNo };
   272→  } catch (e) {
   273→    check("B section completed", false, String(e));
   274→    return null;
   275→  }

   427→async function sectionD(adminToken: string, payoutId: string | null): Promise<void> {
   428→  section("D — Reconciliation (sync + idempotent sweep)");

   430→    if (payoutId) {
   431→      const sync = await api("POST", `/api/admin/payouts/${payoutId}/sync`, { token: adminToken });
   432→      const s = (sync.json.data ?? {}) as { expectedTxns?: { T3?: boolean; T4?: boolean }; cacheMatches?: boolean; status?: string };
   433→      check(
   434→        "D1 payout sync truthful (T3:true, T4:true, cacheMatches:true)",
   435→        sync.status === 200 && s.expectedTxns?.T3 === true && s.expectedTxns?.T4 === true && s.cacheMatches === true,
   436→        `status=${sync.status} expectedTxns=${JSON.stringify(s.expectedTxns)} cacheMatches=${String(s.cacheMatches)} payoutStatus=${s.status}`
   437→      );
   438→    } else {
   439→      check("D1 payout sync (skipped — Section B failed)", false, "no paid payout id");
   440→    }
   441→
   442→    type Sweep = { status: number; json: Json };
   443→    const sweep = (): Promise<Sweep> => api("POST", "/api/admin/payouts/sweep", { token: adminToken });
   444→    const stuckIds = (r: Sweep) => ((r.json.data as { stuck?: Array<{ payoutId: string; alerted?: boolean }> } | undefined)?.stuck ?? []);
   445→    const invOk = (r: Sweep) => ((r.json.data as { invariants?: { ok?: boolean } } | undefined)?.invariants?.ok === true);
   446→
   447→    const s1 = await sweep();
   448→    const inv1 = invOk(s1);
   449→    const ids1 = new Set(stuckIds(s1).map((x) => x.payoutId));
   450→    const new1 = stuckIds(s1).filter((x) => x.alerted).length;
   451→    check("D2 sweep run 1 invariants PASS", s1.status === 200 && inv1, `status=${s1.status} stuckScanned=${ids1.size} newlyAlerted=${new1}`);
   452→
   453→    const s2 = await sweep();
   454→    const ids2 = stuckIds(s2).map((x) => x.payoutId);
   455→    const new2 = stuckIds(s2).filter((x) => x.alerted).length;
   456→    const subset = ids2.every((id) => ids1.has(id));
   457→    check("D3 sweep run 2 idempotent (no NEW stuck alerts, invariants PASS)", s2.status === 200 && invOk(s2) && new2 === 0 && subset, `newlyAlerted=${new2} stuckSubset=${subset} invOk=${String(invOk(s2))}`);
   458→  } catch (e) {
   459→    check("D section completed", false, String(e));
   460→  }
   461→}
