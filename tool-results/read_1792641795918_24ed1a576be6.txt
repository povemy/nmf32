   100→// ---------------------------------------------------------------------------
   101→// EDD (estimated delivery) helpers — Task 14-b. The checkout quote carries a
   102→// working-days label ("2-4 working days"); the shipment record carries
   103→// concrete etaFrom/etaTo dates. Both surface as the same small chip:
   104→//   [ Estimated delivery Oct 18–21 ]
   105→// ---------------------------------------------------------------------------
   106→
   107→function addWorkingDays(from: Date, days: number): Date {
   108→  const d = new Date(from);
   109→  let added = 0;
   110→  while (added < days) {
   111→    d.setDate(d.getDate() + 1);
   112→    const wd = d.getDay();
   113→    if (wd !== 0 && wd !== 6) added += 1; // skip Sat/Sun
   114→  }
   115→  return d;
   116→}

   118→function eddRangeLabel(from: Date, to: Date): string {
   119→  if (from.getFullYear() === to.getFullYear() && from.getMonth() === to.getMonth()) {
   120→    return `${from.toLocaleDateString("en-MY", { month: "short" })} ${from.getDate()}–${to.getDate()}`;
   121→  }
   122→  return `${from.toLocaleDateString("en-MY", { day: "numeric", month: "short" })} – ${to.toLocaleDateString("en-MY", { day: "numeric", month: "short" })}`;
   123→}

   125→/** Working-days eta label from the quote ("2-4 working days") → date window. */
   126→function eddFromEtaLabel(eta: string | null | undefined): { from: Date; to: Date } | null {
   127→  if (!eta) return null;
   128→  const m = /(\d+)\s*(?:-|–)\s*(\d+)\s*working days?/i.exec(eta) ?? /^(\d+)\s*working days?$/i.exec(eta);
   129→  if (!m) return null;
   130→  const a = Number(m[1]);
   131→  const b = Number(m[2] ?? m[1]);
   132→  if (!Number.isFinite(a) || !Number.isFinite(b) || a < 1 || b < 1) return null;
   133→  const now = new Date();
   134→  return { from: addWorkingDays(now, a), to: addWorkingDays(now, Math.max(a, b)) };
   135→}

   137→/** Concrete ISO dates from the shipment record → label (null when absent). */
   138→function eddFromShipment(etaFrom: string | null | undefined, etaTo: string | null | undefined): string | null {
   139→  if (!etaFrom || !etaTo) return null;
   140→  const a = new Date(etaFrom);
   141→  const b = new Date(etaTo);
   142→  if (Number.isNaN(a.getTime()) || Number.isNaN(b.getTime())) return null;
   143→  return eddRangeLabel(a, b);
   144→}

   188→function makeIdemKey(): string {
   189→  try {
   190→    return crypto.randomUUID();
   191→  } catch {
   192→    return `idem_${Date.now()}_${Math.random().toString(36).slice(2)}`;
   193→  }

   196→function loadDraft(): DraftT {
   197→  const savedContact: Record<string, string> = (() => {
   198→    if (typeof window === "undefined") return {};
   199→    try {
   200→      return JSON.parse(window.localStorage.getItem(CONTACT_KEY) ?? "{}") as Record<string, string>;
   201→    } catch {
   202→      return {};
   203→    }
   204→  })();
   205→  const base: DraftT = {
   206→    phase: 1,
   207→    name: savedContact.name ?? "",
   208→    email: savedContact.email ?? "",
   209→    phone: savedContact.phone ?? "",
   210→    addressId: null,
   211→    shippingRateId: "standard",
   212→    provider: "fpx_toyyibpay",
   213→    couponCode: "",
   214→    note: "",
   215→    agreed: false,
   216→    idemKey: makeIdemKey(),
   217→  };
   218→  if (typeof window === "undefined") return base;
   219→  try {
   220→    const raw = JSON.parse(window.sessionStorage.getItem(DRAFT_KEY) ?? "{}") as Partial<DraftT>;
   221→    // Cart-drawer order note (Seller "Cart Experience" feature) pre-fills the
   222→    // checkout note when the shopper hasn't typed one here yet.
   223→    let drawerNote = "";
   224→    try {
   225→      drawerNote = window.localStorage.getItem("sm_cart_note") ?? "";
   226→    } catch {
   227→      /* ignore */
   228→    }
   229→    return {
   230→      ...base,
   231→      ...raw,
   232→      note: raw.note || drawerNote || "",
   233→      phase: raw.phase === 2 || raw.phase === 3 ? raw.phase : 1,
   234→      idemKey: raw.idemKey || base.idemKey, // stable across retries → idempotent resubmits
   235→    };
   236→  } catch {
   237→    return base;
   238→  }
   239→}
