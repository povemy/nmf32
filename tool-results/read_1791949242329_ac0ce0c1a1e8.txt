    79→/** Total unread seller messages — powers the "My messages" header badge.
    80→ *  Polls lightly + refreshes when the messages view is opened/left. */
    81→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    82→  const [unread, setUnread] = useState(0);
    83→  useEffect(() => {
    84→    if (!shopperSignedIn) {
    85→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    86→      setUnread(0);
    87→      return;
    88→    }
    89→    let alive = true;
    90→    const load = () =>
    91→      fetchJson<{ total: number }>("/api/chat/unread")
    92→        .then((d) => alive && setUnread(d.total ?? 0))
    93→        .catch(() => {});
    94→    load();
    95→    const t = setInterval(load, 15000);
    96→    return () => {
    97→      alive = false;
    98→      clearInterval(t);
    99→    };
   100→  }, [shopperSignedIn]);
   101→  // instant re-sync when a thread is read inside the messages view (the view
   102→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   103→  useEffect(() => {
   104→    if (!shopperSignedIn) return;
   105→    const sync = () =>
   106→      fetchJson<{ total: number }>("/api/chat/unread")
   107→        .then((d) => setUnread(d.total ?? 0))
   108→        .catch(() => {});
   109→    window.addEventListener("shopmesh:chat-sync", sync);
   110→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   111→  }, [shopperSignedIn]);
   112→  // opening (or leaving) the messages view should immediately re-sync the badge
   113→  useEffect(() => {
   114→    if (!shopperSignedIn) return;
   115→    const t = setTimeout(
   116→      () =>
   117→        fetchJson<{ total: number }>("/api/chat/unread")
   118→          .then((d) => setUnread(d.total ?? 0))
   119→          .catch(() => {}),
   120→      viewName === "messages" ? 600 : 0
   121→    );
   122→    return () => clearTimeout(t);
   123→  }, [shopperSignedIn, viewName]);
   124→  return unread;
   125→}

   127→export default function ShopperShell() {
   128→  const setSurface = useApp((s) => s.setSurface);
   129→  const activeStoreSlug = useApp((s) => s.activeStoreSlug);
   130→  const setActiveStoreSlug = useApp((s) => s.setActiveStoreSlug);
   131→
   132→  const [view, setView] = useState<ShopperView>({ name: "market" });
   133→  const [ready, setReady] = useState(false);
   134→  const [cartOpen, setCartOpen] = useState(false);
   135→  const [headerSearch, setHeaderSearch] = useState("");
   136→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   137→  // intentionally NOT part of ShopperView — a synthetic render override.
   138→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   139→  const restored = useRef(false);
   140→
   141→  // ---- mount: activeStoreSlug override, else restore last view --------------
   142→  useEffect(() => {
   143→    if (restored.current) return;
   144→    restored.current = true;
   145→    // claim deep-link wins over everything
   146→    try {
   147→      const params = new URLSearchParams(window.location.search);
   148→      const cj = params.get("claim_job");
   149→      const ct = params.get("claim_token");
   150→      if (cj && ct) {
   151→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   152→        setClaimLink({ jobId: cj, token: ct });
   153→        setReady(true);
   154→        return;
   155→      }
   156→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   157→      const sf = params.get("sf");
   158→      if (sf) {
   159→        const sp = params.get("sp");
   160→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   161→        setReady(true);
   162→        return;
   163→      }
   164→    } catch {
   165→      /* ignore */
   166→    }
   167→    if (activeStoreSlug) {
   168→      setView({ name: "storefront", slug: activeStoreSlug });
   169→      setActiveStoreSlug(null);
   170→    } else {
   171→      try {
   172→        const raw = window.localStorage.getItem(VIEW_KEY);
   173→        if (raw) {
   174→          const v = JSON.parse(raw) as ShopperView;
   175→          if (v && typeof v.name === "string") setView(v);
   176→        }
   177→      } catch {
   178→        /* ignore */
   179→      }
   180→    }
   181→    setReady(true);
   182→  }, [activeStoreSlug, setActiveStoreSlug]);
   183→
   184→  // ---- nav helper: persist + scroll to top ----------------------------------
   185→  const nav = useCallback((v: ShopperView) => {
   186→    setView(v);
   187→    try {
   188→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   189→    } catch {
   190→      /* ignore */
   191→    }
   192→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   193→  }, []);
   194→
   195→  // refresh storefront/store-page when slug changes externally
   196→  useEffect(() => {
   197→    if (!ready) return;
   198→    if (activeStoreSlug) {
   199→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   200→      setView({ name: "storefront", slug: activeStoreSlug });
   201→      setActiveStoreSlug(null);
   202→    }
   203→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   204→
   205→  const ctx = useMemo<ShopperCtx>(() => ({ nav, openCart: () => setCartOpen(true) }), [nav]);
   206→
   207→  const showChrome = !isStoreChrome(view);
   208→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   209→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   210→
   211→  return (
   212→    <div className="flex-1 flex flex-col min-h-screen">
   213→      <Toaster position="top-center" richColors />
   214→      <ShopperAuthDialog />
   215→
   216→      {showChrome ? (
   217→        <MarketHeader
   218→          view={view}
   219→          nav={nav}
   220→          openCart={() => setCartOpen(true)}
   221→          search={headerSearch}
   222→          setSearch={setHeaderSearch}
   223→        />
   224→      ) : null}
   225→
   226→      <AnimatePresence mode="wait">
   227→        <motion.main
   228→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : view.name === "me" ? view.tab ?? "" : ""}`}
   229→          initial={{ opacity: 0, y: 8 }}
   230→          animate={{ opacity: 1, y: 0 }}
   231→          exit={{ opacity: 0, y: -8 }}
   232→          transition={{ duration: 0.18, ease: "easeOut" }}
   233→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   234→        >
   235→          {claimLink ? null : (
   236→            <>
   237→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   238→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   239→              {view.name === "category" ? (
   240→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   241→              ) : null}
   242→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   243→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   244→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   245→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   246→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   247→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   248→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   249→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   250→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   251→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   252→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}
   253→            </>
   254→          )}
   255→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}
   256→        </motion.main>
   257→      </AnimatePresence>
   258→
   259→      {showBottomNav ? (
   260→        <BottomNav
   261→          view={view}
   262→          nav={nav}
   263→          openCart={() => setCartOpen(true)}
   264→        />
   265→      ) : null}
   266→
   267→      {showChrome ? (
   268→        <ShopperFooter
   269→          onSeller={() => setSurface("seller")}
   270→          onLookup={() => nav({ name: "lookup" })}
   271→          onMarket={() => nav({ name: "market" })}
   272→          onGenerate={() => nav({ name: "generate" })}
   273→        />
   274→      ) : null}
   275→
   276→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   277→    </div>
   278→  );
   279→}

   285→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   286→  const me = useApp((s) => s.me);
   287→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   288→  const signedIn = Boolean(me.loaded && me.shopper);
   289→  const { count } = useCart();
   290→  const unreadChats = useChatUnread(signedIn, view.name);
   291→
   292→  const items = [
   293→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   294→    {
   295→      key: "orders",
   296→      label: "Orders",
   297→      icon: Package,
   298→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   299→      badge: 0,
   300→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   301→    },
   302→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   303→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   304→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   305→  ];
   306→
   307→  return (
   308→    <nav
   309→      aria-label="Bottom navigation"
   310→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   311→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   312→    >
   313→      <div className="mx-auto grid max-w-lg grid-cols-5">
   314→        {items.map((it) => {
   315→          const Icon = it.icon;
   316→          return (
   317→            <button
   318→              key={it.key}
   319→              type="button"
   320→              onClick={it.onClick}
   321→              aria-label={it.label}
   322→              aria-current={it.active ? "page" : undefined}
   323→              className={cn(
   324→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   325→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   326→              )}
   327→            >
   328→              <span className="relative">
   329→                <Icon className="h-5 w-5" aria-hidden="true" />
   330→                {it.badge > 0 ? (
   331→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   332→                    {it.badge > 9 ? "9+" : it.badge}
   333→                  </span>
   334→                ) : null}
   335→              </span>
   336→              {it.label}
   337→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   338→            </button>
   339→          );
   340→        })}
   341→      </div>
   342→    </nav>
   343→  );
   344→}

   794→function timeAgoShort(iso: string): string {
   795→  const diff = Date.now() - new Date(iso).getTime();
   796→  const m = Math.floor(diff / 60000);
   797→  if (m < 1) return "now";
   798→  if (m < 60) return `${m}m`;
   799→  const h = Math.floor(m / 60);
   800→  if (h < 24) return `${h}h`;
   801→  const d = Math.floor(h / 24);
   802→  if (d < 7) return `${d}d`;
   803→  return new Date(iso).toLocaleDateString(undefined, { day: "numeric", month: "short" });
   804→}

   806→function MessagesView({ ctx, initialThreadId }: { ctx: ShopperCtx; initialThreadId?: string }) {
   807→  const me = useApp((s) => s.me);
   808→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   809→  const shopper = me.loaded ? me.shopper : null;
   810→
   811→  const [threads, setThreads] = useState<ThreadRow[] | null>(null);
   812→  const [activeId, setActiveId] = useState<string | null>(initialThreadId ?? null);
   813→  const [messages, setMessages] = useState<ChatMsg[] | null>(null);
   814→  const [activeThread, setActiveThread] = useState<{ storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null } | null>(null);
   815→  const [context, setContext] = useState<ThreadContextCard | null>(null);
   816→  const [draft, setDraft] = useState("");
   817→  const [sending, setSending] = useState(false);
   818→  const socketRef = useRef<Socket | null>(null);
   819→  const scrollRef = useRef<HTMLDivElement>(null);
   820→
   821→  const loadThreads = useCallback(async () => {
   822→    try {
   823→      const rows = await fetchJson<ThreadRow[]>("/api/chat/threads");
   824→      setThreads(rows);
   825→      return rows;
   826→    } catch {
   827→      setThreads([]);
   828→      return [];
   829→    }
   830→  }, []);
   831→
   832→  const loadMessages = useCallback(async (threadId: string) => {
   833→    try {
   834→      const d = await fetchJson<{
   835→        thread: { storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null };
   836→        context: ThreadContextCard | null;
   837→        messages: ChatMsg[];
   838→      }>(`/api/chat/threads/${threadId}`);
   839→      setMessages(d.messages);
   840→      setActiveThread(d.thread);
   841→      setContext(d.context ?? null);
   842→    } catch {
   843→      setMessages([]);
   844→      setContext(null);
   845→    }
   846→  }, []);
   847→
   848→  // initial load
   849→  useEffect(() => {
   850→    if (!shopper) return;
   851→    let alive = true;
   852→    loadThreads().then(() => {
   853→      if (!alive) return;
   854→      // Only auto-open when resuming an intent (contact-seller deep link).
   855→      // Plain entry shows the list first so unread badges stay visible —
   856→      // auto-reading the newest thread would instantly consume the badge.
   857→      if (initialThreadId) setActiveId(initialThreadId);
   858→    });
   859→    return () => {
   860→      alive = false;
   861→    };
   862→  }, [shopper, initialThreadId, loadThreads]);
   863→
   864→  // load conversation when thread changes
   865→  useEffect(() => {
   866→    if (!activeId) return;
   867→    setMessages(null);
   868→    // the detail GET marks the thread read server-side — refresh the list so
   869→    // the unread pill on the row (and the header badge) drops immediately
   870→    loadMessages(activeId).then(() => {
   871→      loadThreads();
   872→      window.dispatchEvent(new Event("shopmesh:chat-sync"));
   873→    });
   874→  }, [activeId, loadMessages, loadThreads]);
   875→
   876→  // live updates via the socket.io chat hub
   877→  useEffect(() => {
   878→    if (!shopper) return;
   879→    const socket = io("/?XTransformPort=3003", {
   880→      transports: ["websocket", "polling"],
   881→      reconnection: true,
   882→      reconnectionAttempts: 5,
   883→      timeout: 10000,
   884→    });
   885→    socketRef.current = socket;
   886→    socket.on("connect", () => {
   887→      // join all thread rooms so the sidebar stays fresh too
   888→      const rooms = (threads ?? []).map((t) => `thread:${t.id}`);
   889→      socket.emit("join", { rooms });
   890→    });
   891→    socket.on("chat:message", (msg: ChatMsg & { threadId?: string }) => {
   892→      if (msg.threadId && msg.threadId === activeId) {
   893→        setMessages((prev) => (prev && !prev.some((m) => m.id === msg.id) ? [...prev, msg] : prev));
   894→        if (msg.senderType === "seller") {
   895→          // message arrived while the shopper is looking at this conversation —
   896→          // clear the badge straight away instead of waiting for the poller
   897→          void postJson(`/api/chat/threads/${activeId}/read`, {}).catch(() => {});
   898→        }
   899→      }
   900→      loadThreads();
   901→    });
   902→    return () => {
   903→      socket.disconnect();
   904→      socketRef.current = null;
   905→    };
   906→  }, [shopper, threads, activeId, loadThreads]);
   907→
   908→  // auto-scroll to latest
   909→  useEffect(() => {
   910→    const el = scrollRef.current;

   922→      loadThreads();

   924→    } catch (e) {
   925→      toast.error(e instanceof Error ? e.message : "Could not send the message");

   927→      setSending(false);
   928→    }

   934→        <EmptyState
   935→          icon={MessagesSquare}
   936→          title="Sign in to see your messages"
   937→          hint="Create a free SHOPMESH account to chat with sellers about products, orders, deliveries and returns."
   938→          action={
   939→            <Button onClick={() => openShopperAuth(() => ctx.nav({ name: "messages" }))} className="min-h-[44px] bg-brand-gradient-deep text-white hover:brightness-110 gap-2">
   940→              <LogIn className="h-4 w-4" /> Sign in or create account

   952→          <MessagesSquare className="h-6 w-6 text-teal-700" /> Messages
   953→        </h1>
   954→        <p className="text-sm text-muted-foreground">Chat with sellers about your orders — replies arrive in real time.</p>
   955→      </div>
   956→
   957→      <div className="grid gap-4 md:grid-cols-[300px_1fr]">
   958→        {/* thread list */}
