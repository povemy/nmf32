     2→// AFFILIATE HUB & ONE-LEVEL SPONSOR REFERRAL SYSTEM — core domain library.
     3→// docs/affiliate/*.md are the authoritative design docs.
     4→//
     5→// COMMERCIAL MODEL (fixed, spec §1/§8):
     6→//   • Seller-funded commission pool per eligible OrderItem: P = B × R (bps,
     7→//     integer sen). The pool is carved OUT of seller proceeds at checkout
     8→//     (sellerNet already subtracts it — see computeOrderCommissions).
     9→//   • Split of the POOL: S = floor(P × 10 / 100) to the direct affiliate's
    10→//     ONE-LEVEL sponsor; D = P − S to the direct affiliate. No sponsor ⇒
    11→//     D = P, S = 0. D + S = P ALWAYS (never exceeds the pool).
    12→//   • Attribution travels UP exactly one generation. No recruitment bonuses,
    13→//     no click rewards, no registration rewards — liabilities only ever
    14→//     originate from genuine eligible paid sales.
    15→//
    16→// INTEGRITY CONTRACTS:
    17→//   • Pool snapshots freeze on OrderItem at checkout (commission contract).
    18→//   • AffiliateCommission rows post EXACTLY ONCE per order item at payment
    19→//     confirmation (unique orderItemId; duplicate webhooks are no-ops).
    20→//   • The 90/10 SPLIT is computed at POSTING time from the then-current
    21→//     sponsor-override flag + sponsor eligibility, then never rewritten —
    22→//     so the override can be disabled without corrupting existing records.
    23→//   • Every money mutation rides the immutable ledger (lib/ledger.ts).

    85→function clampInt(v: unknown, min: number, max: number, fallback: number): number {
    86→  const n = typeof v === "number" ? v : Number(v);
    87→  if (!Number.isFinite(n)) return fallback;
    88→  return Math.min(max, Math.max(min, Math.round(n)));
    89→}

   111→/** Effective seller affiliate rate (bps) for a product, honouring every gate.
   112→ *  0 ⇒ not enrolled (no commission). Product override ?? store default,
   113→ *  clamped to the platform max; programme master switch must be on. */
   114→export function effectiveAffiliateRateBps(
   115→  s: AffiliateSettings,
   116→  store: { affiliateOptIn: boolean; affiliateDefaultRateBps: number },
   117→  product: { affiliateEnabled: boolean; affiliateRateBps: number | null }
   118→): number {
   119→  if (!s.enabled || !store.affiliateOptIn || !product.affiliateEnabled) return 0;
   120→  const rate = product.affiliateRateBps ?? store.affiliateDefaultRateBps;
   121→  return Math.max(0, Math.min(rate, s.maxRateBps));
   122→}

   124→// ---------------------------------------------------------------------------
   125→// Commission math — the spec's exact formulas (§8).
   126→// ---------------------------------------------------------------------------
   127→/** Pool P = floor(B × R / 10^4) in integer sen.
   128→ *  POST-IMPLEMENTATION AUDIT (P4): the ONE canonical rounding policy for the
   129→ *  entire affiliate money path is FLOOR (commission-definitions.md §9 is
   130→ *  normative: "Where this document conflicts with prose elsewhere, this
   131→ *  document governs"). Previously half-up (platform percent convention) —
   132→ *  a one-sen drift vs the normative doc. Same floor policy is used at the
   133→ *  checkout snapshot, payment posting, refund reversal, reconciliation and
   134→ *  every report. The platform commission's own percent part keeps its
   135→ *  historical half-up convention (separate ledger line, unchanged). */
   136→export function affiliatePoolMinor(baseMinor: number, rateBps: number): number {
   137→  if (baseMinor <= 0 || rateBps <= 0) return 0;
   138→  return Math.floor((baseMinor * rateBps) / 10000);
   139→}
   140→
   141→/** 90/10 split of the POOL: S = floor(P × 10 / 100); D = P − S (D + S = P). */
   142→export function splitAffiliatePool(poolMinor: number): { directMinor: number; sponsorMinor: number } {
   143→  if (poolMinor <= 0) return { directMinor: 0, sponsorMinor: 0 };
   144→  const sponsorMinor = Math.floor((poolMinor * 10) / 100);
   145→  return { directMinor: poolMinor - sponsorMinor, sponsorMinor };
   146→}

   163→export async function requireAffiliate(): Promise<AffiliateCtx> {
   164→  const sess = await getSession();

   166→    throw new ApiError(401, "UNAUTHORIZED", "Affiliate session required");
   167→  }

   188→function slugifyAffiliate(name: string): string {
   189→  const base = name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 24);
   190→  return base || "affiliate";
   191→}

   193→async function uniqueSlug(base: string): Promise<string> {
   194→  let candidate = base;
   195→  for (let i = 0; i < 20; i++) {
   196→    const clash = await db.affiliateProfile.findUnique({ where: { slug: candidate }, select: { id: true } });
   197→    if (!clash) return candidate;
   198→    candidate = `${base}-${randomToken(2)}`;
   199→  }
   200→  return `aff-${randomToken(4)}`;
   201→}

   203→/** Read the FIRST-TOUCH referral cookie and validate it against the touch
   204→ *  table (unexpired, unconsumed). Returns the sponsor profile or null. */
   205→export async function resolveReferralCookie(): Promise<{ affiliateId: string; displayName: string; code: string } | null> {
   206→  try {
   207→    const store = await cookies();
   208→    const raw = store.get(REFERRAL_COOKIE)?.value;
   209→    if (!raw) return null;
   210→    const parsed = parseJson<{ code?: string }>(raw, {});
   211→    if (!parsed?.code) return null;
   212→    const touch = await db.affiliateReferralTouch.findFirst({
   213→      where: { code: parsed.code, consumedAt: null, expiresAt: { gt: new Date() } },
   214→      orderBy: { touchedAt: "desc" },
   215→      select: { affiliateId: true, code: true },
   216→    });
   217→    const profile = touch
   218→      ? await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId }, select: { id: true, displayName: true, code: true, status: true } })
   219→      : null;
   220→    if (!profile || profile.status !== "active") return null;
   221→    return { affiliateId: profile.id, displayName: profile.displayName, code: profile.code };
   222→  } catch {
   223→    return null;
   224→  }
   225→}

   250→/** Activate the current shopper as an affiliate. Idempotent (already-active →
   251→ *  error with the existing profile). Sponsor precedence (spec §5):
   252→ *  1. Valid first-touch referral cookie (30-day window, unconsumed);
   253→ *  2. Else a one-time manual code entry — confirmed explicitly;
   254→ *  3. Else no sponsor (recorded permanently as none).
   255→ *  The relationship locks forever after activation: no click, seller or admin
   256→ *  tool can casually change it (admin corrections are separately audited). */
   257→export async function activateAffiliate(input: ActivationInput) {
   258→  const settings = await getAffiliateSettings();
   259→  if (!settings.enabled) {
   260→    throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   261→  }
   262→
   263→  const existing = await db.affiliateProfile.findUnique({ where: { userId: input.userId } });
   264→  if (existing) {
   265→    throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   266→  }
   267→
   268→  // ---- Sponsor resolution (server-side only) -----------------------------
   269→  let sponsor: { id: string; displayName: string } | null = null;
   270→  let sponsorSource: string | null = null;
   271→
   272→  const touch = await resolveReferralCookie();
   273→  if (touch) {
   274→    const profile = await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId } });
   275→    if (profile && profile.status === "active" && profile.userId !== input.userId) {
   276→      sponsor = { id: profile.id, displayName: profile.displayName };
   277→      sponsorSource = "referral_touch";
   278→    }
   279→  }
   280→  if (!sponsor && input.sponsorCode) {
   281→    const candidate = await findAffiliateByCode(input.sponsorCode);
   282→    if (!candidate) throw new ApiError(400, "INVALID_SPONSOR_CODE", "That referral code was not found");
   283→    if (candidate.userId === input.userId) {
   284→      throw new ApiError(400, "SELF_REFERRAL", "You cannot refer yourself");
   285→    }
   286→    if (candidate.status !== "active") {
   287→      throw new ApiError(400, "SPONSOR_INACTIVE", "That sponsor is not currently eligible");
   288→    }
   289→    if (!input.sponsorConfirmed) {
   290→      throw new ApiError(400, "SPONSOR_CONFIRMATION_REQUIRED", "Confirm the sponsor relationship before activating");
   291→    }
   292→    sponsor = { id: candidate.id, displayName: candidate.displayName };
   293→    sponsorSource = "code_entry";
   294→  }
   295→
   296→  const code = randomToken(5); // 10-hex-char referral code, unpredictable
   297→  const slug = await uniqueSlug(slugifyAffiliate(input.slug || input.displayName || input.fullName));
   298→
   299→  const profile = await db.$transaction(async (tx) => {
   300→    // Re-check inside the tx (concurrent double-submit).
   301→    const dupe = await tx.affiliateProfile.findUnique({ where: { userId: input.userId } });
   302→    if (dupe) throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   303→
   304→    const created = await tx.affiliateProfile.create({
   305→      data: {
   306→        userId: input.userId,
   307→        code,
   308→        slug,
   309→        displayName: (input.displayName || input.fullName).slice(0, 60),
   310→        bio: input.bio?.slice(0, 500) ?? null,
   311→        status: "active",
   312→        termsVersion: AFFILIATE_TERMS_VERSION,
   313→        termsAcceptedAt: new Date(),
   314→        activatedAt: new Date(),
   315→        sponsorId: sponsor?.id ?? null,
   316→        sponsorAssignedAt: sponsor ? new Date() : null,
   317→        sponsorSource,
   318→      },
   319→    });
   320→
   321→    // Consume the referral touch exactly once (relationship now recorded).
   322→    if (sponsor && sponsorSource === "referral_touch" && touch) {
   323→      await tx.affiliateReferralTouch.updateMany({
   324→        where: { code: touch.code, consumedAt: null },
   325→        data: { consumedAt: new Date(), consumedUserId: input.userId },
   326→      });
   327→    }
   328→    return created;
   329→  });
   330→
   331→  return { profile, sponsor, sponsorSource };
   332→}

   334→/** Admin-only sponsor correction — audited, with documented reason. Cycle
   335→ *  guard: with one-level relationships a 2-cycle (A↔B) is the only possible
   336→ *  loop; rejecting `newSponsor.sponsorId === affiliateId` kills it. */
   337→export async function correctSponsor(input: {
   338→  affiliateId: string;
   339→  newSponsorId: string | null;
   340→  reason: string;
   341→  adminId: string;
   342→}) {
   343→  if (!input.reason || input.reason.trim().length < 10) {
   344→    throw new ApiError(400, "VALIDATION", "A documented reason (≥10 chars) is required for sponsor corrections");
   345→  }
   346→  const affiliate = await db.affiliateProfile.findUnique({ where: { id: input.affiliateId } });
   347→  if (!affiliate) throw new ApiError(404, "NOT_FOUND", "Affiliate not found");
   348→  if (input.newSponsorId === affiliate.id) {
   349→    throw new ApiError(400, "SELF_REFERRAL", "An affiliate cannot sponsor themselves");
   350→  }
   351→  if (input.newSponsorId) {
   352→    const sponsor = await db.affiliateProfile.findUnique({ where: { id: input.newSponsorId } });
   353→    if (!sponsor) throw new ApiError(404, "NOT_FOUND", "New sponsor not found");
   354→    if (sponsor.sponsorId === affiliate.id) {
   355→      throw new ApiError(400, "SPONSOR_CYCLE", "This correction would create a sponsor cycle");
   356→    }
   357→  }
   358→  return db.$transaction(async (tx) => {
   359→    const updated = await tx.affiliateProfile.update({
   360→      where: { id: input.affiliateId },
   361→      data: {
   362→        sponsorId: input.newSponsorId,
   363→        sponsorAssignedAt: input.newSponsorId ? new Date() : null,
   364→        sponsorSource: "admin_correction",
   365→      },
   366→    });
   367→    await tx.affiliateSponsorAudit.create({
   368→      data: {
   369→        affiliateId: input.affiliateId,
   370→        fromSponsorId: affiliate.sponsorId,
   371→        toSponsorId: input.newSponsorId,
   372→        reason: input.reason.trim(),
   373→        actorType: "admin",
   374→        actorId: input.adminId,
   375→      },
   376→    });
   377→    return updated;
   378→  });
   379→}

   387→/** Post AffiliateCommission rows for one paid order. Idempotent: unique
   388→ *  orderItemId + find-first guard. The 90/10 DECISION comes from the
   389→ *  checkout-time policy snapshot (OrderItem.affiliateSponsorEnabled); the
   390→ *  sponsor's status at posting decides pending vs held only. */
   391→export async function postAffiliateCommissionsForOrder(
   392→  tx: Tx,
   393→  order: { id: string; orderNumber: string; storeId: string; items: { id: string; productId?: string | null; affiliateDirectId: string | null; affiliateSponsorId: string | null; affiliateLinkId: string | null; affiliateRateBps: number | null; affiliatePoolMinor: number; affiliateSponsorEnabled?: boolean | null; breakdownJson: string }[] },
   394→  paidAt: Date
   395→): Promise<number> {
   396→  const attributed = order.items.filter((i) => i.affiliateDirectId && i.affiliatePoolMinor > 0);
   397→  if (attributed.length === 0) return 0;
   398→
   399→  const settings = await getAffiliateSettings();
   400→  if (!settings.enabled) return 0; // kill switch: no NEW liabilities
   401→
   402→  // Sponsor profiles snapshot (status at posting decides hold vs pending).
   403→  const sponsorIds = [...new Set(attributed.map((i) => i.affiliateSponsorId).filter((s): s is string => Boolean(s)))];
   404→  const sponsors = sponsorIds.length
   405→    ? new Map((await tx.affiliateProfile.findMany({ where: { id: { in: sponsorIds } }, select: { id: true, status: true } })).map((p) => [p.id, p.status]))
   406→    : new Map<string, string>();
   407→
   408→  // Link metadata snapshot (linkType + channel for reporting — spec §14).
   409→  const linkIds = [...new Set(attributed.map((i) => i.affiliateLinkId).filter((s): s is string => Boolean(s)))];
   410→  const links = linkIds.length
   411→    ? new Map((await tx.affiliateLink.findMany({ where: { id: { in: linkIds } }, select: { id: true, type: true, channel: true } })).map((l) => [l.id, l]))
   412→    : new Map<string, { type: string; channel: string }>();
   413→
   414→  let posted = 0;
   415→  for (const item of attributed) {
   416→    const existing = await tx.affiliateCommission.findUnique({ where: { orderItemId: item.id }, select: { id: true } });
   417→    if (existing) continue; // idempotent replay
   418→
   419→    const pool = item.affiliatePoolMinor;
   420→    let sponsorMinor = 0;
   421→    let sponsorHeld = false;
   422→    // POST-IMPLEMENTATION AUDIT (P3 — deterministic policy freeze): the
   423→    // sponsor-override decision comes from the CHECKOUT-TIME snapshot
   424→    // (OrderItem.affiliateSponsorEnabled), never from the live setting. A
   425→    // settings change between checkout and payment confirmation must not
   426→    // retroactively alter an existing order's compensation arrangement:
   427→    //   • override ON at checkout → this order keeps the 90/10 split even if
   428→    //     an admin later disables sponsor compensation for NEW orders;
   429→    //   • override OFF at checkout → the direct promoter receives 100% (D = P)
   430→    //     for this order, even if the override is later enabled.
   431→    // NULL (legacy rows created before the snapshot column existed) falls
   432→    // back to the current setting — the historical behaviour those orders
   433→    // were created under. The sponsor's CURRENT status still decides only
   434→    // whether the 10% share posts as pending or held (never who the
   435→    // beneficiary is), matching commission-definitions.md §4.
   436→    const sponsorOverrideOn = item.affiliateSponsorEnabled ?? settings.sponsorEnabled;
   437→    if (item.affiliateSponsorId && sponsorOverrideOn) {
   438→      const status = sponsors.get(item.affiliateSponsorId) ?? "missing";
   439→      if (status === "active") {
   440→        sponsorMinor = Math.floor((pool * 10) / 100);
   441→      } else {
   442→        // Sponsor under review / gone: HOLD the 10% separately — the direct
   443→        // promoter's 90% must NOT absorb it and must NOT be penalised.
   444→        sponsorMinor = Math.floor((pool * 10) / 100);
   445→        sponsorHeld = true;
   446→      }
   447→    }
   448→    const directMinor = pool - sponsorMinor; // D + S = P always
   449→    const bd = parseJson<{ base?: number }>(item.breakdownJson, {});
   450→
   451→    const commission = await tx.affiliateCommission.create({
   452→      data: {
   453→        orderItemId: item.id,
   454→        orderId: order.id,
   455→        orderNumber: order.orderNumber,
   456→        storeId: order.storeId,
   457→        productId: item.productId ?? null,
   458→        affiliateId: item.affiliateDirectId!,
   459→        sponsorId: item.affiliateSponsorId ?? null,
   460→        linkId: item.affiliateLinkId ?? null,
   461→        linkType: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.type ?? null) : null,
   462→        channel: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.channel ?? null) : null,
   463→        commissionableBaseMinor: Math.max(0, bd.base ?? 0),
   464→        rateBps: item.affiliateRateBps ?? 0,
   465→        poolMinor: pool,
   466→        status: "pending",
   467→        breakdownJson: item.breakdownJson,
   468→      },
   469→    });
   470→
   471→    await tx.affiliateCommissionEntry.create({
   472→      data: {
   473→        commissionId: commission.id,
   474→        beneficiaryAffiliateId: item.affiliateDirectId!,
   475→        role: "direct",
   476→        amountMinor: directMinor,
   477→        status: "pending",
   478→      },
   479→    });
   480→    if (item.affiliateSponsorId && sponsorMinor > 0) {
   481→      await tx.affiliateCommissionEntry.create({
   482→        data: {
   483→          commissionId: commission.id,
   484→          beneficiaryAffiliateId: item.affiliateSponsorId,
   485→          role: "sponsor",
   486→          amountMinor: sponsorMinor,
   487→          status: sponsorHeld ? "held" : "pending",
   488→          holdReason: sponsorHeld ? "sponsor_under_review" : null,
   489→        },
   490→      });
   491→    }
   492→    posted++;
   493→  }
   494→  return posted;
   495→}

   497→// ---------------------------------------------------------------------------
   498→// Lifecycle: pending → approved (order delivered/completed) → available
   499→// (review window elapsed) → paid. Reversals from refunds. Holds from review.
   500→// ---------------------------------------------------------------------------
   501→/** Approve an order's pending entries (authoritative delivered/completed). */
   502→export async function approveAffiliateCommissionsForOrder(orderId: string, tx?: Tx): Promise<number> {
   503→  const client = tx ?? db;
   504→  const commissions = await client.affiliateCommission.findMany({
   505→    where: { orderId, status: "pending" },
   506→    select: { id: true },
   507→  });
   508→  if (commissions.length === 0) return 0;
   509→  const settings = await getAffiliateSettings();
   510→  const now = new Date();
   511→  const availableAt = new Date(now.getTime() + settings.reviewDays * 86400000);
   512→  let approved = 0;
   513→  for (const c of commissions) {
   514→    const res = await client.affiliateCommissionEntry.updateMany({
   515→      where: { commissionId: c.id, status: "pending" },
   516→      data: { status: "approved", availableAt },
   517→    });
   518→    if (res.count > 0) {
   519→      await client.affiliateCommission.update({ where: { id: c.id }, data: { status: "approved", approvedAt: now } });
   520→      approved += res.count;
   521→    }
   522→  }
   523→  return approved;
   524→}

   526→/** Batch sweep (cron): approve commissions on delivered/completed orders +
   527→ *  mature approved entries whose review window elapsed. Bounded batches. */
   528→export async function sweepAffiliateLifecycle(): Promise<{ approved: number; matured: number }> {
   529→  const settings = await getAffiliateSettings();
   530→  if (!settings.enabled) return { approved: 0, matured: 0 };
   531→
   532→  // 1) Orders delivered/completed with pending commissions → approve.
   533→  const pendingCommissions = await db.affiliateCommission.findMany({
   534→    where: { status: "pending" },
   535→    select: { orderId: true },
   536→    distinct: ["orderId"],
   537→    take: 50,
   538→  });
   539→  let approved = 0;
   540→  if (pendingCommissions.length > 0) {
   541→    const orderIds = pendingCommissions.map((c) => c.orderId);
   542→    const eligible = await db.order.findMany({
   543→      where: { id: { in: orderIds }, status: { in: ["delivered", "completed"] } },
   544→      select: { id: true },
   545→    });
   546→    for (const o of eligible) {
   547→      approved += await approveAffiliateCommissionsForOrder(o.id);
   548→    }
   549→  }
   550→
   551→  // 2) Matured approved entries → available (clock transition, CAS-guarded).
   552→  const now = new Date();
   553→  const matured = await db.affiliateCommissionEntry.updateMany({
   554→    where: { status: "approved", availableAt: { lte: now } },
   555→    data: { status: "available" },
   556→  });
   557→  if (matured.count > 0) {
   558→    await db.affiliateCommission.updateMany({
   559→      where: { status: "approved", entries: { some: { status: "available" } } },
   560→      data: { status: "available" },
   561→    });
   562→  }
   563→  return { approved, matured: matured.count };
   564→}
   565→
   566→// ---------------------------------------------------------------------------
   567→// Balances (affiliate-side truth = AffiliateCommissionEntry rows; the ledger
   568→// mirrors totals on PLATFORM:AFFILIATE_LIABILITY for double-entry integrity).
   569→// ---------------------------------------------------------------------------
   570→export type AffiliateBalances = {
   571→  pendingMinor: number;
   572→  approvedMinor: number; // approved but review window not elapsed
   573→  availableMinor: number; // matured, not yet claimed by a payout
   574→  payoutInFlightMinor: number; // claimed by requested/processing payouts
   575→  paidMinor: number;
   576→  reversedMinor: number;
   577→  heldMinor: number;
   578→  recoveryMinor: number; // reversals booked against already-paid entries
   579→  requestableMinor: number; // available − recovery, floored at 0
   580→};
   581→
   582→export async function affiliateBalances(affiliateId: string): Promise<AffiliateBalances> {
   583→  const entries = await db.affiliateCommissionEntry.findMany({
   584→    where: { beneficiaryAffiliateId: affiliateId },
   585→    select: { status: true, amountMinor: true, reversedMinor: true, availableAt: true, payoutId: true },
   586→  });
   587→  const now = Date.now();
   588→  const b: AffiliateBalances = {
   589→    pendingMinor: 0, approvedMinor: 0, availableMinor: 0, payoutInFlightMinor: 0,
   590→    paidMinor: 0, reversedMinor: 0, heldMinor: 0, recoveryMinor: 0, requestableMinor: 0,
   591→  };
   592→  for (const e of entries) {
   593→    const matured = (e.availableAt?.getTime() ?? 0) <= now;
   594→    switch (e.status) {
   595→      case "pending":
   596→      case "approved":
   597→        if (e.status === "pending" || !matured) {
   598→          if (e.status === "pending") b.pendingMinor += e.amountMinor - e.reversedMinor;
   599→          else b.approvedMinor += e.amountMinor - e.reversedMinor;
   600→          break;
   601→        }
   602→        // fallthrough: matured approved
   603→      case "available":
   604→        if (e.payoutId) b.payoutInFlightMinor += e.amountMinor - e.reversedMinor;
   605→        else b.availableMinor += e.amountMinor - e.reversedMinor;
   606→        break;
   607→      case "paid":
   608→        b.paidMinor += e.amountMinor;
   609→        b.recoveryMinor += e.reversedMinor; // late refund after payout
   610→        break;
   611→      case "reversed":
   612→        b.reversedMinor += e.amountMinor;
   613→        break;
   614→      case "held":
   615→        b.heldMinor += e.amountMinor - e.reversedMinor;
   616→        break;
   617→    }
   618→  }
   619→  b.requestableMinor = Math.max(0, b.availableMinor - b.recoveryMinor);
   620→  return b;
   621→}
   622→
   623→// ---------------------------------------------------------------------------
   624→// Refund reversals — proportional against the ORIGINAL immutable allocations
   625→// (spec §11). Shipping-only refunds book no product-commission reversal (the
   626→// pool base excluded shipping by construction). Late refunds after payout
   627→// become recovery entries: reversedMinor accumulates on the PAID entry and
   628→// offsets future earnings — the original payout history is never rewritten.
   629→// ---------------------------------------------------------------------------
   630→export async function reverseAffiliateForRefund(
   631→  tx: Tx,
   632→  input: {
   633→    refundId: string;
   634→    orderId: string;
   635→    refundAmountMinor: number;
   636→    /** Fresh order items (with affiliate snapshots) under the lock. */
   637→    items: { id: string; lineSubtotalMinor: number; refundedMinor: number; breakdownJson: string; affiliatePoolMinor: number }[];
   638→    /** Σ affiliateReversalMinor of prior PROCESSED refunds on this order. */
   639→    priorAffiliateReversalMinor: number;
   640→  }
   641→): Promise<{ affiliateReversalMinor: number; rule: string }> {
   642→  const attributed = input.items.filter((i) => i.affiliatePoolMinor > 0);
   643→  if (attributed.length === 0) return { affiliateReversalMinor: 0, rule: "no_affiliate" };
   644→
   645→  // Customer-paid gross per item (breakdown-proportional, mirroring
   646→  // deriveCommissionReversal's allocation family).
   647→  const grossArr = attributed.map((it) => {
   648→    const bd = parseJson<{ discount_allocated?: number; shipping_allocated?: number }>(it.breakdownJson, {});
   649→    return Math.max(0, it.lineSubtotalMinor - (bd.discount_allocated ?? 0) + (bd.shipping_allocated ?? 0));
   650→  });
   651→  const allocs: number[] = [];
   652→  {
   653→    // floor + remainder-to-largest (same as allocateProportionally)
   654→    const totalGross = grossArr.reduce((s, v) => s + v, 0);
   655→    if (totalGross <= 0) {
   656→      attributed.forEach(() => allocs.push(0));
   657→    } else {
   658→      let allocated = 0;
   659→      for (let i = 0; i < attributed.length; i++) {
   660→        const a = Math.floor((input.refundAmountMinor * grossArr[i]) / totalGross);
   661→        allocs.push(a);
   662→        allocated += a;
   663→      }
   664→      let rem = input.refundAmountMinor - allocated;
   665→      while (rem > 0) {
   666→        let maxIdx = 0;
   667→        for (let i = 1; i < grossArr.length; i++) if (grossArr[i] > grossArr[maxIdx]) maxIdx = i;
   668→        allocs[maxIdx] += 1;
   669→        rem -= 1;
   670→      }
   671→    }
   672→  }
   673→
   674→  // Per-item affiliate reversal, clamped to the remaining reversible pool.
   675→  const orderPool = attributed.reduce((s, it) => s + it.affiliatePoolMinor, 0);
   676→  const remainingReversible = Math.max(0, orderPool - input.priorAffiliateReversalMinor);
   677→  let total = 0;
   678→  const perItem: { item: (typeof attributed)[number]; reversal: number }[] = [];
   679→  for (let i = 0; i < attributed.length; i++) {
   680→    if (grossArr[i] <= 0 || allocs[i] <= 0) {
   681→      perItem.push({ item: attributed[i], reversal: 0 });
   682→      continue;
   683→    }
   684→    const r = Math.min(Math.floor((allocs[i] * attributed[i].affiliatePoolMinor) / grossArr[i]), attributed[i].affiliatePoolMinor);
   685→    perItem.push({ item: attributed[i], reversal: r });
   686→    total += r;
   687→  }
   688→  total = Math.min(Math.max(0, total), input.refundAmountMinor, remainingReversible);
   689→
   690→  // Apply to the commission + entries (proportional split within each pool).
   691→  if (total > 0) {
   692→    // Scale per-item reversals down proportionally if the clamp reduced total.
   693→    const rawTotal = perItem.reduce((s, p) => s + p.reversal, 0);
   694→    const scale = rawTotal > 0 ? total / rawTotal : 0;
   695→    let applied = 0;
   696→    for (const p of perItem) {
   697→      const itemReversal = p.reversal === 0 ? 0 : Math.min(p.reversal, Math.floor(p.reversal * scale));
   698→      if (itemReversal <= 0) continue;
   699→      const commission = await tx.affiliateCommission.findUnique({ where: { orderItemId: p.item.id }, include: { entries: true } });
   700→      if (!commission) continue; // pre-affiliate legacy rows: nothing to reverse
   701→      const direct = commission.entries.find((e) => e.role === "direct");
   702→      const sponsor = commission.entries.find((e) => e.role === "sponsor");
   703→      // Split proportionally to the frozen allocation shares (P4: floor policy
   704→      // — same rounding as the pool computation). The SPONSOR share is the
   705→      // floor-proportional part and the DIRECT share takes the residual
   706→      // (commission-definitions.md §7 step 3: "direct absorbs the rounding
   707→      // residual"), mirroring the forward split where D = P − S. Invariant:
   708→      // Σ(dRev+sRev) === itemReversal whenever caps allow, and never more.
   709→      const sponsorShare = sponsor ? Math.min(sponsor.amountMinor - sponsor.reversedMinor, Math.floor((itemReversal * sponsor.amountMinor) / commission.poolMinor)) : 0;
   710→      const directShare = direct ? Math.min(direct.amountMinor - direct.reversedMinor, itemReversal - sponsorShare) : 0;
   711→      for (const [entry, share] of [[direct, directShare], [sponsor, sponsorShare]] as const) {
   712→        if (!entry || share <= 0) continue;
   713→        const newReversed = Math.min(entry.amountMinor, entry.reversedMinor + share);
   714→        const fullyReversed = newReversed >= entry.amountMinor;
   715→        await tx.affiliateCommissionEntry.update({
   716→          where: { id: entry.id },
   717→          data: {
   718→            reversedMinor: newReversed,
   719→            // PAID entries stay paid — the reversal is a recovery entry that
   720→            // offsets future earnings (spec §11 late refunds); pending/
   721→            // approved/available entries flip to reversed.
   722→            ...(entry.status !== "paid" && fullyReversed ? { status: "reversed" as const } : {}),
   723→          },
   724→        });
   725→      }
   726→      const anyLive = commission.entries.some((e) => e.status !== "reversed");
   727→      if (!anyLive) {
   728→        await tx.affiliateCommission.update({
   729→          where: { id: commission.id },
   730→          data: { status: "reversed", reversedAt: new Date() },
   731→        });
   732→      }
   733→      applied += itemReversal;
   734→    }
   735→    void applied;
   736→  }
   737→
   738→  return { affiliateReversalMinor: total, rule: "breakdown_snapshot_proportional" };
   739→}
   740→export async function requestAffiliatePayout(affiliateId: string): Promise<{ payoutId: string; amountMinor: number }> {
   741→  const settings = await getAffiliateSettings();
   742→  if (!settings.enabled) throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   743→
   744→  return db.$transaction(async (tx) => {
   745→    // One open request per affiliate — re-checked INSIDE the transaction so a
   746→    // concurrent double-submit cannot create two claims.
   747→    const open = await tx.affiliatePayout.findFirst({
   748→      where: { affiliateId, status: { in: ["requested", "processing"] } },
   749→      select: { id: true },
   750→    });
   751→    if (open) throw new ApiError(409, "PAYOUT_OPEN", "You already have a payout request in progress");
   752→
   753→    const balances = await affiliateBalances(affiliateId);
   754→    if (balances.requestableMinor < settings.minPayoutMinor) {
   755→      throw new ApiError(400, "BELOW_MIN_PAYOUT", `Minimum payout is RM ${(settings.minPayoutMinor / 100).toFixed(2)}`);
   756→    }
   757→
   758→    const amountMinor = balances.requestableMinor;
   759→    const now = new Date();
   760→    // Claim every matured, unclaimed entry; bind to the payout row we create.
   761→    const toClaim = await tx.affiliateCommissionEntry.findMany({
   762→      where: {
   763→        beneficiaryAffiliateId: affiliateId,
   764→        status: { in: ["approved", "available"] },
   765→        availableAt: { lte: now },
   766→        payoutId: null,
   767→      },
   768→      select: { id: true, amountMinor: true, reversedMinor: true, commissionId: true },
   769→    });
   770→    const payout = await tx.affiliatePayout.create({
   771→      data: { affiliateId, amountMinor, status: "requested", method: "manual_bank" },
   772→    });
   773→    if (toClaim.length > 0) {
   774→      await tx.affiliateCommissionEntry.updateMany({
   775→        where: { id: { in: toClaim.map((e) => e.id) } },
   776→        data: { payoutId: payout.id },
   777→      });
   778→    }
   779→    // Ledger-derived statement (schema contract: built from entries at
   780→    // request time, never hand-edited) — order references + balance snapshot.
   781→    const orderRefs = toClaim.length
   782→      ? await tx.affiliateCommission.findMany({
   783→          where: { id: { in: [...new Set(toClaim.map((e) => e.commissionId))] } },
   784→          select: { id: true, orderNumber: true, productId: true, storeId: true },
   785→        })
   786→      : [];
   787→    const refMap = new Map(orderRefs.map((c) => [c.id, c]));
   788→    await tx.affiliatePayout.update({
   789→      where: { id: payout.id },
   790→      data: {
   791→        statementJson: toJson({
   792→          builtAt: now.toISOString(),
   793→          balances: balances,
   794→          entries: toClaim.map((e) => ({
   795→            entryId: e.id,
   796→            commissionId: e.commissionId,
   797→            orderNumber: refMap.get(e.commissionId)?.orderNumber ?? null,
   798→            amountMinor: e.amountMinor,
   799→            reversedMinor: e.reversedMinor,
   800→          })),
   801→        }),
   802→      },
   803→    });
   804→    return { payoutId: payout.id, amountMinor };
   805→  });
   806→}
