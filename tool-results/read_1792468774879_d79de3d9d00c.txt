    39→async function ensureAccountTx(code: string, walletId: string | null | undefined, tx: Prisma.TransactionClient) {
    40→  const meta = ACCOUNT_META[code] ?? {
    41→    name: code.startsWith("GATEWAY:") ? `Gateway ${code.slice(8)}` : code.startsWith("SELLER:") ? "Seller Payable" : code,

    45→    where: { code },
    46→    create: { code, name: meta.name, type: meta.type, walletId: walletId ?? null },
    47→    update: {},
    48→  });

    64→// ---------------------------------------------------------------------------
    65→// FIN-1 POSTING CORE — atomic + idempotent.
    66→//
    67→// ATOMICITY: every posting runs inside a single prisma.$transaction (lines,
    68→// account upserts, wallet-cache refresh all commit or roll back together).
    69→// Recipes called with an outer transaction client run INSIDE it — never
    70→// nested — so a webhook/payout flow can group ledger + business writes.
    71→//
    72→// IDEMPOTENCY (dedupKey semantics — read before changing):
    73→//   • dedupKey is an OPTIONAL caller/recipe-supplied unique helper key stored
    74→//     on LedgerTransaction. When null/undefined the row is written with NULL —
    75→//     Postgres unique indexes treat NULLs as distinct, so non-deduped postings
    76→//     keep the legacy always-insert behavior (e.g. repeatable admin
    77→//     adjustments).
    78→//   • Recipes pass EXPLICIT keys only where a posting must happen EXACTLY
    79→//     once per business event (T1 per order, T2 per settlement ref, T3/T4/T4f
    80→//     per payout, T5 per refund row). Generic auto-keys are deliberately NOT
    81→//     derived here: a (txnType,refType,refId) default would falsely dedup
    82→//     legitimately-repeatable postings such as multi partial refunds per
    83→//     order (refType=order for every refund) — that is why postRefund takes
    84→//     the per-Refund-row refundId for its key.
    85→//   • On a P2002 unique violation the EXISTING transaction is fetched and
    86→//     returned with idempotentReplay:true (see wasReplay) instead of throwing
    87→//     — duplicate webhook deliveries / double-clicked admins converge on the
    88→//     original posting and never create a second one.
    89→// ---------------------------------------------------------------------------
    90→
    91→export type PostedTransaction = Prisma.LedgerTransactionGetPayload<{ include: { lines: true } }> & {
    92→  /** True when the dedupKey already existed and the ORIGINAL posting was returned. */
    93→  idempotentReplay?: boolean;
    94→};
    95→
    96→/** True when a posted result is actually the original txn replayed (dedup hit). */
    97→export function wasReplay(t: PostedTransaction): boolean {
    98→  return t.idempotentReplay === true;
    99→}

   101→/** Post a balanced double-entry transaction. Throws if unbalanced. */
   102→export async function postTransaction(input: {
   103→  txnType: string;
   104→  refType?: string | null;
   105→  refId?: string | null;
   106→  description?: string | null;
   107→  currency?: string;
   108→  lines: LedgerLineInput[];
   109→  occurredAt?: Date;
   110→  /** Optional unique idempotency key (see block comment above). NULL = no dedup. */
   111→  dedupKey?: string | null;
   112→  /** Existing transaction client — when omitted the posting wraps itself in db.$transaction. */
   113→  tx?: Prisma.TransactionClient;
   114→}): Promise<PostedTransaction> {
   115→  const run = async (tx: Prisma.TransactionClient): Promise<PostedTransaction> => {
   116→    const debit = input.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   117→    const credit = input.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   118→    if (debit !== credit || debit === 0) {
   119→      throw new Error(`LEDGER_UNBALANCED debit=${debit} credit=${credit} txn=${input.txnType}`);
   120→    }
   121→    for (const l of input.lines) await ensureAccountTx(l.accountCode, l.walletId ?? null, tx);
   122→    // With a dedupKey we may need to RECOVER from a unique violation inside
   123→    // this (possibly caller-provided) transaction. Postgres aborts the whole
   124→    // tx on a failed statement (25P02 — even a SELECT is refused afterwards),
   125→    // so the create is fenced with a SAVEPOINT: on P2002 we rewind to it and
   126→    // the original row becomes readable again. Without this, the
   127→    // replay-instead-of-throw contract below is unreachable inside a real tx.
   128→    const sp = input.dedupKey
   129→      ? `fin1_sp_${Date.now().toString(36)}${Math.floor(Math.random() * 1e9).toString(36)}`
   130→      : null;
   131→    if (sp) await tx.$executeRawUnsafe(`SAVEPOINT ${sp}`);
   132→    try {
   133→      const created = await tx.ledgerTransaction.create({
   134→        data: {
   135→          txnType: input.txnType,
   136→          refType: input.refType ?? null,
   137→          refId: input.refId ?? null,
   138→          description: input.description ?? null,
   139→          occurredAt: input.occurredAt ?? new Date(),
   140→          dedupKey: input.dedupKey ?? null,
   141→          lines: {
   142→            create: input.lines.map((l) => ({
   143→              accountCode: l.accountCode,
   144→              direction: l.direction,
   145→              amountMinor: l.amountMinor,
   146→              walletId: l.walletId ?? null,
   147→              availableAt: l.availableAt ?? null,
   148→              memo: l.memo ?? null,
   149→              // Denormalized ref (FIN-1 helper columns) so ref-scoped line
   150→              // queries don't need a join; mirrors the parent txn.
   151→              refType: input.refType ?? null,
   152→              refId: input.refId ?? null,
   153→            })),
   154→          },
   155→        },
   156→        include: { lines: true },
   157→      });
   158→      if (sp) await tx.$executeRawUnsafe(`RELEASE SAVEPOINT ${sp}`);
   159→      return created;
   160→    } catch (e) {
   161→      // Idempotent replay: the dedupKey already exists → return the ORIGINAL
   162→      // transaction instead of throwing (never double-post, never 500).
   163→      if (input.dedupKey && e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   164→        if (sp) await tx.$executeRawUnsafe(`ROLLBACK TO SAVEPOINT ${sp}`);
   165→        const existing = await tx.ledgerTransaction.findUnique({
   166→          where: { dedupKey: input.dedupKey },
   167→          include: { lines: true },
   168→        });
   169→        if (existing) return { ...existing, idempotentReplay: true };
   170→      }
   171→      throw e;
   172→    }
   173→  };
   174→  return input.tx ? run(input.tx) : db.$transaction(run);
   175→}

   179→export type WalletTruth = {
   180→  pendingMinor: number;
   181→  availableMinor: number;
   182→  lifetimeEarnedMinor: number;
   183→  lifetimeCommissionMinor: number;
   184→};
   185→
   186→/** Compute wallet balances from ledger lines (the source of truth). */
   187→export async function walletTruth(walletId: string, tx?: Prisma.TransactionClient): Promise<WalletTruth> {
   188→  const payable = ACCOUNTS.sellerPayable(walletId);
   189→  const client = tx ?? db;
   190→  const lines = await client.ledgerLine.findMany({
   191→    where: { accountCode: payable },
   192→    select: { direction: true, amountMinor: true, availableAt: true },
   193→  });
   194→  const now = new Date();
   195→  let totalCredits = 0, maturedCredits = 0, totalDebits = 0, lifetimeCommission = 0;
   196→  for (const l of lines) {
   197→    if (l.direction === "credit") {
   198→      totalCredits += l.amountMinor;
   199→      if (!l.availableAt || l.availableAt <= now) maturedCredits += l.amountMinor;
   200→    } else {
   201→      totalDebits += l.amountMinor;
   202→    }
   203→  }
   204→  // lifetime commission from revenue account lines tagged with this wallet? simpler: from txns T1 memo — use commission revenue lines with walletId
   205→  const commissionLines = await client.ledgerLine.findMany({
   206→    where: { accountCode: ACCOUNTS.revenueCommission(), walletId, direction: "credit" },
   207→    select: { amountMinor: true },
   208→  });
   209→  lifetimeCommission = commissionLines.reduce((s, l) => s + l.amountMinor, 0);
   210→  return {
   211→    pendingMinor: totalCredits - maturedCredits,
   212→    availableMinor: maturedCredits - totalDebits,
   213→    lifetimeEarnedMinor: totalCredits - lifetimeCommission,
   214→    lifetimeCommissionMinor: lifetimeCommission,
   215→  };
   216→}

   218→/**
   219→ * Refresh cached wallet columns from ledger truth (nightly invariant in prod;
   220→ * on-write here). Pass the recipe's `tx` so the refresh reads uncommitted
   221→ * lines and commits atomically with the posting itself.
   222→ */
   223→export async function refreshWalletCache(walletId: string, tx?: Prisma.TransactionClient) {
   224→  const truth = await walletTruth(walletId, tx);
   225→  const client = tx ?? db;
   226→  await client.wallet.update({
   227→    where: { id: walletId },
   228→    data: {
   229→      pendingMinor: truth.pendingMinor,
   230→      availableMinor: truth.availableMinor,
   231→      lifetimeEarnedMinor: truth.lifetimeEarnedMinor,
   232→      lifetimeCommissionMinor: truth.lifetimeCommissionMinor,
   233→    },
   234→  });
   235→  return truth;
   236→}

   439→
   440→/** Nightly invariant (ICCO): every txn balanced; wallet cache == ledger truth. */
   441→export async function runLedgerInvariants() {
   442→  const problems: string[] = [];
   443→  const txns = await db.ledgerTransaction.findMany({ include: { lines: true } });
   444→  for (const t of txns) {
   445→    const debit = t.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   446→    const credit = t.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   447→    if (debit !== credit) problems.push(`txn ${t.id} (${t.txnType}) unbalanced: D${debit} C${credit}`);
   448→  }
   449→  const wallets = await db.wallet.findMany();
   450→  for (const w of wallets) {
   451→    const truth = await walletTruth(w.id);
   452→    if (truth.availableMinor !== w.availableMinor || truth.pendingMinor !== w.pendingMinor) {
   453→      problems.push(`wallet ${w.id} cache drift: available cache=${w.availableMinor} truth=${truth.availableMinor}, pending cache=${w.pendingMinor} truth=${truth.pendingMinor}`);
   454→    }
   455→  }
   456→  // clearing + gateway accounts must net to zero-ish (settlements outstand-ing)
   457→  return { ok: problems.length === 0, problems, txnCount: txns.length, walletCount: wallets.length };
   458→}

   489→// ------------------------------------------------------------- LEDGER-TRUTH STATEMENTS
   490→// INTEGRITY RULE (user directive): every number printed on a payout statement
   491→// must be DERIVED from ledger lines for the payout period — never decorated.
   492→// The lines always reconcile to the payout amount via an explicit
   493→// "Retained in wallet" balancing line, so Σ(lines) ≡ net payout.
   494→
   495→export type LedgerStatement = {
   496→  /** Parsed shape of the JSON string returned by buildLedgerStatementFromLedger
   497→   *  (Payout.statementJson column stores the serialized form). */
   498→  sellerName: string;
   499→  periodFrom: string;
   500→  periodTo: string;
   501→  orderCount: number;
   502→  reconciled: true;
   503→  generatedAt: string;
   504→  lines: { label: string; amountMinor: number }[];
   505→};
   506→
   507→/** Aggregate a wallet's ledger activity for [periodFrom, periodTo] and build a
   508→ *  payout statement whose lines sum EXACTLY to `payoutAmountMinor`.
   509→ *  Components (all from ledger truth):
   510→ *    • sales        — payable credits from order_paid (seller net) + matching
   511→ *                     commission-revenue credits (gross sales)
   512→ *    • commission   — commission-revenue credits from order_paid in the period
   513→ *    • refunds      — payable debits from refund txns + commission reversals
   514→ *    • adjustments  — net admin_adjustment movement on the payable account
   515→ *    • retained     — residual balancing line so Σ(lines) ≡ payout (holds, caps,
   516→ *                     prior-balance carry)                                   */
   517→export async function buildLedgerStatementFromLedger(params: {
   518→  walletId: string;
   519→  sellerName: string;
   520→  payoutAmountMinor: number;
   521→  orderCount: number;
   522→  periodFrom: Date;
   523→  periodTo: Date;
   524→}): Promise<string> {
   525→  const { walletId, sellerName, payoutAmountMinor, orderCount, periodFrom, periodTo } = params;
   526→  const payable = ACCOUNTS.sellerPayable(walletId);
   527→  const commissionAccount = ACCOUNTS.revenueCommission();
   528→
   529→  // Wallet-scope lines in the period, joined to their txn type. Period is
   530→  // evaluated on txn.occurredAt (business time) — rows are often backdated
   531→  // (seed) or posted later (payout T3 lands after the statement snapshot).
   532→  const lines = await db.ledgerLine.findMany({
   533→    where: {
   534→      OR: [{ accountCode: payable }, { accountCode: commissionAccount, walletId }],
   535→      txn: { occurredAt: { gte: periodFrom, lte: periodTo } },
   536→    },
   537→    select: { accountCode: true, direction: true, amountMinor: true, txn: { select: { txnType: true } } },
   538→  });
   539→
   540→  let salesNet = 0; // payable credits from order_paid
   541→  let commission = 0; // commission-revenue credits from order_paid
   542→  let refundSellerShare = 0; // payable debits from refund txns
   543→  let refundCommissionReversal = 0; // commission debits from refund txns
   544→  let adjustments = 0; // admin_adjustment net (credits − debits on payable)
   545→
   546→  for (const l of lines) {
   547→    const type = l.txn.txnType;
   548→    if (l.accountCode === payable) {
   549→      if (l.direction === "credit") {
   550→        if (type === "order_paid") salesNet += l.amountMinor;
   551→        else if (type === "admin_adjustment") adjustments += l.amountMinor;
   552→      } else {
   553→        if (type === "refund") refundSellerShare += l.amountMinor;
   554→        else if (type === "admin_adjustment") adjustments -= l.amountMinor;
   555→      }
   556→    } else if (l.accountCode === commissionAccount) {
   557→      if (l.direction === "credit") {
   558→        if (type === "order_paid") commission += l.amountMinor;
   559→      } else if (type === "refund") {
   560→        refundCommissionReversal += l.amountMinor;
   561→      }
   562→    }
   563→  }
   564→
   565→  const sales = salesNet + commission; // gross sales for the period
   566→  const refunds = refundSellerShare + refundCommissionReversal;
   567→  const netMovement = sales - commission - refunds + adjustments;
   568→  // Balancing line so Σ(statement lines) ≡ net payout: when the period's net
   569→  // movement exceeds the payout (pending holds, min-payout cap, prior carry),
   570→  // the difference is "retained"; when a payout draws on prior balance, it's
   571→  // a positive carry line. Either way the statement always reconciles.
   572→  const retained = payoutAmountMinor - netMovement;
   573→
   574→  const linesOut: { label: string; amountMinor: number }[] = [];
   575→  if (sales !== 0) linesOut.push({ label: "Sales (settled to available)", amountMinor: sales });
   576→  if (commission !== 0) linesOut.push({ label: "Platform commission", amountMinor: -commission });
   577→  if (refunds !== 0) linesOut.push({ label: "Refunds", amountMinor: -refunds });
   578→  if (adjustments !== 0) linesOut.push({ label: "Adjustments", amountMinor: adjustments });
   579→  if (retained !== 0) linesOut.push({ label: "Retained in wallet (hold/cap — not paid out)", amountMinor: retained });
   580→
   581→  return toJson({
   582→    sellerName,
   583→    periodFrom: periodFrom.toISOString(),
   584→    periodTo: periodTo.toISOString(),
   585→    orderCount,
   586→    reconciled: true,
   587→    generatedAt: new Date().toISOString(),
   588→    lines: linesOut,
   589→  });
   590→}
