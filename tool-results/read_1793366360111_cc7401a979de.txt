   215→export class ApiFail extends Error {
   216→  status: number;
   217→  code: string;
   218→  constructor(status: number, code: string, message: string) {
   219→    super(message);
   220→    this.status = status;
   221→    this.code = code;
   222→  }
   223→}
   224→
   225→export async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   226→  const res = await fetch(url, {
   227→    cache: "no-store",
   228→    ...init,
   229→    // Device identity headers (cart + shopper token) — keeps guest state working
   230→    // where cookies are unavailable (cross-site iframe preview panels).
   231→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   232→  });
   233→  let body: unknown = null;
   234→  try {
   235→    body = await res.json();
   236→  } catch {
   237→    /* non-json */
   238→  }
   239→  const b = body as { data?: T; error?: { code?: string; message?: string } } | null;
   240→  if (!res.ok || !b || b.error) {
   241→    // 401 → the shopper session is gone (signed out elsewhere / expired /
   242→    // revoked). Signal app-root to re-check identity; guests stay guests.
   243→    if (res.status === 401) signalSessionExpired();
   244→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   245→  }
   246→  return (b.data ?? ({} as T)) as T;
   247→}
   248→
   249→/** Like fetchJson but also returns the paged meta envelope. */
   250→export async function fetchEnvelope<T>(
   251→  url: string,
   252→  init?: RequestInit
   253→): Promise<{ data: T; meta?: { page: number; size: number; total: number } }> {
   254→  const res = await fetch(url, {
   255→    cache: "no-store",
   256→    ...init,
   257→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   258→  });
   259→  let body: unknown = null;
   260→  try {
   261→    body = await res.json();
   262→  } catch {
   263→    /* non-json */
   264→  }
   265→  const b = body as { data?: T; meta?: { page: number; size: number; total: number }; error?: { code?: string; message?: string } } | null;
   266→  if (!res.ok || !b || b.error) {
   267→    if (res.status === 401) signalSessionExpired();
   268→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   269→  }
   270→  return { data: (b.data ?? ({} as T)) as T, meta: b.meta };
   271→}

   311→export function trackOrderRef(orderNumber: string, email: string) {
   312→  if (typeof window === "undefined") return;
   313→  try {
   314→    const raw = JSON.parse(window.localStorage.getItem(ORDER_REFS_KEY) ?? "[]") as OrderRefT[];
   315→    const clean = raw.filter((r) => r.orderNumber !== orderNumber);
   316→    clean.unshift({ orderNumber, email: email.toLowerCase(), at: Date.now() });
   317→    window.localStorage.setItem(ORDER_REFS_KEY, JSON.stringify(clean.slice(0, 25)));
   318→  } catch {
   319→    /* corrupted list — ignore */
   320→  }
   321→}

   326→    const raw = JSON.parse(window.localStorage.getItem(ORDER_REFS_KEY) ?? "[]") as OrderRefT[];
   327→    return Array.isArray(raw) ? raw.filter((r) => r && r.orderNumber && r.email) : [];
   328→  } catch {
   329→    return [];
   330→  }

   333→/** Fetch the account cart and live-refresh on sm-cart-updated events + on any
   334→ *  sign-in / sign-out change (carts are account-scoped — guests have none).
   335→ *
   336→ *  Task 28: cart state lives in a MODULE-LEVEL shared store, so every useCart()
   337→ *  instance (drawer, header badge, checkout) renders from the same snapshot —
   338→ *  one fetch per event instead of one per instance — and checkout-selection
   339→ *  toggles can mutate it optimistically (mutateCartItems): the UI updates in
   340→ *  the same frame the user clicks while the PATCH rides in the background.
   341→ *  Zero network in the render path — same philosophy as the client-side
   342→ *  Messages list filter. */
   343→const EMPTY_CART: CartT = { id: null, storeId: null, storeSlug: null, storeName: null, items: [] };
   344→
   345→type CartState = { cart: CartT; loading: boolean };
   346→let cartState: CartState = { cart: EMPTY_CART, loading: true };
   347→const CART_SERVER_STATE: CartState = { cart: EMPTY_CART, loading: true };
   348→const cartSubs = new Set<() => void>();
   349→
   350→function emitCart() {
   351→  for (const fn of cartSubs) fn();
   352→}
   353→
   354→function subscribeCart(fn: () => void) {
   355→  cartSubs.add(fn);
   356→  return () => {
   357→    cartSubs.delete(fn);
   358→  };
   359→}
   360→
   361→/** Optimistically mutate the shared cart state — no network, no refetch.
   362→ *  Consumers must keep the failure path honest: revert via mutateCartItems
   363→ *  (guarded so it only undoes its own change) and surface a toast. */
   364→export function mutateCartItems(mutate: (items: CartItemT[]) => CartItemT[]) {
   365→  const cur = cartState.cart;
   366→  if (!cur) return;
   367→  cartState = { ...cartState, cart: { ...cur, items: mutate(cur.items) } };
   368→  emitCart();
   369→}
   370→
   371→/** Adopt a server cart payload directly into the shared store (no refetch).
   372→ *  Add-to-cart responses already carry the fresh cart — feeding it here keeps
   373→ *  every consumer (badge, drawer, checkout) in sync in the same frame without
   374→ *  the event-driven /api/cart refetch storm. */
   375→export function applyServerCart(cart: CartT | null | undefined) {
   376→  if (!cart) return;
   377→  cartState = { cart, loading: false };
   378→  emitCart();
   379→}
   380→
   381→/** Read-only snapshot of the shared cart (for non-render code paths, e.g. the
   382→ *  drawer's stock probe building the variant-id list). */
   383→export function cartSnapshot(): CartT {
   384→  return cartState.cart;
   385→}

   387→export function useCart() {
   388→  const me = useApp((s) => s.me);
   389→  const authed = Boolean(me.loaded && me.shopper);
   390→  const { cart, loading } = useSyncExternalStore(
   391→    subscribeCart,
   392→    () => cartState,
   393→    () => CART_SERVER_STATE
   394→  );
   395→
   396→  const refresh = useCallback(async () => {
   397→    try {
   398→      const data = await fetchJson<{ cart: CartT }>("/api/cart");
   399→      cartState = { cart: data.cart ?? EMPTY_CART, loading: false };
   400→    } catch {
   401→      cartState = { cart: EMPTY_CART, loading: false };
   402→    }
   403→    emitCart();
   404→  }, []);
   405→
   406→  useEffect(() => {
   407→    // Show the loader only when there is nothing to paint yet — data-bearing
   408→    // re-syncs (mounts, sign-in flips, sm-cart-updated events) stay silent so
   409→    // no surface ever flashes a skeleton over live data.
   410→    if (cartState.cart.items.length === 0) {
   411→      cartState = { ...cartState, loading: true };
   412→      emitCart();
   413→    }
   414→    void refresh();
   415→    const onEvt = () => void refresh();
   416→    window.addEventListener(CART_EVENT, onEvt);
   417→    return () => window.removeEventListener(CART_EVENT, onEvt);
   418→  }, [refresh, authed]);
   419→
   420→  const count = cart.items.reduce((s, i) => s + i.qty, 0);
   421→  const subtotal = cart.items.reduce((s, i) => s + i.lineTotalMinor, 0);
   422→  // Selection-aware figures — only checked lines proceed to checkout.
   423→  const selectedItems = cart.items.filter((i) => i.selected);
   424→  const selectedCount = selectedItems.reduce((s, i) => s + i.qty, 0);
   425→  const selectedSubtotal = selectedItems.reduce((s, i) => s + i.lineTotalMinor, 0);
   426→  return { cart, count, subtotal, selectedCount, selectedSubtotal, loading, refresh };
   427→}

   437→/** Idempotently inject a Google Fonts stylesheet for the given font ids. */
   438→export function ensureFontsLoaded(ids: (string | null | undefined)[]) {
   439→  if (typeof document === "undefined") return;
   440→  const href = fontLinkHref(ids);
   441→  if (!href) return;
   442→  if (!document.querySelector(`link[data-sm-fonts="${href}"]`)) {
   443→    const link = document.createElement("link");
   444→    link.rel = "stylesheet";
   445→    link.href = href;
   446→    link.dataset.smFonts = href;
   447→    document.head.appendChild(link);
   448→  }
   449→}

   455→export function Stars({ rating, count, className }: { rating: number; count?: number; className?: string }) {
   456→  const full = Math.floor(rating);
   457→  const half = rating - full >= 0.25 && rating - full < 0.75;
   458→  const rounded = rating - full >= 0.75 ? full + 1 : full;
   459→  return (
   460→    <span className={cn("inline-flex items-center gap-0.5", className)} aria-label={`Rated ${rating.toFixed(1)} out of 5`}>
   461→      {[0, 1, 2, 3, 4].map((i) =>
   462→        i < full || (i === full && rounded > full && !half) ? (
   463→          <Star key={i} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   464→        ) : i === full && half ? (
   465→          <span key={i} className="relative inline-flex">
   466→            <Star className="h-3.5 w-3.5 text-muted-foreground/40" />
   467→            <StarHalf className="h-3.5 w-3.5 absolute inset-0 fill-amber-400 text-amber-400" />
   468→          </span>
   469→        ) : (
   470→          <Star key={i} className="h-3.5 w-3.5 text-muted-foreground/40" />
   471→        )
   472→      )}
   473→      <span className="text-xs text-muted-foreground ml-1 tabular-nums">
   474→        {rating > 0 ? rating.toFixed(1) : "New"}
   475→        {typeof count === "number" && count > 0 ? ` (${count})` : ""}
   476→      </span>
   477→    </span>
   478→  );
   479→}

   485→export function ProductCard({
   486→  p,
   487→  onOpen,
   488→  // Task 28: vendor name no longer renders on product cards by default —
   489→  // cards are cleaner and the seller is always one tap away (store link in
   490→  // the cart groups, storefront header). Pass true to opt back in.
   491→  showStore = false,
   492→  hidePrice = false,
   493→  cardStyle,
   494→}: {
   495→  p: ProductCardT;
   496→  onOpen: (p: ProductCardT) => void;
   497→  showStore?: boolean;
   498→  hidePrice?: boolean;
   499→  /** Seller-customized typography — only passed on the seller's own storefront. */
   500→  cardStyle?: CardStyleTokens | null;
   501→}) {
   502→  const discount =
   503→    p.compareAtPriceMinor && p.compareAtPriceMinor > p.priceMinor
   504→      ? Math.round(100 - (p.priceMinor / p.compareAtPriceMinor) * 100)
   505→      : null;
   506→  const nameCss: React.CSSProperties = cardStyle
   507→    ? { fontFamily: fontStack(cardStyle.nameFont), fontSize: `${nameSizePx(cardStyle.nameSize)}px`, textAlign: cardStyle.nameAlign as React.CSSProperties["textAlign"] }
   508→    : {};
   509→  const priceCss: React.CSSProperties = cardStyle
   510→    ? { fontFamily: fontStack(cardStyle.priceFont), fontSize: `${priceSizePx(cardStyle.priceSize)}px`, textAlign: cardStyle.priceAlign as React.CSSProperties["textAlign"] }
   511→    : {};
   512→  return (
   513→    <button
   514→      type="button"
   515→      onClick={() => onOpen(p)}
   516→      className="group text-left rounded-[8px] border bg-card shadow-soft overflow-hidden hover:shadow-md hover:border-teal-600/40 hover:-translate-y-0.5 transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-600/50"
   517→      aria-label={`${p.title} from ${p.storeName}, ${formatMinor(p.priceMinor)}`}
   518→    >
   519→      <div className="relative aspect-square overflow-hidden bg-muted">
   520→        {/* plain <img> (product mockups / placeholder service) — never next/image here */}
   521→        <img
   522→          src={p.imageUrl}
   523→          alt={p.title}
   524→          loading="lazy"
   525→          className="h-full w-full object-cover transition-transform duration-300 group-hover:scale-[1.04]"
   526→        />
   527→        {discount ? (
   528→          <span className="absolute top-1.5 left-1.5 rounded-full bg-rose-600 text-white text-[10px] font-bold px-1.5 py-0.5">
   529→            -{discount}%
   530→          </span>
   531→        ) : null}
   532→      </div>
   533→      <div className="p-2.5 space-y-0.5">
   534→        <h3 className={cn("font-normal leading-snug line-clamp-2 min-h-[2rem]", !cardStyle && "text-xs")} style={nameCss}>{p.title}</h3>
   535→        {!hidePrice ? (
   536→          <div className="flex items-baseline gap-1.5 flex-wrap" style={cardStyle?.priceAlign === "center" ? { justifyContent: "center" } : cardStyle?.priceAlign === "right" ? { justifyContent: "flex-end" } : undefined}>
   537→            <span className={cn("font-bold text-teal-700", !cardStyle && "text-sm")} style={priceCss}>{formatMinor(p.priceMinor)}</span>

   571→export function QtyStepper({
   572→  value,
   573→  onChange,
   574→  min = 1,
   575→  max = 99,
   576→  small = false,
   577→  xs = false,
   578→}: {
   579→  value: number;
   580→  onChange: (v: number) => void;
   581→  min?: number;
   582→  max?: number;
   583→  small?: boolean;
   584→  /** Cart-drawer minimalist size (Task 29): 24px buttons, 11px digits. */
   585→  xs?: boolean;
   586→}) {
   587→  const btn = xs ? "h-6 w-6" : small ? "h-8 w-8" : "h-10 w-10";
   588→  const icon = xs ? "h-3 w-3" : "h-3.5 w-3.5";
   589→  const field = xs ? "w-6" : small ? "w-8" : "w-10";
   590→  return (
   591→    <div className={cn("inline-flex items-center border", xs ? "rounded-md" : "rounded-lg")} role="group" aria-label="Quantity">
   592→      <Button
   593→        type="button"
   594→        variant="ghost"
   595→        size="icon"
   596→        className={cn(btn, "rounded-r-none")}
   597→        disabled={value <= min}
   598→        onClick={() => onChange(Math.max(min, value - 1))}
   599→        aria-label="Decrease quantity"
   600→      >
   601→        <Minus className={icon} />
   602→      </Button>
   603→      <span
   604→        className={cn(
   605→          "flex items-center justify-center font-semibold tabular-nums",
   606→          xs ? "w-6 text-[11px]" : "text-sm",
   607→          field
   608→        )}
   609→        aria-live="polite"
   610→      >
   611→        {value}
   612→      </span>
   613→      <Button
   614→        type="button"
   615→        variant="ghost"
   616→        size="icon"
   617→        className={cn(btn, "rounded-l-none")}
   618→        disabled={value >= max}
   619→        onClick={() => onChange(Math.min(max, value + 1))}
   620→        aria-label="Increase quantity"
   621→      >
   622→        <Plus className={icon} />
   623→      </Button>
   624→    </div>
   625→  );
   626→}

   824→export function OrderFlowTimeline({ history, current }: { history: { toStatus: string; createdAt: string }[]; current: string }) {
   825→  const special = ["cancelled", "refunded", "partially_refunded"].includes(current);
   826→  const reachedAt = new Map<string, string>();
   827→  for (const h of history) reachedAt.set(h.toStatus, h.createdAt);
   828→  const flow = special ? [] : FLOW;
   829→  const currentIdx = flow.indexOf(current === "completed" ? "delivered" : current);
   830→
   831→  return (
   832→    <ol className="space-y-0" aria-label="Order status timeline">
   833→      {flow.map((step, i) => {
   834→        const done = currentIdx >= i && currentIdx !== -1;
   835→        const ts = reachedAt.get(step);
   836→        const isLast = i === flow.length - 1;
   837→        return (
   838→          <li key={step} className="relative flex gap-3 pb-5 last:pb-0">
   839→            {!isLast ? <span className={cn("absolute left-[9px] top-5 bottom-0 w-px", done ? "bg-teal-600" : "bg-border")} /> : null}
   840→            <span
   841→              className={cn(
   842→                "relative z-10 mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-full border-2",
   843→                done ? "border-teal-600 bg-teal-600" : "border-muted-foreground/30 bg-background"
   844→              )}
   845→            >
   846→              {done ? (
   847→                <svg viewBox="0 0 10 8" className="h-2.5 w-2.5 fill-none stroke-white" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden>
   848→                  <path d="M1 4l2.5 2.5L9 1" />
   849→                </svg>
   850→              ) : null}
   851→            </span>
   852→            <div className="min-w-0">
   853→              <p className={cn("text-sm font-medium leading-5", done ? "text-foreground" : "text-muted-foreground")}>
   854→                {statusLabel(step)}
   855→              </p>
   856→              {ts ? (
   857→                <p className="text-xs text-muted-foreground">{new Date(ts).toLocaleString("en-MY", { dateStyle: "medium", timeStyle: "short" })}</p>
   858→              ) : null}
   859→            </div>
   860→          </li>
   861→        );
   862→      })}
   863→      {special ? (
   864→        <li className="flex items-center gap-2">
   865→          <StatusBadge status={current} />
   866→          <span className="text-xs text-muted-foreground">This order was closed with status above.</span>
   867→        </li>
   868→      ) : null}
   869→    </ol>
   870→  );
   871→}

   877→export function dateShort(iso: string | Date | null | undefined): string {
   878→  if (!iso) return "";
   879→  return new Date(iso).toLocaleDateString("en-MY", { day: "numeric", month: "short", year: "numeric" });
   880→}

   882→/** "12 Oct 2026, 2:56 pm" — used by chat timestamps. */
   883→export function fmtDateTime(iso: string | Date | null | undefined): string {
   884→  if (!iso) return "";
   885→  return new Intl.DateTimeFormat("en-MY", { dateStyle: "medium", timeStyle: "short", timeZone: "Asia/Kuala_Lumpur" }).format(new Date(iso));
   886→}

   888→export function discountPercent(price: number, compareAt: number | null | undefined): number | null {
   889→  if (compareAt && compareAt > price) return Math.round(100 - (price / compareAt) * 100);
   890→  return null;
   891→}
