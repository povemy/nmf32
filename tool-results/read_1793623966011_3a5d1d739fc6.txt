    32→/** Registration with 429-aware retry. Back-to-back suite runs share the 5/min
    33→ *  "auth-register" bucket (F-12), so a suite started right after another can
    34→ *  get throttled. Wait the advertised retryAfterSeconds (+1s) and retry (max 2
    35→ *  waits) instead of silently cascading empty-cookie failures downstream. */
    36→async function registerFetch(path: string, body: unknown) {
    37→  for (let attempt = 0; ; attempt++) {
    38→    const r = await jsonFetch(path, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
    39→    if (r.status !== 429 || attempt >= 2) return r;
    40→    const waitS = (r.body?.error?.details?.retryAfterSeconds ?? 12) + 1;
    41→    console.log(`  [register throttle] ${path} 429 — waiting ${waitS}s (attempt ${attempt + 1}/2)`);
    42→    await sleep(waitS * 1000);
    43→  }
    44→}

    86→/** Merge Set-Cookie values from a response into a cookie-jar string. */
    87→function absorbCookie(jar: string, res: Response): string {
    88→  const map = new Map(jar.split("; ").filter(Boolean).map((c) => [c.split("=")[0], c]));
    89→  for (const c of res.headers.getSetCookie?.() ?? []) {
    90→    const [pair] = c.split(";");
    91→    const eq = pair.indexOf("=");
    92→    if (eq > 0) map.set(pair.slice(0, eq), pair);
    93→  }
    94→  return [...map.values()].join("; ");
    95→}
