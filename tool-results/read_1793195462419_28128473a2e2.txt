    38→// F1 [Measured]: under concurrent same-seller confirmations, the unconditional
    39→// `ledgerAccount.upsert` on the SHARED SELLER/GATEWAY/COMMISSION account rows
    40→// was itself a hot-row write inside every Serializable tx (write-write SSI
    41→// conflicts + extra roundtrips). Accounts are effectively immutable once
    42→// created, so: read-first + per-process memo. The common case becomes a
    43→// read (no write, no conflict domain growth) and 1 roundtrip instead of an
    44→// upsert; first-creation still upserts (P2002-safe).
    45→const ensuredAccounts = new Set<string>();
    46→
    47→/** Same upsert, runnable on an explicit transaction client (FIN-1 atomicity). */
    48→async function ensureAccountTx(code: string, walletId: string | null | undefined, tx: Prisma.TransactionClient) {
    49→  if (ensuredAccounts.has(code)) return;
    50→  const meta = ACCOUNT_META[code] ?? {
    51→    name: code.startsWith("GATEWAY:") ? `Gateway ${code.slice(8)}` : code.startsWith("SELLER:") ? "Seller Payable" : code,
    52→    type: code.startsWith("SELLER:") ? "liability" : code.startsWith("GATEWAY:") ? "asset" : "asset",
    53→  };
    54→  const existing = await tx.ledgerAccount.findUnique({ where: { code }, select: { code: true } });
    55→  if (existing) {
    56→    ensuredAccounts.add(code);
    57→    return;
    58→  }
    59→  try {
    60→    await tx.ledgerAccount.upsert({
    61→      where: { code },
    62→      create: { code, name: meta.name, type: meta.type, walletId: walletId ?? null },
    63→      update: {},
    64→    });
    65→    ensuredAccounts.add(code);
    66→  } catch (e) {
    67→    // Raced creation from another tx — the row exists now; treat as ensured.
    68→    if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
    69→      ensuredAccounts.add(code);
    70→      return;
    71→    }
    72→    throw e;
    73→  }
    74→}

    77→  await ensureAccount(ACCOUNTS.sellerPayable(walletId), walletId);

    89→// ---------------------------------------------------------------------------
    90→// FIN-1 POSTING CORE — atomic + idempotent.
    91→//
    92→// ATOMICITY: every posting runs inside a single prisma.$transaction (lines,
    93→// account upserts, wallet-cache refresh all commit or roll back together).
    94→// Recipes called with an outer transaction client run INSIDE it — never
    95→// nested — so a webhook/payout flow can group ledger + business writes.
    96→//
    97→// IDEMPOTENCY (dedupKey semantics — read before changing):
    98→//   • dedupKey is an OPTIONAL caller/recipe-supplied unique helper key stored
    99→//     on LedgerTransaction. When null/undefined the row is written with NULL —
   100→//     Postgres unique indexes treat NULLs as distinct, so non-deduped postings
   101→//     keep the legacy always-insert behavior (e.g. repeatable admin
   102→//     adjustments).
   103→//   • Recipes pass EXPLICIT keys only where a posting must happen EXACTLY
   104→//     once per business event (T1 per order, T2 per settlement ref, T3/T4/T4f
   105→//     per payout, T5 per refund row). Generic auto-keys are deliberately NOT
   106→//     derived here: a (txnType,refType,refId) default would falsely dedup
   107→//     legitimately-repeatable postings such as multi partial refunds per
   108→//     order (refType=order for every refund) — that is why postRefund takes
   109→//     the per-Refund-row refundId for its key.
   110→//   • On a P2002 unique violation the EXISTING transaction is fetched and
   111→//     returned with idempotentReplay:true (see wasReplay) instead of throwing
   112→//     — duplicate webhook deliveries / double-clicked admins converge on the
   113→//     original posting and never create a second one.
   114→// ---------------------------------------------------------------------------
   115→
   116→export type PostedTransaction = Prisma.LedgerTransactionGetPayload<{ include: { lines: true } }> & {
   117→  /** True when the dedupKey already existed and the ORIGINAL posting was returned. */
   118→  idempotentReplay?: boolean;
   119→};
   120→
   121→/** True when a posted result is actually the original txn replayed (dedup hit). */
   122→export function wasReplay(t: PostedTransaction): boolean {
   123→  return t.idempotentReplay === true;
   124→}

   126→/** Post a balanced double-entry transaction. Throws if unbalanced. */
   127→export async function postTransaction(input: {
   128→  txnType: string;
   129→  refType?: string | null;
   130→  refId?: string | null;
   131→  description?: string | null;
   132→  currency?: string;
   133→  lines: LedgerLineInput[];
   134→  occurredAt?: Date;
   135→  /** Optional unique idempotency key (see block comment above). NULL = no dedup. */
   136→  dedupKey?: string | null;
   137→  /** Existing transaction client — when omitted the posting wraps itself in db.$transaction. */
   138→  tx?: Prisma.TransactionClient;
   139→}): Promise<PostedTransaction> {
   140→  const run = async (tx: Prisma.TransactionClient): Promise<PostedTransaction> => {
   141→    const debit = input.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   142→    const credit = input.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   143→    if (debit !== credit || debit === 0) {
   144→      throw new Error(`LEDGER_UNBALANCED debit=${debit} credit=${credit} txn=${input.txnType}`);
   145→    }
   146→    for (const l of input.lines) await ensureAccountTx(l.accountCode, l.walletId ?? null, tx);
   147→    // With a dedupKey we may need to RECOVER from a unique violation inside
   148→    // this (possibly caller-provided) transaction. Postgres aborts the whole
   149→    // tx on a failed statement (25P02 — even a SELECT is refused afterwards),
   150→    // so the create is fenced with a SAVEPOINT: on P2002 we rewind to it and
   151→    // the original row becomes readable again. Without this, the
   152→    // replay-instead-of-throw contract below is unreachable inside a real tx.
   153→    const sp = input.dedupKey
   154→      ? `fin1_sp_${Date.now().toString(36)}${Math.floor(Math.random() * 1e9).toString(36)}`
   155→      : null;
   156→    if (sp) await tx.$executeRawUnsafe(`SAVEPOINT ${sp}`);
   157→    try {
   158→      const created = await tx.ledgerTransaction.create({
   159→        data: {
   160→          txnType: input.txnType,
   161→          refType: input.refType ?? null,
   162→          refId: input.refId ?? null,
   163→          description: input.description ?? null,
   164→          occurredAt: input.occurredAt ?? new Date(),
   165→          dedupKey: input.dedupKey ?? null,
   166→          lines: {
   167→            create: input.lines.map((l) => ({
   168→              accountCode: l.accountCode,
   169→              direction: l.direction,
   170→              amountMinor: l.amountMinor,
   171→              walletId: l.walletId ?? null,
   172→              availableAt: l.availableAt ?? null,
   173→              memo: l.memo ?? null,
   174→              // Denormalized ref (FIN-1 helper columns) so ref-scoped line
   175→              // queries don't need a join; mirrors the parent txn.
   176→              refType: input.refType ?? null,
   177→              refId: input.refId ?? null,
   178→            })),
   179→          },
   180→        },
   181→        include: { lines: true },
   182→      });
   183→      if (sp) await tx.$executeRawUnsafe(`RELEASE SAVEPOINT ${sp}`);
   184→      return created;
   185→    } catch (e) {
   186→      // Idempotent replay: the dedupKey already exists → return the ORIGINAL
   187→      // transaction instead of throwing (never double-post, never 500).
   188→      if (input.dedupKey && e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2002") {
   189→        if (sp) await tx.$executeRawUnsafe(`ROLLBACK TO SAVEPOINT ${sp}`);
   190→        const existing = await tx.ledgerTransaction.findUnique({
   191→          where: { dedupKey: input.dedupKey },
   192→          include: { lines: true },
   193→        });
   194→        if (existing) return { ...existing, idempotentReplay: true };
   195→      }
   196→      throw e;
   197→    }
   198→  };
   199→  return input.tx ? run(input.tx) : db.$transaction(run);
   200→}

   204→export type WalletTruth = {
   205→  pendingMinor: number;
   206→  availableMinor: number;
   207→  lifetimeEarnedMinor: number;
   208→  lifetimeCommissionMinor: number;
   209→};
   210→
   211→/** Compute wallet balances from ledger lines (the source of truth). */
   212→export async function walletTruth(walletId: string, tx?: Prisma.TransactionClient): Promise<WalletTruth> {
   213→  const payable = ACCOUNTS.sellerPayable(walletId);
   214→  const client = tx ?? db;
   215→  const lines = await client.ledgerLine.findMany({
   216→    where: { accountCode: payable },
   217→    select: { direction: true, amountMinor: true, availableAt: true },
   218→  });
   219→  const now = new Date();
   220→  let totalCredits = 0, maturedCredits = 0, totalDebits = 0, lifetimeCommission = 0;
   221→  for (const l of lines) {
   222→    if (l.direction === "credit") {
   223→      totalCredits += l.amountMinor;
   224→      if (!l.availableAt || l.availableAt <= now) maturedCredits += l.amountMinor;
   225→    } else {
   226→      totalDebits += l.amountMinor;
   227→    }
   228→  }
   229→  // lifetime commission from revenue account lines tagged with this wallet? simpler: from txns T1 memo — use commission revenue lines with walletId
   230→  const commissionLines = await client.ledgerLine.findMany({
   231→    where: { accountCode: ACCOUNTS.revenueCommission(), walletId, direction: "credit" },
   232→    select: { amountMinor: true },
   233→  });
   234→  lifetimeCommission = commissionLines.reduce((s, l) => s + l.amountMinor, 0);
   235→  return {
   236→    pendingMinor: totalCredits - maturedCredits,
   237→    availableMinor: maturedCredits - totalDebits,
   238→    lifetimeEarnedMinor: totalCredits - lifetimeCommission,
   239→    lifetimeCommissionMinor: lifetimeCommission,
   240→  };
   241→}

   243→/**
   244→ * Refresh cached wallet columns from ledger truth (nightly invariant in prod;
   245→ * on-write here). Pass the recipe's `tx` so the refresh reads uncommitted
   246→ * lines and commits atomically with the posting itself.
   247→ *
   248→ * CONTENTION NOTE (measured): inside a Serializable tx this UPDATE on the
   249→ * wallet row is the hot-row epicenter of same-seller payment-confirmation
   250→ * storms (performance/REPORT.md §5.4 — 0/15 succeeded). Hot paths (webhook
   251→ * confirmation) now pass no `tx` and run this AFTER the money tx commits —
   252→ * the cache is a derived display value and payout authorization recomputes
   253→ * truth independently, so a failed/late post-commit refresh is safe (the
   254→ * invariant classifier reports time-maturity staleness as benign).
   255→ */
   256→export async function refreshWalletCache(walletId: string, tx?: Prisma.TransactionClient) {
   257→  const truth = await walletTruth(walletId, tx);
   258→  const client = tx ?? db;
   259→  await client.wallet.update({
   260→    where: { id: walletId },
   261→    data: {
   262→      pendingMinor: truth.pendingMinor,
   263→      availableMinor: truth.availableMinor,
   264→      lifetimeEarnedMinor: truth.lifetimeEarnedMinor,
   265→      lifetimeCommissionMinor: truth.lifetimeCommissionMinor,
   266→    },
   267→  });
   268→  return truth;
   269→}
   270→
   271→/**
   272→ * Best-effort cache refresh for display surfaces: never throws — on failure
   273→ * returns the last-known cache and the error, so a wallet PAGE never 500s
   274→ * because a concurrent payment tx briefly holds the wallet row. Money
   275→ * decisions never use this helper (they recompute truth transactionally).
   276→ */
   277→export async function safeRefreshWalletCache(walletId: string) {
   278→  try {
   279→    const truth = await refreshWalletCache(walletId);
   280→    return { refreshed: true as const, truth, error: null };
   281→  } catch (e) {
   282→    const cached = await db.wallet.findUnique({ where: { id: walletId } });
   283→    return {
   284→      refreshed: false as const,
   285→      truth: null,
   286→      error: e instanceof Error ? e.message : String(e),
   287→      cached,
   288→    };
   289→  }
   290→}

   295→ * T1 — order paid: DR GATEWAY:{p} total | CR SELLER:PAYABLE (available_at=paid+hold) | CR COMMISSION

   300→export async function postOrderPaid(input: {
   301→  orderId: string;
   302→  orderNumber: string;
   303→  walletId: string;
   304→  provider: string;
   305→  totalMinor: number;
   306→  sellerNetMinor: number;
   307→  commissionMinor: number;
   308→  holdUntil: Date;
   309→  currency?: string;
   310→  occurredAt?: Date;
   311→  tx?: Prisma.TransactionClient;
   312→  /** Set false on hot paths (payment confirmation) that refresh the cache
   313→   *  AFTER the money tx commits instead of inside it (see refreshWalletCache
   314→   *  contention note). Default true preserves every other caller. */
   315→  refreshCache?: boolean;
   316→}) {
   317→  const t = await postTransaction({
   318→    txnType: "order_paid",
   319→    refType: "order",
   320→    refId: input.orderId,
   321→    description: `Order ${input.orderNumber} paid via ${input.provider}`,
   322→    dedupKey: `order_paid:order:${input.orderId}`,
   323→    lines: [
   324→      { accountCode: ACCOUNTS.gateway(input.provider), direction: "debit", amountMinor: input.totalMinor, memo: `Order ${input.orderNumber}` },
   325→      { accountCode: ACCOUNTS.sellerPayable(input.walletId), direction: "credit", amountMinor: input.sellerNetMinor, walletId: input.walletId, availableAt: input.holdUntil, memo: `Seller net for ${input.orderNumber}` },
   326→      { accountCode: ACCOUNTS.revenueCommission(), direction: "credit", amountMinor: input.commissionMinor, walletId: input.walletId, memo: `Commission for ${input.orderNumber}` },
   327→    ],
   328→    occurredAt: input.occurredAt,
   329→    tx: input.tx,
   330→  });
   331→  if (input.refreshCache !== false) await refreshWalletCache(input.walletId, input.tx);
   332→  return t;
   333→}

   427→/**
   428→ * T5 — refund: DR SELLER:PAYABLE (seller share) + DR COMMISSION (reversal) | CR GATEWAY:{p}

   496→// ------------------------------------------------------------- INVARIANTS
   497→
   498→export type WalletCacheDrift = {
   499→  walletId: string;
   500→  cachedAvailable: number;
   501→  cachedPending: number;
   502→  truthAvailable: number;
   503→  truthPending: number;
   504→  /** available cache − truth, pending cache − truth (signed). */
   505→  availableDelta: number;
   506→  pendingDelta: number;
   507→  /** Sum of holds whose availableAt matured after the cache's last write —
   508→   *  the full explanation when classification is TIME_MATURITY. */
   509→  maturedSinceLastWrite: number;
   510→  lastCacheWriteAt: Date;
   511→};
   512→
   513→export type LedgerInvariantResult = {
   514→  ok: boolean;
   515→  /** HARD problems only: unbalanced transactions + hard cache drift. */
   516→  problems: string[];
   517→  /** CACHE_VALUE_DRIFT / unexplained staleness — totals differ or the
   518→   *  staleness is NOT explained by holds maturing after the last write.
   519→   *  Always investigated; never auto-dismissed. */
   520→  hardDrift: WalletCacheDrift[];
   521→  /** Benign, self-healing staleness: totals equal AND the cache equals
   522→   *  ledger truth recomputed as-of its own last write — holds matured by
   523→   *  clock afterwards. Money correctness is NOT affected (payout
   524→   *  authorization recomputes truth transactionally). */
   525→  timeMaturityStale: WalletCacheDrift[];
   526→  txnCount: number;
   527→  walletCount: number;
   528→};
   529→
   530→/**
   531→ * Ledger invariants, classification-aware (mission: wallet-ledger drift).
   532→ *
   533→ * Wallet caches are event-driven derived values; hold credits mature by the
   534→ * CLOCK (availableAt <= now) with no triggering write, so a cache frozen
   535→ * since before a maturity is EXPECTED to disagree with truth-now while its
   536→ * TOTAL stays correct. The invariant therefore separates:
   537→ *   • unbalanced transactions            → hard, money correctness
   538→ *   • totals-differ / unexplained drift  → hard (CACHE_VALUE_DRIFT)
   539→ *   • explained time-maturity staleness  → benign, reported for visibility
   540→ * (Health checks must distinguish these — a stale display cache is not a
   541→ * lost-money event, and a nightly full-ledger scan on every poll is itself a
   542→ * hazard; see /api/health for the TTL-wrapped consumption.)
   543→ */
   544→export async function runLedgerInvariants(): Promise<LedgerInvariantResult> {
   545→  const problems: string[] = [];
   546→  const txns = await db.ledgerTransaction.findMany({ include: { lines: true } });
   547→  for (const t of txns) {
   548→    const debit = t.lines.filter((l) => l.direction === "debit").reduce((s, l) => s + l.amountMinor, 0);
   549→    const credit = t.lines.filter((l) => l.direction === "credit").reduce((s, l) => s + l.amountMinor, 0);
   550→    if (debit !== credit) problems.push(`txn ${t.id} (${t.txnType}) unbalanced: D${debit} C${credit}`);
   551→  }
   552→
   553→  const wallets = await db.wallet.findMany();
   554→  const payableCodes = wallets.map((w) => ACCOUNTS.sellerPayable(w.id));
   555→  const lines = payableCodes.length
   556→    ? await db.ledgerLine.findMany({
   557→        where: { accountCode: { in: payableCodes } },
   558→        select: { accountCode: true, direction: true, amountMinor: true, availableAt: true },
   559→      })
   560→    : [];
   561→  const byWallet = new Map<string, { direction: string; amountMinor: number; availableAt: Date | null }[]>();
   562→  for (const l of lines) {
   563→    // accountCode embeds the walletId: SELLER:{walletId}:PAYABLE
   564→    const wid = l.accountCode.slice("SELLER:".length, -":PAYABLE".length);
   565→    const arr = byWallet.get(wid) ?? [];
   566→    arr.push({ direction: l.direction, amountMinor: l.amountMinor, availableAt: l.availableAt });
   567→    byWallet.set(wid, arr);
   568→  }
   569→
   570→  const hardDrift: WalletCacheDrift[] = [];
   571→  const timeMaturityStale: WalletCacheDrift[] = [];
   572→  const now = new Date();
   573→  for (const w of wallets) {
   574→    const wLines = byWallet.get(w.id) ?? [];
   575→    let totalCredits = 0,
   576→      maturedNow = 0,
   577→      totalDebits = 0,
   578→      maturedAtWrite = 0,
   579→      maturedSinceWrite = 0;
   580→    const writeAt = w.updatedAt;
   581→    for (const l of wLines) {
   582→      if (l.direction === "credit") {
   583→        totalCredits += l.amountMinor;
   584→        const matured = !l.availableAt || l.availableAt <= now;
   585→        if (matured) maturedNow += l.amountMinor;
   586→        if (!l.availableAt || l.availableAt <= writeAt) maturedAtWrite += l.amountMinor;
   587→        else if (l.availableAt <= now) maturedSinceWrite += l.amountMinor;
   588→      } else {
   589→        totalDebits += l.amountMinor;
   590→      }
   591→    }
   592→    // Truth as-of NOW and as-of the cache's last write. Debits apply in both
   593→    // views (they are event-driven, not clock-driven).
   594→    const truthNowAvail = maturedNow - totalDebits;
   595→    const truthNowPend = totalCredits - maturedNow;
   596→    const truthAtWriteAvail = maturedAtWrite - totalDebits;
   597→    const truthAtWritePend = totalCredits - maturedAtWrite;
   598→
   599→    if (truthNowAvail === w.availableMinor && truthNowPend === w.pendingMinor) continue; // MATCH
   600→
   601→    const base: WalletCacheDrift = {
   602→      walletId: w.id,
   603→      cachedAvailable: w.availableMinor,
   604→      cachedPending: w.pendingMinor,
   605→      truthAvailable: truthNowAvail,
   606→      truthPending: truthNowPend,
   607→      availableDelta: w.availableMinor - truthNowAvail,
   608→      pendingDelta: w.pendingMinor - truthNowPend,
   609→      maturedSinceLastWrite: maturedSinceWrite,
   610→      lastCacheWriteAt: writeAt,
   611→    };
   612→    const totalsEqual = w.availableMinor + w.pendingMinor === truthNowAvail + truthNowPend;
   613→    if (totalsEqual && truthAtWriteAvail === w.availableMinor && truthAtWritePend === w.pendingMinor) {
   614→      timeMaturityStale.push(base); // benign, clock-driven, self-heals on next event/read
   615→    } else {
   616→      hardDrift.push(base);
   617→      problems.push(
   618→        `wallet ${w.id} cache drift (${totalsEqual ? "UNEXPLAINED staleness" : "CACHE_VALUE_DRIFT — totals differ"}): available cache=${w.availableMinor} truth=${truthNowAvail}, pending cache=${w.pendingMinor} truth=${truthNowPend}`
   619→      );
   620→    }
   621→  }
   622→  return { ok: problems.length === 0, problems, hardDrift, timeMaturityStale, txnCount: txns.length, walletCount: wallets.length };
   623→}

   654→// ------------------------------------------------------------- LEDGER-TRUTH STATEMENTS
   655→// INTEGRITY RULE (user directive): every number printed on a payout statement
   656→// must be DERIVED from ledger lines for the payout period — never decorated.
   657→// The lines always reconcile to the payout amount via an explicit
   658→// "Retained in wallet" balancing line, so Σ(lines) ≡ net payout.
   659→
   660→export type LedgerStatement = {
   661→  /** Parsed shape of the JSON string returned by buildLedgerStatementFromLedger
   662→   *  (Payout.statementJson column stores the serialized form). */
   663→  sellerName: string;
   664→  periodFrom: string;
   665→  periodTo: string;
   666→  orderCount: number;
   667→  reconciled: true;
   668→  generatedAt: string;
   669→  lines: { label: string; amountMinor: number }[];
   670→};
   671→
   672→/** Aggregate a wallet's ledger activity for [periodFrom, periodTo] and build a
   673→ *  payout statement whose lines sum EXACTLY to `payoutAmountMinor`.
   674→ *  Components (all from ledger truth):
   675→ *    • sales        — payable credits from order_paid (seller net) + matching
   676→ *                     commission-revenue credits (gross sales)
   677→ *    • commission   — commission-revenue credits from order_paid in the period
   678→ *    • refunds      — payable debits from refund txns + commission reversals
   679→ *    • adjustments  — net admin_adjustment movement on the payable account
   680→ *    • retained     — residual balancing line so Σ(lines) ≡ payout (holds, caps,
   681→ *                     prior-balance carry)                                   */
   682→export async function buildLedgerStatementFromLedger(params: {
   683→  walletId: string;
   684→  sellerName: string;
   685→  payoutAmountMinor: number;
   686→  orderCount: number;
   687→  periodFrom: Date;
   688→  periodTo: Date;
   689→}): Promise<string> {
   690→  const { walletId, sellerName, payoutAmountMinor, orderCount, periodFrom, periodTo } = params;
   691→  const payable = ACCOUNTS.sellerPayable(walletId);
   692→  const commissionAccount = ACCOUNTS.revenueCommission();
   693→
   694→  // Wallet-scope lines in the period, joined to their txn type. Period is
   695→  // evaluated on txn.occurredAt (business time) — rows are often backdated
   696→  // (seed) or posted later (payout T3 lands after the statement snapshot).
   697→  const lines = await db.ledgerLine.findMany({
   698→    where: {
   699→      OR: [{ accountCode: payable }, { accountCode: commissionAccount, walletId }],
   700→      txn: { occurredAt: { gte: periodFrom, lte: periodTo } },
   701→    },
   702→    select: { accountCode: true, direction: true, amountMinor: true, txn: { select: { txnType: true } } },
   703→  });
   704→
   705→  let salesNet = 0; // payable credits from order_paid
   706→  let commission = 0; // commission-revenue credits from order_paid
   707→  let refundSellerShare = 0; // payable debits from refund txns
   708→  let refundCommissionReversal = 0; // commission debits from refund txns
   709→  let adjustments = 0; // admin_adjustment net (credits − debits on payable)
   710→
   711→  for (const l of lines) {
   712→    const type = l.txn.txnType;
   713→    if (l.accountCode === payable) {
   714→      if (l.direction === "credit") {
   715→        if (type === "order_paid") salesNet += l.amountMinor;
   716→        else if (type === "admin_adjustment") adjustments += l.amountMinor;
   717→      } else {
   718→        if (type === "refund") refundSellerShare += l.amountMinor;
   719→        else if (type === "admin_adjustment") adjustments -= l.amountMinor;
   720→      }
   721→    } else if (l.accountCode === commissionAccount) {
   722→      if (l.direction === "credit") {
   723→        if (type === "order_paid") commission += l.amountMinor;
   724→      } else if (type === "refund") {
   725→        refundCommissionReversal += l.amountMinor;
   726→      }
   727→    }
   728→  }
   729→
   730→  const sales = salesNet + commission; // gross sales for the period
   731→  const refunds = refundSellerShare + refundCommissionReversal;
   732→  const netMovement = sales - commission - refunds + adjustments;
   733→  // Balancing line so Σ(statement lines) ≡ net payout: when the period's net
   734→  // movement exceeds the payout (pending holds, min-payout cap, prior carry),
   735→  // the difference is "retained"; when a payout draws on prior balance, it's
   736→  // a positive carry line. Either way the statement always reconciles.
   737→  const retained = payoutAmountMinor - netMovement;
   738→
   739→  const linesOut: { label: string; amountMinor: number }[] = [];
   740→  if (sales !== 0) linesOut.push({ label: "Sales (settled to available)", amountMinor: sales });
   741→  if (commission !== 0) linesOut.push({ label: "Platform commission", amountMinor: -commission });
   742→  if (refunds !== 0) linesOut.push({ label: "Refunds", amountMinor: -refunds });
   743→  if (adjustments !== 0) linesOut.push({ label: "Adjustments", amountMinor: adjustments });
   744→  if (retained !== 0) linesOut.push({ label: "Retained in wallet (hold/cap — not paid out)", amountMinor: retained });
   745→
   746→  return toJson({
   747→    sellerName,
   748→    periodFrom: periodFrom.toISOString(),
   749→    periodTo: periodTo.toISOString(),
   750→    orderCount,
   751→    reconciled: true,
   752→    generatedAt: new Date().toISOString(),
   753→    lines: linesOut,
   754→  });
   755→}
