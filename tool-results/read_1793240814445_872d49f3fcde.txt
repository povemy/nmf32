   715→// LATE-PAYMENT RECONCILIATION (mission: final operational gaps, Phase 2).
   716→// A payment.succeeded webhook that arrives AFTER the expiry sweep released the
   717→// group's stock is refused with 409 GROUP_EXPIRED (money never enters the
   718→// ledger). For a REAL PSP the buyer's funds are already captured at the
   719→// provider — this table is the durable, traceable REFUND OBLIGATION for those
   720→// funds: who must be refunded, how much (the webhook amount = captured funds),
   721→// and where the provider interaction stands. It is an obligation tracker, NOT
   722→// a ledger posting — ledger postings remain exclusively under claim-first
   723→// payment success. Duplicate webhooks converge on ONE row per (provider, ref)
   724→// via the unique key; retries can never duplicate the obligation.
   725→model PaymentReconciliation {
   726→  id                String    @id @default(cuid())
   727→  provider          String
   728→  // PaymentGroup.ref — the PSP checkout session the late payment belongs to.
   729→  providerRef       String
   730→  // PSP's own payment object id when the event carries one (real adapters
   731→  // MUST send it; the mock rails identify by providerRef).
   732→  providerPaymentId String?
   733→  // Amount captured at the provider per the verified webhook (this is what
   734→  // must be RETURNED — not the group total when they diverge; divergence is
   735→  // recorded in resolutionNote for operator verification).
   736→  amountMinor       Int
   737→  currency          String    @default("MYR")
   738→  reason            String    @default("late_payment_after_expiry")
   739→  // requested   — obligation recorded, no successful provider attempt yet
   740→  // processing   — attempt in flight (durable intent written pre-call)
   741→  // succeeded    — refund confirmed FROM the provider's authoritative record
   742→  // failed       — provider REJECTED the refund (permanent; operator item)
   743→  // unknown      — provider outcome pending/timed out (re-checked next tick)
   744→  status            String    @default("requested")
   745→  providerRefundId  String?
   746→  attempts          Int       @default(0)
   747→  lastAttemptAt     DateTime?
   748→  lastError         String?
   749→  resolutionNote    String?
   750→  createdAt         DateTime  @default(now())
   751→  updatedAt         DateTime  @updatedAt
   752→
   753→  @@unique([provider, providerRef])
   754→  @@index([status])
   755→  @@index([updatedAt])
   756→}

  1341→  showFreeShippingBar     Boolean  @default(true)
  1342→  freeShipThresholdMinor  Int      @default(10000)
  1343→  // AOV — upsell / cross-sell strip (auto best-sellers or manual picks)
  1344→  upsellEnabled           Boolean  @default(false)
