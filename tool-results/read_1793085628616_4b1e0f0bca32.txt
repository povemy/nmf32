    53→    53→function prettyValue(raw: string): string {
    54→    54→  try {
    55→    55→    return JSON.stringify(JSON.parse(raw), null, 0);
    56→    56→  } catch {
    57→    57→    return raw;
    58→    58→  }

    66→    66→  const [draft, setDraft] = useState("");
    67→    67→  const [saving, setSaving] = useState(false);
    68→    68→
    69→    69→  const openEdit = (r: SettingRow) => {
    70→    70→    setEditing(r);
    71→    71→    setDraft(r.valueJson ?? "");
    72→    72→  };
    73→    73→
    74→    74→  const save = async () => {
    75→    75→    if (!editing) return;
    76→    76→    try {
    77→    77→      JSON.parse(draft);
    78→    78→    } catch {
    79→    79→      toast.error("Invalid JSON", {
    80→    80→        description: "Use quoted strings, numbers, booleans or objects — e.g. \"friday\" or 1000.",
    81→    81→      });
    82→    82→      return;
    83→    83→    }
    84→    84→    setSaving(true);
    85→    85→    try {
    86→    86→      await apiSend("/api/admin/settings", "PUT", { settings: [{ key: editing.key, valueJson: draft }] });

    88→    88→      setEditing(null);
    89→    89→      void refetch();
    90→    90→    } catch (e) {
    91→    91→      toast.error("Save failed", { description: e instanceof Error ? e.message : String(e) });
