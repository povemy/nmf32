   323→// FIN-3-c — seller in-app notification bell. Feeds from the NotificationLog
   324→// rows (channel "inapp") the finance flows already write for the seller email:
   325→// order.paid (FIN-2-b) + payout requested/approved/paid/failed/rejected
   326→// (FIN-2-a/FIN-3-b). The shopper bell is a different subsystem (Notification
   327→// model via /api/me/notifications) and is intentionally untouched.
   328→
   329→type SellerNotificationRow = { id: string; subject: string; createdAt: string; event: string };
   330→
   331→type SellerNotificationsPayload = { notifications: SellerNotificationRow[]; recentCount: number };
   332→
   333→/** House scrollbar style (same treatment as the shopper bell panel). */
   334→const SELLER_BELL_SCROLL =
   335→  "max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent";
   336→
   337→/** 24h recency = client-side "new" heuristic: NotificationLog has no read /
   338→ *  seen column (adding fake read-tracking was out of scope), so "within the
   339→ *  last 24h" is the honest approximation of "unread". */
   340→const RECENT_WINDOW_MS = 24 * 60 * 60 * 1000;
   341→
   342→function SellerNotificationBell() {
   343→  const [open, setOpen] = useState(false);
   344→  const [data, setData] = useState<SellerNotificationsPayload | null>(null);
   345→  const [loading, setLoading] = useState(false);
   346→  const [errored, setErrored] = useState(false);
   347→
   348→  const load = useCallback(async () => {
   349→    setLoading(true);
   350→    try {
   351→      const d = await apiGet<SellerNotificationsPayload>("/api/store/notifications");
   352→      setData(d);
   353→      setErrored(false);
   354→    } catch (e) {
   355→      // Silent by design — a failed bell must never break the seller shell:
   356→      // keep the last data (no badge flicker), warn for debugging only.
   357→      setErrored(true);
   358→      console.warn("[seller-bell] notifications fetch failed", e instanceof Error ? e.message : e);
   359→    } finally {
   360→      setLoading(false);
   361→    }
   362→  }, []);
   363→
   364→  // Badge freshness without aggressive polling: fetch once on mount (dot only
   365→  // needs recentCount) and refetch whenever the popover opens.
   366→  useEffect(() => {
   367→    void load();
   368→  }, [load]);
   369→
   370→  const onOpenChange = (v: boolean) => {
   371→    setOpen(v);
   372→    if (v) void load();
   373→  };
   374→
   375→  const recent = data?.recentCount ?? 0;
   376→  const rows = data?.notifications ?? null;
   377→
   378→  return (
   379→    <Popover open={open} onOpenChange={onOpenChange}>
   380→      <PopoverTrigger asChild>
   381→        <Button
   382→          variant="ghost"
   383→          size="icon"
   384→          className="relative rounded-full border border-slate-200 bg-white text-slate-500 shadow-sm hover:border-teal-400 hover:bg-white hover:text-teal-700"
   385→          aria-label="Seller notifications"
   386→          aria-haspopup="dialog"
   387→          aria-expanded={open}
   388→        >
   389→          <Bell className="h-4 w-4" aria-hidden="true" />
   390→          {recent > 0 ? (
   391→            <span
   392→              className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-amber-500 ring-2 ring-white"
   393→              aria-hidden="true"
   394→            />
   395→          ) : null}
   396→        </Button>
   397→      </PopoverTrigger>
   398→      <PopoverContent align="end" className="w-80 rounded-2xl p-0 sm:w-96" role="dialog" aria-label="Seller notifications panel">
   399→        <div className="flex items-center gap-2 border-b border-slate-200/80 px-4 py-3">
   400→          <h2 className="text-sm font-bold text-slate-900">Notifications</h2>
   401→          {recent > 0 ? (
   402→            <span className="rounded-full bg-amber-50 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 tabular-nums">
   403→              {recent} new
   404→            </span>
   405→          ) : null}
   406→        </div>
   407→
   408→        {loading && !rows ? (
   409→          <div className="space-y-3 p-4" aria-label="Loading notifications">
   410→            {Array.from({ length: 3 }).map((_, i) => (
   411→              <div key={i} className="flex items-center gap-3">
   412→                <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   413→                <div className="flex-1 space-y-1.5">
   414→                  <Skeleton className="h-3.5 w-3/4" />
   415→                  <Skeleton className="h-3 w-1/3" />
   416→                </div>
   417→              </div>
   418→            ))}
   419→          </div>
   420→        ) : !rows || rows.length === 0 ? (
   421→          <div className="flex flex-col items-center gap-2 px-6 py-10 text-center" aria-label="No notifications">
   422→            <span className="rounded-full bg-muted p-4">
   423→              <Bell className="h-6 w-6 text-muted-foreground" aria-hidden="true" />
   424→            </span>
   425→            <p className="text-sm font-semibold">{errored && !data ? "Notifications unavailable" : "No notifications yet"}</p>
   426→            <p className="max-w-[16rem] text-xs text-muted-foreground">
   427→              {errored && !data
   428→                ? "Couldn't load right now — it will retry next time you open the bell."
   429→                : "Payout and order updates will appear here."}
   430→            </p>
   431→          </div>
   432→        ) : (
   433→          <ul className={cn("divide-y divide-slate-100", SELLER_BELL_SCROLL)} aria-label="Notification list">
   434→            {rows.map((n) => {
   435→              const isNew = Date.now() - new Date(n.createdAt).getTime() < RECENT_WINDOW_MS;
   436→              return (
   437→                <li key={n.id} className="flex items-start gap-3 px-4 py-3">
   438→                  <div className="min-w-0 flex-1">
   439→                    <p className="truncate text-[13px] font-medium" title={n.subject}>
   440→                      {n.subject}
   441→                    </p>
   442→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   443→                      <span title={new Date(n.createdAt).toLocaleString("en-MY")}>{ageOf(n.createdAt)}</span>
   444→                      <span aria-hidden="true"> · </span>
   445→                      <span className="lowercase">{n.event}</span>
   446→                    </p>
   447→                  </div>
   448→                  {isNew ? (
   449→                    <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-amber-500" role="img" aria-label="Recent" />
   450→                  ) : null}
   451→                </li>
   452→              );
   453→            })}
   454→          </ul>
   455→        )}
   456→      </PopoverContent>
   457→    </Popover>
   458→  );
   459→}

   463→/** Total unread shopper messages — powers the "Messages" nav badge. */
   464→function useSellerChatUnread(active: SellerNav["view"]) {
   465→  const [unread, setUnread] = useState(0);
   466→  useEffect(() => {
   467→    let alive = true;
   468→    const load = () =>
   469→      apiGet<{ total: number }>("/api/store/chat/unread")
   470→        .then((d) => alive && setUnread(d.total ?? 0))
   471→        .catch(() => {});
   472→    load();
   473→    const t = setInterval(load, 15000);
   474→    return () => {
   475→      alive = false;
   476→      clearInterval(t);
   477→    };
   478→  }, [active]);
   479→  // instant re-sync when a thread is read inside the messages view (the view
   480→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   481→  useEffect(() => {
   482→    const sync = () =>
   483→      apiGet<{ total: number }>("/api/store/chat/unread")
   484→        .then((d) => setUnread(d.total ?? 0))
   485→        .catch(() => {});
   486→    window.addEventListener("nomofee:chat-sync", sync);
   487→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   488→  }, []);
   489→  return unread;
   490→}
