   113→/** Total unread seller messages — powers the "My messages" header badge.
   114→ *  Polls lightly + refreshes when the messages view is opened/left. */
   115→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
   116→  const [unread, setUnread] = useState(0);
   117→  useEffect(() => {
   118→    if (!shopperSignedIn) {
   119→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
   120→      setUnread(0);
   121→      return;
   122→    }
   123→    let alive = true;
   124→    const load = () =>
   125→      fetchJson<{ total: number }>("/api/chat/unread")
   126→        .then((d) => alive && setUnread(d.total ?? 0))
   127→        .catch(() => {});
   128→    load();
   129→    const t = setInterval(load, 15000);
   130→    return () => {
   131→      alive = false;
   132→      clearInterval(t);
   133→    };
   134→  }, [shopperSignedIn]);
   135→  // instant re-sync when a thread is read inside the messages view (the view
   136→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   137→  useEffect(() => {
   138→    if (!shopperSignedIn) return;
   139→    const sync = () =>
   140→      fetchJson<{ total: number }>("/api/chat/unread")
   141→        .then((d) => setUnread(d.total ?? 0))
   142→        .catch(() => {});
   143→    window.addEventListener("nomofee:chat-sync", sync);
   144→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   145→  }, [shopperSignedIn]);
   146→  // opening (or leaving) the messages view should immediately re-sync the badge
   147→  useEffect(() => {
   148→    if (!shopperSignedIn) return;
   149→    const t = setTimeout(
   150→      () =>
   151→        fetchJson<{ total: number }>("/api/chat/unread")
   152→          .then((d) => setUnread(d.total ?? 0))
   153→          .catch(() => {}),
   154→      viewName === "messages" ? 600 : 0
   155→    );
   156→    return () => clearTimeout(t);
   157→  }, [shopperSignedIn, viewName]);
   158→  return unread;
   159→}

   161→// ---- platform chrome config (global footer visibility) ---------------------
   162→// Admin → Settings → "Show global footer" (Setting ui.footer_visible) governs
   163→// the shopper footer. Public endpoint + 60 s module cache: toggling takes
   164→// effect on the next app load without per-navigation refetch chatter.
   165→let appConfigCache: { at: number; footerVisible: boolean } | null = null;
   166→
   167→function useFooterVisible() {
   168→  const [visible, setVisible] = useState(true);
   169→  useEffect(() => {
   170→    if (appConfigCache && Date.now() - appConfigCache.at < 60_000) {
   171→      // eslint-disable-next-line react-hooks/set-state-in-effect -- cache hit syncs an external module cache; the fresh-fetch path below sets state in a callback
   172→      setVisible(appConfigCache.footerVisible);
   173→      return;
   174→    }
   175→    let cancelled = false;
   176→    fetch("/api/sf/app-config", { cache: "no-store" })
   177→      .then((r) => (r.ok ? r.json() : null))
   178→      .then((b) => {
   179→        const v = (b as { data?: { footerVisible?: boolean } } | null)?.data?.footerVisible;
   180→        if (typeof v === "boolean" && !cancelled) {
   181→          appConfigCache = { at: Date.now(), footerVisible: v };
   182→          setVisible(v);
   183→        }
   184→      })
   185→      .catch(() => undefined);
   186→    return () => {
   187→      cancelled = true;
   188→    };
   189→  }, []);
   190→  return visible;
   191→}

   206→  // viewRef mirrors `view` so nav() can push the PREVIOUS view without reading
   207→  // state inside the setState updater (StrictMode-safe).
   208→  const viewStack = useRef<ShopperView[]>([]);
   209→  const viewRef = useRef<ShopperView>({ name: "market" });
   210→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   211→  // intentionally NOT part of ShopperView — a synthetic render override.
   212→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   213→  const restored = useRef(false);
   214→
   215→  // ---- mount: activeStoreSlug override, else restore last view --------------
   216→  useEffect(() => {
   217→    if (restored.current) return;
   218→    restored.current = true;
   219→    // claim deep-link wins over everything
   220→    try {
   221→      const params = new URLSearchParams(window.location.search);
   222→      const cj = params.get("claim_job");
   223→      const ct = params.get("claim_token");
   224→      if (cj && ct) {
   225→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   226→        setClaimLink({ jobId: cj, token: ct });
   227→        setReady(true);
   228→        return;
   229→      }
   230→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   231→      const sf = params.get("sf");
   232→      if (sf) {
   233→        const sp = params.get("sp");
   234→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   235→        setReady(true);
   236→        return;
   237→      }
   238→    } catch {
   239→      /* ignore */
   240→    }
   241→    if (activeStoreSlug) {
   242→      setView({ name: "storefront", slug: activeStoreSlug });
   243→      setActiveStoreSlug(null);
   244→    } else {
   245→      try {
   246→        const raw = window.localStorage.getItem(VIEW_KEY);
   247→        if (raw) {
   248→          const v = JSON.parse(raw) as ShopperView;
   249→          if (v && typeof v.name === "string") setView(v);
   250→        }
   251→      } catch {
   252→        /* ignore */
   253→      }
   254→    }
   255→    setReady(true);
   256→  }, [activeStoreSlug, setActiveStoreSlug]);
   257→
   258→  // ---- nav helper: persist + scroll to top ----------------------------------
   259→  useEffect(() => {
   260→    viewRef.current = view;
   261→  }, [view]);
   262→
   263→  const nav = useCallback((v: ShopperView) => {
   264→    if (viewRef.current.name !== v.name || JSON.stringify(viewRef.current) !== JSON.stringify(v)) {
   265→      viewStack.current.push(viewRef.current);
   266→      if (viewStack.current.length > 25) viewStack.current.shift();
   267→    }
   268→    setView(v);
   269→    try {
   270→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   271→    } catch {
   272→      /* ignore */
   273→    }
   274→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   275→  }, []);
   276→
   277→  const back = useCallback(() => {
   278→    const prev = viewStack.current.pop() ?? { name: "market" as const };
   279→    setView(prev);
   280→    try {
   281→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(prev));
   282→    } catch {
   283→      /* ignore */
   284→    }
   285→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   286→  }, []);
   287→
   288→  // refresh storefront/store-page when slug changes externally
   289→  useEffect(() => {
   290→    if (!ready) return;
   291→    if (activeStoreSlug) {
   292→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   293→      setView({ name: "storefront", slug: activeStoreSlug });
   294→      setActiveStoreSlug(null);
   295→    }
   296→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   297→
   298→  const ctx = useMemo<ShopperCtx>(
   299→    () => ({
   300→      nav,
   301→      back,
   302→      openCart: (storeSlug?: string) => {
   303→        setCartStoreSlug(storeSlug ?? null);
   304→        setCartOpen(true);
   305→      },
   306→    }),

   308→  );
   309→
   310→  // ---- global inbox presence (ONE socket on every view) ---------------------
   311→  // Away from the messages page nothing used to listen for pushes, so the
   312→  // Message/Chat tab badges only moved on a 15s poll — incoming messages felt
   313→  // silent. This keeps the user's inbox room joined on every view: a push
   314→  // fires "nomofee:chat-sync" (all badge hooks re-sync instantly) and, when
   315→  // the user is NOT on the messages page, raises a toast with a jump-in link.

   325→          description: preview,
   326→          action: msg.threadId
   327→            ? { label: "Open chat", onClick: () => nav({ name: "messages", threadId: msg.threadId }) }
   328→            : undefined,
   329→        });
   330→      }
   331→    },
   332→    onActivity: () => window.dispatchEvent(new Event("nomofee:chat-sync")),
   333→  });
   334→
   335→  const showChrome = !isStoreChrome(view);
   336→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   337→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   338→  // Admin-controlled global footer switch (default ON; checkout never shows it).
   339→  const footerVisible = useFooterVisible();
   340→
   341→  return (
   342→    <div className="flex-1 flex flex-col min-h-screen">
   343→      <Toaster position="top-center" richColors />
   344→      <ShopperAuthDialog />
   345→

   348→          view={view}

   351→          search={headerSearch}

   355→
   356→      <AnimatePresence mode="wait">
   357→        <motion.main

"" : view.name === "me" ? view.tab ?? "" : ""}`}

   365→          {claimLink ? null : (
   366→            <>
   367→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   368→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   369→              {view.name === "category" ? (
   370→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   371→              ) : null}
   372→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   373→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}

   415→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   416→  const me = useApp((s) => s.me);
   417→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   418→  const signedIn = Boolean(me.loaded && me.shopper);
   419→  const { count } = useCart();
   420→  const unreadChats = useChatUnread(signedIn, view.name);
   421→
   422→  const items = [
   423→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   424→    {
   425→      key: "orders",
   426→      label: "Orders",
   427→      icon: Package,
   428→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   429→      badge: 0,
   430→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   431→    },
   432→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   433→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   434→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   435→  ];
   436→
   437→  return (
   438→    <nav
   439→      aria-label="Bottom navigation"
   440→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   441→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   442→    >
   443→      <div className="mx-auto grid max-w-lg grid-cols-5">
   444→        {items.map((it) => {
   445→          const Icon = it.icon;
   446→          return (
   447→            <button
   448→              key={it.key}
   449→              type="button"
   450→              onClick={it.onClick}
   451→              aria-label={it.label}
   452→              aria-current={it.active ? "page" : undefined}
   453→              className={cn(
   454→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   455→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   456→              )}
   457→            >
   458→              <span className="relative">
   459→                <Icon className="h-5 w-5" aria-hidden="true" />
   460→                {it.badge > 0 ? (
   461→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   462→                    {it.badge > 9 ? "9+" : it.badge}
   463→                  </span>
   464→                ) : null}
   465→              </span>
   466→              {it.label}
   467→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   468→            </button>
   469→          );
   470→        })}
   471→      </div>
   472→    </nav>
   473→  );
   474→}

   884→function ShopperAuthDialog() {
   885→  const open = useApp((s) => s.shopperAuthOpen);
   886→  const close = useApp((s) => s.closeShopperAuth);
   887→  const after = useApp((s) => s.afterShopperAuth);
   888→  const refreshMe = useApp((s) => s.refreshMe);
   889→  const [tab, setTab] = useState("login");
   890→  const [busy, setBusy] = useState(false);
   891→
   892→  useEffect(() => {
   893→    if (open) setTab("login");
   894→  }, [open]);
   895→
   896→  const finish = async () => {
   897→    await refreshMe();
   898→    close();
   899→    after?.();
   900→  };
   901→
   902→  const submitLogin = async (e: React.FormEvent<HTMLFormElement>) => {
   903→    e.preventDefault();
   904→    if (busy) return;
   905→    const f = new FormData(e.currentTarget);
   906→    setBusy(true);
   907→    try {
   908→      const res = await postJson<{ token?: string }>("/api/auth/shopper/login", { email: f.get("email"), password: f.get("password") });
   909→      setSessionToken(res?.token); // localStorage bridge for cookie-less contexts
   910→      toast.success("Welcome back!");
   911→      await finish();
   912→    } catch (err) {
   913→      toast.error(err instanceof Error ? err.message : "Sign in failed");
   914→    } finally {
   915→      setBusy(false);
   916→    }
   917→  };
   918→
   919→  const submitRegister = async (e: React.FormEvent<HTMLFormElement>) => {
   920→    e.preventDefault();
   921→    if (busy) return;
   922→    const f = new FormData(e.currentTarget);
   923→    setBusy(true);
   924→    try {
   925→      const res = await postJson<{ token?: string }>("/api/auth/shopper/register", {
   926→        fullName: f.get("fullName"),
   927→        email: f.get("email"),
   928→        password: f.get("password"),
   929→      });
   930→      setSessionToken(res?.token); // localStorage bridge for cookie-less contexts
   931→      toast.success("Account created!", { description: "You can now chat with sellers and track orders." });
   932→      await finish();
   933→    } catch (err) {
   934→      toast.error(err instanceof Error ? err.message : "Registration failed");
   935→    } finally {
   936→      setBusy(false);
   937→    }
   938→  };
   939→
   940→  return (
   941→    <Dialog open={open} onOpenChange={(o) => (!o ? close() : null)}>
   942→      <DialogContent className="max-w-sm">
   943→        <DialogHeader>
   944→          <DialogTitle className="flex items-center gap-2">
   945→            <User className="h-5 w-5 text-teal-700" /> NOMOFEE shopper account
   946→          </DialogTitle>
   947→          <DialogDescription>Sign in to chat with sellers, or create a free account in seconds.</DialogDescription>
   948→        </DialogHeader>
   949→        <Tabs value={tab} onValueChange={setTab}>
   950→          <TabsList className="grid w-full grid-cols-2">
   951→            <TabsTrigger value="login">Sign in</TabsTrigger>
   952→            <TabsTrigger value="register">Create account</TabsTrigger>
   953→          </TabsList>
   954→          <TabsContent value="login">
   955→            <form onSubmit={submitLogin} className="space-y-3 pt-2">
   956→              <Input name="email" type="email" required placeholder="you@example.com" autoComplete="email" className="min-h-[44px]" aria-label="Email" />
   957→              <Input name="password" type="password" required placeholder="Password" autoComplete="current-password" className="min-h-[44px]" aria-label="Password" />
   958→              <Button type="submit" disabled={busy} className="w-full min-h-[44px] bg-brand-gradient-deep text-white hover:brightness-110 gap-2">
   959→                {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <LogIn className="h-4 w-4" />} Sign in
   960→              </Button>
   961→            </form>
   962→          </TabsContent>
   963→          <TabsContent value="register">
   964→            <form onSubmit={submitRegister} className="space-y-3 pt-2">
   965→              <Input name="fullName" required placeholder="Full name" autoComplete="name" className="min-h-[44px]" aria-label="Full name" />
   966→              <Input name="email" type="email" required placeholder="you@example.com" autoComplete="email" className="min-h-[44px]" aria-label="Email" />
   967→              <Input name="password" type="password" required minLength={8} placeholder="Password (min 8 characters)" autoComplete="new-password" className="min-h-[44px]" aria-label="Password" />
   968→              <Button type="submit" disabled={busy} className="w-full min-h-[44px] bg-brand-gradient-deep text-white hover:brightness-110 gap-2">
   969→                {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <User className="h-4 w-4" />} Create free account
   970→              </Button>

  1790→  function bumpQty(itemId: string, nextQty: number) {
  1791→    const it = cartSnapshot().items.find((x) => x.id === itemId);
  1792→    if (!it) return;
  1793→    // Client-side guard rails: 1..stock (moq compliance is the stepper's min;
  1794→    // checkout/create re-validates authoritatively either way).
  1795→    const clamped = Math.max(1, Math.min(nextQty, Math.max(1, it.stock)));
  1796→    if (clamped === it.qty) return;
  1797→    if (!qtyTimers.current.has(itemId)) qtyRevertPoint.current.set(itemId, it.qty);
  1798→    mutateCartItems((items) =>
  1799→      items.map((x) => (x.id === itemId ? { ...x, qty: clamped, lineTotalMinor: x.unitPriceMinor * clamped } : x))
  1800→    );
  1801→    const t = qtyTimers.current.get(itemId);
  1802→    if (t) clearTimeout(t);
  1803→    qtyTimers.current.set(
  1804→      itemId,
  1805→      setTimeout(() => {
  1806→        qtyTimers.current.delete(itemId);
  1807→        const intended = cartSnapshot().items.find((x) => x.id === itemId)?.qty ?? clamped;
  1808→        patchJson<{ cart: CartT }>(`/api/cart/items/${itemId}`, { qty: intended })
  1809→          .then((res) => {
  1810→            qtyRevertPoint.current.set(itemId, intended);
  1811→            if (qtyTimers.current.has(itemId)) return; // newer taps pending — they will settle
  1812→            const serverItem = res.cart.items.find((x) => x.id === itemId);
  1813→            const localItem = cartSnapshot().items.find((x) => x.id === itemId);
  1814→            // The server clamps on stock changes (another device bought the
  1815→            // last units mid-burst). Adopt its number if we're still pristine.
  1816→            if (serverItem && localItem && serverItem.qty !== localItem.qty) {
  1817→              mutateCartItems((items) =>
  1818→                items.map((x) =>
  1819→                  x.id === itemId
  1820→                    ? { ...x, qty: serverItem.qty, stock: serverItem.stock, moq: serverItem.moq, lineTotalMinor: serverItem.lineTotalMinor }
  1821→                    : x
  1822→                )
  1823→              );
  1824→              toast.warning("Quantity adjusted", { description: `${localItem.title}: only ${serverItem.stock} left in stock` });
  1825→            }
  1826→          })
  1827→          .catch(() => {
  1828→            if (qtyTimers.current.has(itemId)) return; // a newer PATCH owns the outcome
  1829→            const revertTo = qtyRevertPoint.current.get(itemId);
  1830→            mutateCartItems((items) =>
  1831→              items.map((x) => {
  1832→                if (x.id !== itemId) return x;
  1833→                if (revertTo == null || x.qty !== intended) return x; // guarded revert
  1834→                return { ...x, qty: revertTo, lineTotalMinor: x.unitPriceMinor * revertTo };
  1835→              })
  1836→            );
  1837→            toast.error("Could not update quantity");
