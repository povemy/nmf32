   380→function RequestPayoutDialog({
   381→  open,
   382→  onOpenChange,
   383→  availableMinor,
   384→  minPayoutMinor,
   385→  bankAccounts,
   386→  onDone,
   387→}: {
   388→  open: boolean;
   389→  onOpenChange: (o: boolean) => void;
   390→  availableMinor: number;
   391→  minPayoutMinor: number;
   392→  bankAccounts: BankAccount[];
   393→  onDone: () => void;
   394→}) {
   395→  // Fresh mount on open (parent conditionally renders it) — defaults to the
   396→  // full available balance without a setState-in-effect.
   397→  const [amountMinor, setAmountMinor] = useState<number | null>(availableMinor);
   398→  const [busy, setBusy] = useState(false);
   399→
   400→  const bank = bankAccounts.find((b) => b.verified) ?? bankAccounts[0] ?? null;
   401→  const amount = amountMinor ?? 0;
   402→  const valid = amount >= minPayoutMinor && amount <= availableMinor;
   403→
   404→  const submit = async () => {
   405→    if (!valid) return;
   406→    setBusy(true);
   407→    const ok = await runMutation(
   408→      () => apiSend("POST", "/api/store/payouts", { amountMinor: amount }),
   409→      "Payout request submitted — NOMOFEE finance will review it"
   410→    );
   411→    setBusy(false);
   412→    if (ok) {
   413→      onOpenChange(false);
   414→      onDone();
   415→    }
   416→  };
   417→
   418→  return (
   419→    <Dialog open={open} onOpenChange={onOpenChange}>
   420→      <DialogContent className="max-w-md">
   421→        <DialogHeader>
   422→          <DialogTitle>Request a payout</DialogTitle>
   423→        </DialogHeader>
   424→        <div className="space-y-3.5">
   425→          <div className="space-y-1.5">
   426→            <Label htmlFor="pr-amount">Amount *</Label>
   427→            <MoneyInput id="pr-amount" valueMinor={amountMinor} onChangeMinor={setAmountMinor} placeholder="0.00" />
   428→            <div className="flex items-center justify-between text-xs text-muted-foreground">
   429→              <span>Min {formatMinor(minPayoutMinor)}</span>
   430→              <button
   431→                type="button"
   432→                className="font-semibold text-teal-700 hover:underline"
   433→                onClick={() => setAmountMinor(availableMinor)}
   434→              >
   435→                Use full balance ({formatMinor(availableMinor)})
   436→              </button>
   437→            </div>
   438→          </div>
   439→          <div className="rounded-xl border bg-muted/30 p-3 text-xs text-muted-foreground">
   440→            <p className="flex items-center gap-1.5 font-semibold text-foreground">
   441→              <Landmark className="h-3.5 w-3.5" /> Paying to
   442→            </p>
   443→            {bank ? (
   444→              <p className="mt-1">
   445→                {bank.bankName} · {bank.holderName} · {bank.accountNumberMasked}
   446→                {bank.verified ? " (verified)" : " (verification pending)"}
   447→              </p>
   448→            ) : (
   449→              <p className="mt-1">No bank account on file — add one first.</p>
   450→            )}
   451→            <p className="mt-2">
   452→              Your funds stay in the available balance while the request is reviewed. NOMOFEE transfers to this account once approved.
   453→            </p>
   454→          </div>
   455→          <Button onClick={submit} disabled={busy || !valid || !bank} className="w-full min-h-[44px] gap-2">
   456→            {busy ? <Wallet2 className="h-4 w-4 animate-pulse" /> : <Banknote className="h-4 w-4" />}
   457→            Submit request
   458→          </Button>
   459→        </div>
   460→      </DialogContent>
   461→    </Dialog>
   462→  );
   463→}

   467→function AddBankDialog({ open, onOpenChange, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; onDone: () => void }) {
   468→  const [holderName, setHolderName] = useState("");
   469→  const [bankName, setBankName] = useState("");
   470→  const [accountNumber, setAccountNumber] = useState("");
   471→  const [busy, setBusy] = useState(false);
   472→
   473→  const submit = async () => {
   474→    const digits = accountNumber.replace(/\s+/g, "");
   475→    if (!holderName.trim() || !bankName || digits.length < 4) {
   476→      return runMutation(() => Promise.reject(new Error("Holder, bank and account number (min 4 digits) are required")), "");
   477→    }
   478→    setBusy(true);
   479→    const ok = await runMutation(
   480→      () => apiSend("POST", "/api/store/bank-accounts", { holderName: holderName.trim(), bankName, accountNumber }),
   481→      "Bank account added — verification by admin pending"
   482→    );
   483→    setBusy(false);
   484→    if (ok) {
   485→      setHolderName("");
   486→      setBankName("");
   487→      setAccountNumber("");
   488→      onOpenChange(false);
   489→      onDone();
   490→    }
   491→  };
   492→
   493→  return (
   494→    <Dialog open={open} onOpenChange={onOpenChange}>
   495→      <DialogContent className="max-w-md">
   496→        <DialogHeader>
   497→          <DialogTitle>Add bank account</DialogTitle>
   498→        </DialogHeader>
   499→        <div className="space-y-3.5">
   500→          <div className="space-y-1.5">
   501→            <Label htmlFor="wa-holder">Account holder name *</Label>
   502→            <Input id="wa-holder" value={holderName} onChange={(e) => setHolderName(e.target.value)} placeholder="Lumina Living Sdn Bhd" />
   503→          </div>
   504→          <div className="space-y-1.5">
   505→            <Label>Bank *</Label>

   587→function LedgerActivityDetail({
   588→  activityId,

   590→  onNavigate,

   597→
   598→  if (loading) return <RowsSkeleton rows={6} />;
   599→  if (error || !data) {
   600→    return (
   601→      <div className="space-y-3">
   602→        <Button variant="ghost" onClick={onBack} className="gap-1.5">
   603→          <ArrowLeft className="h-4 w-4" /> Back to wallet

   608→  }
   609→
   610→  const { line, txn, walletAmountMinor, related, journal } = data;
   611→  const myDirection = walletAmountMinor >= 0 ? "credit" : "debit";
   612→  const walletAmount = Math.abs(walletAmountMinor);
   613→
   614→  return (
   615→    <div className="space-y-4">

   627→          {myDirection} {formatMinor(walletAmount)}
   628→        </span>
   629→        <span className="text-xs text-muted-foreground">{fmtDateTime(txn.occurredAt)}</span>
   630→      </div>
   631→
   632→      {/* what is this activity about */}
   633→      <Card>
   634→        <CardHeader className="pb-2">
   635→          <CardTitle className="flex items-center gap-2 text-sm">

   645→                {myDirection === "credit" ? "+" : "−"}
   646→                {formatMinor(walletAmount)}

   650→              <span className="text-xs text-muted-foreground">Clearing hold</span>
