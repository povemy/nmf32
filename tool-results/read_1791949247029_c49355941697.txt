   111→function makeIdemKey(): string {
   112→  try {
   113→    return crypto.randomUUID();
   114→  } catch {
   115→    return `idem_${Date.now()}_${Math.random().toString(36).slice(2)}`;
   116→  }
   117→}
   118→
   119→function loadDraft(): DraftT {
   120→  const savedContact: Record<string, string> = (() => {
   121→    if (typeof window === "undefined") return {};
   122→    try {
   123→      return JSON.parse(window.localStorage.getItem(CONTACT_KEY) ?? "{}") as Record<string, string>;
   124→    } catch {
   125→      return {};
   126→    }
   127→  })();
   128→  const base: DraftT = {
   129→    phase: 1,
   130→    name: savedContact.name ?? "",
   131→    email: savedContact.email ?? "",
   132→    phone: savedContact.phone ?? "",
   133→    addressId: null,
   134→    shippingRateId: "standard",
   135→    provider: "fpx_toyyibpay",
   136→    couponCode: "",
   137→    note: "",
   138→    agreed: false,
   139→    idemKey: makeIdemKey(),
   140→  };
   141→  if (typeof window === "undefined") return base;
   142→  try {
   143→    const raw = JSON.parse(window.sessionStorage.getItem(DRAFT_KEY) ?? "{}") as Partial<DraftT>;
   144→    return {
   145→      ...base,
   146→      ...raw,
   147→      phase: raw.phase === 2 || raw.phase === 3 ? raw.phase : 1,
   148→      idemKey: raw.idemKey || base.idemKey, // stable across retries → idempotent resubmits
   149→    };
   150→  } catch {
   151→    return base;
   152→  }
   153→}
   154→
   155→function persistDraft(d: DraftT) {
   156→  if (typeof window !== "undefined") window.sessionStorage.setItem(DRAFT_KEY, JSON.stringify(d));
   157→}
   158→
   159→function clearDraft() {
   160→  if (typeof window !== "undefined") window.sessionStorage.removeItem(DRAFT_KEY);
   161→}

   204→const STEPS = [
   205→  { n: 1 as const, label: "Information", icon: User },
   206→  { n: 2 as const, label: "Delivery", icon: Truck },
   207→  { n: 3 as const, label: "Payment", icon: CreditCard },

   209→
   210→function Stepper({ phase, onJump }: { phase: 1 | 2 | 3; onJump: (n: 1 | 2) => void }) {
   211→  return (
   212→    <ol className="flex min-w-0 items-center gap-2 overflow-hidden sm:gap-3" aria-label="Checkout progress">
   213→      {STEPS.map((s, i) => {
   214→        const done = phase > s.n;
   215→        const current = phase === s.n;
   216→        return (
   217→          <li key={s.n} className="flex min-w-0 flex-1 items-center gap-2 last:flex-none sm:gap-3">
   218→            <button
   219→              type="button"
   220→              disabled={!done}
   221→              onClick={() => done && onJump(s.n)}
   222→              className="flex min-w-0 items-center gap-2 disabled:cursor-default"
   223→              aria-current={current ? "step" : undefined}
   224→              aria-label={`Step ${s.n}: ${s.label}${done ? " (completed — click to edit)" : ""}`}
   225→            >
   226→              <span
   227→                className={cn(
   228→                  "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors",
   229→                  done && "border-transparent bg-brand-gradient text-white shadow-sm",
   230→                  current && "border-teal-600 bg-teal-50 text-teal-700",
   231→                  !done && !current && "border-muted-foreground/25 bg-card text-muted-foreground"
   232→                )}
   233→              >
   234→                {done ? <Check className="h-4 w-4" aria-hidden="true" /> : s.n}
   235→              </span>
   236→              <span
   237→                className={cn(
   238→                  "truncate text-xs font-bold sm:text-sm",
   239→                  current ? "text-teal-800" : done ? "text-foreground" : "text-muted-foreground"
   240→                )}
   241→              >
   242→                {s.label}
   243→              </span>
   244→            </button>
   245→            {i < STEPS.length - 1 ? (
   246→              <span className={cn("h-0.5 min-w-4 flex-1 rounded-full", phase > s.n ? "bg-brand-gradient" : "bg-border")} aria-hidden="true" />
   247→            ) : null}
   248→          </li>
   249→        );
   250→      })}
   251→    </ol>
   252→  );
   253→}

   305→function validateForm(form: typeof emptyForm): Record<string, string | null> {
   306→  return {
   307→    name: form.name.trim().length >= 2 ? null : "Recipient name is required",
   308→    phone: !form.phone.trim() || PHONE_RE.test(form.phone.trim()) ? null : "Enter a valid phone number",
   309→    line1: form.line1.trim() ? null : "Street address is required",
   310→    city: form.city.trim() ? null : "City is required",
   311→    state: form.state ? null : "State is required",
   312→    postcode: POSTCODE_RE.test(form.postcode.trim()) ? null : "5-digit postcode required",
   313→  };
   314→}

   320→export function CheckoutView({ ctx }: { ctx: ShopperCtx }) {
   321→  const { cart, loading: cartLoading, refresh: refreshCart } = useCart();
   322→  const items = cart?.items ?? [];
   323→
   324→  const me = useApp((s) => s.me);
   325→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   326→  const shopper = me.loaded ? me.shopper : null;
   327→
   328→  const [draft, setDraft] = useState<DraftT>(loadDraft);
   329→  useEffect(() => persistDraft(draft), [draft]);
   330→
   331→  // Contact details come from the ACCOUNT (carts/checkout are account-scoped);
   332→  // only fill blanks so the shopper can still override per-order (gifts etc.).
   333→  useEffect(() => {
   334→    if (!shopper) return;
   335→    setDraft((d) => ({
   336→      ...d,
   337→      name: d.name.trim() || shopper.fullName,
   338→      email: d.email.trim() || shopper.email,
   339→      phone: d.phone.trim() || shopper.phone || "",
   340→    }));
   341→  }, [shopper]);
   342→
   343→  const [addresses, setAddresses] = useState<AddressT[] | null>(null);
   344→  const [addressesLoading, setAddressesLoading] = useState(true);
   345→  const [formMode, setFormMode] = useState<null | { kind: "new" } | { kind: "edit"; address: AddressT }>(null);
   346→  const [form, setForm] = useState(emptyForm);
   347→  const [formErrors, setFormErrors] = useState<Record<string, string | null>>({});
   348→  const [formBusy, setFormBusy] = useState(false);
   349→  const [deleteArm, setDeleteArm] = useState<string | null>(null);
   350→
   351→  const [quote, setQuote] = useState<QuoteT | null>(null);
   352→  const [quoting, setQuoting] = useState(false);
   353→  const [couponInput, setCouponInput] = useState("");
   354→  const [couponErr, setCouponErr] = useState<string | null>(null);
   355→  const [tried, setTried] = useState(false);
   356→  const [submitting, setSubmitting] = useState(false);
   357→
   358→  const selectedAddr = useMemo(
   359→    () => addresses?.find((a) => a.id === draft.addressId) ?? null,
   360→    [addresses, draft.addressId]
   361→  );
   362→
   363→  // load the device address book once
   364→  useEffect(() => {
   365→    let alive = true;
   366→    fetchJson<{ addresses: AddressT[] }>("/api/checkout/addresses")
   367→      .then((d) => {
   368→        if (!alive) return;
   369→        setAddresses(d.addresses);
   370→        setDraft((dft) => {
   371→          if (dft.addressId && d.addresses.some((a) => a.id === dft.addressId)) return dft;
   372→          const def = d.addresses.find((a) => a.isDefault) ?? d.addresses[0];
   373→          return def ? { ...dft, addressId: def.id } : dft;
   374→        });
   375→      })
   376→      .catch(() => alive && setAddresses([]))
   377→      .finally(() => alive && setAddressesLoading(false));
   378→    return () => {
   379→      alive = false;
   380→    };
   381→  }, []);
   382→
   383→  const payloadItems = useMemo(() => items.map((i) => ({ variantId: i.variantId, qty: i.qty })), [items]);
   384→  const destState = selectedAddr?.state ?? null;
   385→
   386→  const refreshQuote = useCallback(
   387→    async (code: string, rateId: string, state: string | null) => {
   388→      if (payloadItems.length === 0) return;
   389→      setQuoting(true);
   390→      try {
   391→        const q = await postJson<QuoteT>("/api/checkout/quote", {
   392→          storeSlug: cart?.storeSlug,
   393→          items: payloadItems,
   394→          couponCode: code || undefined,
   395→          shippingRateId: rateId,
   396→          state: state ?? undefined,
   397→        });
   398→        setQuote(q);
   399→      } catch {
   400→        setQuote(null);
   401→      } finally {
   402→        setQuoting(false);
   403→      }
   404→    },
   405→    [cart?.storeSlug, payloadItems]
   406→  );
   407→
   408→  // auto-quote on cart / shipping / destination / coupon change
   409→  useEffect(() => {
   410→    refreshQuote(draft.couponCode, draft.shippingRateId, destState);
   411→  }, [refreshQuote, draft.couponCode, draft.shippingRateId, destState]);
   412→
   413→  // delivery option became invalid for this destination → fall back to Standard (server rule)
   414→  useEffect(() => {
   415→    if (!quote?.shippingOptions) return;
   416→    const sel = quote.shippingOptions.find((o) => o.id === draft.shippingRateId);
   417→    if (sel && !sel.available) {
   418→      setDraft((d) => ({ ...d, shippingRateId: "standard" }));
   419→      toast.error(sel.reason ?? "Selected delivery is unavailable — switched to Standard");
   420→    }
   421→  }, [quote, draft.shippingRateId]);
   422→
   423→  const stockIssues = useMemo(
   424→    () => (quote?.items ?? []).filter((i) => i.qty > i.stock),
   425→    [quote]
   426→  );
   427→
   428→  // ------------------------------------------------------------- coupon
   429→  async function applyCoupon() {
   430→    const code = couponInput.trim().toUpperCase();
   431→    if (!code) return;
   432→    setCouponErr(null);
   433→    setQuoting(true);
   434→    try {
   435→      const q = await postJson<QuoteT>("/api/checkout/quote", {
   436→        storeSlug: cart?.storeSlug,
   437→        items: payloadItems,
   438→        couponCode: code,
   439→        shippingRateId: draft.shippingRateId,
   440→        state: destState ?? undefined,
   441→      });
   442→      if (q.coupon?.valid) {
   443→        setDraft((d) => ({ ...d, couponCode: code }));
   444→        setQuote(q);
   445→        toast.success(`Coupon ${code} applied`, {
   446→          description: q.coupon.freeShip ? "Free shipping unlocked" : `You save ${formatMinor(q.coupon.discountMinor ?? 0)}`,
   447→        });
   448→      } else {
   449→        setCouponErr(q.coupon?.reason ?? "Invalid coupon");
   450→      }
   451→    } catch (e) {
   452→      setCouponErr(e instanceof Error ? e.message : "Could not validate coupon");
   453→    } finally {
   454→      setQuoting(false);
   455→    }
   456→  }
   457→
   458→  function removeCoupon() {
   459→    setDraft((d) => ({ ...d, couponCode: "" }));
   460→    setCouponInput("");
   461→    setCouponErr(null);
   462→  }
   463→
   464→  // ------------------------------------------------------------- address book ops
   465→  async function saveAddress() {
   466→    const errs = validateForm(form);
   467→    setFormErrors(errs);
   468→    if (Object.values(errs).some(Boolean)) return;
   469→    setFormBusy(true);
   470→    try {
   471→      const payload = { ...form, name: form.name.trim(), line1: form.line1.trim(), city: form.city.trim(), postcode: form.postcode.trim() };
   472→      const res =
   473→        formMode?.kind === "edit"
   474→          ? await patchJson<{ address: AddressT; addresses: AddressT[] }>(`/api/checkout/addresses/${formMode.address.id}`, payload)
   475→          : await postJson<{ address: AddressT; addresses: AddressT[] }>("/api/checkout/addresses", payload);
   476→      setAddresses(res.addresses);
   477→      setDraft((d) => ({ ...d, addressId: res.address.id }));
   478→      setFormMode(null);
   479→      setForm(emptyForm);
   480→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   481→      toast.success(formMode?.kind === "edit" ? "Address updated" : "Address saved", {
   482→        description: `${res.address.line1}, ${res.address.city} ${res.address.postcode}`,
   483→      });
   484→    } catch (e) {
   485→      toast.error(e instanceof Error ? e.message : "Could not save address");
   486→    } finally {
   487→      setFormBusy(false);
   488→    }
   489→  }
   490→
   491→  async function removeAddress(id: string) {
   492→    try {
   493→      const res = await deleteJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`);
   494→      setAddresses(res.addresses);
   495→      setDeleteArm(null);
   496→      setDraft((d) => (d.addressId === id ? { ...d, addressId: res.addresses.find((a) => a.isDefault)?.id ?? res.addresses[0]?.id ?? null } : d));
   497→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   498→      toast.success("Address removed");
   499→    } catch (e) {
   500→      toast.error(e instanceof Error ? e.message : "Could not remove address");
   501→    }
   502→  }
   503→
   504→  async function makeDefault(id: string) {
   505→    try {
   506→      const res = await patchJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`, { isDefault: true });
   507→      setAddresses(res.addresses);
   508→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
   509→      toast.success("Default address updated");
   510→    } catch (e) {
   511→      toast.error(e instanceof Error ? e.message : "Could not update address");
   512→    }
   513→  }
   514→
   515→  // ------------------------------------------------------------- phase validation
   516→  const contactErrors = {
   517→    name: draft.name.trim().length >= 2 ? null : "Full name is required",
   518→    email: EMAIL_RE.test(draft.email.trim()) ? null : "Enter a valid email address",
   519→    phone: !draft.phone.trim() || PHONE_RE.test(draft.phone.trim()) ? null : "Enter a valid phone number",
   520→  };
   521→  const contactValid = !Object.values(contactErrors).some(Boolean);
   522→  const addressPicked = !!selectedAddr;
   523→  const phase1Valid = contactValid && addressPicked;
   524→
   525→  function goToDelivery() {

  1156→      setPayment(p);

  1158→      if (p.status === "succeeded") {
  1159→        setPhase((ph) => (ph === "awaiting" ? "success" : ph));
  1160→      } else if (p.status === "failed") {

  1177→  // 10:00 countdown

  1374→function OrderCard({ o, compact = false }: { o: LookupResultT; compact?: boolean }) {
