   402→  const body: SimBody = useMemo(
   403→    () => ({

   405→      qty: Math.max(1, Number(qty || "1")),
   406→      discountMinor: toMinor(discount || "0"),
   407→      shippingMinor: toMinor(shipping || "0"),
   408→      categoryId: categoryId === "none" ? null : categoryId,
   409→      sellerId: sellerId === "none" ? null : sellerId,
   410→      productId: null,
   411→    }),
   412→    [price, qty, discount, shipping, categoryId, sellerId]
   413→  );
   414→  const debouncedBody = useDebounced(JSON.stringify(body), 450);
   415→
   416→  const [result, setResult] = useState<SimResult | null>(null);
   417→  const [err, setErr] = useState<string | null>(null);
   418→  const [running, setRunning] = useState(false);
   419→
   420→  useEffect(() => {
   421→    let cancelled = false;
   422→    const run = async () => {
   423→      setRunning(true);
   424→      setErr(null);
   425→      try {
   426→        const res = await apiSend<SimResult>("/api/admin/commission/simulate", "POST", JSON.parse(debouncedBody) as SimBody);
   427→        if (!cancelled) setResult(res);
   428→      } catch (e) {
   429→        if (!cancelled) {
   430→          setResult(null);
   431→          setErr(e instanceof Error ? e.message : "Simulation failed");
   432→        }
   433→      } finally {
   434→        if (!cancelled) setRunning(false);
   435→      }
   436→    };
   437→    void run();
   438→    return () => {
   439→      cancelled = true;
   440→    };
   441→  }, [debouncedBody]);
   442→
   443→  return (
   444→    <div className="space-y-4">
   445→      <PageHeader title="Commission Simulator" description="Pure what-if tool — nothing is persisted." />
   446→
   447→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   448→        <Card className="py-0">
   449→          <CardHeader className="px-4 pt-4 sm:px-6 sm:pt-5">
   450→            <CardTitle className="text-base">Inputs</CardTitle>
