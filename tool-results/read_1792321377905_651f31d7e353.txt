    54→/** INDEPENDENT ORACLE — exact (a × b) / d with ROUND_HALF_UP, pure BigInt. */
    55→function oracleMulDivHalfUp(a: bigint, b: bigint, d: bigint): bigint {
    56→  const p = a * b * 2n;
    57→  return p >= 0n ? (p + d) / (2n * d) : -((-p + d) / (2n * d));
    58→}
    59→
    60→/** Mulberry32 — deterministic PRNG so runs are reproducible. */
    61→function rng(seed: number) {
    62→  let t = seed >>> 0;
    63→  return () => {
    64→    t = (t + 0x6d2b79f5) >>> 0;
    65→    let r = Math.imul(t ^ (t >>> 15), 1 | t);
    66→    r = (r + Math.imul(r ^ (r >>> 7), 61 | r)) ^ r;
    67→    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
    68→  };
    69→}

   251→test("3-seller cart, randomized: Σ(sellerNet) + Σ(commission) ≡ subtotal − discount + shipping", () => {
   252→  const rand = rng(8080);
   253→  for (let iter = 0; iter < 2000; iter++) {
   254→    const sellers = 1 + Math.floor(rand() * 3);
   255→    let subtotal = 0;
   256→    const lines: { seller: number; lineSubtotal: number }[] = [];
   257→    for (let s = 0; s < sellers; s++) {
   258→      const nItems = 1 + Math.floor(rand() * 4);
   259→      for (let j = 0; j < nItems; j++) {
   260→        const unit = 100 + Math.floor(rand() * 50_000);
   261→        const qty = 1 + Math.floor(rand() * 5);
   262→        lines.push({ seller: s, lineSubtotal: unit * qty });
   263→        subtotal += unit * qty;
   264→      }
   265→    }
   266→    const discount = Math.floor(rand() * Math.floor(subtotal / 3));
   267→    const shipping = Math.floor(rand() * 2_500) * sellers;
   268→    const bps = 1 + Math.floor(rand() * (BPS_DENOM - 1));
   269→
   270→    // POLICY P1 pipeline (mirrors computeOrderCommissions): allocate, round once.
   271→    const discounts = allocateProportionally(discount, lines.map((l) => l.lineSubtotal));
   272→    const shippingAlloc = allocateProportionally(shipping, lines.map((l) => l.lineSubtotal));
   273→    let commission = 0;
   274→    let sellerNet = 0;
   275→    for (let i = 0; i < lines.length; i++) {
   276→      const base = lines[i].lineSubtotal - discounts[i];
   277→      const c = bpsOf(base, bps);
   278→      commission += c;
   279→      sellerNet += base + shippingAlloc[i] - c;
   280→    }
   281→    const buyerTotal = subtotal - discount + shipping;
   282→    assert.equal(
   283→      sellerNet + commission,
   284→      buyerTotal,
   285→      `iter ${iter}: nets+commission ${sellerNet + commission} ≠ buyer total ${buyerTotal}`
   286→    );
   287→  }
   288→});
   289→test("min-commission floor bump moves money commission↔seller 1:1 (identity preserved)", () => {
   290→  // base 1000 sen × 8% = 80; floor forces 200 → seller net drops by exactly 120
   291→  let base = 1000;
   292→  let commission = bpsOf(base, 800);
   293→  const sellerNetBefore = base - commission;
   294→  const floor = 200;
   295→  const bump = floor - commission;
