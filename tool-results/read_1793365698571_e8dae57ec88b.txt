   123→/** 850 → "8.5" (1 decimal) — seed value for the percent inputs. */
   124→function bpsToPercentInput(bps: number): string {
   125→  return (bps / 100).toFixed(1);
   126→}
   127→
   128→/** Percent text → basis points, or null when blank/unparseable/out of range. */
   129→function percentTextToBps(text: string, maxRateBps: number): number | null {
   130→  const trimmed = text.trim();
   131→  if (trimmed === "") return null;
   132→  const pct = Number(trimmed);
   133→  if (!Number.isFinite(pct)) return null;
   134→  const bps = Math.round(pct * 100);
   135→  if (bps < 0 || bps > maxRateBps) return null;
   136→  return bps;
   137→}

   209→function ProgrammeTab() {
   210→  const { data, loading, error, refetch } = useApi<ProgrammeData>("/api/store/affiliate");
   211→  const [optIn, setOptIn] = useState(false);
   212→  const [rateText, setRateText] = useState("");
   213→  const [busy, setBusy] = useState(false);
   214→
   215→  // hydrate the controls from the server state (and re-seed after every save —
   216→  // the refetch writes back the persisted opt-in + default rate)
   217→  useEffect(() => {
   218→    if (!data) return;
   219→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   220→    setOptIn(data.optIn);
   221→  }, [data?.optIn]);
   222→  useEffect(() => {
   223→    if (!data) return;
   224→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   225→    setRateText(bpsToPercentInput(data.defaultRateBps));
   226→  }, [data?.defaultRateBps]);
   227→
   228→  if (loading) {
   229→    return (
   230→      <div className="space-y-4">
   231→        <Skeleton className="h-24 w-full" />
   232→        <Skeleton className="h-40 w-full" />
   233→      </div>
   234→    );
   235→  }
   236→  if (error || !data) {
   237→    return (
   238→      <Card>
   239→        <CardContent className="p-6 text-sm text-rose-600">Could not load the affiliate programme{error ? `: ${error}` : ""}</CardContent>
   240→      </Card>
   241→    );
   242→  }
   243→
   244→  const paused = !data.programmeEnabled;
   245→  const maxPercent = fmtCommissionPercent(data.maxRateBps / 100);
   246→
   247→  const saveOptIn = async (next: boolean) => {
   248→    setOptIn(next);
   249→    const ok = await runMutation(
   250→      () => apiSend("PATCH", "/api/store/affiliate", { optIn: next }),
   251→      next ? "Your store joined the affiliate programme" : "Your store left the affiliate programme"
   252→    );
   253→    if (!ok) setOptIn(!next);
   254→    else refetch();
   255→  };
   256→
   257→  const saveRate = async () => {
   258→    const bps = percentTextToBps(rateText, data.maxRateBps);
   259→    if (bps == null) {
   260→      toast.error(`Rate must be between 0% and ${maxPercent}`);
   261→      return;
   262→    }
   263→    setBusy(true);
   264→    const ok = await runMutation(
   265→      () => apiSend("PATCH", "/api/store/affiliate", { defaultRateBps: bps }),
   266→      `Default commission rate saved at ${fmtCommissionPercent(bps / 100)}`
   267→    );
   268→    setBusy(false);
   269→    if (ok) refetch();
   270→  };
   271→
   272→  return (
   273→    <div className="space-y-4">
   274→      {paused ? (
   275→        <Alert>
   276→          <CirclePause />
   277→          <AlertTitle>Affiliate programme paused</AlertTitle>

   279→            The affiliate programme is currently paused platform-wide. Your opt-in, default rate and product enrollments are preserved — controls
   280→            are disabled until it resumes.
   281→          </AlertDescription>
   282→        </Alert>
   283→      ) : null}
   284→
   285→      {/* summary chips */}
   286→      <div className="flex flex-wrap items-center gap-2">
   287→        <Badge
   288→          variant="outline"
   289→          className={cn(
   290→            "border font-semibold",
   291→            data.optIn ? "bg-emerald-50 text-emerald-700 border-emerald-200/70" : "bg-slate-100 text-slate-600 border-slate-200/70"
   292→          )}
   293→        >
   294→          {data.optIn ? "Store opted in" : "Store not opted in"}
   295→        </Badge>
   296→        <Badge variant="outline" className="border bg-slate-100 font-semibold text-slate-600 border-slate-200/70">
   297→          Enrolled products · {data.enrolledProducts} of {data.totalProducts}

   313→                When enabled, buyers can earn commission promoting your enrolled products. Commission is deducted from your proceeds at checkout —
   314→                never from platform revenue.
   315→              </p>
   316→            </div>
   317→            <Switch checked={optIn} onCheckedChange={saveOptIn} disabled={paused} aria-label="Affiliate programme participation" />
   318→          </div>
   319→        </CardContent>
   320→      </Card>

   322→      {/* default rate */}

   363→type EnrolledFilter = "all" | "enrolled" | "notEnrolled";
   364→
   365→function ProductsTab() {
   366→  const [search, setSearch] = useState("");
   367→  const [query, setQuery] = useState("");
   368→  const [filter, setFilter] = useState<EnrolledFilter>("all");
   369→  const [page, setPage] = useState(1);
   370→  const { run: debouncedSearch } = useDebouncedCallback((v: string) => {
   371→    setQuery(v);
   372→    setPage(1);
   373→  }, 350);
   374→
   375→  const params = new URLSearchParams();
   376→  if (filter === "enrolled") params.set("enrolled", "true");
   377→  if (filter === "notEnrolled") params.set("enrolled", "false");
   378→  if (query.trim()) params.set("q", query.trim());
   379→  params.set("page", String(page));
   380→  params.set("size", "10");
   381→
   382→  const { data, loading, error, refetch } = useApi<AffiliateProductsPage>(`/api/store/affiliate/products?${params.toString()}`);
   383→
   384→  const totalPages = Math.max(1, Math.ceil((data?.total ?? 0) / 10));
   385→  const filtering = query.trim() !== "" || filter !== "all";
   386→
   387→  return (
   388→    <div className="space-y-4">
   389→      <p className="text-sm text-muted-foreground">
   390→        Choose which products buyers can promote, and set per-product commission rates. Products without their own rate use your store default.
   391→      </p>
   392→
   393→      {!loading && data && !data.storeOptIn ? (
   394→        <Alert>
   395→          <HandCoins />
   396→          <AlertDescription className="text-xs">
   397→            Your store has not opted in yet — enable participation on the Programme tab before enrolling products. Per-product rates can still be
   398→            prepared ahead.
   399→          </AlertDescription>
   400→        </Alert>
   401→      ) : null}
   402→
   403→      <div className="flex flex-wrap items-center gap-2">
   404→        <div className="relative min-w-52 flex-1">
   405→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   406→          <Input
   407→            placeholder="Search products…"
   408→            className="pl-8"
   409→            value={search}
   410→            onChange={(e) => {
   411→              setSearch(e.target.value);
   412→              debouncedSearch(e.target.value);
   413→            }}
   414→          />
   415→        </div>
   416→        <ToggleGroup
   417→          type="single"
   418→          variant="outline"
   419→          value={filter}
   420→          onValueChange={(v) => {
   421→            if (!v) return;
   422→            setFilter(v as EnrolledFilter);
   423→            setPage(1);
   424→          }}
   425→          aria-label="Filter products by enrollment"
   426→        >
   427→          <ToggleGroupItem value="all" className="min-h-[44px] px-3">
   428→            All
   429→          </ToggleGroupItem>
   430→          <ToggleGroupItem value="enrolled" className="min-h-[44px] px-3">
   431→            Enrolled
   432→          </ToggleGroupItem>
   433→          <ToggleGroupItem value="notEnrolled" className="min-h-[44px] px-3">
   434→            Not enrolled
   435→          </ToggleGroupItem>
   436→        </ToggleGroup>
   437→      </div>
   438→
   439→      {loading ? (
   440→        <RowsSkeleton rows={5} />
   441→      ) : error || !data ? (
   442→        <Card>
   443→          <CardContent className="p-6 text-sm text-rose-600">Could not load products{error ? `: ${error}` : ""}</CardContent>
   444→        </Card>
   445→      ) : data.products.length === 0 ? (
   446→        <EmptyState
   447→          icon={<Package className="h-8 w-8" />}
   448→          title={filtering ? "No matching products" : "No products yet"}
   449→          hint={filtering ? "Try clearing the search or switching the enrollment filter." : "Create products first — then enroll them so buyers can earn commission promoting them."}
   450→        />
   451→      ) : (
   452→        <div className="space-y-2">
   453→          {data.products.map((p) => (
   454→            <AffiliateProductRow
   455→              key={p.id}
   456→              product={p}
   457→              maxRateBps={data.maxRateBps}
   458→              storeDefaultRateBps={data.storeDefaultRateBps}
   459→              storeOptIn={data.storeOptIn}
   460→              onChanged={refetch}
   461→            />
   462→          ))}
   463→        </div>
   464→      )}
   465→
   466→      {!loading && data && data.products.length > 0 ? (
   467→        <div className="flex items-center justify-between text-sm text-muted-foreground">
   468→          <span>
   469→            {data.total} product{data.total === 1 ? "" : "s"} · page {page} of {totalPages}
   470→          </span>
   471→          <div className="flex gap-2">
   472→            <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage((p) => p - 1)}>
   473→              Previous
   474→            </Button>
   475→            <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage((p) => p + 1)}>
   476→              Next
   477→            </Button>
   478→          </div>
   479→        </div>
   480→      ) : null}
   481→    </div>
   482→  );
   483→}

   485→function AffiliateProductRow({
   486→  product,
   487→  maxRateBps,
   488→  storeDefaultRateBps,
   489→  storeOptIn,
   490→  onChanged,
   491→}: {
   492→  product: AffiliateProduct;
   493→  maxRateBps: number;
   494→  storeDefaultRateBps: number;
   495→  storeOptIn: boolean;
   496→  onChanged: () => void;
   497→}) {
   498→  const [enabled, setEnabled] = useState(product.affiliateEnabled);
   499→  const [rateText, setRateText] = useState(product.rateSource === "product" ? bpsToPercentInput(product.effectiveRateBps) : "");
   500→  const [busy, setBusy] = useState(false);
   501→
   502→  // re-seed the inline controls whenever the server row changes (post-save
   503→  // refetch writes new affiliateEnabled / rateSource / effectiveRateBps)
   504→  useEffect(() => {
   505→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   506→    setEnabled(product.affiliateEnabled);
   507→  }, [product.affiliateEnabled]);
   508→  useEffect(() => {
   509→    // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   510→    setRateText(product.rateSource === "product" ? bpsToPercentInput(product.effectiveRateBps) : "");
   511→  }, [product.rateSource, product.effectiveRateBps]);
   512→
   513→  const toggleEnrollment = async (next: boolean) => {
   514→    setEnabled(next);
   515→    const ok = await runMutation(
   516→      () => apiSend("PATCH", `/api/store/affiliate/products/${product.id}`, { enabled: next }),
   517→      next ? `${product.title} enrolled — buyers can now earn commission on it` : `${product.title} paused from the affiliate programme`
   518→    );
   519→    if (ok) onChanged();
   520→    else setEnabled(!next);
   521→  };
   522→
   523→  const applyRate = async () => {
   524→    const bps = percentTextToBps(rateText, maxRateBps);
   525→    if (bps == null) {
   526→      toast.error(`Rate must be between 0% and ${fmtCommissionPercent(maxRateBps / 100)}`);
   527→      return;
   528→    }
   529→    setBusy(true);
   530→    const ok = await runMutation(
   531→      () => apiSend("PATCH", `/api/store/affiliate/products/${product.id}`, { rateBps: bps }),
   532→      `${product.title} commission rate set to ${fmtCommissionPercent(bps / 100)}`
   533→    );
   534→    setBusy(false);
   535→    if (ok) onChanged();
   536→  };
   537→
   538→  // X = "reset to store default": drops the drafted text and, when the product
   539→  // carries its own override, sends rateBps: null so it inherits the store rate.
   540→  const resetRate = async () => {
   541→    setRateText("");
   542→    if (product.rateSource !== "product") return;
   543→    setBusy(true);
   544→    const ok = await runMutation(
   545→      () => apiSend("PATCH", `/api/store/affiliate/products/${product.id}`, { rateBps: null }),
   546→      `${product.title} reset to the store default rate`
   547→    );
   548→    setBusy(false);
   549→    if (ok) onChanged();
   550→  };
   551→
   552→  const canReset = rateText.trim() !== "" || product.rateSource === "product";
   553→
   554→  return (
   555→    <div className="rounded-xl border bg-card p-3 sm:p-4">
   556→      <div className="flex flex-wrap items-center gap-3">
   557→        <span className="flex h-10 w-10 shrink-0 items-center justify-center overflow-hidden rounded-lg border bg-muted">
   558→          {product.image ? (

   594→            />
   595→            <span className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-sm text-muted-foreground">%</span>
   596→          </div>
   597→          {canReset ? (
   598→            <Button
   599→              variant="ghost"
   600→              size="icon"
   601→              className="h-8 w-8 text-muted-foreground"
   602→              onClick={resetRate}
   603→              disabled={busy}
   604→              aria-label={`Reset ${product.title} to the store default rate`}
   605→              title="Reset to store default rate"
   606→            >
   607→              <X className="h-4 w-4" />
   608→            </Button>
   609→          ) : null}
   610→          <Button variant="outline" size="sm" onClick={applyRate} disabled={busy || rateText.trim() === ""} className="gap-1.5">
   611→            {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : null} Apply
   612→          </Button>
   613→        </div>
   614→      </div>
   615→    </div>
   616→  );
