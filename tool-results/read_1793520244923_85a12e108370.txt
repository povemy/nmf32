     1→     1→// ============================================================================
     2→     2→// Administrator TOTP MFA (Phase 1 / C4) — enrollment storage, login challenges
     3→     3→// and code verification. Stdlib crypto ONLY (no new deps).
     4→     4→//
     5→     5→// Layers:
     6→     6→//   • TOTP (RFC 6238 / RFC 4226): HMAC-SHA1, 30s steps, 6 digits, ±1 step
     7→     7→//     verification window, over a base32 (RFC 4648) 20-byte secret.
     8→     8→//   • Secret at rest: AES-256-GCM, key = sha256(MFA_ENCRYPTION_KEY). Format
     9→     9→//     "base64(iv).base64(tag).base64(ct)". Missing key: dev → one-time-warned
    10→    10→//     default key; production → 503 MFA_NOT_CONFIGURED (fail-closed, mirrors
    11→    11→//     the PSP_WEBHOOK_SECRET pattern).
    12→    12→//   • Recovery codes: 10 × randomToken(5) (10 hex chars), stored as a JSON
    13→    13→//     array of sha256(code) digests, one-time use (removed on use + persisted).
    14→    14→//   • Login challenges: in-memory Map (module-level, single process — same
    15→    15→//     architecture as the in-memory rate limiter), randomToken(32), 5-min TTL,
    16→    16→//     PROGRESSIVE cooldowns (3 fails → 60 s lock; each repeat doubles it,
    17→    17→//     capped at 10 min — Phase 2.1 removed the old 8-fail "permanent" lock so
    18→    18→//     no one can permanently lock out an administrator), PLUS a cross-challenge
    19→    19→//     per-admin throttle that pauses challenge MINTING for admins who keep
    20→    20→//     failing (prevents challenge-cycling to reset the counter).
    21→    21→//   • TOTP replay protection (Phase 2.1): the accepted time-step counter is
    22→    22→//     persisted on the enrollment (lastUsedTotpCounter) and consumed with a
    23→    23→//     CAS updateMany — the same valid code can never verify twice, and the
    24→    24→//     enrollment-confirm code is burned the same way.
    25→    25→//
    26→    26→// DB guard: getMfaEnrollment() treats Prisma P2021/P2022 (table/column does
    27→    27→// not exist) as UNENROLLED, so the app runs BEFORE migration
    28→    28→// 20261101000002 is applied (dev) while migrated production gets full
    29→    29→// enforcement.
    30→    30→//
    31→    31→// NEVER log the TOTP secret, the otpauth URL, recovery codes or submitted
    32→    32→// codes — audit actions carry counts/status only.

    44→    44→export const TOTP_STEP_SECONDS = 30;
    45→    45→export const TOTP_DIGITS = 6;
    46→    46→/** Verification window: codes from counter-1 … counter+1 are accepted. */
    47→    47→export const TOTP_WINDOW = 1;

    50→    50→/* --------------------------------------------------- base32 (RFC 4648) */
    51→    51→
    52→    52→const B32_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
    53→    53→
    54→    54→/** RFC 4648 base32, no padding. */
    55→    55→export function base32Encode(buf: Buffer): string {
    56→    56→  let bits = 0;
    57→    57→  let value = 0;
    58→    58→  let out = "";
    59→    59→  for (const byte of buf) {
    60→    60→    value = (value << 8) | byte;
    61→    61→    bits += 8;
    62→    62→    while (bits >= 5) {
    63→    63→      out += B32_ALPHABET[(value >>> (bits - 5)) & 31];
    64→    64→      bits -= 5;
    65→    65→    }
    66→    66→  }
    67→    67→  // Final partial group — pad with zero bits on the right.
    68→    68→  if (bits > 0) out += B32_ALPHABET[(value << (5 - bits)) & 31];
    69→    69→  return out;
    70→    70→}
    71→    71→
    72→    72→/** RFC 4648 base32 decode (tolerates padding + whitespace + case). Throws on
    73→    73→ *  characters outside the alphabet. */
    74→    74→export function base32Decode(s: string): Buffer {
    75→    75→  const clean = s.replace(/=+$/g, "").replace(/\s+/g, "").toUpperCase();
    76→    76→  let bits = 0;
    77→    77→  let value = 0;
    78→    78→  const bytes: number[] = [];
    79→    79→  for (const ch of clean) {
    80→    80→    const idx = B32_ALPHABET.indexOf(ch);
    81→    81→    if (idx < 0) throw new Error(`invalid base32 character '${ch}'`);
    82→    82→    value = (value << 5) | idx;
    83→    83→    bits += 5;
    84→    84→    if (bits >= 8) {
    85→    85→      bytes.push((value >>> (bits - 8)) & 255);
    86→    86→      bits -= 8;
    87→    87→    }
    88→    88→  }
    89→    89→  return Buffer.from(bytes);
    90→    90→}

    92→    92→/* ------------------------------------------------------- TOTP (RFC 6238) */
    93→    93→
    94→    94→/** HOTP code (RFC 4226): HMAC-SHA1 over the 8-byte big-endian counter,
    95→    95→ *  dynamic truncation, `digits` decimal digits (zero-padded). */
    96→    96→export function totpCode(secret: Buffer, counter: number, digits = TOTP_DIGITS): string {
    97→    97→  const counterBuf = Buffer.alloc(8);
    98→    98→  counterBuf.writeBigUInt64BE(BigInt(counter));
    99→    99→  const mac = createHmac("sha1", secret).update(counterBuf).digest();
   100→   100→  const offset = mac[mac.length - 1] & 0x0f;
   101→   101→  const bin =
   102→   102→    ((mac[offset] & 0x7f) << 24) | (mac[offset + 1] << 16) | (mac[offset + 2] << 8) | mac[offset + 3];
   103→   103→  return (bin % 10 ** digits).toString().padStart(digits, "0");
   104→   104→}
   105→   105→
   106→   106→/** TOTP counter for a Unix timestamp (seconds). */
   107→   107→export function totpCounterAt(timeSeconds: number, step = TOTP_STEP_SECONDS): number {
   108→   108→  return Math.floor(timeSeconds / step);
   109→   109→}

   111→   111→function safeEqualStr(a: string, b: string): boolean {
   112→   112→  const ab = Buffer.from(a, "utf8");
   113→   113→  const bb = Buffer.from(b, "utf8");
   114→   114→  return ab.length === bb.length && timingSafeEqual(ab, bb);
   115→   115→}

   117→   117→/** Verify a 6-digit code against the secret with a ±window-step window at
   118→   118→ *  `nowMs` (default: current time). Non-digit input simply fails. */
   119→   119→export function verifyTotp(
   120→   120→  secret: Buffer,
   121→   121→  code: string,
   122→   122→  nowMs = Date.now(),
   123→   123→  window = TOTP_WINDOW,
   124→   124→  step = TOTP_STEP_SECONDS
   125→   125→): boolean {
   126→   126→  return verifyTotpDetailed(secret, code, nowMs, window, step).ok;
   127→   127→}
   128→   128→
   129→   129→/** PHASE 2.1 (§5-C): verifyTotp that also reports WHICH time-step counter
   130→   130→ *  matched — required for replay protection (the accepted counter is persisted
   131→   131→ *  and never accepted again). `counter` is present only when ok. */
   132→   132→export function verifyTotpDetailed(
   133→   133→  secret: Buffer,
   134→   134→  code: string,
   135→   135→  nowMs = Date.now(),
   136→   136→  window = TOTP_WINDOW,
   137→   137→  step = TOTP_STEP_SECONDS
   138→   138→): { ok: boolean; counter?: number } {
   139→   139→  const normalized = (code ?? "").replace(/[^0-9]/g, "");
   140→   140→  if (normalized.length !== TOTP_DIGITS) return { ok: false };
   141→   141→  const counter = totpCounterAt(Math.floor(nowMs / 1000), step);
   142→   142→  for (let i = -window; i <= window; i++) {
   143→   143→    if (safeEqualStr(totpCode(secret, counter + i), normalized)) return { ok: true, counter: counter + i };
   144→   144→  }
   145→   145→  return { ok: false };
   146→   146→}

   162→   162→/* ------------------------------------------- secret encryption (AES-GCM) */
   163→   163→
   164→   164→const DEV_FALLBACK_KEY = "sm_mfa_dev_key";
   165→   165→let devKeyWarned = false;
   166→   166→
   167→   167→/** True when MFA crypto is usable in this environment: an explicit
   168→   168→ *  MFA_ENCRYPTION_KEY is set, OR we are outside production (dev default key). */
   169→   169→export function mfaEncryptionConfigured(): boolean {
   170→   170→  const key = process.env.MFA_ENCRYPTION_KEY;
   171→   171→  if (key && key.trim() !== "") return true;
   172→   172→  return process.env.NODE_ENV !== "production";
   173→   173→}
   174→   174→
   175→   175→/** AES-256-GCM key = sha256(MFA_ENCRYPTION_KEY). Fail-closed in production
   176→   176→ *  when the key is unset (503 MFA_NOT_CONFIGURED); dev gets a one-time-warned
   177→   177→ *  default so local flows work end-to-end. */
   178→   178→function mfaKey(): Buffer {
   179→   179→  const envKey = process.env.MFA_ENCRYPTION_KEY;
   180→   180→  if (envKey && envKey.trim() !== "") {
   181→   181→    return createHash("sha256").update(envKey, "utf8").digest();
   182→   182→  }
   183→   183→  if (process.env.NODE_ENV === "production") {
   184→   184→    throw new ApiError(503, "MFA_NOT_CONFIGURED", "MFA_ENCRYPTION_KEY is not set — two-factor authentication is unavailable.");
   185→   185→  }
   186→   186→  if (!devKeyWarned) {
   187→   187→    devKeyWarned = true;
   188→   188→    console.warn("[mfa] MFA_ENCRYPTION_KEY not set — using the development default key. Set MFA_ENCRYPTION_KEY in production.");
   189→   189→  }
   190→   190→  return createHash("sha256").update(DEV_FALLBACK_KEY, "utf8").digest();
   191→   191→}

   193→   193→/** Encrypt a secret → "base64(iv).base64(tag).base64(ct)". */
   194→   194→export function encryptSecret(plaintext: string): string {
   195→   195→  const iv = randomBytes(12);
   196→   196→  const cipher = createCipheriv("aes-256-gcm", mfaKey(), iv);
   197→   197→  const ct = Buffer.concat([cipher.update(plaintext, "utf8"), cipher.final()]);
   198→   198→  const tag = cipher.getAuthTag();
   199→   199→  return `${iv.toString("base64")}.${tag.toString("base64")}.${ct.toString("base64")}`;
   200→   200→}
   201→   201→
   202→   202→/** Decrypt "base64(iv).base64(tag).base64(ct)". Throws on tampering (GCM tag
   203→   203→ *  mismatch), malformed input or a changed key. */
   204→   204→export function decryptSecret(enc: string): string {
   205→   205→  const parts = enc.split(".");
   206→   206→  if (parts.length !== 3) throw new Error("malformed MFA secret ciphertext");
   207→   207→  const decipher = createDecipheriv("aes-256-gcm", mfaKey(), Buffer.from(parts[0], "base64"));
   208→   208→  decipher.setAuthTag(Buffer.from(parts[1], "base64"));
   209→   209→  const pt = Buffer.concat([decipher.update(Buffer.from(parts[2], "base64")), decipher.final()]);
   210→   210→  return pt.toString("utf8");
   211→   211→}

   213→   213→/** Verify a TOTP code against an enrollment's ENCRYPTED secret (±1 step).
   214→   214→ *  Tampered/undecryptable secrets fail closed (false); ApiError 503
   215→   215→ *  (unconfigured production key) still propagates. */
   216→   216→export function verifyTotpForSecretEnc(secretEnc: string, code: string, nowMs = Date.now()): boolean {
   217→   217→  try {
   218→   218→    const secret = base32Decode(decryptSecret(secretEnc));
   219→   219→    return verifyTotp(secret, code, nowMs);
   220→   220→  } catch (e) {
   221→   221→    if (e instanceof ApiError) throw e;
   222→   222→    return false;
   223→   223→  }
   224→   224→}

   243→   243→
   244→   244→let replayWarned = false;
   245→   245→
   246→   246→/** CAS-guarded consumption of a TOTP time-step counter: the update lands only
   247→   247→ *  when the stored counter is OLDER than `counter` (or null). Returns false
   248→   248→ *  when the counter was already used (replay) or a concurrent verification
   249→   249→ *  consumed it first — exactly one parallel replay can ever win.
   250→   250→ *  Pre-migration deployments (table OR column missing): protection is skipped
   251→   251→ *  with a loud one-time warning — identical fail-open contract to the Phase 1
   252→   252→ *  table-absence guard (production always deploys migrations first). */
   253→   253→export async function consumeTotpCounter(adminId: string, counter: number): Promise<boolean> {
   254→   254→  try {
   255→   255→    const res = await db.adminMfaEnrollment.updateMany({
   256→   256→      where: {
   257→   257→        adminId,
   258→   258→        OR: [{ lastUsedTotpCounter: null }, { lastUsedTotpCounter: { lt: counter } }],
   259→   259→      },
   260→   260→      data: { lastUsedTotpCounter: counter },
   261→   261→    });
   262→   262→    return res.count === 1;
   263→   263→  } catch (e) {
   264→   264→    if (isPrismaMissingTableError(e)) {
   265→   265→      if (!replayWarned) {
   266→   266→        replayWarned = true;
   267→   267→        console.warn(
   268→   268→          "[mfa] TOTP replay protection unavailable — apply migrations 20261101000002 + 20261103000000 to enable it"
   269→   269→        );
   270→   270→      }
   271→   271→      return true; // column/table absent (unmigrated dev) — fail-open, warn once
   272→   272→    }
   273→   273→    throw e;
   274→   274→  }
   275→   275→}

   281→   281→export function sha256Hex(input: string): string {
   282→   282→  return createHash("sha256").update(input, "utf8").digest("hex");
   283→   283→}

   285→   285→/** Canonical recovery-code form: lowercase alphanumerics only. */
   286→   286→export function normalizeRecoveryCode(raw: string): string {
   287→   287→  return (raw ?? "").toLowerCase().replace(/[^a-z0-9]/g, "");

   290→   290→/** "abcdefghij" → "abcde-fghij" (display grouping only). */
   291→   291→export function formatRecoveryCode(code: string): string {
   292→   292→  return code.length === 10 ? `${code.slice(0, 5)}-${code.slice(5)}` : code;
   293→   293→}
   294→   294→
   295→   295→/** 10 recovery codes (10 hex chars each) + the JSON array of sha256 digests
   296→   296→ *  to persist. Plaintext codes exist ONLY in the caller's response. */
   297→   297→export function generateRecoveryCodes(count = RECOVERY_CODE_COUNT): { codes: string[]; hashesJson: string } {
   298→   298→  const codes = Array.from({ length: count }, () => randomToken(5));
   299→   299→  return { codes, hashesJson: JSON.stringify(codes.map((c) => sha256Hex(c))) };
   300→   300→}
   301→   301→
   302→   302→/** Parse the stored hashesJson → string[] of sha256 hex digests (defensive:
   303→   303→ *  non-string / non-digest entries dropped; malformed JSON → []). */
   304→   304→export function parseRecoveryHashes(hashesJson: string): string[] {
   305→   305→  try {
   306→   306→    const parsed = JSON.parse(hashesJson ?? "[]");
   307→   307→    if (!Array.isArray(parsed)) return [];
   308→   308→    return parsed.filter((h): h is string => typeof h === "string" && /^[0-9a-f]{64}$/.test(h));
   309→   309→  } catch {
   310→   310→    return [];
   311→   311→  }
   312→   312→}
   313→   313→
   314→   314→/** Index of the matching recovery code in the stored hashes, or -1. */
   315→   315→export function matchRecoveryCode(hashesJson: string, rawCode: string): number {
   316→   316→  const normalized = normalizeRecoveryCode(rawCode);
   317→   317→  if (normalized === "") return -1;
   318→   318→  return parseRecoveryHashes(hashesJson).indexOf(sha256Hex(normalized));
   319→   319→}
   320→   320→
   321→   321→/** Pure removal: returns the updated hashesJson with the hash at `index`
   322→   322→ *  removed (one-time-use burn). Out-of-range index → input unchanged. */
   323→   323→export function removeRecoveryHash(hashesJson: string, index: number): string {
   324→   324→  const hashes = parseRecoveryHashes(hashesJson);
   325→   325→  if (index < 0 || index >= hashes.length) return hashesJson;
   326→   326→  return JSON.stringify([...hashes.slice(0, index), ...hashes.slice(index + 1)]);
   327→   327→}

   400→   400→/** Pure status check: valid | locked (with retryAfterSeconds) | expired. */
   401→   401→export function challengeStatus(
   402→   402→  state: MfaChallengeState,
   403→   403→  now: number
   404→   404→): { status: "valid" | "locked" | "expired"; retryAfterSeconds?: number } {
   405→   405→  if (now >= state.expiresAt) return { status: "expired" };
   406→   406→  if (now < state.lockedUntil) {
   407→   407→    return { status: "locked", retryAfterSeconds: Math.max(1, Math.ceil((state.lockedUntil - now) / 1000)) };
   408→   408→  }
   409→   409→  return { status: "valid" };
   410→   410→}
   411→   411→
   412→   412→/** Pure failure accounting: after a WRONG code at `now`, what does the
   413→   413→ *  challenge look like and which failure reason applies?
   414→   414→ *
   415→   415→ *  PHASE 2.1 (§5-A): PROGRESSIVE COOLDOWNS — no permanent lockout exists any
   416→   416→ *  more. Every lock is temporary and doubles with each repetition:
   417→   417→ *    • lock #1 (3 consecutive failures) → 60 s cooldown
   418→   418→ *    • lock #2 → 120 s · lock #3 → 240 s · lock #4 → 480 s
   419→   419→ *    • any lock after 8 TOTAL failures, or lock #5+ → 600 s cap
   420→   420→ *  A challenge that hits the cap simply outlives its own 5-minute TTL — the
   421→   421→ *  attacker gains nothing, the legitimate administrator is never permanently
   422→   422→ *  locked out by anyone else's activity (challenges require the password to
   423→   423→ *  obtain in the first place, and a fresh sign-in always starts clean). */
   424→   424→export function recordChallengeFailure(
   425→   425→  state: MfaChallengeState,
   426→   426→  now: number
   427→   427→): { next: MfaChallengeState; outcome: "INVALID" | "LOCKED"; cooldownMs: number; retryAfterSeconds: number } {
   428→   428→  const totalFailures = state.totalFailures + 1;
   429→   429→  const failures = state.failures + 1;
   430→   430→  const lockNow = failures >= MFA_FAIL_LOCK_THRESHOLD || totalFailures >= MFA_TOTAL_FAIL_LIMIT;
   431→   431→  if (lockNow) {
   432→   432→    const lockCount = state.lockCount + 1;
   433→   433→    const escalated = totalFailures >= MFA_TOTAL_FAIL_LIMIT || lockCount >= 5;
   434→   434→    const cooldownMs = escalated
   435→   435→      ? MFA_LOCK_MAX_MS
   436→   436→      : Math.min(MFA_FAIL_LOCK_MS * 2 ** (lockCount - 1), MFA_LOCK_MAX_MS);
   437→   437→    return {
   438→   438→      next: { ...state, failures: 0, totalFailures, lockCount, lockedUntil: Math.max(state.lockedUntil, now + cooldownMs) },
   439→   439→      outcome: "LOCKED",
   440→   440→      cooldownMs,
   441→   441→      retryAfterSeconds: Math.max(1, Math.ceil(cooldownMs / 1000)),
   442→   442→    };
   443→   443→  }
   444→   444→  return { next: { ...state, failures, totalFailures }, outcome: "INVALID", cooldownMs: 0, retryAfterSeconds: 0 };
   445→   445→}
   446→   446→
   447→   447→// Module-level in-memory challenge store — single process, consistent with the
   448→   448→// existing in-memory rate limiter architecture (lib/rate-limit.ts).
   449→   449→const challenges = new Map<string, MfaChallengeState>();
   450→   450→let lastChallengeSweep = 0;
   451→   451→
   452→   452→function sweepChallenges(now: number): void {
   453→   453→  if (now - lastChallengeSweep < 60_000) return;
   454→   454→  lastChallengeSweep = now;
   455→   455→  for (const [token, state] of challenges) {
   456→   456→    if (now >= state.expiresAt) challenges.delete(token);
   457→   457→  }
   458→   458→}
   459→   459→
   460→   460→/* --------------------------------------- per-admin cross-challenge throttle */
   461→   461→// PHASE 2.1 (§5-A): without this, an attacker holding a stolen password could
   462→   462→// cycle FRESH challenges (each new sign-in mints one) to reset the per-challenge
   463→   463→// failure counter and grind codes indefinitely. The throttle accumulates MFA
   464→   464→// failures PER ADMIN across challenges and applies its own escalating cooldown
   465→   465→// to challenge MINTING. It cannot be armed by unauthenticated traffic: a
   466→   466→// challenge only exists after the password verified, so only someone who
   467→   467→// already passed the password gate can accrue failures here — a legit admin
   468→   468→// who types a wrong code recovers after at most ADMIN_MFA_THROTTLE_MAX_MS.
   469→   469→const adminThrottle = new Map<string, { failures: number; lockCount: number; lockedUntil: number }>();
   470→   470→let lastThrottleSweep = 0;
   471→   471→
   472→   472→/** Cross-challenge failures (within the window) that trigger an admin cooldown. */
   473→   473→export const ADMIN_MFA_THROTTLE_FAIL_LIMIT = 6;
   474→   474→/** Failure accrual window (rolling, refreshed by each failure). */
   475→   475→export const ADMIN_MFA_THROTTLE_WINDOW_MS = 15 * 60 * 1000;
   476→   476→/** Per-admin mint cooldown cap. */
   477→   477→export const ADMIN_MFA_THROTTLE_MAX_MS = 10 * 60 * 1000;
   478→   478→
   479→   479→function sweepAdminThrottle(now: number): void {
   480→   480→  if (now - lastThrottleSweep < 60_000) return;
   481→   481→  lastThrottleSweep = now;
   482→   482→  for (const [adminId, t] of adminThrottle) {
   483→   483→    if (t.lockedUntil < now - ADMIN_MFA_THROTTLE_WINDOW_MS) adminThrottle.delete(adminId);
   484→   484→  }
   485→   485→}
   486→   486→
   487→   487→/** Mint-time gate: is challenge creation currently cooled down for this admin? */
   488→   488→export function adminMfaThrottleStatus(adminId: string, now = Date.now()): { locked: boolean; retryAfterSeconds?: number } {
   489→   489→  sweepAdminThrottle(now);
   490→   490→  const t = adminThrottle.get(adminId);
   491→   491→  if (!t || now >= t.lockedUntil) return { locked: false };
   492→   492→  return { locked: true, retryAfterSeconds: Math.max(1, Math.ceil((t.lockedUntil - now) / 1000)) };
   493→   493→}
   494→   494→
   495→   495→/** Record an MFA failure against the admin (cross-challenge). */
   496→   496→function noteAdminMfaFailure(adminId: string, now: number): void {
   497→   497→  const t = adminThrottle.get(adminId) ?? { failures: 0, lockCount: 0, lockedUntil: 0 };
   498→   498→  t.failures += 1;
   499→   499→  if (t.failures >= ADMIN_MFA_THROTTLE_FAIL_LIMIT) {
   500→   500→    t.lockCount += 1;
   501→   501→    t.failures = 0;
   502→   502→    const cooldown = Math.min(MFA_FAIL_LOCK_MS * 2 ** (t.lockCount - 1), ADMIN_MFA_THROTTLE_MAX_MS);
   503→   503→    t.lockedUntil = Math.max(t.lockedUntil, now + cooldown);
   504→   504→  }
   505→   505→  adminThrottle.set(adminId, t);
   506→   506→}
   507→   507→
   508→   508→/** Mint a login challenge for a (password-verified) admin. TTL 5 min.
   509→   509→ *  PHASE 2.1: refuses (429 MFA_ACCOUNT_COOLDOWN) while the cross-challenge
   510→   510→ *  throttle holds a cooldown for this admin. */
   511→   511→export async function createMfaChallenge(adminId: string): Promise<{ challenge: string; expiresAt: number }> {
   512→   512→  const now = Date.now();
   513→   513→  sweepChallenges(now);
   514→   514→  const throttle = adminMfaThrottleStatus(adminId, now);
   515→   515→  if (throttle.locked) {
   516→   516→    throw new ApiError(
   517→   517→      429,
   518→   518→      "MFA_ACCOUNT_COOLDOWN",
   519→   519→      `Too many incorrect codes — new sign-in attempts are paused for ${throttle.retryAfterSeconds}s`,
   520→   520→      { retryAfterSeconds: throttle.retryAfterSeconds }
   521→   521→    );
   522→   522→  }
   523→   523→  const challenge = randomToken(32);
   524→   524→  const expiresAt = now + MFA_CHALLENGE_TTL_MS;
   525→   525→  challenges.set(challenge, freshChallengeState(adminId, expiresAt));
   526→   526→  return { challenge, expiresAt };
   527→   527→}

   532→   532→  adminId: string,
   533→   533→  action: "admin.mfa.enrolled" | "admin.mfa.confirmed" | "admin.mfa.disabled" | "admin.mfa.recovery_used" | "admin.mfa.verify_failed",
   534→   534→  meta: Record<string, unknown>,

   541→   541→      actorLabel = admin?.email ?? null;

   548→   548→      entityType: "admin_mfa",
   549→   549→      entityId: adminId,
   550→   550→      meta,
   551→   551→    });
   552→   552→  } catch {
   553→   553→    // Audit is best-effort — a failed log write must never break the auth flow.
   554→   554→  }
   555→   555→}

   558→   558→
   559→   559→export type MfaVerifyResult =
   560→   560→  | { ok: true; adminId: string; usedRecoveryCode: boolean }
   561→   561→  | {
   562→   562→      ok: false;
   563→   563→      reason: "INVALID" | "EXPIRED" | "LOCKED";
   564→   564→      /** Failure-arm extras (informational, optional): how many wrong codes
   565→   565→       *  remain before the next lock, and how long a lock still runs. */
   566→   566→      attemptsRemaining?: number;
   567→   567→      retryAfterSeconds?: number;
   568→   568→    };
   569→   569→
   570→   570→/** Verify a login challenge: TOTP (±1 step) OR one-time recovery code.
   571→   571→ *  Progressive lockout per challenge; success consumes the challenge.
   572→   572→ *  NEVER logs the submitted code. */
   573→   573→export async function verifyMfaChallenge(challengeToken: string, code: string): Promise<MfaVerifyResult> {
   574→   574→  const token = (challengeToken ?? "").trim();
   575→   575→  const now = Date.now();
   576→   576→  sweepChallenges(now);
   577→   577→  const state = challenges.get(token);
   578→   578→  // Unknown token: already used, swept, or garbage — one uniform answer.
   579→   579→  if (!state) return { ok: false, reason: "EXPIRED" };
   580→   580→
   581→   581→  const status = challengeStatus(state, now);
   582→   582→  if (status.status === "expired") {
   583→   583→    challenges.delete(token);
   584→   584→    return { ok: false, reason: "EXPIRED" };
   585→   585→  }
   586→   586→  if (status.status === "locked") {
   587→   587→    return { ok: false, reason: "LOCKED", retryAfterSeconds: status.retryAfterSeconds };
   588→   588→  }
   589→   589→
   590→   590→  // Login enforcement only applies to CONFIRMED enrollments. An enrollment
   591→   591→  // that vanished (or was never confirmed) can never satisfy this challenge.
   592→   592→  const enrollment = await getMfaEnrollment(state.adminId);
   593→   593→  if (!enrollment || !enrollment.confirmedAt) {
   594→   594→    challenges.delete(token);
   595→   595→    return { ok: false, reason: "EXPIRED" };
   596→   596→  }
   597→   597→
   598→   598→  const attempt = (code ?? "").trim();
   599→   599→
   600→   600→  // 1) TOTP path (6-digit authenticator code) — PHASE 2.1 (§5-C): the matched
   601→   601→  //    time-step counter is CAS-consumed; a replay of the SAME valid code (or
   602→   602→  //    an older still-in-window code) is rejected exactly like a wrong code.
   603→   603→  let secret: Buffer;
   604→   604→  try {
   605→   605→    secret = base32Decode(decryptSecret(enrollment.secretEnc));
   606→   606→  } catch (e) {
   607→   607→    if (e instanceof ApiError) throw e; // 503 MFA_NOT_CONFIGURED — fail-closed
   608→   608→    // Undecryptable secret (corrupt row / rotated key): the challenge can
   609→   609→    // never succeed — retire it and fail closed.
   610→   610→    challenges.delete(token);
   611→   611→    return { ok: false, reason: "INVALID" };
   612→   612→  }
   613→   613→  const totp = verifyTotpDetailed(secret, attempt, now);
   614→   614→  if (totp.ok) {
   615→   615→    const consumed = await consumeTotpCounter(state.adminId, totp.counter!);
   616→   616→    if (consumed) {
   617→   617→      challenges.delete(token);
   618→   618→      return { ok: true, adminId: state.adminId, usedRecoveryCode: false };
   619→   619→    }
   620→   620→    // Replayed time step — uniform INVALID (never reveal WHY it failed).
   621→   621→    await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", { replay: true });
   622→   622→    const replayRec = recordChallengeFailure(state, now);
   623→   623→    challenges.set(token, replayRec.next);
   624→   624→    noteAdminMfaFailure(state.adminId, now);
   625→   625→    if (replayRec.outcome === "LOCKED") {
   626→   626→      return { ok: false, reason: "LOCKED", retryAfterSeconds: replayRec.retryAfterSeconds, attemptsRemaining: 0 };
   627→   627→    }
   628→   628→    return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(replayRec.next) };
   629→   629→  }
   630→   630→
   631→   631→  // 2) Recovery-code path (one-time; hash burned BEFORE success is reported).
   632→   632→  const recoveryIndex = matchRecoveryCode(enrollment.recoveryHashes, attempt);
   633→   633→  if (recoveryIndex >= 0) {
   634→   634→    const remainingHashes = removeRecoveryHash(enrollment.recoveryHashes, recoveryIndex);
   635→   635→    // Optimistic-concurrency burn: the update only lands if the stored hash
   636→   636→    // list is still the one we matched against — two parallel replays of the
   637→   637→    // same code cannot BOTH consume it.
   638→   638→    const burn = await db.adminMfaEnrollment.updateMany({
   639→   639→      where: { adminId: state.adminId, recoveryHashes: enrollment.recoveryHashes },
   640→   640→      data: { recoveryHashes: remainingHashes },
   641→   641→    });
   642→   642→    if (burn.count === 0) {
   643→   643→      // Concurrent verify already burned this code — treat as a wrong code.
   644→   644→      const rec = recordChallengeFailure(state, now);
   645→   645→      challenges.set(token, rec.next);
   646→   646→      noteAdminMfaFailure(state.adminId, now);
   647→   647→      if (rec.outcome === "LOCKED") {
   648→   648→        await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", { cooldown: rec.cooldownMs });
   649→   649→        return { ok: false, reason: "LOCKED", retryAfterSeconds: rec.retryAfterSeconds };
   650→   650→      }
   651→   651→      return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(rec.next) };
   652→   652→    }
   653→   653→    challenges.delete(token);
   654→   654→    await auditMfaEvent(state.adminId, "admin.mfa.recovery_used", { via: "login" });
   655→   655→    return { ok: true, adminId: state.adminId, usedRecoveryCode: true };
   656→   656→  }
   657→   657→
   658→   658→  // 3) Wrong code — progressive lockout (audit only when a lock fires, to
   659→   659→  //    avoid per-attempt spam). Every failure also feeds the cross-challenge
   660→   660→  //    per-admin throttle so challenge-cycling cannot reset the counters.
   661→   661→  const rec = recordChallengeFailure(state, now);
   662→   662→  challenges.set(token, rec.next);
   663→   663→  noteAdminMfaFailure(state.adminId, now);
   664→   664→  if (rec.outcome === "LOCKED") {
   665→   665→    await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", {
   666→   666→      cooldown: rec.cooldownMs,
   667→   667→      totalFailures: rec.next.totalFailures,
   668→   668→    });
   669→   669→    return { ok: false, reason: "LOCKED", retryAfterSeconds: rec.retryAfterSeconds, attemptsRemaining: 0 };
   670→   670→  }
   671→   671→  return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(rec.next) };
   672→   672→}
   673→   673→
   674→   674→function remainingAttempts(state: MfaChallengeState): number {
   675→   675→  return Math.max(0, Math.min(MFA_FAIL_LOCK_THRESHOLD - state.failures, MFA_TOTAL_FAIL_LIMIT - state.totalFailures));
   676→   676→}

   678→   678→/* ------------------------------------- MANDATORY ENROLLMENT GATE (§5-B) */
   679→   679→
   680→   680→let gateWarned = false;
   681→   681→
   682→   682→/** Roles that hold any FINANCIAL capability must have a CONFIRMED TOTP
   683→   683→ *  enrollment before they may exercise privileged admin functionality.
   684→   684→ *  Derived from the ADMIN_CAPS matrix (pure — see admin-caps.ts), so the
   685→   685→ *  privileged set stays true if caps evolve: currently {superadmin, finance,
   686→   686→ *  ops} (ops holds refunds.manage + disputes.manage).
   687→   687→ *
   688→   688→ *  Return values:
   689→   689→ *    enrollmentRequired: false — no restriction (non-privileged role, OR
   690→   690→ *      privileged with a confirmed enrollment).
   691→   691→ *    enrollmentRequired: true — RESTRICTED: the admin may only use the MFA
   692→   692→ *      bootstrap routes (enroll/confirm/status + me/logout) until enrollment
   693→   693→ *      is confirmed. Bootstrap is safe: /api/admin/mfa/enroll needs only the
   694→   694→ *      CURRENT PASSWORD (fresh superadmin on a migrated deployment can always
   695→   695→ *      self-recover into full access — NOMOFEE can never be fully locked out).
   696→   696→ *    Table missing (pre-migration deployment) — no enforcement surface yet;
   697→   697→ *      returns false with a loud one-time warning, identical to the Phase 1
   698→   698→ *      contract (enforcement begins the moment migration 20261101000002 lands).
   699→   699→ */
   700→   700→export async function mfaEnrollmentGate(role: string, adminId: string): Promise<{ enrollmentRequired: boolean }> {
   701→   701→  if (!roleRequiresMfa(role)) return { enrollmentRequired: false };
   702→   702→  let row: { confirmedAt: Date | null } | null;
   703→   703→  try {
   704→   704→    row = await db.adminMfaEnrollment.findUnique({ where: { adminId }, select: { confirmedAt: true } });
   705→   705→  } catch (e) {
   706→   706→    if (isPrismaMissingTableError(e)) {
   707→   707→      if (!gateWarned) {
   708→   708→        gateWarned = true;
   709→   709→        console.warn("[mfa] enrollment table missing — mandatory-enrollment gate inactive until 20261101000002 is applied");
   710→   710→      }
   711→   711→      return { enrollmentRequired: false };
   712→   712→    }
   713→   713→    throw e;
   714→   714→  }
   715→   715→  return { enrollmentRequired: !row?.confirmedAt };
   716→   716→}
