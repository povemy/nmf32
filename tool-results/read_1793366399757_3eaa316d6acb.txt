     2→// AFFILIATE HUB & ONE-LEVEL SPONSOR REFERRAL SYSTEM — core domain library.
     3→// docs/affiliate/*.md are the authoritative design docs.
     4→//
     5→// COMMERCIAL MODEL (fixed, spec §1/§8):
     6→//   • Seller-funded commission pool per eligible OrderItem: P = B × R (bps,
     7→//     integer sen). The pool is carved OUT of seller proceeds at checkout
     8→//     (sellerNet already subtracts it — see computeOrderCommissions).
     9→//   • Split of the POOL: S = floor(P × 10 / 100) to the direct affiliate's
    10→//     ONE-LEVEL sponsor; D = P − S to the direct affiliate. No sponsor ⇒
    11→//     D = P, S = 0. D + S = P ALWAYS (never exceeds the pool).
    12→//   • Attribution travels UP exactly one generation. No recruitment bonuses,
    13→//     no click rewards, no registration rewards — liabilities only ever
    14→//     originate from genuine eligible paid sales.
    15→//
    16→// INTEGRITY CONTRACTS:
    17→//   • Pool snapshots freeze on OrderItem at checkout (commission contract).
    18→//   • AffiliateCommission rows post EXACTLY ONCE per order item at payment
    19→//     confirmation (unique orderItemId; duplicate webhooks are no-ops).
    20→//   • The 90/10 SPLIT is computed at POSTING time from the then-current
    21→//     sponsor-override flag + sponsor eligibility, then never rewritten —
    22→//     so the override can be disabled without corrupting existing records.
    23→//   • Every money mutation rides the immutable ledger (lib/ledger.ts).

    73→function clampInt(v: unknown, min: number, max: number, fallback: number): number {
    74→  const n = typeof v === "number" ? v : Number(v);
    75→  if (!Number.isFinite(n)) return fallback;
    76→  return Math.min(max, Math.max(min, Math.round(n)));
    77→}

    79→export async function getAffiliateSettings(): Promise<AffiliateSettings> {
    80→  const rows = await db.setting.findMany({ where: { key: { in: Object.values(AFFILIATE_SETTING_KEYS) } } });
    81→  const map = new Map(rows.map((r) => [r.key, r.valueJson]));
    82→  const bool = (k: string, d: boolean) => {

    84→    return typeof v === "boolean" ? v : d;
    85→  };

    87→    enabled: bool(AFFILIATE_SETTING_KEYS.enabled, DEFAULTS.enabled),
    88→    sponsorEnabled: bool(AFFILIATE_SETTING_KEYS.sponsorEnabled, DEFAULTS.sponsorEnabled),

    97→/** Effective seller affiliate rate (bps) for a product, honouring every gate.
    98→ *  0 ⇒ not enrolled (no commission). Product override ?? store default,
    99→ *  clamped to the platform max; programme master switch must be on. */
   100→export function effectiveAffiliateRateBps(
   101→  s: AffiliateSettings,
   102→  store: { affiliateOptIn: boolean; affiliateDefaultRateBps: number },
   103→  product: { affiliateEnabled: boolean; affiliateRateBps: number | null }
   104→): number {
   105→  if (!s.enabled || !store.affiliateOptIn || !product.affiliateEnabled) return 0;
   106→  const rate = product.affiliateRateBps ?? store.affiliateDefaultRateBps;
   107→  return Math.max(0, Math.min(rate, s.maxRateBps));
   108→}

   113→/** Pool P = B × R in integer sen, half-up rounding (platform convention). */
   114→export function affiliatePoolMinor(baseMinor: number, rateBps: number): number {
   115→  if (baseMinor <= 0 || rateBps <= 0) return 0;
   116→  return divHalfUp(baseMinor * rateBps, 10000);
   117→}
   118→
   119→/** 90/10 split of the POOL: S = floor(P × 10 / 100); D = P − S (D + S = P). */
   120→export function splitAffiliatePool(poolMinor: number): { directMinor: number; sponsorMinor: number } {
   121→  if (poolMinor <= 0) return { directMinor: 0, sponsorMinor: 0 };
   122→  const sponsorMinor = Math.floor((poolMinor * 10) / 100);
   123→  return { directMinor: poolMinor - sponsorMinor, sponsorMinor };
   124→}

   166→function slugifyAffiliate(name: string): string {
   167→  const base = name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 24);
   168→  return base || "affiliate";
   169→}

   171→async function uniqueSlug(base: string): Promise<string> {
   172→  let candidate = base;
   173→  for (let i = 0; i < 20; i++) {
   174→    const clash = await db.affiliateProfile.findUnique({ where: { slug: candidate }, select: { id: true } });
   175→    if (!clash) return candidate;
   176→    candidate = `${base}-${randomToken(2)}`;
   177→  }
   178→  return `aff-${randomToken(4)}`;
   179→}

   181→/** Read the FIRST-TOUCH referral cookie and validate it against the touch
   182→ *  table (unexpired, unconsumed). Returns the sponsor profile or null. */
   183→export async function resolveReferralCookie(): Promise<{ affiliateId: string; displayName: string; code: string } | null> {
   184→  try {
   185→    const store = await cookies();
   186→    const raw = store.get(REFERRAL_COOKIE)?.value;
   187→    if (!raw) return null;
   188→    const parsed = parseJson<{ code?: string }>(raw, {});
   189→    if (!parsed?.code) return null;
   190→    const touch = await db.affiliateReferralTouch.findFirst({
   191→      where: { code: parsed.code, consumedAt: null, expiresAt: { gt: new Date() } },
   192→      orderBy: { touchedAt: "desc" },
   193→      select: { affiliateId: true, code: true },
   194→    });
   195→    const profile = touch
   196→      ? await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId }, select: { id: true, displayName: true, code: true } })
   197→      : null;
   198→    if (!profile || profile.status !== "active") return null;
   199→    return { affiliateId: profile.id, displayName: profile.displayName, code: profile.code };
   200→  } catch {
   201→    return null;
   202→  }
   203→}

   228→/** Activate the current shopper as an affiliate. Idempotent (already-active →
   229→ *  error with the existing profile). Sponsor precedence (spec §5):
   230→ *  1. Valid first-touch referral cookie (30-day window, unconsumed);
   231→ *  2. Else a one-time manual code entry — confirmed explicitly;
   232→ *  3. Else no sponsor (recorded permanently as none).
   233→ *  The relationship locks forever after activation: no click, seller or admin
   234→ *  tool can casually change it (admin corrections are separately audited). */
   235→export async function activateAffiliate(input: ActivationInput) {
   236→  const settings = await getAffiliateSettings();
   237→  if (!settings.enabled) {
   238→    throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   239→  }
   240→
   241→  const existing = await db.affiliateProfile.findUnique({ where: { userId: input.userId } });
   242→  if (existing) {
   243→    throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   244→  }
   245→
   246→  // ---- Sponsor resolution (server-side only) -----------------------------
   247→  let sponsor: { id: string; displayName: string } | null = null;
   248→  let sponsorSource: string | null = null;
   249→
   250→  const touch = await resolveReferralCookie();
   251→  if (touch) {
   252→    const profile = await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId } });
   253→    if (profile && profile.status === "active" && profile.userId !== input.userId) {
   254→      sponsor = { id: profile.id, displayName: profile.displayName };
   255→      sponsorSource = "referral_touch";
   256→    }
   257→  }
   258→  if (!sponsor && input.sponsorCode) {
   259→    const candidate = await findAffiliateByCode(input.sponsorCode);
   260→    if (!candidate) throw new ApiError(400, "INVALID_SPONSOR_CODE", "That referral code was not found");
   261→    if (candidate.userId === input.userId) {
   262→      throw new ApiError(400, "SELF_REFERRAL", "You cannot refer yourself");
   263→    }
   264→    if (candidate.status !== "active") {
   265→      throw new ApiError(400, "SPONSOR_INACTIVE", "That sponsor is not currently eligible");
   266→    }
   267→    if (!input.sponsorConfirmed) {
   268→      throw new ApiError(400, "SPONSOR_CONFIRMATION_REQUIRED", "Confirm the sponsor relationship before activating");
   269→    }
   270→    sponsor = { id: candidate.id, displayName: candidate.displayName };
   271→    sponsorSource = "code_entry";
   272→  }
   273→
   274→  const code = randomToken(5); // 10-hex-char referral code, unpredictable
   275→  const slug = await uniqueSlug(slugifyAffiliate(input.slug || input.displayName || input.fullName));
   276→
   277→  const profile = await db.$transaction(async (tx) => {
   278→    // Re-check inside the tx (concurrent double-submit).
   279→    const dupe = await tx.affiliateProfile.findUnique({ where: { userId: input.userId } });
   280→    if (dupe) throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   281→
   282→    const created = await tx.affiliateProfile.create({
   283→      data: {
   284→        userId: input.userId,
   285→        code,
   286→        slug,
   287→        displayName: (input.displayName || input.fullName).slice(0, 60),
   288→        bio: input.bio?.slice(0, 500) ?? null,
   289→        status: "active",
   290→        termsVersion: AFFILIATE_TERMS_VERSION,
   291→        termsAcceptedAt: new Date(),
   292→        activatedAt: new Date(),
   293→        sponsorId: sponsor?.id ?? null,
   294→        sponsorAssignedAt: sponsor ? new Date() : null,
   295→        sponsorSource,
   296→      },
   297→    });
   298→
   299→    // Consume the referral touch exactly once (relationship now recorded).
   300→    if (sponsor && sponsorSource === "referral_touch" && touch) {
   301→      await tx.affiliateReferralTouch.updateMany({
   302→        where: { code: touch.code, consumedAt: null },
   303→        data: { consumedAt: new Date(), consumedUserId: input.userId },
   304→      });
   305→    }
   306→    return created;
   307→  });
   308→
   309→  return { profile, sponsor, sponsorSource };
   310→}

   312→/** Admin-only sponsor correction — audited, with documented reason. Cycle
   313→ *  guard: with one-level relationships a 2-cycle (A↔B) is the only possible
   314→ *  loop; rejecting `newSponsor.sponsorId === affiliateId` kills it. */
   315→export async function correctSponsor(input: {
   316→  affiliateId: string;
   317→  newSponsorId: string | null;
   318→  reason: string;
   319→  adminId: string;
   320→}) {
   321→  if (!input.reason || input.reason.trim().length < 10) {
   322→    throw new ApiError(400, "VALIDATION", "A documented reason (≥10 chars) is required for sponsor corrections");
   323→  }
   324→  const affiliate = await db.affiliateProfile.findUnique({ where: { id: input.affiliateId } });
   325→  if (!affiliate) throw new ApiError(404, "NOT_FOUND", "Affiliate not found");
   326→  if (input.newSponsorId === affiliate.id) {
   327→    throw new ApiError(400, "SELF_REFERRAL", "An affiliate cannot sponsor themselves");
   328→  }
   329→  if (input.newSponsorId) {
   330→    const sponsor = await db.affiliateProfile.findUnique({ where: { id: input.newSponsorId } });
   331→    if (!sponsor) throw new ApiError(404, "NOT_FOUND", "New sponsor not found");
   332→    if (sponsor.sponsorId === affiliate.id) {
   333→      throw new ApiError(400, "SPONSOR_CYCLE", "This correction would create a sponsor cycle");
   334→    }
   335→  }
   336→  return db.$transaction(async (tx) => {
   337→    const updated = await tx.affiliateProfile.update({
   338→      where: { id: input.affiliateId },
   339→      data: {
   340→        sponsorId: input.newSponsorId,
   341→        sponsorAssignedAt: input.newSponsorId ? new Date() : null,
   342→        sponsorSource: "admin_correction",
   343→      },
   344→    });
   345→    await tx.affiliateSponsorAudit.create({
   346→      data: {
   347→        affiliateId: input.affiliateId,
   348→        fromSponsorId: affiliate.sponsorId,
   349→        toSponsorId: input.newSponsorId,
   350→        reason: input.reason.trim(),
   351→        actorType: "admin",
   352→        actorId: input.adminId,
   353→      },
   354→    });
   355→    return updated;
   356→  });
   357→}

   365→/** Post AffiliateCommission rows for one paid order. Idempotent: unique
   366→ *  orderItemId + find-first guard. Split computed HERE (posting time) from
   367→ *  the then-current sponsor flag + sponsor eligibility. */
   368→export async function postAffiliateCommissionsForOrder(
   369→  tx: Tx,
   370→  order: { id: string; orderNumber: string; storeId: string; items: { id: string; productId?: string | null; affiliateDirectId: string | null; affiliateSponsorId: string | null; affiliateLinkId: string | null; affiliateRateBps: number | null; affiliatePoolMinor: number; breakdownJson: string }[] },
   371→  paidAt: Date
   372→): Promise<number> {
   373→  const attributed = order.items.filter((i) => i.affiliateDirectId && i.affiliatePoolMinor > 0);
   374→  if (attributed.length === 0) return 0;
   375→
   376→  const settings = await getAffiliateSettings();
   377→  if (!settings.enabled) return 0; // kill switch: no NEW liabilities
   378→
   379→  // Sponsor profiles snapshot (status at posting decides hold vs pending).
   380→  const sponsorIds = [...new Set(attributed.map((i) => i.affiliateSponsorId).filter((s): s is string => Boolean(s)))];
   381→  const sponsors = sponsorIds.length
   382→    ? new Map((await tx.affiliateProfile.findMany({ where: { id: { in: sponsorIds } }, select: { id: true, status: true } })).map((p) => [p.id, p.status]))
   383→    : new Map<string, string>();
   384→
   385→  // Link metadata snapshot (linkType + channel for reporting — spec §14).
   386→  const linkIds = [...new Set(attributed.map((i) => i.affiliateLinkId).filter((s): s is string => Boolean(s)))];
   387→  const links = linkIds.length
   388→    ? new Map((await tx.affiliateLink.findMany({ where: { id: { in: linkIds } }, select: { id: true, type: true, channel: true } })).map((l) => [l.id, l]))
   389→    : new Map<string, { type: string; channel: string }>();
   390→
   391→  let posted = 0;
   392→  for (const item of attributed) {
   393→    const existing = await tx.affiliateCommission.findUnique({ where: { orderItemId: item.id }, select: { id: true } });
   394→    if (existing) continue; // idempotent replay
   395→
   396→    const pool = item.affiliatePoolMinor;
   397→    let sponsorMinor = 0;
   398→    let sponsorHeld = false;
   399→    if (item.affiliateSponsorId && settings.sponsorEnabled) {
   400→      const status = sponsors.get(item.affiliateSponsorId) ?? "missing";
   401→      if (status === "active") {
   402→        sponsorMinor = Math.floor((pool * 10) / 100);
   403→      } else {
   404→        // Sponsor under review / gone: HOLD the 10% separately — the direct
   405→        // promoter's 90% must NOT absorb it and must NOT be penalised.
   406→        sponsorMinor = Math.floor((pool * 10) / 100);
   407→        sponsorHeld = true;
   408→      }
   409→    }
   410→    const directMinor = pool - sponsorMinor; // D + S = P always
   411→    const bd = parseJson<{ base?: number }>(item.breakdownJson, {});
   412→
   413→    const commission = await tx.affiliateCommission.create({
   414→      data: {
   415→        orderItemId: item.id,
   416→        orderId: order.id,
   417→        orderNumber: order.orderNumber,
   418→        storeId: order.storeId,
   419→        productId: item.productId ?? null,
   420→        affiliateId: item.affiliateDirectId!,
   421→        sponsorId: item.affiliateSponsorId ?? null,
   422→        linkId: item.affiliateLinkId ?? null,
   423→        linkType: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.type ?? null) : null,
   424→        channel: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.channel ?? null) : null,
   425→        commissionableBaseMinor: Math.max(0, bd.base ?? 0),
   426→        rateBps: item.affiliateRateBps ?? 0,
   427→        poolMinor: pool,
   428→        status: "pending",
   429→        breakdownJson: item.breakdownJson,
   430→      },
   431→    });
   432→
   433→    await tx.affiliateCommissionEntry.create({
   434→      data: {
   435→        commissionId: commission.id,
   436→        beneficiaryAffiliateId: item.affiliateDirectId!,
   437→        role: "direct",
   438→        amountMinor: directMinor,
   439→        status: "pending",
   440→      },
   441→    });
   442→    if (item.affiliateSponsorId && sponsorMinor > 0) {
   443→      await tx.affiliateCommissionEntry.create({
   444→        data: {
   445→          commissionId: commission.id,
   446→          beneficiaryAffiliateId: item.affiliateSponsorId,
   447→          role: "sponsor",
   448→          amountMinor: sponsorMinor,
   449→          status: sponsorHeld ? "held" : "pending",
   450→          holdReason: sponsorHeld ? "sponsor_under_review" : null,
   451→        },
   452→      });
   453→    }
   454→    posted++;
   455→  }
   456→  return posted;
   457→}

   459→// ---------------------------------------------------------------------------
   460→// Lifecycle: pending → approved (order delivered/completed) → available
   461→// (review window elapsed) → paid. Reversals from refunds. Holds from review.
   462→// ---------------------------------------------------------------------------
   463→/** Approve an order's pending entries (authoritative delivered/completed). */
   464→export async function approveAffiliateCommissionsForOrder(orderId: string, tx?: Tx): Promise<number> {
   465→  const client = tx ?? db;
   466→  const commissions = await client.affiliateCommission.findMany({
   467→    where: { orderId, status: "pending" },
   468→    select: { id: true },
   469→  });
   470→  if (commissions.length === 0) return 0;
   471→  const settings = await getAffiliateSettings();
   472→  const now = new Date();
   473→  const availableAt = new Date(now.getTime() + settings.reviewDays * 86400000);
   474→  let approved = 0;
   475→  for (const c of commissions) {
   476→    const res = await client.affiliateCommissionEntry.updateMany({
   477→      where: { commissionId: c.id, status: "pending" },
   478→      data: { status: "approved", availableAt },
   479→    });
   480→    if (res.count > 0) {
   481→      await client.affiliateCommission.update({ where: { id: c.id }, data: { status: "approved", approvedAt: now } });
   482→      approved += res.count;
   483→    }
   484→  }
   485→  return approved;
   486→}

   488→/** Batch sweep (cron): approve commissions on delivered/completed orders +
   489→ *  mature approved entries whose review window elapsed. Bounded batches. */
   490→export async function sweepAffiliateLifecycle(): Promise<{ approved: number; matured: number }> {
   491→  const settings = await getAffiliateSettings();
   492→  if (!settings.enabled) return { approved: 0, matured: 0 };
   493→
   494→  // 1) Orders delivered/completed with pending commissions → approve.
   495→  const pendingCommissions = await db.affiliateCommission.findMany({
   496→    where: { status: "pending" },
   497→    select: { orderId: true },
   498→    distinct: ["orderId"],
   499→    take: 50,
   500→  });
   501→  let approved = 0;
   502→  if (pendingCommissions.length > 0) {
   503→    const orderIds = pendingCommissions.map((c) => c.orderId);
   504→    const eligible = await db.order.findMany({
   505→      where: { id: { in: orderIds }, status: { in: ["delivered", "completed"] } },
   506→      select: { id: true },
   507→    });
   508→    for (const o of eligible) {
   509→      approved += await approveAffiliateCommissionsForOrder(o.id);
   510→    }
   511→  }
   512→
   513→  // 2) Matured approved entries → available (clock transition, CAS-guarded).
   514→  const now = new Date();
   515→  const matured = await db.affiliateCommissionEntry.updateMany({
   516→    where: { status: "approved", availableAt: { lte: now } },
   517→    data: { status: "available" },
   518→  });
   519→  if (matured.count > 0) {
   520→    await db.affiliateCommission.updateMany({
   521→      where: { status: "approved", entries: { some: { status: "available" } } },
   522→      data: { status: "available" },
   523→    });
   524→  }
   525→  return { approved, matured: matured.count };
   526→}
   527→
   528→// ---------------------------------------------------------------------------
   529→// Balances (affiliate-side truth = AffiliateCommissionEntry rows; the ledger
   530→// mirrors totals on PLATFORM:AFFILIATE_LIABILITY for double-entry integrity).
   531→// ---------------------------------------------------------------------------
   532→export type AffiliateBalances = {
   533→  pendingMinor: number;
   534→  approvedMinor: number; // approved but review window not elapsed
   535→  availableMinor: number; // matured, not yet claimed by a payout
   536→  payoutInFlightMinor: number; // claimed by requested/processing payouts
   537→  paidMinor: number;
   538→  reversedMinor: number;
   539→  heldMinor: number;
   540→  recoveryMinor: number; // reversals booked against already-paid entries
   541→  requestableMinor: number; // available − recovery, floored at 0
   542→};
   543→
   544→export async function affiliateBalances(affiliateId: string): Promise<AffiliateBalances> {
   545→  const entries = await db.affiliateCommissionEntry.findMany({
   546→    where: { beneficiaryAffiliateId: affiliateId },
   547→    select: { status: true, amountMinor: true, reversedMinor: true, availableAt: true, payoutId: true },
   548→  });
   549→  const now = Date.now();
   550→  const b: AffiliateBalances = {
   551→    pendingMinor: 0, approvedMinor: 0, availableMinor: 0, payoutInFlightMinor: 0,
   552→    paidMinor: 0, reversedMinor: 0, heldMinor: 0, recoveryMinor: 0, requestableMinor: 0,
   553→  };
   554→  for (const e of entries) {
   555→    const matured = (e.availableAt?.getTime() ?? 0) <= now;
   556→    switch (e.status) {
   557→      case "pending":
   558→      case "approved":
   559→        if (e.status === "pending" || !matured) {
   560→          if (e.status === "pending") b.pendingMinor += e.amountMinor - e.reversedMinor;
   561→          else b.approvedMinor += e.amountMinor - e.reversedMinor;
   562→          break;
   563→        }
   564→        // fallthrough: matured approved
   565→      case "available":
   566→        if (e.payoutId) b.payoutInFlightMinor += e.amountMinor - e.reversedMinor;
   567→        else b.availableMinor += e.amountMinor - e.reversedMinor;
   568→        break;
   569→      case "paid":
   570→        b.paidMinor += e.amountMinor;
   571→        b.recoveryMinor += e.reversedMinor; // late refund after payout
   572→        break;
   573→      case "reversed":
   574→        b.reversedMinor += e.amountMinor;
   575→        break;
   576→      case "held":
   577→        b.heldMinor += e.amountMinor - e.reversedMinor;
   578→        break;
   579→    }
   580→  }
   581→  b.requestableMinor = Math.max(0, b.availableMinor - b.recoveryMinor);
   582→  return b;
   583→}

   585→// ---------------------------------------------------------------------------
   586→// Refund reversals — proportional against the ORIGINAL immutable allocations
   587→// (spec §11). Shipping-only refunds book no product-commission reversal (the
   588→// pool base excluded shipping by construction). Late refunds after payout
   589→// become recovery entries: reversedMinor accumulates on the PAID entry and
   590→// offsets future earnings — the original payout history is never rewritten.
   591→// ---------------------------------------------------------------------------
   592→export async function reverseAffiliateForRefund(
   593→  tx: Tx,
   594→  input: {
   595→    refundId: string;
   596→    orderId: string;
   597→    refundAmountMinor: number;
   598→    /** Fresh order items (with affiliate snapshots) under the lock. */
   599→    items: { id: string; lineSubtotalMinor: number; refundedMinor: number; breakdownJson: string; affiliatePoolMinor: number }[];
   600→    /** Σ affiliateReversalMinor of prior PROCESSED refunds on this order. */
   601→    priorAffiliateReversalMinor: number;
   602→  }
   603→): Promise<{ affiliateReversalMinor: number; rule: string }> {
   604→  const attributed = input.items.filter((i) => i.affiliatePoolMinor > 0);
   605→  if (attributed.length === 0) return { affiliateReversalMinor: 0, rule: "no_affiliate" };
   606→
   607→  // Customer-paid gross per item (breakdown-proportional, mirroring
   608→  // deriveCommissionReversal's allocation family).
   609→  const grossArr = attributed.map((it) => {
   610→    const bd = parseJson<{ discount_allocated?: number; shipping_allocated?: number }>(it.breakdownJson, {});
   611→    return Math.max(0, it.lineSubtotalMinor - (bd.discount_allocated ?? 0) + (bd.shipping_allocated ?? 0));
   612→  });
   613→  const allocs: number[] = [];
   614→  {
   615→    // floor + remainder-to-largest (same as allocateProportionally)
   616→    const totalGross = grossArr.reduce((s, v) => s + v, 0);
   617→    if (totalGross <= 0) {
   618→      attributed.forEach(() => allocs.push(0));
   619→    } else {
   620→      let allocated = 0;
   621→      for (let i = 0; i < attributed.length; i++) {
   622→        const a = Math.floor((input.refundAmountMinor * grossArr[i]) / totalGross);
   623→        allocs.push(a);
   624→        allocated += a;
   625→      }
   626→      let rem = input.refundAmountMinor - allocated;
   627→      while (rem > 0) {
   628→        let maxIdx = 0;
   629→        for (let i = 1; i < grossArr.length; i++) if (grossArr[i] > grossArr[maxIdx]) maxIdx = i;
   630→        allocs[maxIdx] += 1;
   631→        rem -= 1;
   632→      }
   633→    }
   634→  }
   635→
   636→  // Per-item affiliate reversal, clamped to the remaining reversible pool.
   637→  const orderPool = attributed.reduce((s, it) => s + it.affiliatePoolMinor, 0);
   638→  const remainingReversible = Math.max(0, orderPool - input.priorAffiliateReversalMinor);
   639→  let total = 0;
   640→  const perItem: { item: (typeof attributed)[number]; reversal: number }[] = [];
   641→  for (let i = 0; i < attributed.length; i++) {
   642→    if (grossArr[i] <= 0 || allocs[i] <= 0) {
   643→      perItem.push({ item: attributed[i], reversal: 0 });
   644→      continue;
   645→    }
   646→    const r = Math.min(divHalfUp(allocs[i] * attributed[i].affiliatePoolMinor, grossArr[i]), attributed[i].affiliatePoolMinor);
   647→    perItem.push({ item: attributed[i], reversal: r });
   648→    total += r;
   649→  }
   650→  total = Math.min(Math.max(0, total), input.refundAmountMinor, remainingReversible);
   651→
   652→  // Apply to the commission + entries (proportional split within each pool).
   653→  if (total > 0) {
   654→    // Scale per-item reversals down proportionally if the clamp reduced total.
   655→    const rawTotal = perItem.reduce((s, p) => s + p.reversal, 0);
   656→    const scale = rawTotal > 0 ? total / rawTotal : 0;
   657→    let applied = 0;
   658→    for (const p of perItem) {
   659→      const itemReversal = p.reversal === 0 ? 0 : Math.min(p.reversal, Math.round(p.reversal * scale));
   660→      if (itemReversal <= 0) continue;
   661→      const commission = await tx.affiliateCommission.findUnique({ where: { orderItemId: p.item.id }, include: { entries: true } });
   662→      if (!commission) continue; // pre-affiliate legacy rows: nothing to reverse
   663→      const direct = commission.entries.find((e) => e.role === "direct");
   664→      const sponsor = commission.entries.find((e) => e.role === "sponsor");
   665→      // Split proportionally to the frozen allocation shares.
   666→      const directShare = direct ? Math.min(direct.amountMinor - direct.reversedMinor, divHalfUp(itemReversal * direct.amountMinor, commission.poolMinor)) : 0;
   667→      const sponsorShare = sponsor ? Math.min(sponsor.amountMinor - sponsor.reversedMinor, itemReversal - directShare) : 0;
   668→      for (const [entry, share] of [[direct, directShare], [sponsor, sponsorShare]] as const) {
   669→        if (!entry || share <= 0) continue;
   670→        const newReversed = Math.min(entry.amountMinor, entry.reversedMinor + share);
   671→        const fullyReversed = newReversed >= entry.amountMinor;
   672→        await tx.affiliateCommissionEntry.update({
   673→          where: { id: entry.id },
   674→          data: {
   675→            reversedMinor: newReversed,
   676→            // PAID entries stay paid — the reversal is a recovery entry that
   677→            // offsets future earnings (spec §11 late refunds); pending/
   678→            // approved/available entries flip to reversed.
   679→            ...(entry.status !== "paid" && fullyReversed ? { status: "reversed" as const } : {}),
   680→          },
   681→        });
   682→      }
   683→      const anyLive = commission.entries.some((e) => e.status !== "reversed");
   684→      if (!anyLive) {
   685→        await tx.affiliateCommission.update({
   686→          where: { id: commission.id },
   687→          data: { status: "reversed", reversedAt: new Date() },
   688→        });
   689→      }
   690→      applied += itemReversal;
   691→    }
   692→    void applied;
   693→  }
   694→
   695→  return { affiliateReversalMinor: total, rule: "breakdown_snapshot_proportional" };
   696→}
   697→export async function requestAffiliatePayout(affiliateId: string): Promise<{ payoutId: string; amountMinor: number }> {
   698→  const settings = await getAffiliateSettings();
   699→  if (!settings.enabled) throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   700→
   701→  return db.$transaction(async (tx) => {
   702→    // One open request per affiliate — re-checked INSIDE the transaction so a
   703→    // concurrent double-submit cannot create two claims.
   704→    const open = await tx.affiliatePayout.findFirst({
   705→      where: { affiliateId, status: { in: ["requested", "processing"] } },
   706→      select: { id: true },
   707→    });
   708→    if (open) throw new ApiError(409, "PAYOUT_OPEN", "You already have a payout request in progress");
   709→
   710→    const balances = await affiliateBalances(affiliateId);
   711→    if (balances.requestableMinor < settings.minPayoutMinor) {
   712→      throw new ApiError(400, "BELOW_MIN_PAYOUT", `Minimum payout is RM ${(settings.minPayoutMinor / 100).toFixed(2)}`);
   713→    }
   714→
   715→    const amountMinor = balances.requestableMinor;
   716→    const now = new Date();
   717→    // Claim every matured, unclaimed entry; bind to the payout row we create.
   718→    const toClaim = await tx.affiliateCommissionEntry.findMany({
   719→      where: {
   720→        beneficiaryAffiliateId: affiliateId,
   721→        status: { in: ["approved", "available"] },
   722→        availableAt: { lte: now },
   723→        payoutId: null,
   724→      },
   725→      select: { id: true, amountMinor: true, reversedMinor: true, commissionId: true },
   726→    });
   727→    const payout = await tx.affiliatePayout.create({
   728→      data: { affiliateId, amountMinor, status: "requested", method: "manual_bank" },
   729→    });
   730→    if (toClaim.length > 0) {
   731→      await tx.affiliateCommissionEntry.updateMany({
   732→        where: { id: { in: toClaim.map((e) => e.id) } },
   733→        data: { payoutId: payout.id },
   734→      });
   735→    }
   736→    // Ledger-derived statement (schema contract: built from entries at
   737→    // request time, never hand-edited) — order references + balance snapshot.
   738→    const orderRefs = toClaim.length
   739→      ? await tx.affiliateCommission.findMany({
   740→          where: { id: { in: [...new Set(toClaim.map((e) => e.commissionId))] } },
   741→          select: { id: true, orderNumber: true, productId: true, storeId: true },
   742→        })
   743→      : [];
   744→    const refMap = new Map(orderRefs.map((c) => [c.id, c]));
   745→    await tx.affiliatePayout.update({
   746→      where: { id: payout.id },
   747→      data: {
   748→        statementJson: toJson({
   749→          builtAt: now.toISOString(),
   750→          balances: balances,
   751→          entries: toClaim.map((e) => ({
   752→            entryId: e.id,
   753→            commissionId: e.commissionId,
   754→            orderNumber: refMap.get(e.commissionId)?.orderNumber ?? null,
   755→            amountMinor: e.amountMinor,
   756→            reversedMinor: e.reversedMinor,
   757→          })),
   758→        }),
   759→      },
   760→    });
   761→    return { payoutId: payout.id, amountMinor };
   762→  });
   763→}
