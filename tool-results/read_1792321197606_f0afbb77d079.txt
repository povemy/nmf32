    81→    81→/** Total unread seller messages — powers the "My messages" header badge.
    82→    82→ *  Polls lightly + refreshes when the messages view is opened/left. */
    83→    83→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    84→    84→  const [unread, setUnread] = useState(0);
    85→    85→  useEffect(() => {
    86→    86→    if (!shopperSignedIn) {
    87→    87→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    88→    88→      setUnread(0);
    89→    89→      return;
    90→    90→    }
    91→    91→    let alive = true;
    92→    92→    const load = () =>
    93→    93→      fetchJson<{ total: number }>("/api/chat/unread")
    94→    94→        .then((d) => alive && setUnread(d.total ?? 0))
    95→    95→        .catch(() => {});
    96→    96→    load();
    97→    97→    const t = setInterval(load, 15000);
    98→    98→    return () => {
    99→    99→      alive = false;
   100→   100→      clearInterval(t);
   101→   101→    };
   102→   102→  }, [shopperSignedIn]);
   103→   103→  // instant re-sync when a thread is read inside the messages view (the view
   104→   104→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   105→   105→  useEffect(() => {
   106→   106→    if (!shopperSignedIn) return;
   107→   107→    const sync = () =>
   108→   108→      fetchJson<{ total: number }>("/api/chat/unread")
   109→   109→        .then((d) => setUnread(d.total ?? 0))
   110→   110→        .catch(() => {});
   111→   111→    window.addEventListener("nomofee:chat-sync", sync);
   112→   112→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   113→   113→  }, [shopperSignedIn]);
   114→   114→  // opening (or leaving) the messages view should immediately re-sync the badge
   115→   115→  useEffect(() => {
   116→   116→    if (!shopperSignedIn) return;
   117→   117→    const t = setTimeout(
   118→   118→      () =>
   119→   119→        fetchJson<{ total: number }>("/api/chat/unread")
   120→   120→          .then((d) => setUnread(d.total ?? 0))
   121→   121→          .catch(() => {}),
   122→   122→      viewName === "messages" ? 600 : 0
   123→   123→    );
   124→   124→    return () => clearTimeout(t);
   125→   125→  }, [shopperSignedIn, viewName]);
   126→   126→  return unread;
   127→   127→}
