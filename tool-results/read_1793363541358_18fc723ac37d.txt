   212→export class ApiFail extends Error {
   213→  status: number;
   214→  code: string;
   215→  constructor(status: number, code: string, message: string) {
   216→    super(message);
   217→    this.status = status;
   218→    this.code = code;
   219→  }
   220→}
   221→
   222→export async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   223→  const res = await fetch(url, {
   224→    cache: "no-store",
   225→    ...init,
   226→    // Device identity headers (cart + shopper token) — keeps guest state working
   227→    // where cookies are unavailable (cross-site iframe preview panels).
   228→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   229→  });
   230→  let body: unknown = null;
   231→  try {
   232→    body = await res.json();
   233→  } catch {
   234→    /* non-json */
   235→  }
   236→  const b = body as { data?: T; error?: { code?: string; message?: string } } | null;
   237→  if (!res.ok || !b || b.error) {
   238→    // 401 → the shopper session is gone (signed out elsewhere / expired /
   239→    // revoked). Signal app-root to re-check identity; guests stay guests.
   240→    if (res.status === 401) signalSessionExpired();
   241→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   242→  }
   243→  return (b.data ?? ({} as T)) as T;
   244→}
   245→
   246→/** Like fetchJson but also returns the paged meta envelope. */
   247→export async function fetchEnvelope<T>(
   248→  url: string,
   249→  init?: RequestInit
   250→): Promise<{ data: T; meta?: { page: number; size: number; total: number } }> {
   251→  const res = await fetch(url, {
   252→    cache: "no-store",
   253→    ...init,
   254→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   255→  });
   256→  let body: unknown = null;
   257→  try {
   258→    body = await res.json();
   259→  } catch {
   260→    /* non-json */
   261→  }
   262→  const b = body as { data?: T; meta?: { page: number; size: number; total: number }; error?: { code?: string; message?: string } } | null;
   263→  if (!res.ok || !b || b.error) {
   264→    if (res.status === 401) signalSessionExpired();
   265→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   266→  }
   267→  return { data: (b.data ?? ({} as T)) as T, meta: b.meta };
   268→}

   308→export function trackOrderRef(orderNumber: string, email: string) {
   309→  if (typeof window === "undefined") return;
   310→  try {
   311→    const raw = JSON.parse(window.localStorage.getItem(ORDER_REFS_KEY) ?? "[]") as OrderRefT[];
   312→    const clean = raw.filter((r) => r.orderNumber !== orderNumber);
   313→    clean.unshift({ orderNumber, email: email.toLowerCase(), at: Date.now() });
   314→    window.localStorage.setItem(ORDER_REFS_KEY, JSON.stringify(clean.slice(0, 25)));
   315→  } catch {
   316→    /* corrupted list — ignore */
   317→  }
   318→}

   320→export function readOrderRefs(): OrderRefT[] {
   321→  if (typeof window === "undefined") return [];
   322→  try {
   323→    const raw = JSON.parse(window.localStorage.getItem(ORDER_REFS_KEY) ?? "[]") as OrderRefT[];
   324→    return Array.isArray(raw) ? raw.filter((r) => r && r.orderNumber && r.email) : [];
   325→  } catch {
   326→    return [];
   327→  }

   330→/** Fetch the account cart and live-refresh on sm-cart-updated events + on any
   331→ *  sign-in / sign-out change (carts are account-scoped — guests have none).
   332→ *
   333→ *  Task 28: cart state lives in a MODULE-LEVEL shared store, so every useCart()
   334→ *  instance (drawer, header badge, checkout) renders from the same snapshot —
   335→ *  one fetch per event instead of one per instance — and checkout-selection
   336→ *  toggles can mutate it optimistically (mutateCartItems): the UI updates in
   337→ *  the same frame the user clicks while the PATCH rides in the background.
   338→ *  Zero network in the render path — same philosophy as the client-side
   339→ *  Messages list filter. */
   340→const EMPTY_CART: CartT = { id: null, storeId: null, storeSlug: null, storeName: null, items: [] };
   341→
   342→type CartState = { cart: CartT; loading: boolean };
   343→let cartState: CartState = { cart: EMPTY_CART, loading: true };
   344→const CART_SERVER_STATE: CartState = { cart: EMPTY_CART, loading: true };
   345→const cartSubs = new Set<() => void>();
   346→
   347→function emitCart() {
   348→  for (const fn of cartSubs) fn();
   349→}
   350→
   351→function subscribeCart(fn: () => void) {
   352→  cartSubs.add(fn);
   353→  return () => {
   354→    cartSubs.delete(fn);
   355→  };
   356→}
   357→
   358→/** Optimistically mutate the shared cart state — no network, no refetch.
   359→ *  Consumers must keep the failure path honest: revert via mutateCartItems
   360→ *  (guarded so it only undoes its own change) and surface a toast. */
   361→export function mutateCartItems(mutate: (items: CartItemT[]) => CartItemT[]) {
   362→  const cur = cartState.cart;
   363→  if (!cur) return;
   364→  cartState = { ...cartState, cart: { ...cur, items: mutate(cur.items) } };
   365→  emitCart();
   366→}
   367→
   368→/** Adopt a server cart payload directly into the shared store (no refetch).
   369→ *  Add-to-cart responses already carry the fresh cart — feeding it here keeps
   370→ *  every consumer (badge, drawer, checkout) in sync in the same frame without
   371→ *  the event-driven /api/cart refetch storm. */
   372→export function applyServerCart(cart: CartT | null | undefined) {
   373→  if (!cart) return;
   374→  cartState = { cart, loading: false };
   375→  emitCart();
   376→}

   378→/** Read-only snapshot of the shared cart (for non-render code paths, e.g. the
   379→ *  drawer's stock probe building the variant-id list). */
   380→export function cartSnapshot(): CartT {
   381→  return cartState.cart;
   382→}

   384→export function useCart() {
   385→  const me = useApp((s) => s.me);
   386→  const authed = Boolean(me.loaded && me.shopper);
   387→  const { cart, loading } = useSyncExternalStore(
   388→    subscribeCart,
   389→    () => cartState,
   390→    () => CART_SERVER_STATE
   391→  );
   392→
   393→  const refresh = useCallback(async () => {
   394→    try {
   395→      const data = await fetchJson<{ cart: CartT }>("/api/cart");
   396→      cartState = { cart: data.cart ?? EMPTY_CART, loading: false };
   397→    } catch {
   398→      cartState = { cart: EMPTY_CART, loading: false };
   399→    }
   400→    emitCart();
   401→  }, []);
   402→
   403→  useEffect(() => {
   404→    // Show the loader only when there is nothing to paint yet — data-bearing
   405→    // re-syncs (mounts, sign-in flips, sm-cart-updated events) stay silent so
   406→    // no surface ever flashes a skeleton over live data.
   407→    if (cartState.cart.items.length === 0) {
   408→      cartState = { ...cartState, loading: true };
   409→      emitCart();
   410→    }
   411→    void refresh();
   412→    const onEvt = () => void refresh();
   413→    window.addEventListener(CART_EVENT, onEvt);
   414→    return () => window.removeEventListener(CART_EVENT, onEvt);
   415→  }, [refresh, authed]);
   416→
   417→  const count = cart.items.reduce((s, i) => s + i.qty, 0);
   418→  const subtotal = cart.items.reduce((s, i) => s + i.lineTotalMinor, 0);
   419→  // Selection-aware figures — only checked lines proceed to checkout.
   420→  const selectedItems = cart.items.filter((i) => i.selected);
   421→  const selectedCount = selectedItems.reduce((s, i) => s + i.qty, 0);
   422→  const selectedSubtotal = selectedItems.reduce((s, i) => s + i.lineTotalMinor, 0);
   423→  return { cart, count, subtotal, selectedCount, selectedSubtotal, loading, refresh };
   424→}

   434→/** Idempotently inject a Google Fonts stylesheet for the given font ids. */
   435→export function ensureFontsLoaded(ids: (string | null | undefined)[]) {
   436→  if (typeof document === "undefined") return;
   437→  const href = fontLinkHref(ids);
   438→  if (!href) return;
   439→  if (!document.querySelector(`link[data-sm-fonts="${href}"]`)) {
   440→    const link = document.createElement("link");
   441→    link.rel = "stylesheet";
   442→    link.href = href;
   443→    link.dataset.smFonts = href;
   444→    document.head.appendChild(link);
   445→  }
   446→}

   452→export function Stars({ rating, count, className }: { rating: number; count?: number; className?: string }) {
   453→  const full = Math.floor(rating);
   454→  const half = rating - full >= 0.25 && rating - full < 0.75;
   455→  const rounded = rating - full >= 0.75 ? full + 1 : full;
   456→  return (
   457→    <span className={cn("inline-flex items-center gap-0.5", className)} aria-label={`Rated ${rating.toFixed(1)} out of 5`}>
   458→      {[0, 1, 2, 3, 4].map((i) =>
   459→        i < full || (i === full && rounded > full && !half) ? (
   460→          <Star key={i} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   461→        ) : i === full && half ? (
   462→          <span key={i} className="relative inline-flex">
   463→            <Star className="h-3.5 w-3.5 text-muted-foreground/40" />
   464→            <StarHalf className="h-3.5 w-3.5 absolute inset-0 fill-amber-400 text-amber-400" />
   465→          </span>
   466→        ) : (
   467→          <Star key={i} className="h-3.5 w-3.5 text-muted-foreground/40" />
   468→        )
   469→      )}
   470→      <span className="text-xs text-muted-foreground ml-1 tabular-nums">
   471→        {rating > 0 ? rating.toFixed(1) : "New"}
   472→        {typeof count === "number" && count > 0 ? ` (${count})` : ""}
   473→      </span>
   474→    </span>
   475→  );
   476→}

   482→export function ProductCard({
   483→  p,
   484→  onOpen,
   485→  // Task 28: vendor name no longer renders on product cards by default —
   486→  // cards are cleaner and the seller is always one tap away (store link in
   487→  // the cart groups, storefront header). Pass true to opt back in.
   488→  showStore = false,
   489→  hidePrice = false,
   490→  cardStyle,
   491→}: {
   492→  p: ProductCardT;
   493→  onOpen: (p: ProductCardT) => void;
   494→  showStore?: boolean;
   495→  hidePrice?: boolean;
   496→  /** Seller-customized typography — only passed on the seller's own storefront. */
   497→  cardStyle?: CardStyleTokens | null;
   498→}) {
   499→  const discount =
   500→    p.compareAtPriceMinor && p.compareAtPriceMinor > p.priceMinor
   501→      ? Math.round(100 - (p.priceMinor / p.compareAtPriceMinor) * 100)
   502→      : null;
   503→  const nameCss: React.CSSProperties = cardStyle
   504→    ? { fontFamily: fontStack(cardStyle.nameFont), fontSize: `${nameSizePx(cardStyle.nameSize)}px`, textAlign: cardStyle.nameAlign as React.CSSProperties["textAlign"] }
   505→    : {};
   506→  const priceCss: React.CSSProperties = cardStyle
   507→    ? { fontFamily: fontStack(cardStyle.priceFont), fontSize: `${priceSizePx(cardStyle.priceSize)}px`, textAlign: cardStyle.priceAlign as React.CSSProperties["textAlign"] }
   508→    : {};
   509→  return (
   510→    <button
   511→      type="button"
   512→      onClick={() => onOpen(p)}
   513→      className="group text-left rounded-[8px] border bg-card shadow-soft overflow-hidden hover:shadow-md hover:border-teal-600/40 hover:-translate-y-0.5 transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-600/50"
   514→      aria-label={`${p.title} from ${p.storeName}, ${formatMinor(p.priceMinor)}`}
   515→    >
   516→      <div className="relative aspect-square overflow-hidden bg-muted">
   517→        {/* plain <img> (product mockups / placeholder service) — never next/image here */}
   518→        <img
   519→          src={p.imageUrl}
   520→          alt={p.title}
   521→          loading="lazy"
   522→          className="h-full w-full object-cover transition-transform duration-300 group-hover:scale-[1.04]"
   523→        />
   524→        {discount ? (
   525→          <span className="absolute top-1.5 left-1.5 rounded-full bg-rose-600 text-white text-[10px] font-bold px-1.5 py-0.5">
   526→            -{discount}%
   527→          </span>
   528→        ) : null}
   529→      </div>
   530→      <div className="p-2.5 space-y-0.5">
   531→        <h3 className={cn("font-normal leading-snug line-clamp-2 min-h-[2rem]", !cardStyle && "text-xs")} style={nameCss}>{p.title}</h3>
   532→        {!hidePrice ? (
   533→          <div className="flex items-baseline gap-1.5 flex-wrap" style={cardStyle?.priceAlign === "center" ? { justifyContent: "center" } : cardStyle?.priceAlign === "right" ? { justifyContent: "flex-end" } : undefined}>
   534→            <span className={cn("font-bold text-teal-700", !cardStyle && "text-sm")} style={priceCss}>{formatMinor(p.priceMinor)}</span>
   535→            {p.compareAtPriceMinor && p.compareAtPriceMinor > p.priceMinor ? (
   536→              <span className="text-[11px] text-muted-foreground line-through">{formatMinor(p.compareAtPriceMinor)}</span>
   537→            ) : null}
   538→          </div>

   568→export function QtyStepper({
   569→  value,
   570→  onChange,
   571→  min = 1,
   572→  max = 99,
   573→  small = false,
   574→  xs = false,
   575→}: {
   576→  value: number;
   577→  onChange: (v: number) => void;
   578→  min?: number;
   579→  max?: number;
   580→  small?: boolean;
   581→  /** Cart-drawer minimalist size (Task 29): 24px buttons, 11px digits. */
   582→  xs?: boolean;
   583→}) {
   584→  const btn = xs ? "h-6 w-6" : small ? "h-8 w-8" : "h-10 w-10";
   585→  const icon = xs ? "h-3 w-3" : "h-3.5 w-3.5";
   586→  const field = xs ? "w-6" : small ? "w-8" : "w-10";
   587→  return (
   588→    <div className={cn("inline-flex items-center border", xs ? "rounded-md" : "rounded-lg")} role="group" aria-label="Quantity">
   589→      <Button
   590→        type="button"
   591→        variant="ghost"
   592→        size="icon"
   593→        className={cn(btn, "rounded-r-none")}
   594→        disabled={value <= min}
   595→        onClick={() => onChange(Math.max(min, value - 1))}
   596→        aria-label="Decrease quantity"
   597→      >
   598→        <Minus className={icon} />
   599→      </Button>
   600→      <span
   601→        className={cn(
   602→          "flex items-center justify-center font-semibold tabular-nums",
   603→          xs ? "w-6 text-[11px]" : "text-sm",
   604→          field
   605→        )}
   606→        aria-live="polite"
   607→      >
   608→        {value}
   609→      </span>
   610→      <Button
   611→        type="button"
   612→        variant="ghost"
   613→        size="icon"
   614→        className={cn(btn, "rounded-l-none")}
   615→        disabled={value >= max}
   616→        onClick={() => onChange(Math.min(max, value + 1))}
   617→        aria-label="Increase quantity"
   618→      >
   619→        <Plus className={icon} />
   620→      </Button>
   621→    </div>
   622→  );
   623→}

   821→export function OrderFlowTimeline({ history, current }: { history: { toStatus: string; createdAt: string }[]; current: string }) {
   822→  const special = ["cancelled", "refunded", "partially_refunded"].includes(current);
   823→  const reachedAt = new Map<string, string>();
   824→  for (const h of history) reachedAt.set(h.toStatus, h.createdAt);
   825→  const flow = special ? [] : FLOW;
   826→  const currentIdx = flow.indexOf(current === "completed" ? "delivered" : current);
   827→
   828→  return (
   829→    <ol className="space-y-0" aria-label="Order status timeline">
   830→      {flow.map((step, i) => {
   831→        const done = currentIdx >= i && currentIdx !== -1;
   832→        const ts = reachedAt.get(step);
   833→        const isLast = i === flow.length - 1;
   834→        return (
   835→          <li key={step} className="relative flex gap-3 pb-5 last:pb-0">
   836→            {!isLast ? <span className={cn("absolute left-[9px] top-5 bottom-0 w-px", done ? "bg-teal-600" : "bg-border")} /> : null}
   837→            <span
   838→              className={cn(
   839→                "relative z-10 mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-full border-2",
   840→                done ? "border-teal-600 bg-teal-600" : "border-muted-foreground/30 bg-background"
   841→              )}
   842→            >
   843→              {done ? (
   844→                <svg viewBox="0 0 10 8" className="h-2.5 w-2.5 fill-none stroke-white" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden>
   845→                  <path d="M1 4l2.5 2.5L9 1" />
   846→                </svg>
   847→              ) : null}
   848→            </span>
   849→            <div className="min-w-0">
   850→              <p className={cn("text-sm font-medium leading-5", done ? "text-foreground" : "text-muted-foreground")}>
   851→                {statusLabel(step)}
   852→              </p>
   853→              {ts ? (
   854→                <p className="text-xs text-muted-foreground">{new Date(ts).toLocaleString("en-MY", { dateStyle: "medium", timeStyle: "short" })}</p>
   855→              ) : null}
   856→            </div>
   857→          </li>
   858→        );
   859→      })}
   860→      {special ? (
   861→        <li className="flex items-center gap-2">
   862→          <StatusBadge status={current} />
   863→          <span className="text-xs text-muted-foreground">This order was closed with status above.</span>
   864→        </li>
   865→      ) : null}
   866→    </ol>
   867→  );
   868→}

   874→export function dateShort(iso: string | Date | null | undefined): string {
   875→  if (!iso) return "";

   879→/** "12 Oct 2026, 2:56 pm" — used by chat timestamps. */
   880→export function fmtDateTime(iso: string | Date | null | undefined): string {
   881→  if (!iso) return "";
   882→  return new Intl.DateTimeFormat("en-MY", { dateStyle: "medium", timeStyle: "short", timeZone: "Asia/Kuala_Lumpur" }).format(new Date(iso));
   883→}

   885→export function discountPercent(price: number, compareAt: number | null | undefined): number | null {
   886→  if (compareAt && compareAt > price) return Math.round(100 - (price / compareAt) * 100);
   887→  return null;
   888→}
