    53→const EMERALD_STATUSES = new Set([
    54→  "active", "live", "paid", "approved", "verified", "completed", "delivered", "success",
    55→  "succeeded", "sent", "processed", "resolved", "issued", "ok", "done", "match",

    63→  "suspended", "rejected", "banned", "failed", "dead", "terminated", "urgent", "high",

    68→export function statusTone(status?: string | null): Tone {
    69→  const s = (status ?? "").toLowerCase();
    70→  if (EMERALD_STATUSES.has(s)) return "emerald";
    71→  if (AMBER_STATUSES.has(s)) return "amber";
    72→  if (RED_STATUSES.has(s)) return "red";
    73→  if (TEAL_STATUSES.has(s)) return "teal";
    74→  return "gray";
    75→}

   133→export function fmtDT(iso?: string | Date | null): string {
   134→  if (!iso) return "—";
   135→  const d = typeof iso === "string" ? new Date(iso) : iso;
   136→  if (Number.isNaN(d.getTime())) return "—";
   137→  return d.toLocaleString("en-MY", {
   138→    day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit", hour12: false,
   139→  });
   140→}
   141→
   142→export function fmtDate(iso?: string | Date | null): string {
   143→  if (!iso) return "—";
   144→  const d = typeof iso === "string" ? new Date(iso) : iso;
   145→  if (Number.isNaN(d.getTime())) return "—";
   146→  return d.toLocaleDateString("en-MY", { day: "2-digit", month: "short", year: "numeric" });
   147→}
   148→
   149→export function timeAgo(iso?: string | Date | null): string {
   150→  if (!iso) return "—";
   151→  const d = typeof iso === "string" ? new Date(iso) : iso;
   152→  if (Number.isNaN(d.getTime())) return "—";
   153→  const secs = Math.floor((Date.now() - d.getTime()) / 1000);
   154→  if (secs < 60) return "just now";
   155→  if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   156→  if (secs < 86400) return `${Math.floor(secs / 3600)}h ago`;
   157→  if (secs < 86400 * 30) return `${Math.floor(secs / 86400)}d ago`;
   158→  return fmtDate(d);
   159→}

   221→export function KpiCard({
   222→  label,
   223→  value,
   224→  sub,
   225→  icon: Icon,
   226→  onClick,
   227→  tone = "plain",
   228→}: {
   229→  label: string;
   230→  value: ReactNode;
   231→  sub?: ReactNode;
   232→  icon?: LucideIcon;
   233→  onClick?: () => void;
   234→  tone?: Tone | "plain";
   235→}) {
   236→  const clickable = Boolean(onClick);
   237→  // Tone-tinted icon chip (recipe 4): soft 50-tint background + 600/700 text.
   238→  const chipTone =
   239→    tone === "emerald"
   240→      ? "bg-emerald-50 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-400"
   241→      : tone === "amber"
   242→        ? "bg-amber-50 text-amber-600 dark:bg-amber-950/40 dark:text-amber-400"
   243→        : tone === "red"
   244→          ? "bg-rose-50 text-rose-600 dark:bg-red-950/40 dark:text-red-400"
   245→          : tone === "teal"
   246→            ? "bg-teal-50 text-teal-700 dark:bg-teal-950/40 dark:text-teal-300"
   247→            : "bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-300";
   248→  return (
   249→    <div
   250→      role={clickable ? "button" : undefined}
   251→      tabIndex={clickable ? 0 : undefined}
   252→      onClick={onClick}
   253→      onKeyDown={(e) => {
   254→        if (clickable && (e.key === "Enter" || e.key === " ")) {
   255→          e.preventDefault();
   256→          onClick?.();
   257→        }
   258→      }}
   259→      className={cn(
   260→        "rounded-xl border border-slate-200/80 bg-card p-4 shadow-soft-card transition-shadow",
   261→        clickable && "cursor-pointer hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-500"
   262→      )}
   263→    >
   264→      <div className="flex items-center justify-between gap-2">
   265→        <p className="truncate text-[11px] font-bold uppercase tracking-wider text-slate-400">{label}</p>
   266→        {Icon ? (
   267→          <span className={cn("flex h-8 w-8 shrink-0 items-center justify-center rounded-lg", chipTone)}>
   268→            <Icon className="h-4 w-4" />
   269→          </span>
   270→        ) : null}
   271→      </div>
   272→      <div className="mt-2 text-2xl font-extrabold tracking-tight tabular-nums text-slate-900">{value}</div>
   273→      {sub ? <div className="mt-1 text-xs text-slate-400">{sub}</div> : null}
   274→    </div>
   275→  );
   276→}

   478→function TypingConfirmInner({

   505→        </AlertDialogDescription>
   506→      </AlertDialogHeader>

   643→const BREAKDOWN_MONEY_KEYS = new Set([
   644→  "base", "percent_part", "fixed_fee_allocated", "discount_allocated", "shipping_allocated",
   645→]);
   646→
   647→function breakdownValue(key: string, v: unknown): string | null {
   648→  if (v === null || v === undefined) return null;
   649→  if (key === "flags") {
   650→    const arr = Array.isArray(v) ? v.map(String) : [];
   651→    return arr.length ? arr.join(", ") : null;
   652→  }
   653→  if (BREAKDOWN_MONEY_KEYS.has(key)) return formatMinor(Number(v));
   654→  if (typeof v === "number") return key === "percent" ? `${v}%` : String(v);
   655→  if (typeof v === "string") return v;
   656→  return JSON.stringify(v);
   657→}

   659→/** Renders a commission breakdownJson payload as key:value chips. */
   660→export function BreakdownChips({ json, className }: { json?: string | null; className?: string }) {
   661→  const data = parseJsonSafe<Record<string, unknown>>(json, {});
   662→  const entries = Object.entries(data).filter(([k, v]) => breakdownValue(k, v) !== null);
   663→  if (entries.length === 0) return <span className="text-xs text-muted-foreground">No breakdown snapshot</span>;
   664→  return (
   665→    <div className={cn("flex flex-wrap gap-1.5", className)}>
   666→      {entries.map(([k, v]) => {
   667→        const val = breakdownValue(k, v) as string;
   668→        const highlight = k === "rule_source";
   669→        return (
   670→          <span
   671→            key={k}
   672→            className={cn(
   673→              "inline-flex items-center gap-1 rounded-full border px-1.5 py-0.5 text-[11px]",
   674→              highlight ? TONE_CLASS.teal : "bg-muted/60 text-muted-foreground"
   675→            )}
   676→          >
   677→            <span className="font-medium capitalize">{k.replace(/_/g, " ")}</span>
   678→            <span className="tabular-nums">{val}</span>
   679→          </span>
   680→        );
   681→      })}
   682→    </div>
   683→  );

   686→/* -------------------------------------------------------------- diff view */

   688→function flatObj(v: unknown): Record<string, unknown> {
   689→  return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : {};
   690→}
   691→
   692→function diffVal(key: string, v: unknown): string {
   693→  if (v === null || v === undefined || v === "") return "—";
   694→  if (typeof v === "object") return JSON.stringify(v);
   695→  if (key.endsWith("Minor")) return formatMinor(Number(v));
   696→  if (typeof v === "boolean") return v ? "true" : "false";
   697→  return String(v);
   698→}

   700→/** Field-by-field before/after viewer for audit log rows. */
   701→export function DiffViewer({ beforeJson, afterJson }: { beforeJson?: string | null; afterJson?: string | null }) {
   702→  const before = flatObj(parseJsonSafe<unknown>(beforeJson, {}));
   703→  const after = flatObj(parseJsonSafe<unknown>(afterJson, {}));
   704→  const keys = Array.from(new Set([...Object.keys(before), ...Object.keys(after)])).sort();
   705→  if (keys.length === 0) {
   706→    return <p className="text-xs text-muted-foreground">No structured before/after snapshot recorded for this entry.</p>;
   707→  }
   708→  return (
   709→    <div className="overflow-hidden rounded-lg border">
   710→      <Table>
   711→        <TableHeader>
   712→          <TableRow>
   713→            <TableHead className="w-40">Field</TableHead>
   714→            <TableHead>Before</TableHead>
   715→            <TableHead>After</TableHead>
   716→          </TableRow>
   717→        </TableHeader>
   718→        <TableBody>
   719→          {keys.map((k) => {
   720→            const b = k in before ? diffVal(k, before[k]) : null;
   721→            const a = k in after ? diffVal(k, after[k]) : null;
   722→            const changed = b !== a;
   723→            const added = b === null;
   724→            const removed = a === null;
   725→            return (
   726→              <TableRow key={k} className={cn(changed && !removed && "bg-amber-50/60 dark:bg-amber-950/20")}>
   727→                <TableCell className="font-mono text-[11px] font-medium">{k}</TableCell>
   728→                <TableCell className={cn("text-xs tabular-nums", removed && "text-red-600 line-through dark:text-red-400")}>
   729→                  {b ?? "—"}
   730→                </TableCell>
   731→                <TableCell className={cn("text-xs tabular-nums", added && "font-medium text-emerald-700 dark:text-emerald-400")}>
   732→                  {a ?? "—"}
   733→                </TableCell>
   734→              </TableRow>
   735→            );
   736→          })}
   737→        </TableBody>
   738→      </Table>
   739→    </div>
   740→  );
   741→}
