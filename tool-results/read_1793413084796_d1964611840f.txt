    57→export function webhookSecret(): string {
    58→  const secret = process.env.PSP_WEBHOOK_SECRET;
    59→  if (secret && secret.length > 0) return secret;
    60→  if (process.env.NODE_ENV === "production") {
    61→    throw new ApiError(
    62→      503,
    63→      "WEBHOOK_SECRET_UNSET",
    64→      "PSP_WEBHOOK_SECRET is not configured — payment webhook processing is disabled (fail closed)"
    65→    );
    66→  }
    67→  return "sm_webhook_secret_demo";
    68→}

    76→  return hmacSha256Hex(webhookSecret(), `${providerRef}.${eventType}.${amountMinor}.${ts}`);
    77→}

   114→/**
   115→ * Verify signature (caller may pre-compute it, as simulate does), log the raw
   116→ * webhook, then process idempotently. Throws ApiError with proper status codes.
   117→ *
   118→ * RACE SAFETY (why claim-first): the success paths' FIRST write inside the
   119→ * Serializable transaction is a CONDITIONAL updateMany (status flip guarded by
   120→ * the pre-status). Two concurrent deliveries of the same event race on that
   121→ * row lock; the loser matches zero rows, writes nothing and returns the
   122→ * duplicate result — money movement, order transitions and notifications can
   123→ * only ever run for the winner. The ledger dedupKeys (FIN-1) backstop this at
   124→ * the posting layer, and the notification dedupKeys backstop the mail log.
   125→ */
   126→export async function processPaymentWebhook(input: {
   127→  provider: string;
   128→  body: { providerRef?: string; eventType?: string; amountMinor?: number; ts?: number; sig?: string; reason?: string; paymentId?: string };
   129→  rawBody: string;
   130→  headerSignature?: string | null;
   131→}): Promise<ProcessResult> {
   132→  // (2b.2) Provider sandbox enforcement FIRST — fail closed. resolvePspProvider
   133→  // falls back to the mock provider for unknown ids, so in production an
   134→  // unregistered provider id can never sneak through: mock is sandboxOnly → 403.
   135→  const psp = resolvePspProvider(input.provider);
   136→  if (psp.sandboxOnly && process.env.NODE_ENV === "production") {
   137→    throw new ApiError(403, "PROVIDER_SANDBOX_ONLY", `Provider "${input.provider}" is sandbox-only and cannot process webhooks in production`);
   138→  }
   139→
   140→  const providerRef = String(input.body.providerRef ?? "");
   141→  const eventType = String(input.body.eventType ?? "");
   142→  const amountMinor = Math.round(Number(input.body.amountMinor ?? 0));
   143→  const ts = Math.round(Number(input.body.ts ?? 0));
   144→  if (!providerRef || !eventType) {
   145→    throw new ApiError(400, "BAD_REQUEST", "providerRef and eventType are required");
   146→  }
   147→
   148→  // (2b.1) Timing-safe compare + freshness. signatureValid in the log means
   149→  // "authenticated as a LIVE event": correct MAC AND inside the replay window —
   150→  // a stale-but-correctly-signed delivery is recorded as signatureValid=false
   151→  // (replay protection is part of webhook authentication semantics) and
   152→  // rejected with STALE_EVENT below, AFTER the row is logged.
   153→  const providedSig = input.headerSignature ?? (input.body.sig ? String(input.body.sig) : null);
   154→  const expected = expectedSignature(providerRef, eventType, amountMinor, ts);
   155→  const sigMatches = !!providedSig && safeEqual(providedSig, expected);
   156→  const fresh = Math.abs(Date.now() - ts) <= WEBHOOK_TS_WINDOW_MS;
   157→  const signatureValid = sigMatches && fresh;
   158→
   159→  // OPTION A first: the ref may be a multi-seller payment group.
   160→  const group = await db.paymentGroup.findUnique({
   161→    where: { ref: providerRef },
   162→    include: {
   163→      orders: {
   164→        orderBy: { createdAt: "asc" },
   165→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   166→      },
   167→    },
   168→  });
   169→
   170→  const intent = group ? null : await db.paymentIntent.findUnique({ where: { providerRef } });
   171→
   172→  // ALWAYS log the webhook first (even for invalid signature / unknown ref) —
   173→  // invalid/stale rows stay in the log with their signatureValid flag.
   174→  const log = await db.paymentWebhookLog.create({
   175→    data: {
   176→      provider: input.provider,
   177→      providerRef,
   178→      eventType,
   179→      payloadJson: input.rawBody,
   180→      signatureValid,
   181→      processed: false,
   182→      orderId: group?.orders[0]?.id ?? intent?.orderId ?? null,
   183→    },
   184→  });
   185→
   186→  if (!sigMatches) {
   187→    throw new ApiError(400, "BAD_SIGNATURE", "Webhook signature verification failed");
   188→  }
   189→  if (!fresh) {
   190→    throw new ApiError(400, "STALE_EVENT", `Webhook timestamp is outside the ±${WEBHOOK_TS_WINDOW_MS / 60_000}-minute acceptance window (replay protection)`);
   191→  }
   192→
   193→  // ------------------------------------------------------------------ FIN-2b: inbound payout events
   194→  if (eventType === "payout.succeeded" || eventType === "payout.failed") {
   195→    return processPayoutEvent({ provider: input.provider, providerRef, eventType, amountMinor, reason: input.body.reason ?? null, logId: log.id });
   196→  }
   197→
   198→  // ------------------------------------------------------------------ OPTION A group path
   199→  if (group) {
   200→    // LATE PAYMENT AFTER EXPIRY — defined recovery policy (F3 remediation):
   201→    // the expiry sweep has already claimed this group, transitioned its orders
   202→    // to a terminal unpaid state and RESTOCKED the items. Fulfilling now would
   203→    // sell unreserved stock; accepting the money without goods would steal.
   204→    // The ONLY safe response is a definitive refusal: no money moves, no order
   205→    // is resurrected, the PSP sees a permanent failure (mock PSP surfaces it
   206→    // to the buyer as an expired payment window). Never silently fulfill.
   207→    //
   208→    // DURABLE REFUND OBLIGATION (mission: final operational gaps): a REAL PSP
   209→    // has already CAPTURED the buyer's funds when this verified event arrives —
   210→    // the 409 below protects the order/inventory transition, it does not return
   211→    // those funds. Record the obligation BEFORE refusing so a crash between the
   212→    // two leaves at worst "case exists, webhook answered 5xx" → PSP redelivery
   213→    // converges on the unique (provider, providerRef) row. Re-reservation is
   214→    // deliberately NOT attempted (business rule: released stock may be re-sold;
   215→    // fulfilling a stale payment without buyer re-consent is not legitimate).
   216→    if (eventType === "payment.succeeded" && group.status === "expired") {
   217→      await recordLatePaymentObligation({
   218→        provider: input.provider,
   219→        providerRef,
   220→        providerPaymentId: input.body.paymentId ?? null,
   221→        amountMinor,
   222→        groupTotalMinor: group.totalMinor,
   223→        currency: group.currency,
   224→      });
   225→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   226→      throw new ApiError(409, "GROUP_EXPIRED", `Payment group ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   227→    }
   228→    if (eventType === "payment.failed") {
   229→      // No money moves on failure — a single conditional update is fine outside
   230→      // a Serializable tx (an already-succeeded group ignores the failed event).
   231→      if (group.status === "succeeded") {
   232→        await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   233→        return {
   234→          ok: true,
   235→          duplicate: true,
   236→          eventType,
   237→          orderNumber: group.orders[0]?.orderNumber ?? "",
   238→          orderNumbers: group.orders.map((o) => o.orderNumber),
   239→          orderStatus: "paid",
   240→          intentStatus: "succeeded",
   241→          isGroup: true,
   242→        };
   243→      }
   244→      await db.paymentGroup.update({ where: { id: group.id }, data: { status: "failed" } });
   245→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   246→      return {
   247→        ok: true,
   248→        duplicate: false,
   249→        eventType,
   250→        orderNumber: group.orders[0]?.orderNumber ?? "",
   251→        orderNumbers: group.orders.map((o) => o.orderNumber),
   252→        orderStatus: group.orders[0]?.status ?? "payment_pending",
   253→        intentStatus: "failed",
   254→        isGroup: true,
   255→      };
   256→    }
   257→
   258→    // Unknown-event + amount checks BEFORE the claim (unchanged order) — a
   259→    // malformed delivery must never even attempt the status flip.
   260→    if (eventType !== "payment.succeeded") {
   261→      throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   262→    }
   263→
   264→    if (amountMinor !== group.totalMinor) {
   265→      throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match payment group ${group.totalMinor}`);
   266→    }
   267→
   268→    const settings = await getCommissionSettings();
   269→    const paidAt = new Date();
   270→
   271→    // SUCCESS PATH — one Serializable transaction around claim + money.
   272→    // The conditional claim is the FIRST write: count 0 ⇒ another delivery
   273→    // already succeeded ⇒ nothing this tx wrote (the claim was the only
   274→    // intended write) ⇒ return the duplicate sentinel; committing an empty tx
   275→    // is equivalent to rollback-by-return.
   276→    //
   277→    // F1 REMEDIATION (measured): concurrent same-seller confirmations used to
   278→    // mass-abort here (P2034 SSI conflicts on the wallet-cache row — 0/15 in
   279→    // the measured storm). Two fixes, both evidence-driven:
   280→    //   1. the wallet-cache refresh moved OUT of this tx (post-commit,
   281→    //      best-effort) — the wallet row is no longer in this tx's conflict
   282→    //      domain; payout authorization never trusted the cache anyway;
   283→    //   2. bounded P2034 retry of the WHOLE transaction (2 retries, jittered
   284→    //      backoff). SSI conflicts are transient by definition; the claim's
   285→    //      conditional update + ledger dedupKeys make a full retry safe (no
   286→    //      double-posting is possible). PSP calls stay outside — nothing
   287→    //      non-transactional is ever retried.
   288→    const walletIdsTouched = new Set<string>();
   289→    // F1 [Measured]: notification/audit fan-out (notify, dispatchStoreWebhooks,
   290→    // logActivity, shopper nudge) used to run INSIDE the Serializable tx — but
   291→    // on the GLOBAL db client, i.e. on ADDITIONAL pool connections while the tx
   292→    // holds its own. Under concurrent confirmations that multiplies connection
   293→    // demand inside the tx window (15 webhooks × ≥2 connections vs pool of 10)
   294→    // → measured P2024 connection-starvation + P2028 (dev.log). Money + order
   295→    // state stay transactional; the non-transactional side effects fire
   296→    // POST-COMMIT. dedupKeys make a redelivery unable to multiply mail; a
   297→    // crash in the post-commit window can only delay/lose OBSERVABILITY rows,
   298→    // never financial state — the correct trade for a bounded money tx.
   299→    const postedOrders: { order: (typeof group.orders)[number]; sellerEmail: string | null }[] = [];
   300→    let outcome: { claimed: boolean } | undefined;
   301→    for (let attempt = 1; ; attempt++) {
   302→      try {
   303→        outcome = await db.$transaction(
   304→      async (tx) => {
   305→        // Bounded-failure policy (measured): a row lock held by another session
   306→        // stalls statements past Prisma's interactive-tx timer (measured 120 s
   307→        // = the server's 2-min statement_timeout). Session-scoped 10 s bound,
   308→        // self-reverting at commit/rollback — lock contention fails FAST, the
   309→        // PSP redelivers, and no pool connection is burned for 2 minutes.
   310→        await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   311→        const claim = await tx.paymentGroup.updateMany({
   312→          where: { id: group.id, status: { notIn: ["succeeded", "expired"] } },
   313→          data: { status: "succeeded", paidAt },
   314→        });
   315→        if (claim.count === 0) return { claimed: false as const };
   316→
   317→        for (const order of group.orders) {
   318→          await transitionOrder({
   319→            orderId: order.id,
   320→            toStatus: "paid",
   321→            actor: "system",
   322→            note: `Payment ${providerRef} succeeded via ${group.provider} (group payment)`,
   323→            timestamps: { paidAt },
   324→            tx,
   325→          });
   326→
   327→          // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   328→          const seller = order.store.seller;
   329→          const wallet = seller?.wallet;
   330→          if (seller && wallet) {
   331→            const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   332→            const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   333→            const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   334→            const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   335→            // AFFILIATE — seller-funded pool (T1 4th line; 0 ⇒ legacy 3-line
   336→            // posting stays byte-compatible) + the commission rows themselves
   337→            // (exactly once per order item: unique orderItemId idempotency).
   338→            const affiliateMinor = order.items.reduce((s, i) => s + i.affiliatePoolMinor, 0);
   339→            if (wallet) walletIdsTouched.add(wallet.id);
   340→            await postOrderPaid({
   341→              orderId: order.id,
   342→              orderNumber: order.orderNumber,
   343→              walletId: wallet.id,
   344→              provider: group.provider,
   345→              totalMinor: order.totalMinor,
   346→              sellerNetMinor,
   347→              commissionMinor,
   348→              affiliateMinor,
   349→              holdUntil,
   350→              currency: order.currency,
   351→              occurredAt: paidAt,
   352→              tx,
   353→              refreshCache: false, // post-commit below — keeps the wallet row out of this tx's conflict domain
   354→            });
   355→            await postAffiliateCommissionsForOrder(tx, order, paidAt);
   356→          }
   357→          postedOrders.push({ order, sellerEmail: seller?.email ?? null });
   358→        }
   359→
   360→        // Server-side cart clearing — only the PURCHASED variants leave the cart,
   361→        // and only on verified payment success (never on failure/pending). Lines
   362→        // the buyer unchecked (kept for later) survive: selection-aware carts.
   363→        // PaymentGroups are account-scoped (shopperId always set). Inside the tx:
   364→        // a rollback never strands paid items out of the cart inconsistently.
   365→        const purchasedVariantIds = [
   366→          ...new Set(group.orders.flatMap((o) => o.items.map((it) => it.variantId)).filter((v): v is string => Boolean(v))),
   367→        ];
   368→        await tx.cartItem.deleteMany({
   369→          where: { cart: { shopperId: group.shopperId }, variantId: { in: purchasedVariantIds } },
   370→        });
   371→
   372→        return { claimed: true as const };
   373→      },
   374→      // timeout: measured evidence (dev.log) — ~18-query Serializable tx over
   375→      // remote RTT exceeds Prisma's 5 s default under concurrency (P2028 ×13).
   376→      // 15 s is a BOUNDED policy, not the unbounded external-lock stall.
   377→      { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, timeout: 15_000 }
   378→        );
   379→        break;
   380→      } catch (e) {
   381→        if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2034" && attempt < 3) {
   382→          await new Promise((r) => setTimeout(r, 50 + Math.random() * 100));
   383→          continue;
   384→        }
   385→        throw e;
   386→      }
   387→    }
   388→
   389→    if (!outcome?.claimed) {
   390→      // Claim lost — WHY decides the answer. A concurrent delivery that
   391→      // succeeded → genuine duplicate (ok). But a group the expiry sweep
   392→      // claimed between our load and our claim must NOT be answered "ok":
   393→      // the money must not be captured for goods we already restocked.
   394→      // (Late-payment recovery policy — see the expired check above.)
   395→      const nowStatus = (await db.paymentGroup.findUnique({ where: { id: group.id }, select: { status: true } }))?.status;
   396→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   397→      if (nowStatus === "expired") {
   398→        // The sweep won the claim while this delivery was in flight — same
   399→        // late-payment policy as above, including the durable refund obligation.
   400→        await recordLatePaymentObligation({
   401→          provider: input.provider,
   402→          providerRef,
   403→          providerPaymentId: input.body.paymentId ?? null,
   404→          amountMinor,
   405→          groupTotalMinor: group.totalMinor,
   406→          currency: group.currency,
   407→        });
   408→        throw new ApiError(409, "GROUP_EXPIRED", `Payment group ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   409→      }
   410→      return {
   411→        ok: true,
   412→        duplicate: true,
   413→        eventType,
   414→        orderNumber: group.orders[0]?.orderNumber ?? "",
   415→        orderNumbers: group.orders.map((o) => o.orderNumber),
   416→        orderStatus: "paid",
   417→        intentStatus: "succeeded",
   418→        isGroup: true,
   419→      };
   420→    }
   421→
   422→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   423→
   424→    // POST-COMMIT fan-out (see F1 note above) — non-transactional side effects.
   425→    for (const { order, sellerEmail } of postedOrders) {
   426→      try {
   427→        await dispatchStoreWebhooks({
   428→          storeId: order.storeId,
   429→          event: "order.paid",
   430→          payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency, paymentGroupRef: providerRef },
   431→        });
   432→        if (sellerEmail) {
   433→          await notify({ event: "order.paid", channel: "email", recipient: sellerEmail, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   434→          await notify({ event: "order.paid", channel: "inapp", recipient: sellerEmail, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   435→        }
   436→        await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   437→        await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   438→        await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: group.provider, totalMinor: order.totalMinor, paymentGroup: true } });
   439→      } catch (e) {
   440→        // Observability/notification loss never fails the money path.
   441→        console.error(`[post-commit] fan-out failed for order ${order.orderNumber} (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   442→      }
   443→    }
   444→
   445→    // F1 — post-commit wallet-cache refresh (best-effort, OUTSIDE the money
   446→    // tx). A failure here leaves the display cache stale — classified as
   447→    // benign time-maturity/event staleness by the ledger invariant, invisible
   448→    // to payout authorization (which recomputes truth transactionally), and
   449→    // self-healing on the next wallet event or wallet-page read.
   450→    for (const wid of walletIdsTouched) {
   451→      try {
   452→        await refreshWalletCache(wid);
   453→      } catch (e) {
   454→        console.error(`[wallet-cache] post-commit refresh failed for ${wid} (benign, self-heals):`, e instanceof Error ? e.message : e);
   455→      }
   456→    }
   457→
   458→    // (FIN-1 1.6 wiring) T2 auto-settle — AFTER the success tx commits. The
   459→    // recipe is idempotent (dedupKey gateway_settlement:settlement:{ref}), so a
   460→    // crash here is healed by the next delivery's claim-miss path... which is
   461→    // why the call lives outside the tx: the PSP settles buyer funds to the
   462→    // platform bank account; T2 clears the GATEWAY:{provider} receivable into
   463→    // BANK_MAIN. Default setting false = dormant (documented decision).
   464→    if (await autoSettleEnabled()) {
   465→      await postGatewaySettlement({ provider: group.provider, amountMinor: group.totalMinor, ref: providerRef, occurredAt: paidAt });
   466→    }
   467→
   468→    return {
   469→      ok: true,
   470→      duplicate: false,
   471→      eventType,
   472→      orderNumber: group.orders[0]?.orderNumber ?? "",
   473→      orderNumbers: group.orders.map((o) => o.orderNumber),
   474→      orderStatus: "paid",
   475→      intentStatus: "succeeded",
   476→      isGroup: true,
   477→    };
   478→  }
   479→
   480→  // ------------------------------------------------------------------ legacy single-order path
   481→  if (!intent) {
   482→    throw new ApiError(404, "INTENT_NOT_FOUND", "Payment intent not found");
   483→  }
   484→
   485→  if (eventType === "payment.failed") {
   486→    if (intent.status === "succeeded") {
   487→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   488→      const order = await db.order.findUnique({ where: { id: intent.orderId } });
   489→      return {
   490→        ok: true,
   491→        duplicate: true,
   492→        eventType,
   493→        orderNumber: order?.orderNumber ?? "",
   494→        orderStatus: order?.status ?? "",
   495→        intentStatus: intent.status,
   496→      };
   497→    }
   498→    await db.paymentIntent.update({ where: { id: intent.id }, data: { status: "failed" } });
   499→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   500→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   501→    return {
   502→      ok: true,
   503→      duplicate: false,
   504→      eventType,
   505→      orderNumber: order?.orderNumber ?? "",
   506→      orderStatus: order?.status ?? "",
   507→      intentStatus: "failed",
   508→    };
   509→  }
   510→
   511→  if (eventType !== "payment.succeeded") {
   512→    throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   513→  }
   514→
   515→  if (amountMinor !== intent.amountMinor) {
   516→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match intent ${intent.amountMinor}`);
   517→  }
   518→
   519→  const settings = await getCommissionSettings();
   520→  const paidAt = new Date();
   521→
   522→  // SUCCESS PATH — same claim-first Serializable pattern as the group path,
   523→  // with the same F1 remediation (post-commit wallet refresh + bounded P2034
   524→  // retry of the whole transaction).
   525→  const walletIdsTouched = new Set<string>();
   526→  // NOTE: array holder (same pattern as the group path's postedOrders) — a
   527→  // `let x: T | null = null` assigned inside the $transaction callback gets
   528→  // CFA-narrowed to `null` at post-commit use sites (TS never sees the callback
   529→  // assignment), making `if (postedOrder)` `never` and every property access an
   530→  // error. Array contents are not flow-narrowed, so the post-commit fan-out
   531→  // below reads the real types.
   532→  const postedOrderRef: { order: Prisma.OrderGetPayload<{ include: { items: true; store: { include: { seller: { include: { wallet: true } } } } } }>; sellerEmail: string | null }[] = [];
   533→  let outcome: { claimed: boolean } | undefined;
   534→  for (let attempt = 1; ; attempt++) {
   535→    try {
   536→      outcome = await db.$transaction(
   537→    async (tx) => {
   538→      // Bounded-failure policy (see group path) — 10 s per-statement bound.
   539→      await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   540→      const claim = await tx.paymentIntent.updateMany({
   541→        where: { id: intent.id, status: { not: "succeeded" } },
   542→        data: { status: "succeeded" },
   543→      });
   544→      if (claim.count === 0) return { claimed: false as const };
   545→
   546→      const order = await tx.order.findUnique({
   547→        where: { id: intent.orderId },
   548→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   549→      });
   550→      if (!order) throw new ApiError(404, "ORDER_NOT_FOUND", "Order for payment intent not found");
   551→
   552→      await transitionOrder({ orderId: order.id, toStatus: "paid", actor: "system", note: `Payment ${providerRef} succeeded via ${intent.provider}`, timestamps: { paidAt }, tx });
   553→
   554→      // Server-side cart clearing — only the PURCHASED variants leave the cart,
   555→      // and only on verified payment success (never on failure/pending, never
   556→      // client-driven). Account carts clear by shopperId; the token is a legacy/
   557→      // reconciliation fallback for rows that predate account-scoped carts.
   558→      // Selection-aware: unchecked (kept-for-later) lines are never removed here.
   559→      const purchasedVariantIds = [...new Set(order.items.map((it) => it.variantId).filter((v): v is string => Boolean(v)))];
   560→      if (order.shopperId) {
   561→        await tx.cartItem.deleteMany({
   562→          where: { cart: { shopperId: order.shopperId }, variantId: { in: purchasedVariantIds } },
   563→        });
   564→      } else if (order.cartToken) {
   565→        await tx.cartItem.deleteMany({
   566→          where: { cart: { token: order.cartToken }, variantId: { in: purchasedVariantIds } },
   567→        });
   568→      }
   569→
   570→      // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   571→      const seller = order.store.seller;
   572→      const wallet = seller?.wallet;
   573→      if (seller && wallet) {
   574→        const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   575→        const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   576→        const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   577→        const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   578→        // AFFILIATE — pool + commission rows (see group-path note).
   579→        const affiliateMinor = order.items.reduce((s, i) => s + i.affiliatePoolMinor, 0);
   580→        walletIdsTouched.add(wallet.id);
   581→        await postOrderPaid({
   582→          orderId: order.id,
   583→          orderNumber: order.orderNumber,
   584→          walletId: wallet.id,
   585→          provider: intent.provider,
   586→          totalMinor: intent.amountMinor,
   587→          sellerNetMinor,
   588→          commissionMinor,
   589→          affiliateMinor,
   590→          holdUntil,
   591→          currency: order.currency,
   592→          occurredAt: paidAt,
   593→          tx,
   594→          refreshCache: false, // post-commit below (F1)
   595→        });
   596→        await postAffiliateCommissionsForOrder(tx, order, paidAt);
   597→      }
   598→
   599→      // Notification/audit fan-out moved POST-COMMIT (group-path F1 note):
   600→      // these ran on the global db client inside the tx window, multiplying
   601→      // pool-connection demand under concurrency (measured P2024/P2028).
   602→      postedOrderRef.push({ order, sellerEmail: seller?.email ?? null });
   603→
   604→      return { claimed: true as const };
   605→    },
   606→    // timeout: same measured P2028 evidence as the group path.
   607→    { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, timeout: 15_000 }
   608→      );
   609→      break;
   610→    } catch (e) {
   611→      if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2034" && attempt < 3) {
   612→        await new Promise((r) => setTimeout(r, 50 + Math.random() * 100));
   613→        continue;
   614→      }
   615→      throw e;
   616→    }
   617→  }
   618→
   619→  if (!outcome?.claimed) {
   620→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   621→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   622→    return {
   623→      ok: true,
   624→      duplicate: true,
   625→      eventType,
   626→      orderNumber: order?.orderNumber ?? "",
   627→      orderStatus: order?.status ?? "",
   628→      intentStatus: "succeeded",
   629→    };
   630→  }
   631→
   632→  await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   633→
   634→  // POST-COMMIT fan-out (see group-path F1 note) — non-transactional side effects.
   635→  for (const { order, sellerEmail } of postedOrderRef) {
   636→    try {
   637→      await dispatchStoreWebhooks({
   638→        storeId: order.storeId,
   639→        event: "order.paid",
   640→        payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency },
   641→      });
   642→      if (sellerEmail) {
   643→        await notify({ event: "order.paid", channel: "email", recipient: sellerEmail, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   644→        await notify({ event: "order.paid", channel: "inapp", recipient: sellerEmail, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   645→      }
   646→      await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   647→      await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   648→      await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: intent.provider, totalMinor: intent.amountMinor } });
   649→    } catch (e) {
   650→      console.error(`[post-commit] fan-out failed for order ${order.orderNumber} (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   651→    }
   652→  }
   653→
   654→  // F1 — post-commit wallet-cache refresh (same rationale as the group path).
   655→  for (const wid of walletIdsTouched) {
   656→    try {
   657→      await refreshWalletCache(wid);
   658→    } catch (e) {
   659→      console.error(`[wallet-cache] post-commit refresh failed for ${wid} (benign, self-heals):`, e instanceof Error ? e.message : e);
   660→    }
   661→  }
   662→
   663→  // T2 auto-settle (see group path) — after commit, idempotent recipe.
   664→  if (await autoSettleEnabled()) {
   665→    await postGatewaySettlement({ provider: intent.provider, amountMinor: intent.amountMinor, ref: providerRef, occurredAt: paidAt });
   666→  }
   667→
   668→  return {
   669→    ok: true,
   670→    duplicate: false,
   671→    eventType,
   672→    orderNumber: (await db.order.findUnique({ where: { id: intent.orderId } }))?.orderNumber ?? "",
   673→    orderStatus: "paid",
   674→    intentStatus: "succeeded",
   675→  };
   676→}
   677→
   678→/**
   679→ * FIN-2b (2b.4) — inbound PAYOUT lifecycle events from the PSP.
   680→ *
   681→ * The provider confirms (or fails) a bank transfer we initiated. providerRef
   682→ * resolution: try Payout.referenceNo first (the canonical platform reference —
   683→ * what the mock PSP and payout-simulate echo); fall back to Payout.providerRef
   684→ * (the transfer ref an admin recorded at mark-paid — some providers echo that
   685→ * instead). Both lookups are documented PSP integration surface.
   686→ *
   687→ * Same claim-first race safety as the payment paths: assertTransition guards
   688→ * the semantics, the conditional updateMany serializes concurrent deliveries,
   689→ * and T4/T4f post inside the same Serializable tx as the status flip.
   690→ */
   691→async function processPayoutEvent(args: {
   692→  provider: string;
   693→  providerRef: string;
   694→  eventType: "payout.succeeded" | "payout.failed";
   695→  amountMinor: number;
   696→  reason: string | null;
   697→  logId: string;
   698→}): Promise<ProcessResult> {
   699→  const payout =
   700→    (await db.payout.findUnique({ where: { referenceNo: args.providerRef } })) ??
   701→    (await db.payout.findFirst({ where: { providerRef: args.providerRef } }));
   702→  if (!payout) {
   703→    throw new ApiError(404, "PAYOUT_NOT_FOUND", `No payout matches providerRef ${args.providerRef}`);
   704→  }
   705→
   706→  const toStatus = args.eventType === "payout.succeeded" ? "paid" : "failed";
   707→  const dedupKey = `${toStatus === "paid" ? "payout.paid" : "payout.failed"}:${payout.id}`;
   708→
   709→  // Idempotent replay: the provider re-delivered an event we already applied
   710→  // (e.g. payout.succeeded for an already-paid payout) → ok + conflict flag,
   711→  // never a raw 409 at the provider. Decision is recorded (processed=true).
   712→  if (payout.status === toStatus) {
   713→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   714→    return {
   715→      ok: true,
   716→      duplicate: true,
   717→      payoutConflict: true,
   718→      eventType: args.eventType,
   719→      orderNumber: "",
   720→      orderStatus: payout.status,
   721→      intentStatus: "",
   722→      payoutStatus: payout.status,
   723→    };
   724→  }
   725→
   726→  // State machine guard (cheap, no locks): an illegal transition (e.g.
   727→  // requested→paid — the money was never reserved by T3) is a hard 409. The
   728→  // decision is recorded on the webhook row before throwing.
   729→  assertTransition(payout.status, toStatus);
   730→
   731→  // Amount guard BEFORE the claim (mirror of the payment paths).
   732→  if (args.amountMinor !== payout.amountMinor) {
   733→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${args.amountMinor} does not match payout ${payout.amountMinor}`);
   734→  }
   735→
   736→  const seller = await db.seller.findUnique({ where: { id: payout.sellerId } });
   737→  if (!seller) throw new ApiError(404, "NOT_FOUND", "Seller for payout not found");
   738→
   739→  const at = new Date();
   740→  // T4f reason: use the provider-supplied reason when present.
   741→  const reason = args.reason ?? "provider reported failure";
   742→  const subject = toStatus === "paid" ? `Payout ${payout.referenceNo} on the way` : `Payout ${payout.referenceNo} failed`;
   743→
   744→  const outcome = await db.$transaction(
   745→    async (tx) => {
   746→      // Conditional claim — expectedFrom is "processing" per the state machine
   747→      // (only processing payouts are awaiting provider confirmation). count 0
   748→      // ⇒ lost race / concurrent admin action ⇒ empty-tx-return (rollback-by-
   749→      // return: the claim was the only intended write and it wrote nothing).
   750→      const claim = await tx.payout.updateMany({
   751→        where: { id: payout.id, status: "processing" },
   752→        data: { status: toStatus },
   753→      });
   754→      if (claim.count === 0) return { claimed: false as const };
   755→
   756→      if (toStatus === "paid") {
   757→        await tx.payout.update({ where: { id: payout.id }, data: { paidAt: at } });
   758→        // T4 — clearing → bank; refreshWalletCache runs inside the recipe (tx).
   759→        await postPayoutConfirmed({
   760→          walletId: payout.walletId,
   761→          payoutId: payout.id,
   762→          referenceNo: payout.referenceNo,
   763→          amountMinor: payout.amountMinor,
   764→          tx,
   765→        });
   766→      } else {
   767→        await tx.payout.update({ where: { id: payout.id }, data: { failReason: reason } });
   768→        // T4f — reversal of T3 (clearing → back to seller payable).
   769→        await postPayoutFailed({
   770→          walletId: payout.walletId,
   771→          payoutId: payout.id,
   772→          referenceNo: payout.referenceNo,
   773→          amountMinor: payout.amountMinor,
   774→          reason,
   775→          tx,
   776→        });
   777→      }
   778→
   779→      // Seller notifications (email + inapp) with stable per-event dedupKeys;
   780→      // subjects mirror the admin mark-paid / fail routes.
   781→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "email", recipient: seller.email, subject, dedupKey });
   782→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "inapp", recipient: seller.email, subject, dedupKey });
   783→      await logActivity({
   784→        actorType: "system",
   785→        action: toStatus === "paid" ? "payout.provider_confirmed" : "payout.provider_failed",
   786→        entityType: "payout",
   787→        entityId: payout.id,
   788→        meta: { providerRef: args.providerRef, referenceNo: payout.referenceNo, amountMinor: payout.amountMinor },
   789→      });
   790→
   791→      return { claimed: true as const };
   792→    },
   793→    { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
   794→  );
   795→
   796→  const freshPayout = await db.payout.findUnique({ where: { id: payout.id } });
   797→  const payoutStatus = freshPayout?.status ?? payout.status;
   798→
   799→  if (!outcome.claimed) {
   800→    // Lost the race (e.g. an admin transitioned the payout concurrently) —
   801→    // answer ok with payoutConflict so the provider's retry converges; never
   802→    // throw raw 5xx at a PSP. Decision recorded (processed=true).
   803→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   804→    return {
   805→      ok: true,
   806→      duplicate: true,
   807→      payoutConflict: true,
   808→      eventType: args.eventType,
   809→      orderNumber: "",
   810→      orderStatus: payoutStatus,
   811→      intentStatus: "",
   812→      payoutStatus,
   813→    };
   814→  }
   815→
   816→  await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   817→  return {
   818→    ok: true,
   819→    duplicate: false,
   820→    eventType: args.eventType,
   821→    orderNumber: "",
   822→    orderStatus: payoutStatus,
   823→    intentStatus: "",
   824→    payoutStatus,
   825→  };
   826→}
   827→
   828→export function webhookPayloadJson(body: unknown): string {
   829→  return toJson(body);
   830→}
