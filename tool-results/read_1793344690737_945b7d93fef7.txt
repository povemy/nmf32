     3→// ---------------------------------------------------------------------------
     4→// Conversion Funnel — interactive 3D funnel diagram (HTML5 Canvas) + stage
     5→// streamgraph (recharts, centered "silhouette" stream) for the Insights page.
     6→//
     7→// Data: GET /api/store/insights/funnel?window=today|7d|30d|1y[&productId=<id>]
     8→// (seller-session scoped; stage numbers are server-computed — this component
     9→// never re-derives step rates or drop-off).
    10→//
    11→// Scope: store-wide (5 stages) or, via the product selector, ONE product's
    12→// funnel (4 stages: product view → add to cart → checkout → paid) with sales
    13→// context (units sold, gross product revenue). The 3D canvas, callout chips,
    14→// hover tooltip, legend, daily stacked bar and streamgraph are all
    15→// scope-generic — stage count and palette follow the active scope.
    16→//
    17→// Canvas rendering notes:
    18→//   • The funnel is a tapered horizontal ribbon: thickness ∝ stage count
    19→//     (min clamp keeps small stages visible/hoverable). Sections WIDEN when a
    20→//     later stage exceeds an earlier one — the honest mixed-basis case
    21→//     (sessions vs orders), never clamped to fake a funnel shape.
    22→//   • Pseudo-3D: each section gets an extruded top face (lighter tint) and the
    23→//     last section a right cap (darker shade).
    24→//   • Icons are white-stroke SVG data-URL <Image>s drawn on the slab faces.
    25→//   • Callout pointer chips sit above the ribbon (wide viewports); on narrow
    26→//     viewports the count pills on the ribbon + the DOM legend carry the info.
    27→//   • Hover: hit-test rects stored per section → lift + glow + DOM tooltip
    28→//     with count / prev / step rate / drop-off-or-gained.
    29→//   • Animations: thickness/height ease toward targets via requestAnimationFrame
    30→//     (prefers-reduced-motion snaps instantly).
    31→//   • The canvas is aria-hidden; a visually-hidden table mirrors the stage data
    32→//     for screen readers.

   185→/** Hex mix toward white (t>0) or black (t<0) — for the 3D face shading. */
   186→function shade(hex: string, t: number): string {
   187→  const n = parseInt(hex.slice(1), 16);
   188→  const r = (n >> 16) & 255;
   189→  const g = (n >> 8) & 255;
   190→  const b = n & 255;
   191→  const m = t >= 0 ? 255 : 0;
   192→  const k = Math.abs(t);
   193→  const c = (v: number) => Math.round(v + (m - v) * k);
   194→  return `rgb(${c(r)},${c(g)},${c(b)})`;
   195→}
   196→function withAlpha(hex: string, a: number): string {
   197→  const n = parseInt(hex.slice(1), 16);
   198→  return `rgba(${(n >> 16) & 255},${(n >> 8) & 255},${n & 255},${a})`;
   199→}

   367→type StageRow = FunnelData["stages"][number];
   368→type HitRect = { i: number; x0: number; x1: number; y0: number; y1: number };
   369→
   370→function FunnelCanvas({ stages, meta, emptyCaption }: { stages: StageRow[]; meta: StageMeta[]; emptyCaption?: string }) {
   371→  const wrapRef = useRef<HTMLDivElement | null>(null);
   372→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   373→  const [width, setWidth] = useState(0);
   374→  const [hover, setHover] = useState<number | null>(null);
   375→  const [tip, setTip] = useState<{ i: number; x: number; y: number } | null>(null);
   376→  const [iconTick, setIconTick] = useState(0);
   377→
   378→  const iconsRef = useRef<Map<string, HTMLImageElement>>(new Map());
   379→  const hitRef = useRef<HitRect[]>([]);
   380→  const animRef = useRef<{ thick: number[]; lift: number[]; raf: number | null }>({ thick: [], lift: [], raf: null });
   381→  const hoverRef = useRef<number | null>(null);
   382→  useEffect(() => {
   383→    hoverRef.current = hover;
   384→  }, [hover]);
   385→
   386→  const reducedMotion = useMemo(
   387→    () => typeof window !== "undefined" && window.matchMedia("(prefers-reduced-motion: reduce)").matches,
   388→    []
   389→  );
   390→
   391→  // Load the SVG icons once (both scopes — keys overlap, dedup keeps one
   392→  // image per key); each completed load triggers a redraw.
   393→  useEffect(() => {
   394→    for (const m of [...STORE_STAGE_META, ...PRODUCT_STAGE_META]) {
   395→      if (iconsRef.current.has(m.key)) continue;
   396→      const img = new Image();
   397→      img.onload = () => setIconTick((t) => t + 1);
   398→      img.src = m.icon;
   399→      iconsRef.current.set(m.key, img);
   400→    }
   401→  }, []);
   402→
   403→  // Responsive width.
   404→  useEffect(() => {
   405→    const el = wrapRef.current;
   406→    if (!el) return;
   407→    const ro = new ResizeObserver((entries) => {
   408→      for (const e of entries) setWidth(Math.max(280, Math.round(e.contentRect.width)));
   409→    });
   410→    ro.observe(el);
   411→    setWidth(Math.max(280, Math.round(el.getBoundingClientRect().width)));
   412→    return () => ro.disconnect();
   413→  }, []);
   414→
   415→  const draw = useCallback(() => {
   416→    const canvas = canvasRef.current;
   417→    if (!canvas || width < 100) return;
   418→    const wide = width >= 640;
   419→    const H = wide ? 340 : 300;
   420→    const dpr = Math.min(2, window.devicePixelRatio || 1);
   421→    canvas.width = Math.round(width * dpr);
   422→    canvas.height = Math.round(H * dpr);
   423→    canvas.style.height = `${H}px`;
   424→    const ctx = canvas.getContext("2d");
   425→    if (!ctx) return;
   426→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   427→    ctx.clearRect(0, 0, width, H);
   428→
   429→    // ----- geometry -------------------------------------------------------
   430→    const DX = 24;
   431→    const DY = 12;
   432→    const topPad = wide ? 92 : 16;
   433→    const bottomPad = wide ? 34 : 26;
   434→    const fx0 = 14;
   435→    const fx1 = width - 14 - DX;
   436→    const fy = topPad + (H - topPad - bottomPad) / 2;
   437→    const maxThick = Math.max(60, H - topPad - bottomPad - 20);
   438→    const minThick = 18;
   439→
   440→    const counts = stages.map((s) => s.count);
   441→    const maxCount = Math.max(1, ...counts);
   442→    const targets = stages.map((s) => minThick + (s.count / maxCount) * (maxThick - minThick));
   443→    const anim = animRef.current;
   444→    if (anim.thick.length !== stages.length) {
   445→      anim.thick = reducedMotion ? [...targets] : stages.map(() => 0);
   446→      anim.lift = stages.map(() => 0);
   447→    }
   448→
   449→    const n = stages.length;
   450→    const cx = (i: number) => fx0 + ((i + 0.5) * (fx1 - fx0)) / n;
   451→    const bound = (i: number) => (i === 0 ? fx0 : i === n ? fx1 : (cx(i - 1) + cx(i)) / 2);
   452→
   453→    const empty = counts.every((c) => c === 0);
   454→    const thick = anim.thick;
   455→    const lift = anim.lift;
   456→
   457→    // Thickness at a boundary: flat over the owning stage's centre, linear
   458→    // between adjacent stage centres.
   459→    const thAt = (i: number) => {
   460→      if (i <= 0) return thick[0] ?? minThick;
   461→      if (i >= n) return thick[n - 1] ?? minThick;
   462→      return ((thick[i - 1] ?? minThick) + (thick[i] ?? minThick)) / 2;
   463→    };
   464→
   465→    hitRef.current = [];
   466→
   467→    // ----- draw ------------------------------------------------------------
   468→    for (let i = 0; i < n; i++) {
   469→      const bL = bound(i);
   470→      const bR = bound(i + 1);
   471→      const thL = Math.max(minThick * 0.6, thAt(i));
   472→      const thR = Math.max(minThick * 0.6, thAt(i + 1));
   473→      const sm = meta[i]!;
   474→      const color = empty ? "#cbd5e1" : sm.color;
   475→      const hov = hoverRef.current === i;
   476→      const ly = -lift[i]!;
   477→
   478→      // hit rect (generous: front face + the extruded depth column)
   479→      const maxTh = Math.max(thL, thR);
   480→      hitRef.current.push({ i, x0: bL, x1: bR + DX, y0: fy - maxTh / 2 - 8 + ly, y1: fy + maxTh / 2 + 6 + ly });
   481→
   482→      const frontPath = new Path2D();
   483→      frontPath.moveTo(bL, fy - thL / 2 + ly);
   484→      frontPath.lineTo(bR, fy - thR / 2 + ly);
   485→      frontPath.lineTo(bR, fy + thR / 2 + ly);
   486→      frontPath.lineTo(bL, fy + thL / 2 + ly);
   487→      frontPath.closePath();
   488→
   489→      const topPath = new Path2D();
   490→      topPath.moveTo(bL, fy - thL / 2 + ly);
   491→      topPath.lineTo(bR, fy - thR / 2 + ly);
   492→      topPath.lineTo(bR + DX, fy - thR / 2 - DY + ly);
   493→      topPath.lineTo(bL + DX, fy - thL / 2 - DY + ly);
   494→      topPath.closePath();
   495→
   496→      // shadow under hovered slab
   497→      if (hov) {
   498→        ctx.save();
   499→        ctx.shadowColor = "rgba(15,23,42,0.28)";
   500→        ctx.shadowBlur = 20;
   501→        ctx.shadowOffsetY = 10;
   502→        ctx.fillStyle = shade(color, hov ? 0.08 : -0.12);
   503→        ctx.fill(frontPath);
   504→        ctx.restore();
   505→      }
   506→
   507→      // top (3D) face — lighter tint; hovered pops brighter
   508→      const topFill = ctx.createLinearGradient(bL, fy - thL / 2 - DY + ly, bL, fy - thL / 2 + ly);
   509→      topFill.addColorStop(0, shade(color, hov ? 0.62 : 0.45));
   510→      topFill.addColorStop(1, shade(color, hov ? 0.2 : 0.08));
   511→      ctx.fillStyle = topFill;
   512→      ctx.fill(topPath);
   513→
   514→      // front face — the stage's colour gradient
   515→      const frontFill = ctx.createLinearGradient(0, fy - thL / 2 + ly, 0, fy + thL / 2 + ly);
   516→      frontFill.addColorStop(0, shade(color, hov ? 0.18 : 0.04));
   517→      frontFill.addColorStop(0.5, color);
   518→      frontFill.addColorStop(1, shade(color, hov ? -0.1 : -0.22));
   519→      ctx.fillStyle = frontFill;
   520→      ctx.fill(frontPath);
   521→
   522→      // right cap on the last section (depth shading)
   523→      if (i === n - 1) {
   524→        const capPath = new Path2D();
   525→        capPath.moveTo(bR, fy - thR / 2 + ly);
   526→        capPath.lineTo(bR + DX, fy - thR / 2 - DY + ly);
   527→        capPath.lineTo(bR + DX, fy + thR / 2 - DY + ly);
   528→        capPath.lineTo(bR, fy + thR / 2 + ly);
   529→        capPath.closePath();
   530→        ctx.fillStyle = shade(color, -0.32);
   531→        ctx.fill(capPath);
   532→      }
   533→
   534→      // hover outline
   535→      if (hov) {
   536→        ctx.strokeStyle = withAlpha("#ffffff", 0.85);
   537→        ctx.lineWidth = 1.5;
   538→        ctx.stroke(frontPath);
   539→        ctx.stroke(topPath);
   540→      }
   541→
   542→      // ----- icon + count on the slab ---------------------------------------
   543→      const icx = cx(i);
   544→      const t = thick[i] ?? minThick;
   545→      const stage = stages[i]!;
   546→      const cnt = stage.count.toLocaleString("en-MY");
   547→      const icon = iconsRef.current.get(sm.key);
   548→
   549→      if (t >= 30) {
   550→        ctx.font = FONT(700, 14);

   559→        ctx.textAlign = "left";
   560→        ctx.textBaseline = "middle";

   571→        ctx.fillStyle = "rgba(255,255,255,0.96)";
   572→        ctx.fill();
   573→        ctx.strokeStyle = withAlpha(color, 0.55);
   574→        ctx.lineWidth = 1;
   575→        ctx.stroke();
   576→        ctx.fillStyle = shade(color, -0.18);
   577→        ctx.textAlign = "left";

   595→        const chipH = 27;
   596→        let chipX = icx - chipW / 2;
   597→        chipX = Math.max(4, Math.min(width - chipW - 4, chipX));
   598→        const chipY = level;
   599→
   600→        // pointer line: chip → slab top edge
   601→        const slabTop = fy - (thick[i] ?? minThick) / 2 + ly;
   602→        ctx.strokeStyle = withAlpha(color, 0.6);
   603→        ctx.lineWidth = 1;
   604→        ctx.beginPath();
   605→        const pxLine = chipX + chipW / 2;
   606→        ctx.moveTo(pxLine, chipY + chipH);
   607→        ctx.lineTo(pxLine, slabTop - 7);

   621→        ctx.fill();
   622→        ctx.strokeStyle = withAlpha(color, hov ? 0.95 : 0.5);
   623→        ctx.lineWidth = hov ? 1.6 : 1;
   624→        ctx.stroke();
   625→        // colour dot
   626→        ctx.beginPath();

   628→        ctx.fillStyle = color;
   629→        ctx.fill();

   632→        ctx.textAlign = "left";
   633→        ctx.fillStyle = "#374151";

   653→  }, [stages, meta, emptyCaption, width, reducedMotion, iconTick]);
   654→
   655→  // Animation loop — eases thickness toward targets and lift toward hover state.
   656→  useEffect(() => {
   657→    const targets = stages.map((s) => {
   658→      const counts = stages.map((x) => x.count);
   659→      const maxCount = Math.max(1, ...counts);
   660→      const wide = width >= 640;
   661→      const H = wide ? 340 : 300;
   662→      const topPad = wide ? 92 : 16;
   663→      const bottomPad = wide ? 34 : 26;
   664→      const maxThick = Math.max(60, H - topPad - bottomPad - 20);
   665→      return 18 + (s.count / maxCount) * (maxThick - 18);
   666→    });
   667→    const anim = animRef.current;
   668→    if (anim.thick.length !== stages.length) anim.thick = reducedMotion ? [...targets] : stages.map(() => 0);
   669→    if (anim.lift.length !== stages.length) anim.lift = stages.map(() => 0);
   670→
   671→    if (reducedMotion) {
   672→      anim.thick = [...targets];
   673→      anim.lift = stages.map((_, i) => (hoverRef.current === i ? 6 : 0));
   674→      draw();
   675→      return;
   676→    }
   677→
   678→    const step = () => {
   679→      let moving = false;
   680→      for (let i = 0; i < targets.length; i++) {
   681→        const dt = targets[i]! - (anim.thick[i] ?? 0);
   682→        const dl = (hoverRef.current === i ? 6 : 0) - (anim.lift[i] ?? 0);
   683→        if (Math.abs(dt) > 0.4) {
   684→          anim.thick[i] = (anim.thick[i] ?? 0) + dt * 0.16;
   685→          moving = true;
   686→        } else anim.thick[i] = targets[i]!;
   687→        if (Math.abs(dl) > 0.3) {
   688→          anim.lift[i] = (anim.lift[i] ?? 0) + dl * 0.28;
   689→          moving = true;
   690→        } else anim.lift[i] = hoverRef.current === i ? 6 : 0;
   691→      }
   692→      draw();
   693→      anim.raf = moving ? requestAnimationFrame(step) : null;
   694→    };
   695→    anim.raf = requestAnimationFrame(step);
   696→    return () => {
   697→      if (anim.raf != null) cancelAnimationFrame(anim.raf);
   698→      anim.raf = null;
   699→    };
   700→  }, [stages, width, hover, reducedMotion, draw, iconTick]);
   701→
   702→  // Pointer events → hover hit-testing + tooltip.
   703→  const onPointer = useCallback((e: React.PointerEvent<HTMLCanvasElement>) => {
   704→    const canvas = canvasRef.current;
   705→    if (!canvas) return;
   706→    const rect = canvas.getBoundingClientRect();
   707→    const x = e.clientX - rect.left;
   708→    const y = e.clientY - rect.top;
   709→    let found: number | null = null;
   710→    for (const h of hitRef.current) {
   711→      if (x >= h.x0 && x <= h.x1 && y >= h.y0 && y <= h.y1) {
   712→        found = h.i;
   713→        break;
   714→      }
   715→    }
   716→    setHover(found);
   717→    setTip(found != null ? { i: found, x: e.clientX - rect.left, y: e.clientY - rect.top } : null);
   718→  }, []);
   719→
   720→  const stage = tip != null ? stages[tip.i] : null;
   721→  const sm = tip != null ? meta[tip.i] : null;
   722→
   723→  return (
   724→    <div ref={wrapRef} className="relative w-full">
   725→      <canvas
   726→        ref={canvasRef}
   727→        className="block w-full touch-none select-none"
   728→        style={{ cursor: hover != null ? "pointer" : "default" }}
   729→        aria-hidden="true"
   730→        onPointerMove={onPointer}
   731→        onPointerLeave={() => {
   732→          setHover(null);
   733→          setTip(null);
   734→        }}
   735→      />
   736→      {/* screen-reader mirror of the diagram */}
   737→      <table className="sr-only">
   738→        <caption>Conversion funnel stages for the selected window</caption>
   739→        <thead>
   740→          <tr>
   741→            <th scope="col">Stage</th>
   742→            <th scope="col">Count</th>
   743→            <th scope="col">Previous window</th>
   744→            <th scope="col">Step rate</th>
   745→            <th scope="col">Drop-off</th>
   746→          </tr>
   747→        </thead>
   748→        <tbody>
   749→          {stages.map((s) => (
   750→            <tr key={s.key}>
   751→              <th scope="row">{s.label}</th>
   752→              <td>{s.count}</td>
   753→              <td>{s.prev}</td>
   754→              <td>{s.stepRate == null ? "not computable" : `${s.stepRate}%`}</td>
   755→              <td>{s.dropOff == null ? "—" : s.dropOff > 0 ? `${s.dropOff} (${s.dropOffPct ?? "—"}%)` : s.dropOff < 0 ? `gained ${-s.dropOff}` : "0"}</td>
   756→            </tr>
   757→          ))}
   758→        </tbody>
   759→      </table>
   760→
   761→      {stage && sm && tip ? (
   762→        <div
   763→          role="tooltip"
   764→          className="pointer-events-none absolute z-10 w-52 rounded-xl border bg-popover/95 p-3 text-xs shadow-lg backdrop-blur"
   765→          style={{
   766→            left: Math.max(6, Math.min(Math.max(280, width) - 216, tip.x + 16)),
   767→            top: Math.max(6, tip.y - 64),
   768→          }}
   769→        >
   770→          <p className="flex items-center gap-1.5 font-semibold">
   771→            <span className="h-2 w-2 rounded-full" style={{ backgroundColor: sm.color }} aria-hidden="true" />
   772→            {stage.label}
   773→          </p>
   774→          <p className="mt-1 tabular-nums">
   775→            <span className="font-bold" style={{ color: shade(sm.color, -0.15) }}>{stage.count.toLocaleString("en-MY")}</span>
   776→            <span className="text-muted-foreground"> in window · prev {stage.prev.toLocaleString("en-MY")}</span>
   777→          </p>
   778→          <p className="mt-0.5 tabular-nums text-muted-foreground">
   779→            Step {stage.stepRate == null ? "—" : `${stage.stepRate}%`}
   780→            {stage.dropOff != null ? (
   781→              <span className={stage.dropOff > 0 ? "text-rose-600" : stage.dropOff < 0 ? "text-emerald-600" : ""}>
   782→                {stage.dropOff > 0
   783→                  ? ` · drop ${stage.dropOff.toLocaleString("en-MY")}${stage.dropOffPct != null ? ` (${stage.dropOffPct}%)` : ""}`
   784→                  : stage.dropOff < 0
   785→                    ? ` · gained ${(-stage.dropOff).toLocaleString("en-MY")}`
   786→                    : " · drop 0"}
   787→              </span>
   788→            ) : null}
   789→          </p>
   790→        </div>
   791→      ) : null}
   792→    </div>
   793→  );
   794→}

   796→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   797→  ctx.beginPath();
   798→  ctx.moveTo(x + r, y);
   799→  ctx.lineTo(x + w - r, y);
   800→  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
   801→  ctx.lineTo(x + w, y + h - r);
   802→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
   803→  ctx.lineTo(x + r, y + h);
   804→  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
   805→  ctx.lineTo(x, y + r);
   806→  ctx.quadraticCurveTo(x, y, x + r, y);
   807→  ctx.closePath();
   808→}

   812→/** Stream series field per scope: the store funnel carries 5 stages (incl.
   813→ *  visits), the product funnel 4 (starts at product views — a store visit is
   814→ *  not product-scoped). Module-level consts → stable useMemo deps. */
   815→const STORE_STREAM_KEYS = ["visits", "productViews", "cartAdds", "checkouts", "paidOrders"] as const;
   816→const PRODUCT_STREAM_KEYS = ["productViews", "cartAdds", "checkouts", "paidOrders"] as const;
   817→
   818→type StreamPoint = Record<string, number | string>;
   819→
   820→/** Stage flow over time. Weekly/Monthly/Yearly → centred streamgraph (recharts
   821→ *  stacked areas over an invisible negative-offset base — the "silhouette"
   822→ *  stream). Daily → the window is a single day, so a stacked bar of today's
   823→ *  stages replaces the degenerate one-point stream. Scope-generic: the stage
   824→ *  count follows the active funnel (5 store stages / 4 product stages). */
   825→function FunnelStream({ data, view }: { data: FunnelData; view: FunnelViewKey }) {
   826→  const meta = stageMetaFor(data.scope);
   827→  const keys = data.scope === "product" ? PRODUCT_STREAM_KEYS : STORE_STREAM_KEYS;
   828→  const series = useMemo(
   829→    () =>
   830→      data.byDay.map((d): StreamPoint => {
   831→        const vals = keys.map((k) => Number(d[k] ?? 0));
   832→        const total = vals.reduce((a, b) => a + b, 0);
   833→        const o: StreamPoint = { day: d.day, off: -total / 2, total };
   834→        vals.forEach((v, i) => {
   835→          o[`v${i}`] = v;
   836→        });
   837→        return o;
   838→      }),
   839→    [data.byDay, keys]
   840→  );
   841→  const hasFlow = series.some((p) => Number(p.total) > 0);
   842→
   843→  if (view === "daily") {
   844→    const p = series[series.length - 1];
   845→    const vals = meta.map((_, i) => (p ? Number(p[`v${i}`] ?? 0) : 0));
   846→    const total = vals.reduce((a, b) => a + b, 0);
   847→    return (
   848→      <div>
   849→        <p className="mb-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Today&apos;s stages</p>
   850→        {total === 0 ? (
   851→          <p className="rounded-xl border border-dashed py-6 text-center text-xs text-muted-foreground">
   852→            {data.scope === "product" ? "No activity for this product today yet." : "No activity today yet."}
   853→          </p>
   854→        ) : (
   855→          <div className="space-y-2">
   856→            <div
   857→              className="flex h-9 w-full overflow-hidden rounded-lg"
   858→              role="img"
   859→              aria-label={data.scope === "product" ? "Today's product funnel stages stacked bar" : "Today's funnel stages stacked bar"}
   860→            >
   861→              {vals.map((v, i) =>
   862→                v <= 0 ? null : (
   863→                  <div
   864→                    key={i}
   865→                    className="h-full min-w-[4px] transition-[flex-grow] duration-500"
   866→                    style={{ flexGrow: v, backgroundColor: meta[i]!.color }}
   867→                    title={`${meta[i]!.label}: ${v}`}
   868→                  />
   869→                )
   870→              )}
   871→            </div>
   872→            <p className="text-[11px] text-muted-foreground">
   873→              {total.toLocaleString("en-MY")} stage events today · segments scale to their share.
   874→            </p>
   875→          </div>
   876→        )}
   877→      </div>
   878→    );
   879→  }
   880→
   881→  return (
   882→    <div>
   883→      <p className="mb-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Stage flow over time</p>
   884→      {hasFlow ? (
   885→        <div className="h-[190px] w-full" role="img" aria-label="Funnel stage flow streamgraph by day">
   886→          <ResponsiveContainer width="100%" height="100%">
   887→            <AreaChart data={series} margin={{ top: 4, right: 8, left: 8, bottom: 0 }}>
   888→              {meta.map((m, i) => (
   889→                <defs key={m.key}>
   890→                  <linearGradient id={`fsGrad${i}`} x1="0" y1="0" x2="0" y2="1">
   891→                    <stop offset="0%" stopColor={m.color} stopOpacity={0.72} />
   892→                    <stop offset="100%" stopColor={m.color} stopOpacity={0.28} />
   893→                  </linearGradient>
   894→                </defs>
   895→              ))}
   896→              <XAxis
   897→                dataKey="day"
   898→                tickFormatter={(v: string) => String(v).slice(5)}
   899→                fontSize={10}
   900→                tickLine={false}
   901→                axisLine={false}
   902→                minTickGap={30}
   903→                stroke="currentColor"
   904→                className="text-muted-foreground"
   905→              />
   906→              <Tooltip content={<StreamTip meta={meta} />} />
   907→              {/* invisible negative base centred around zero → silhouette stream */}
   908→              <Area dataKey="off" stackId="s" type="monotone" stroke="none" fill="transparent" isAnimationActive={false} />
   909→              {meta.map((m, i) => (
   910→                <Area
   911→                  key={m.key}
   912→                  dataKey={`v${i}`}
   913→                  stackId="s"
   914→                  type="monotone"
   915→                  stroke={m.color}
   916→                  strokeWidth={1.4}
   917→                  fill={`url(#fsGrad${i})`}
   918→                  isAnimationActive={false}
   919→                />
   920→              ))}
   921→            </AreaChart>
   922→          </ResponsiveContainer>
   923→        </div>
   924→      ) : (
   925→        <p className="rounded-xl border border-dashed py-6 text-center text-xs text-muted-foreground">
   926→          {data.scope === "product"
   927→            ? "No stage activity for this product in this window yet — the stream appears as tracking and orders accrue."
   928→            : "No stage activity in this window yet — the stream appears as visitor tracking and orders accrue."}
   929→        </p>
   930→      )}
   931→    </div>
   932→  );
   933→}
