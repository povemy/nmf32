     3→// Design contract (agent-ctx/SI-PLAN.md + docs/seller-insights/metric-definitions.md):
     4→//  • Money metrics derive ONLY from authoritative Order/OrderItem/Refund records

    24→export type InsightsWindow = {
    25→  key: "today" | "7d" | "30d" | "custom";
    26→  from: Date;
    27→  to: Date;
    28→  prevFrom: Date;
    29→  prevTo: Date;
    30→  days: number;
    31→};
    32→
    33→function klDayKey(d: Date): string {
    34→  return new Date(d.getTime() + KL_OFFSET_MS).toISOString().slice(0, 10);
    35→}
    36→function klStartOfToday(): Date {
    37→  const kl = new Date(new Date().getTime() + KL_OFFSET_MS);
    38→  return new Date(Date.UTC(kl.getUTCFullYear(), kl.getUTCMonth(), kl.getUTCDate()) - KL_OFFSET_MS);
    39→}
    40→function klDayStart(ymd: string): Date {
    41→  const kl = new Date(`${ymd}T00:00:00.000Z`); // interpreted as KL wall clock
    42→  return new Date(kl.getTime() - KL_OFFSET_MS);
    43→}

    45→export function parseInsightsWindow(url: URL): InsightsWindow {
    46→  const raw = url.searchParams.get("window") ?? "30d";
    47→  const now = new Date();
    48→  let from: Date;
    49→  let to: Date = now;
    50→  if (raw === "today") {
    51→    from = klStartOfToday();
    52→  } else if (raw === "7d" || raw === "30d") {
    53→    from = new Date(now.getTime() - (raw === "7d" ? 7 : 30) * 86_400_000);
    54→  } else if (raw === "custom") {
    55→    const fromStr = url.searchParams.get("from") ?? "";
    56→    const toStr = url.searchParams.get("to") ?? "";
    57→    if (!/^\d{4}-\d{2}-\d{2}$/.test(fromStr) || !/^\d{4}-\d{2}-\d{2}$/.test(toStr)) {
    58→      throw new ApiError(400, "VALIDATION", "custom window requires from=YYYY-MM-DD and to=YYYY-MM-DD");
    59→    }
    60→    from = klDayStart(fromStr);
    61→    to = new Date(klDayStart(toStr).getTime() + 86_400_000); // inclusive end day
    62→    if (from >= to) throw new ApiError(400, "VALIDATION", "custom window `from` must be before `to`");
    63→    if (to.getTime() - from.getTime() > MAX_WINDOW_DAYS * 86_400_000) {
    64→      throw new ApiError(400, "VALIDATION", `custom window cannot exceed ${MAX_WINDOW_DAYS} days`);
    65→    }
    66→    if (to > now) to = now;
    67→  } else {
    68→    throw new ApiError(400, "VALIDATION", "window must be one of today|7d|30d|custom");
    69→  }
    70→  const len = to.getTime() - from.getTime();
    71→  return {
    72→    key: raw,
    73→    from,
    74→    to,
    75→    prevFrom: new Date(from.getTime() - len),
    76→    prevTo: from,
    77→    days: Math.max(1, Math.round(len / 86_400_000)),
    78→  };
    79→}

    82→// TTL cache (house pattern: globalThis + ?fresh=1 bypass, like /api/health).

    84→type CacheEntry = { at: number; value: unknown };
    85→const gCache = globalThis as typeof globalThis & { __siInsightsCache?: Map<string, CacheEntry> };
    86→function cacheStore(): Map<string, CacheEntry> {
    87→  if (!gCache.__siInsightsCache) gCache.__siInsightsCache = new Map();
    88→  return gCache.__siInsightsCache;

    90→export async function insightsCached<T>(key: string, url: URL, compute: () => Promise<T>, ttlMs = 45_000): Promise<T> {
    91→  if (url.searchParams.get("fresh") === "1") return compute();
    92→  const store = cacheStore();
    93→  const hit = store.get(key);
    94→  if (hit && Date.now() - hit.at < ttlMs) return hit.value as T;
    95→  const value = await compute();
    96→  store.set(key, { at: Date.now(), value });
    97→  if (store.size > 500) {
    98→    // bounded: drop the 200 oldest entries
    99→    const entries = [...store.entries()].sort((a, b) => a[1].at - b[1].at).slice(0, 200);
   100→    for (const [k] of entries) store.delete(k);
   101→  }
   102→  return value;
   103→}

   116→function pct(value: number, prev: number): { pct: number | null; isNew: boolean } {
   117→  if (prev <= 0) return { pct: null, isNew: value > 0 };
   118→  return { pct: Math.round(((value - prev) / prev) * 1000) / 10, isNew: false };
   119→}
   120→function rate(num: number, den: number): number | null {
   121→  if (den <= 0) return null;
   122→  return Math.round((num / den) * 1000) / 10;
   123→}

   125→type EventRow = { eventType: string; sessionId: string | null; visitorId: string | null; shopperId: string | null; productId: string | null; qty: number | null; channel: string | null; createdAt: Date };

   127→/** Bounded event fetch for a store window (≤ 20k rows; sets `truncated` if hit). */
   128→async function eventsFor(storeId: string, from: Date, to: Date): Promise<{ rows: EventRow[]; truncated: boolean }> {
   129→  const rows = await db.analyticsEvent.findMany({
   130→    where: { storeId, createdAt: { gte: from, lt: to } },
   131→    select: { eventType: true, sessionId: true, visitorId: true, shopperId: true, productId: true, qty: true, channel: true, createdAt: true },
   132→    orderBy: { createdAt: "asc" },
   133→    take: 20_001,
   134→  });
   135→  const truncated = rows.length > 20_000;
   136→  return { rows: truncated ? rows.slice(0, 20_000) : rows, truncated };
   137→}
   138→
   139→function sessionKey(e: EventRow): string | null {
   140→  if (e.sessionId) return e.sessionId;
   141→  if (e.shopperId) return `u:${e.shopperId}`;
   142→  return null;
   143→}

   186→export type SalesAgg = {
   187→  paidOrders: number;
   188→  unitsSold: number;
   189→  grossProductSalesMinor: number;
   190→  discountsMinor: number;
   191→  shippingChargedMinor: number;
   192→  grossSalesMinor: number; // order GMV: products + shipping − discounts (+ tax)
   193→  sellerProceedsMinor: number;
   194→  refundedItemsMinor: number; // cumulative refundedMinor on items of these orders
   195→};
   196→
   197→export function aggregateSales(orders: PaidOrderRow[]): SalesAgg {
   198→  const agg: SalesAgg = {
   199→    paidOrders: orders.length, unitsSold: 0, grossProductSalesMinor: 0, discountsMinor: 0,
   200→    shippingChargedMinor: 0, grossSalesMinor: 0, sellerProceedsMinor: 0, refundedItemsMinor: 0,
   201→  };
   202→  for (const o of orders) {
   203→    agg.grossSalesMinor += o.totalMinor;
   204→    for (const it of o.items) {
   205→      agg.unitsSold += it.qty;
   206→      agg.grossProductSalesMinor += it.lineSubtotalMinor;
   207→      agg.discountsMinor += it.discountMinor;
   208→      agg.shippingChargedMinor += it.shippingMinor;
   209→      agg.sellerProceedsMinor += it.sellerNetMinor;
   210→      agg.refundedItemsMinor += it.refundedMinor;
   211→    }
   212→  }
   213→  return agg;
   214→}

   239→export async function trafficSummary(storeId: string, win: InsightsWindow): Promise<TrafficSummary> {
   240→  const [cur, prev] = await Promise.all([eventsFor(storeId, win.from, win.to), eventsFor(storeId, win.prevFrom, win.prevTo)]);
   241→
   242→  const views = cur.rows.filter((e) => e.eventType === "store_view");
   243→  const pviews = cur.rows.filter((e) => e.eventType === "product_view");
   244→  const sessions = new Set(views.map(sessionKey).filter((x): x is string => Boolean(x)));
   245→  const visitors = new Set(cur.rows.map((e) => e.visitorId).filter((x): x is string => Boolean(x)));
   246→  const prevViews = prev.rows.filter((e) => e.eventType === "store_view");
   247→  const prevSessions = new Set(prevViews.map(sessionKey).filter((x): x is string => Boolean(x)));
   248→  const prevVisitors = new Set(prev.rows.map((e) => e.visitorId).filter((x): x is string => Boolean(x)));
   249→
   250→  // Daily series (KL day buckets)
   251→  const dayMap = new Map<string, { storeViews: number; productViews: number }>();
   252→  for (let d = new Date(win.from); d < win.to; d = new Date(d.getTime() + 86_400_000)) {
   253→    dayMap.set(klDayKey(d), { storeViews: 0, productViews: 0 });
   254→  }
   255→  for (const e of [...views, ...pviews]) {
   256→    const k = klDayKey(e.createdAt);
   257→    const slot = dayMap.get(k) ?? { storeViews: 0, productViews: 0 };
   258→    if (e.eventType === "store_view") slot.storeViews += 1;
   259→    else slot.productViews += 1;
   260→    dayMap.set(k, slot);
   261→  }
   262→
   263→  // Channels (store_view events only; sessions = distinct session per channel)
   264→  const chanMap = new Map<string, { views: number; sessions: Set<string> }>();
   265→  for (const e of views) {
   266→    const ch = e.channel ?? "direct";
   267→    const slot = chanMap.get(ch) ?? { views: 0, sessions: new Set<string>() };
   268→    slot.views += 1;
   269→    const sk = sessionKey(e);
   270→    if (sk) slot.sessions.add(sk);
   271→    chanMap.set(ch, slot);
   272→  }
   273→
   274→  // Top viewed products (join titles/images from the store's own products only)
   275→  const viewCounts = new Map<string, number>();
   276→  for (const e of pviews) {
   277→    if (e.productId) viewCounts.set(e.productId, (viewCounts.get(e.productId) ?? 0) + 1);
   278→  }
   279→  const topIds = [...viewCounts.entries()].sort((a, b) => b[1] - a[1]).slice(0, 8).map(([id]) => id);
   280→  const products = topIds.length
   281→    ? await db.product.findMany({ where: { id: { in: topIds }, storeId }, select: { id: true, title: true, slug: true, images: { orderBy: { sortOrder: "asc" }, take: 1, select: { url: true } } } })
   282→    : [];
   283→  const byId = new Map(products.map((p) => [p.id, p]));
   284→
   285→  return {
   286→    storeViews: views.length,
   287→    storeViewsPrev: prevViews.length,
   288→    sessions: sessions.size,
   289→    sessionsPrev: prevSessions.size,
   290→    uniqueVisitors: visitors.size,
   291→    uniqueVisitorsPrev: prevVisitors.size,
   292→    productViews: pviews.length,
   293→    productViewsPrev: prev.rows.filter((e) => e.eventType === "product_view").length,
   294→    byDay: [...dayMap.entries()].map(([d, v]) => ({ day: d, ...v })),
   295→    channels: [...chanMap.entries()].map(([channel, v]) => ({ channel, views: v.views, sessions: v.sessions.size })).sort((a, b) => b.views - a.views),
   296→    topViewedProducts: topIds
   297→      .map((id) => {
   298→        const p = byId.get(id);
   299→        return p ? { productId: id, title: p.title, slug: p.slug, imageUrl: p.images[0]?.url ?? null, views: viewCounts.get(id) ?? 0 } : null;
   300→      })
   301→      .filter((x): x is NonNullable<typeof x> => Boolean(x)),
   302→    truncated: cur.truncated || prev.truncated,
   303→  };
   304→}

   306→export type FunnelStage = { key: string; label: string; count: number; prev: number; stepRate: number | null };
   307→export type Funnel = { stages: FunnelStage[]; basis: string[] };
   308→
   309→export async function funnelFor(storeId: string, win: InsightsWindow): Promise<Funnel> {
   310→  const [cur, prev, ordersCur, ordersPrev] = await Promise.all([
   311→    eventsFor(storeId, win.from, win.to),
   312→    eventsFor(storeId, win.prevFrom, win.prevTo),
   313→    db.order.count({ where: { storeId, createdAt: { gte: win.from, lt: win.to } } }),
   314→    db.order.count({ where: { storeId, createdAt: { gte: win.prevFrom, lt: win.prevTo } } }),
   315→  ]);
   316→  const paidCur = await db.order.count({ where: { storeId, paidAt: { gte: win.from, lt: win.to } } });
   317→  const paidPrev = await db.order.count({ where: { storeId, paidAt: { gte: win.prevFrom, lt: win.prevTo } } });
   318→
   319→  const sessionSets = (rows: EventRow[]) => {
   320→    const store = new Set<string>();
   321→    const product = new Set<string>();
   322→    const cart = new Set<string>();
   323→    for (const e of rows) {
   324→      const sk = sessionKey(e);
   325→      if (!sk) continue;
   326→      if (e.eventType === "store_view") store.add(sk);
   327→      if (e.eventType === "product_view") product.add(sk);
   328→      if (e.eventType === "add_to_cart") cart.add(sk);
   329→    }
   330→    return { store, product, cart };
   331→  };
   332→  const s = sessionSets(cur.rows);
   333→  const sp = sessionSets(prev.rows);
   334→
   335→  const stages: FunnelStage[] = [
   336→    { key: "store_visit", label: "Store visit (sessions)", count: s.store.size, prev: sp.store.size, stepRate: null },
   337→    { key: "product_view", label: "Viewed a product", count: s.product.size, prev: sp.product.size, stepRate: rate(s.product.size, s.store.size) },
   338→    { key: "add_to_cart", label: "Added to cart", count: s.cart.size, prev: sp.cart.size, stepRate: rate(s.cart.size, s.product.size) },
   339→    { key: "begin_checkout", label: "Checkout started (orders)", count: ordersCur, prev: ordersPrev, stepRate: rate(ordersCur, s.cart.size) },
   340→    { key: "purchase", label: "Paid orders", count: paidCur, prev: paidPrev, stepRate: rate(paidCur, ordersCur) },
   341→  ];
   342→  return {
   343→    stages,
   344→    basis: [
   345→      "Visit → cart stages count DISTINCT pseudonymous sessions that performed each stage inside the window (estimates from first-party cookies; login not required).",
   346→      "Checkout started = orders created in the window (authoritative Order.createdAt, any status). Paid orders = orders with paidAt in the window.",
   347→      "The cart → checkout step mixes bases (sessions vs orders) — read it as an indication, not an exact visitor conversion.",
   348→    ],
   349→  };
   350→}

   355→export type CartActivity = {
   356→  addEvents: number; addEventsPrev: number;
   357→  addUnits: number; addUnitsPrev: number;
   358→  removeEvents: number; removeEventsPrev: number;
   359→  removeUnits: number; removeUnitsPrev: number;
   360→  byDay: Array<{ day: string; addUnits: number; removeUnits: number }>;
   361→  topAdded: Array<{ productId: string; title: string; imageUrl: string | null; addUnits: number; addEvents: number }>;

   364→export async function cartActivity(storeId: string, win: InsightsWindow): Promise<CartActivity> {
   365→  const [cur, prev] = await Promise.all([eventsFor(storeId, win.from, win.to), eventsFor(storeId, win.prevFrom, win.prevTo)]);
   366→  const adds = cur.rows.filter((e) => e.eventType === "add_to_cart");
   367→  const removes = cur.rows.filter((e) => e.eventType === "remove_from_cart");
   368→  const prevAdds = prev.rows.filter((e) => e.eventType === "add_to_cart");
   369→  const prevRemoves = prev.rows.filter((e) => e.eventType === "remove_from_cart");
   370→
   371→  const dayMap = new Map<string, { addUnits: number; removeUnits: number }>();
   372→  for (let d = new Date(win.from); d < win.to; d = new Date(d.getTime() + 86_400_000)) {
   373→    dayMap.set(klDayKey(d), { addUnits: 0, removeUnits: 0 });
   374→  }
   375→  for (const e of [...adds, ...removes]) {
   376→    const k = klDayKey(e.createdAt);
   377→    const slot = dayMap.get(k) ?? { addUnits: 0, removeUnits: 0 };
   378→    if (e.eventType === "add_to_cart") slot.addUnits += e.qty ?? 0;
   379→    else slot.removeUnits += e.qty ?? 0;
   380→    dayMap.set(k, slot);
   381→  }
   382→
   383→  const addMap = new Map<string, { units: number; events: number }>();
   384→  for (const e of adds) {
   385→    if (!e.productId) continue;
   386→    const slot = addMap.get(e.productId) ?? { units: 0, events: 0 };
   387→    slot.units += e.qty ?? 0;
   388→    slot.events += 1;
   389→    addMap.set(e.productId, slot);
   390→  }
   391→  const topIds = [...addMap.entries()].sort((a, b) => b[1].units - a[1].units).slice(0, 8).map(([id]) => id);
   392→  const products = topIds.length
   393→    ? await db.product.findMany({ where: { id: { in: topIds }, storeId }, select: { id: true, title: true, images: { orderBy: { sortOrder: "asc" }, take: 1, select: { url: true } } } })
   394→    : [];
   395→  const byId = new Map(products.map((p) => [p.id, p]));
   396→
   397→  return {
   398→    addEvents: adds.length, addEventsPrev: prevAdds.length,
   399→    addUnits: adds.reduce((s, e) => s + (e.qty ?? 0), 0),
   400→    addUnitsPrev: prevAdds.reduce((s, e) => s + (e.qty ?? 0), 0),
   401→    removeEvents: removes.length, removeEventsPrev: prevRemoves.length,
   402→    removeUnits: removes.reduce((s, e) => s + (e.qty ?? 0), 0),
   403→    removeUnitsPrev: prevRemoves.reduce((s, e) => s + (e.qty ?? 0), 0),
   404→    byDay: [...dayMap.entries()].map(([day, v]) => ({ day, ...v })),
   405→    topAdded: topIds
   406→      .map((id) => {
   407→        const p = byId.get(id);
   408→        return p ? { productId: id, title: p.title, imageUrl: p.images[0]?.url ?? null, addUnits: addMap.get(id)?.units ?? 0, addEvents: addMap.get(id)?.events ?? 0 } : null;
   409→      })
   410→      .filter((x): x is NonNullable<typeof x> => Boolean(x)),
   411→  };
   412→}

   424→/**
   425→ * CURRENT-CART snapshot — live CartItem state (NOT a sum of historical events).
   426→ * Eligibility (documented rule, mirrors the shopper cart's purchasable check):
   427→ * variant.isActive AND product.status = active AND store.status = live.
   428→ * Removed/purchased lines are physically deleted by those operations, so they
   429→ * are excluded by construction. A cart with 5 units of one product counts
   430→ * 5 units but 1 distinct cart. Mixed-vendor carts attribute each line to the
   431→ * line's own product store.
   432→ */
   433→export async function currentCartSnapshot(storeId: string): Promise<CartSnapshot> {
   434→  const [eligible, ineligible] = await Promise.all([
   435→    db.cartItem.findMany({
   436→      where: {
   437→        variant: { isActive: true, product: { storeId, status: "active", store: { status: "live" } } },
   438→      },
   439→      select: { cartId: true, qty: true, variant: { select: { productId: true } } },
   440→    }),
   441→    db.cartItem.count({
   442→      where: {
   443→        variant: { product: { storeId }, OR: [{ isActive: false }, { product: { status: { not: "active" } } }, { product: { store: { status: { not: "live" } } } }] },
   444→      },
   445→    }),
   446→  ]);
   447→
   448→  const carts = new Set<string>();
   449→  const perProduct = new Map<string, { units: number; carts: Set<string> }>();
   450→  let totalUnits = 0;
   451→  for (const it of eligible) {
   452→    carts.add(it.cartId);
   453→    totalUnits += it.qty;
   454→    const p = perProduct.get(it.variant.productId) ?? { units: 0, carts: new Set<string>() };
   455→    p.units += it.qty;
   456→    p.carts.add(it.cartId);
   457→    perProduct.set(it.variant.productId, p);
   458→  }
   459→
   460→  const topIds = [...perProduct.entries()].sort((a, b) => b[1].units - a[1].units).slice(0, 50).map(([id]) => id);
   461→  const products = topIds.length
   462→    ? await db.product.findMany({ where: { id: { in: topIds }, storeId }, select: { id: true, title: true, slug: true, images: { orderBy: { sortOrder: "asc" }, take: 1, select: { url: true } } } })
   463→    : [];
   464→  const byId = new Map(products.map((p) => [p.id, p]));
   465→
   466→  return {
   467→    sampledAt: new Date().toISOString(),
   468→    activeCarts: carts.size,
   469→    totalUnits,
   470→    totalLines: eligible.length,
   471→    ineligibleUnits: ineligible,
   472→    byProduct: topIds
   473→      .map((id) => {
   474→        const p = byId.get(id);
   475→        return p ? { productId: id, title: p.title, slug: p.slug, imageUrl: p.images[0]?.url ?? null, units: perProduct.get(id)?.units ?? 0, carts: perProduct.get(id)?.carts.size ?? 0 } : null;
   476→      })
   477→      .filter((x): x is NonNullable<typeof x> => Boolean(x)),
   478→    eligibilityNote:
   479→      "Live cart state in the database right now. Only purchasable lines count (variant active, product active, store live); units parked on unavailable items are reported separately as unavailable. Purchased/removed lines are deleted by those operations and therefore excluded. Historical add/remove activity is a separate metric.",
   480→  };
   481→}

   486→export type ProductPerfRow = {
   487→  productId: string;
   488→  title: string;
   489→  slug: string;
   490→  imageUrl: string | null;
   491→  status: string;
   492→  views: number;
   493→  viewSessions: number;
   494→  addUnits: number;
   495→  addEvents: number;
   496→  currentCartUnits: number;
   497→  currentCarts: number;
   498→  orderedUnits: number; // units in orders CREATED in window (any status)
   499→  purchasedUnits: number; // units in orders PAID in window
   500→  grossSalesMinor: number;
   501→  netSalesMinor: number; // lineSubtotal − discount − refunded (cumulative on those items)
   502→  refundedMinor: number;
   503→  viewToCartRate: number | null; // sessions that viewed AND added / sessions that viewed
   504→  checkoutToPurchaseRate: number | null; // purchasedUnits / orderedUnits
   505→};

   507→export type ProductSort = "views" | "adds" | "cartQty" | "purchases" | "revenue" | "conversion";

   509→export async function productPerformance(
   510→  storeId: string,
   511→  win: InsightsWindow,
   512→  sort: ProductSort,
   513→  page: number,
   514→  size: number
   515→): Promise<{ rows: ProductPerfRow[]; total: number }> {
   516→  const products = await db.product.findMany({
   517→    where: { storeId },
   518→    select: {
   519→      id: true, title: true, slug: true, status: true,
   520→      images: { orderBy: { sortOrder: "asc" }, take: 1, select: { url: true } },
   521→    },
   522→  });
   523→
   524→  const [events, createdOrders, paidOrders] = await Promise.all([
   525→    eventsFor(storeId, win.from, win.to),
   526→    db.order.findMany({
   527→      where: { storeId, createdAt: { gte: win.from, lt: win.to } },
   528→      select: { items: { select: { productId: true, qty: true } } },
   529→    }),
   530→    paidOrdersFor(storeId, win.from, win.to),
   531→  ]);
   532→
   533→  const snap = await currentCartSnapshot(storeId);
   534→
   535→  // Per-product accumulators
   536→  type Acc = {
   537→    views: number; viewSessions: Set<string>; addSessions: Set<string>;
   538→    addUnits: number; addEvents: number;
   539→    orderedUnits: number; purchasedUnits: number; gross: number; net: number; refunded: number;
   540→  };
   541→  const acc = new Map<string, Acc>();
   542→  const accFor = (id: string): Acc => {
   543→    let a = acc.get(id);
   544→    if (!a) {
   545→      a = { views: 0, viewSessions: new Set(), addSessions: new Set(), addUnits: 0, addEvents: 0, orderedUnits: 0, purchasedUnits: 0, gross: 0, net: 0, refunded: 0 };
   546→      acc.set(id, a);
   547→    }
   548→    return a;
   549→  };
   550→
   551→  for (const e of events.rows) {
   552→    if (!e.productId) continue;
   553→    const a = accFor(e.productId);
   554→    const sk = sessionKey(e);
   555→    if (e.eventType === "product_view") {
   556→      a.views += 1;
   557→      if (sk) a.viewSessions.add(sk);
   558→    } else if (e.eventType === "add_to_cart") {
   559→      a.addUnits += e.qty ?? 0;
   560→      a.addEvents += 1;
   561→      if (sk) a.addSessions.add(sk);
   562→    }
   563→  }
   564→  for (const o of createdOrders) {
   565→    for (const it of o.items) {
   566→      if (!it.productId) continue;
   567→      accFor(it.productId).orderedUnits += it.qty;
   568→    }
   569→  }
   570→  for (const o of paidOrders) {
   571→    for (const it of o.items) {
   572→      if (!it.productId) continue;
   573→      const a = accFor(it.productId);
   574→      a.purchasedUnits += it.qty;
   575→      a.gross += it.lineSubtotalMinor;
   576→      a.net += it.lineSubtotalMinor - it.discountMinor - it.refundedMinor;
   577→      a.refunded += it.refundedMinor;
   578→    }
   579→  }
   580→  const cartUnits = new Map(snap.byProduct.map((p) => [p.productId, p.units]));
   581→  const cartCarts = new Map(snap.byProduct.map((p) => [p.productId, p.carts]));
   582→
   583→  const rows: ProductPerfRow[] = products.map((p) => {
   584→    const a = acc.get(p.id);
   585→    const cart = { units: cartUnits.get(p.id) ?? 0, carts: cartCarts.get(p.id) ?? 0 };
   586→    return {
   587→      productId: p.id,
   588→      title: p.title,
   589→      slug: p.slug,
   590→      imageUrl: p.images[0]?.url ?? null,
   591→      status: p.status,
   592→      views: a?.views ?? 0,
   593→      viewSessions: a?.viewSessions.size ?? 0,
   594→      addUnits: a?.addUnits ?? 0,
   595→      addEvents: a?.addEvents ?? 0,
   596→      currentCartUnits: cart.units,
   597→      currentCarts: cart.carts,
   598→      orderedUnits: a?.orderedUnits ?? 0,
   599→      purchasedUnits: a?.purchasedUnits ?? 0,
   600→      grossSalesMinor: a?.gross ?? 0,
   601→      netSalesMinor: a?.net ?? 0,
   602→      refundedMinor: a?.refunded ?? 0,
   603→      viewToCartRate: a && a.viewSessions.size > 0 ? rate(a.addSessions.size, a.viewSessions.size) : null,
   604→      checkoutToPurchaseRate: a && a.orderedUnits > 0 ? rate(a.purchasedUnits, a.orderedUnits) : null,
   605→    };
   606→  });
   607→
   608→  const sorters: Record<ProductSort, (a: ProductPerfRow, b: ProductPerfRow) => number> = {
   609→    views: (a, b) => b.views - a.views,
   610→    adds: (a, b) => b.addUnits - a.addUnits,
   611→    cartQty: (a, b) => b.currentCartUnits - a.currentCartUnits,
   612→    purchases: (a, b) => b.purchasedUnits - a.purchasedUnits,
   613→    revenue: (a, b) => b.grossSalesMinor - a.grossSalesMinor,
   614→    conversion: (a, b) => (b.viewToCartRate ?? -1) - (a.viewToCartRate ?? -1),
   615→  };
   616→  rows.sort(sorters[sort] ?? sorters.views);
   617→
   618→  return { rows: rows.slice((page - 1) * size, page * size), total: rows.length };
   619→}

   624→export type SalesTrend = {
   625→  bucket: string;
   626→  grossSalesMinor: number;
   627→  netProductSalesMinor: number;
   628→  orders: number;
   629→  units: number;
   630→  refundsMinor: number;
   631→};
   632→
   633→export async function salesTrend(
   634→  storeId: string,
   635→  win: InsightsWindow,
   636→  granularity: "day" | "week" | "month"
   637→): Promise<{ trend: SalesTrend[]; granularity: string }> {
   638→  const [orders, refundRows] = await Promise.all([
   639→    paidOrdersFor(storeId, win.from, win.to),
   640→    db.refund.findMany({
   641→      where: { status: "processed", processedAt: { gte: win.from, lt: win.to }, order: { storeId } },
   642→      select: { amountMinor: true, processedAt: true },
   643→    }),
   644→  ]);
   645→
   646→  const bucketKey = (d: Date): string => {
   647→    const kl = new Date(d.getTime() + KL_OFFSET_MS);
   648→    const ymd = kl.toISOString().slice(0, 10);
   649→    if (granularity === "month") return ymd.slice(0, 7);
   650→    if (granularity === "week") {
   651→      const dayIdx = (kl.getUTCDay() + 6) % 7; // Monday = 0
   652→      const monday = new Date(kl.getTime() - dayIdx * 86_400_000);
   653→      return monday.toISOString().slice(0, 10);
   654→    }
   655→    return ymd;
   656→  };
   657→
   658→  const buckets = new Map<string, SalesTrend>();
   659→  for (const o of orders) {
   660→    if (!o.paidAt) continue;
   661→    const k = bucketKey(o.paidAt);
   662→    const slot = buckets.get(k) ?? { bucket: k, grossSalesMinor: 0, netProductSalesMinor: 0, orders: 0, units: 0, refundsMinor: 0 };
   663→    slot.orders += 1;
   664→    slot.grossSalesMinor += o.totalMinor;
   665→    for (const it of o.items) {
   666→      slot.units += it.qty;
   667→      slot.netProductSalesMinor += it.lineSubtotalMinor - it.discountMinor;
   668→    }
   669→    buckets.set(k, slot);
   670→  }
   671→  for (const r of refundRows) {
   672→    const k = bucketKey(r.processedAt ?? new Date());
   673→    const slot = buckets.get(k) ?? { bucket: k, grossSalesMinor: 0, netProductSalesMinor: 0, orders: 0, units: 0, refundsMinor: 0 };
   674→    slot.refundsMinor += r.amountMinor;
   675→    slot.netProductSalesMinor -= r.amountMinor;
   676→    buckets.set(k, slot);
   677→  }
   678→  return {
   679→    trend: [...buckets.entries()].sort((a, b) => a[0].localeCompare(b[0])).map(([, v]) => v),
   680→    granularity,
   681→  };
   682→}

   691→export async function fulfillmentInsights(storeId: string): Promise<FulfillmentInsights> {
   692→  const [unfulfilled, shipped] = await Promise.all([
   693→    db.order.count({ where: { storeId, status: "paid", paidAt: { not: null } } }),
   694→    db.order.findMany({

   707→    avgPaidToShippedHours: durations.length > 0 ? Math.round((durations.reduce((s, h) => s + h, 0) / durations.length) * 10) / 10 : null,
   708→    shippedSample: durations.length,
   709→    note: "Average hours from payment confirmation to seller shipping (most recent 200 shipped orders). Delivery-time metrics are not shown: trustworthy carrier delivery timestamps are not recorded.",
   710→  };

   739→  notes: string[];

   742→
   743→function geoKey(dimension: "country" | "state", o: PaidOrderRow): string {
   744→  if (dimension === "country") return o.shippingCountryCode || "ZZ";
   745→  try {
   746→    const addr = JSON.parse(o.shippingAddressJson || "{}") as { state?: unknown };
   747→    return typeof addr.state === "string" && addr.state.trim() ? addr.state.trim() : "Unspecified";
   748→  } catch {
   749→    return "Unspecified";
   750→  }
   751→}
   752→
   753→export async function marketReport(storeId: string, win: InsightsWindow, dimension: "country" | "state"): Promise<MarketReport> {
   754→  const [cur, prev, refundsCur, refundsPrev] = await Promise.all([
   755→    paidOrdersFor(storeId, win.from, win.to),
   756→    paidOrdersFor(storeId, win.prevFrom, win.prevTo),
   757→    processedRefundsFor(storeId, win.from, win.to),
   758→    processedRefundsFor(storeId, win.prevFrom, win.prevTo),
   759→  ]);
   760→
   761→  type M = { orders: number; units: number; gross: number; discounts: number; shipping: number };
   762→  const agg = new Map<string, M>();
   763→  const aggFor = (k: string): M => {
   764→    let m = agg.get(k);
   765→    if (!m) {
   766→      m = { orders: 0, units: 0, gross: 0, discounts: 0, shipping: 0 };
   767→      agg.set(k, m);
   768→    }
   769→    return m;
   770→  };
   771→  for (const o of cur) {
   772→    const m = aggFor(geoKey(dimension, o));
   773→    m.orders += 1;
   774→    for (const it of o.items) {
   775→      m.units += it.qty;
   776→      m.gross += it.lineSubtotalMinor;
   777→      m.discounts += it.discountMinor;
   778→      m.shipping += it.shippingMinor;
   779→    }
   780→  }
   781→  const prevAgg = new Map<string, number>(); // prev paid orders per key
   782→  for (const o of prev) {
   783→    const k = geoKey(dimension, o);
   784→    prevAgg.set(k, (prevAgg.get(k) ?? 0) + 1);
   785→  }
   786→
   787→  // Refunds are order-level (no per-geo split in the authoritative model) —
   788→  // attributed to the order's own geography via a second pass.
   789→  const refundRowsCur = await db.refund.findMany({
   790→    where: { status: "processed", processedAt: { gte: win.from, lt: win.to }, order: { storeId } },
   791→    select: { amountMinor: true, order: { select: { shippingCountryCode: true, shippingAddressJson: true } } },
   792→  });
   793→  const refundsByGeo = new Map<string, number>();
   794→  for (const r of refundRowsCur) {
   795→    const k = dimension === "country" ? r.order.shippingCountryCode || "ZZ" : (() => {
   796→      try {
   797→        const addr = JSON.parse(r.order.shippingAddressJson || "{}") as { state?: unknown };
   798→        return typeof addr.state === "string" && addr.state.trim() ? addr.state.trim() : "Unspecified";
   799→      } catch { return "Unspecified"; }
   800→    })();
   801→    refundsByGeo.set(k, (refundsByGeo.get(k) ?? 0) + r.amountMinor);
   802→  }
   803→
   804→  const totalNet = [...agg.values()].reduce((s, m) => s + m.gross - m.discounts, 0) - refundsCur.amountMinor;
   805→  const totalOrders = cur.length;
   806→
   807→  const sourceByCode = new Map<string, Set<string>>();
   808→  for (const o of cur) {
   809→    const k = geoKey(dimension, o);
   810→    const set = sourceByCode.get(k) ?? new Set<string>();
   811→    if (dimension === "country") set.add(o.shippingCountrySource || "");
   812→    sourceByCode.set(k, set);
   813→  }
   814→
   815→  const rows: MarketRow[] = [...agg.entries()].map(([code, m]) => {
   816→    const net = m.gross - m.discounts - (refundsByGeo.get(code) ?? 0);
   817→    const growth = pct(m.orders, prevAgg.get(code) ?? 0);
   818→    const sources = [...(sourceByCode.get(code) ?? new Set())].filter(Boolean);
   819→    return {
   820→      code,
   821→      label: dimension === "country" ? COUNTRY_LABELS[code] ?? code : code,
   822→      paidOrders: m.orders,
   823→      paidOrdersPrev: prevAgg.get(code) ?? 0,
   824→      unitsSold: m.units,
   825→      grossProductSalesMinor: m.gross,
   826→      discountsMinor: m.discounts,
   827→      refundsMinor: refundsByGeo.get(code) ?? 0,
   828→      netProductSalesMinor: net,
   829→      shippingChargedMinor: m.shipping,
   830→      aovMinor: m.orders > 0 ? Math.round((m.gross + m.shipping - m.discounts) / m.orders) : null,
   831→      shareNetPct: totalNet > 0 ? Math.round((net / totalNet) * 1000) / 10 : null,
   832→      shareOrdersPct: totalOrders > 0 ? Math.round((m.orders / totalOrders) * 1000) / 10 : null,
   833→      growthPct: growth.pct,
   834→      isNew: growth.isNew,
   835→      source: sources.length === 1 ? sources[0] : sources.length > 1 ? "mixed" : "",
   836→    };
   837→  });
   838→  rows.sort((a, b) => b.netProductSalesMinor - a.netProductSalesMinor);
   839→
   840→  const movers = { risers: [] as Array<{ code: string; label: string; growthPct: number }>, fallers: [] as Array<{ code: string; label: string; growthPct: number }> };
   841→  const MIN_SAMPLE = 5; // insufficient-sample guard: no market claims on tiny data
   842→  for (const r of rows) {
   843→    if (r.growthPct == null || r.paidOrders < MIN_SAMPLE) continue;
   844→    if (r.growthPct >= 20) movers.risers.push({ code: r.code, label: r.label, growthPct: r.growthPct });
   845→    if (r.growthPct <= -20) movers.fallers.push({ code: r.code, label: r.label, growthPct: r.growthPct });
   846→  }
   847→
   848→  const notes = [
   849→    dimension === "country"
   850→      ? "Geographic dimension = destination country recorded at checkout (shipping snapshot). Historical orders before this feature are labelled 'backfill' (derived from the frozen Malaysian postcode + state) — see metric definitions."
   851→      : "Geographic dimension = destination state recorded in the order's frozen shipping address snapshot at checkout time.",
   852→    "Billing country is not recorded anywhere in the current data model — that dimension is unavailable (not silently substituted).",
   853→    "Growth compares paid-order counts against the equivalent previous period. Markets with fewer than 5 paid orders are excluded from mover highlights (insufficient sample) and growth shows as 'new' when the previous period had none.",
   854→    refundsPrev.count > 0 || refundsCur.count > 0
   855→      ? "Refunds are reported on their own processed date (order-level) and attributed to the order's recorded geography; they can relate to orders paid in an earlier period."
   856→      : "No processed refunds in this window.",
   857→    "All amounts are in MYR minor units (sen). The platform is single-currency — no conversion is applied anywhere.",
   858→  ];
   859→
   860→  return {
   861→    dimension,
   862→    rows,
   863→    totals: { paidOrders: totalOrders, netProductSalesMinor: totalNet },
   864→    notes,
   865→    movers,
   866→  };
   867→}

   876→  discountsMinor: number;

   885→export async function marketProducts(

   887→  win: InsightsWindow,
   888→  dimension: "country" | "state",
   889→  code: string
   890→): Promise<{ rows: MarketProductRow[]; label: string; notes: string[] }> {
   891→  const orders = await paidOrdersFor(storeId, win.from, win.to);

   894→  const events = await eventsFor(storeId, win.from, win.to);
   895→  const viewsByProduct = new Map<string, number>();
   896→  const addsByProduct = new Map<string, number>();
   897→  for (const e of events.rows) {
   898→    if (!e.productId) continue;
   899→    if (e.eventType === "product_view") viewsByProduct.set(e.productId, (viewsByProduct.get(e.productId) ?? 0) + 1);
   900→    if (e.eventType === "add_to_cart") addsByProduct.set(e.productId, (addsByProduct.get(e.productId) ?? 0) + (e.qty ?? 0));
   901→  }
   902→
   903→  type P = { units: number; gross: number; discounts: number; refunded: number; orders: Set<string> };
   904→  const perProduct = new Map<string, P>();
   905→  for (const o of filtered) {
   906→    for (const it of o.items) {
   907→      if (!it.productId) continue;
   908→      const p = perProduct.get(it.productId) ?? { units: 0, gross: 0, discounts: 0, refunded: 0, orders: new Set<string>() };
   909→      p.units += it.qty;
   910→      p.gross += it.lineSubtotalMinor;
   911→      p.discounts += it.discountMinor;
   912→      p.refunded += it.refundedMinor;
   913→      p.orders.add(o.id);
   914→      perProduct.set(it.productId, p);
   915→    }
   916→  }
   917→
   918→  const ids = [...perProduct.keys()];
   919→  const products = ids.length
   920→    ? await db.product.findMany({ where: { id: { in: ids }, storeId }, select: { id: true, title: true, slug: true, images: { orderBy: { sortOrder: "asc" }, take: 1, select: { url: true } } } })
   921→    : [];
   922→  const byId = new Map(products.map((p) => [p.id, p]));
   923→
   924→  const rows: MarketProductRow[] = ids
   925→    .map((id) => {
   926→      const p = perProduct.get(id)!;
   927→      const prod = byId.get(id);
   928→      return {
   929→        productId: id,
   930→        title: prod?.title ?? "(deleted product)",
   931→        slug: prod?.slug ?? "",
   932→        imageUrl: prod?.images[0]?.url ?? null,
   933→        unitsSold: p.units,
   934→        grossProductSalesMinor: p.gross,
   935→        discountsMinor: p.discounts,
   936→        refundedMinor: p.refunded,
   937→        netProductSalesMinor: p.gross - p.discounts - p.refunded,
   938→        paidOrders: p.orders.size,
   939→        avgSellingPriceMinor: p.units > 0 ? Math.round(p.gross / p.units) : null,
   940→        views: viewsByProduct.get(id) ?? 0,
   941→        addUnits: addsByProduct.get(id) ?? 0,
   942→      };
   943→    })
   944→    .filter((r) => r.title !== "(deleted product)")
   945→    .sort((a, b) => b.netProductSalesMinor - a.netProductSalesMinor || b.unitsSold - a.unitsSold);
   946→
   947→  return {
   948→    rows,
   949→    label: dimension === "country" ? COUNTRY_LABELS[code] ?? code : code,
   950→    notes: [
   951→      "Product views and cart-add units are STORE-WIDE for the selected window — visitor geography is not tracked, so per-country view/cart attribution does not exist and is never invented (no per-country conversion rates are shown for this reason).",
   952→      "Sales columns are attributed strictly by the order's recorded destination geography.",
   953→    ],
   954→  };
   955→}

   971→export type Observation = { kind: string; text: string };
   972→
   973→/** Evidence-gated observations — only emitted when the measurements support them. */
   974→export function buildObservations(input: {
   975→  traffic: TrafficSummary;
   976→  funnel: Funnel;
   977→  snapshot: CartSnapshot;
   978→  sales: { paidOrders: number; paidOrdersPrev: number; unitsSold: number; grossSalesMinor: number };
   979→  addUnits: number;
   980→  addUnitsPrev: number;
   981→}): Observation[] {
   982→  const out: Observation[] = [];
   983→  const { traffic, funnel, snapshot, sales } = input;
   984→
   985→  // Store-level view→cart signal (per-product rates live in productPerformance)
   986→  const storeViews = traffic.storeViews;
   987→  const addSessions = funnel.stages.find((s) => s.key === "add_to_cart")?.count ?? 0;
   988→  const viewSessions = funnel.stages.find((s) => s.key === "product_view")?.count ?? 0;
   989→  if (storeViews >= 30 && viewSessions > 0 && addSessions / viewSessions < 0.05) {
   990→    out.push({
   991→      kind: "low_cart_interest",
   992→      text: `Your products were viewed in ${viewSessions} session${viewSessions === 1 ? "" : "s"} this period, but very few of those sessions added anything to the cart. Product price, photos or description may be worth reviewing.`,
   993→    });
   994→  }
   995→
   996→  // Cart interest not converting
   997→  if (snapshot.totalUnits >= 5 && sales.paidOrders === 0) {
   998→    out.push({
   999→      kind: "cart_not_converting",
  1000→      text: `${snapshot.totalUnits} unit${snapshot.totalUnits === 1 ? "" : "s"} of your products are currently sitting in ${snapshot.activeCarts} active shopper cart${snapshot.activeCarts === 1 ? "" : "s"}, with no paid orders in this period.`,
  1001→    });
  1002→  }
  1003→
  1004→  // Cart adds trend vs previous period
  1005→  if (input.addUnits >= 10 && input.addUnitsPrev > 0) {
  1006→    const delta = (input.addUnits - input.addUnitsPrev) / input.addUnitsPrev;
  1007→    if (delta >= 0.2) {
  1008→      out.push({ kind: "adds_up", text: `Cart additions increased ${Math.round(delta * 100)}% compared with the previous period (${input.addUnitsPrev} → ${input.addUnits} units added).` });
  1009→    } else if (delta <= -0.2) {
  1010→      out.push({ kind: "adds_down", text: `Cart additions decreased ${Math.round(-delta * 100)}% compared with the previous period (${input.addUnitsPrev} → ${input.addUnits} units added).` });
  1011→    }
  1012→  }
  1013→
  1014→  // Top cart quantity
  1015→  const top = snapshot.byProduct[0];
  1016→  if (top && top.units >= 3) {
  1017→    out.push({ kind: "top_in_carts", text: `"${top.title}" currently has the largest quantity sitting in shopper carts (${top.units} unit${top.units === 1 ? "" : "s"} across ${top.carts} cart${top.carts === 1 ? "" : "s"}).` });
  1018→  }
  1019→
  1020→  // Sales trend
  1021→  if (sales.paidOrders >= 3 && sales.paidOrdersPrev > 0) {
  1022→    const delta = (sales.paidOrders - sales.paidOrdersPrev) / sales.paidOrdersPrev;
  1023→    if (delta >= 0.2) out.push({ kind: "sales_up", text: `Paid orders increased ${Math.round(delta * 100)}% versus the previous period (${sales.paidOrdersPrev} → ${sales.paidOrders}).` });
  1024→    if (delta <= -0.2) out.push({ kind: "sales_down", text: `Paid orders decreased ${Math.round(-delta * 100)}% versus the previous period (${sales.paidOrdersPrev} → ${sales.paidOrders}).` });
  1025→  }
  1026→  return out;
  1027→}
