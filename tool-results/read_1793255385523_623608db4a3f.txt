   671→function Avatar({ profile, size = "md" }: { profile: ProfileT | null; size?: "sm" | "md" | "lg" }) {
   672→  // Task 33: the banner avatar shrinks with the mobile banner (−40% height
   673→  // goal) and returns to the full 64px on desktop.
   674→  const px = size === "lg" ? "h-11 w-11 text-base md:h-16 md:w-16 md:text-xl" : size === "md" ? "h-10 w-10 text-sm" : "h-8 w-8 text-xs";
   675→  const gradKey = profile?.avatarUrl?.startsWith("grad:") ? profile.avatarUrl : null;
   676→  const gradClass =
   677→    gradKey === "grad:amber"
   678→      ? "from-amber-400 to-orange-500"
   679→      : gradKey === "grad:rose"
   680→        ? "from-rose-400 to-pink-600"
   681→        : gradKey === "grad:violet"
   682→          ? "from-violet-400 to-purple-600"
   683→          : gradKey === "grad:emerald"
   684→            ? "from-emerald-400 to-teal-600"
   685→            : gradKey === "grad:slate"
   686→              ? "from-slate-400 to-slate-600"
   687→              : "from-teal-500 to-teal-800";
   688→  if (profile?.avatarUrl && !gradKey && /^https?:\/\/|^\//.test(profile.avatarUrl)) {
   689→    return <img src={profile.avatarUrl} alt={`${profile.fullName} avatar`} className={cn(px, "shrink-0 rounded-full object-cover ring-2 ring-white/70")} />;
   690→  }
   691→  const initial = (profile?.fullName ?? "?").trim().slice(0, 1).toUpperCase();
   692→  return (
   693→    <span className={cn(px, "flex shrink-0 items-center justify-center rounded-full bg-gradient-to-br font-extrabold text-white ring-2 ring-white/70", gradClass)} aria-hidden="true">
   694→      {initial}
   695→    </span>
   696→  );
   697→}
