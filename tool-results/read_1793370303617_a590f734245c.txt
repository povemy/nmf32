   220→// data hook — profile + wallet + purchase counts + like/viewed counters
   221→// (Task 34: backed by the module-level Me-hub cache in @/lib/me-hub-cache —

   225→
   226→function useMeData(enabled: boolean) {
   227→  const me = useApp((s) => s.me);
   228→  const shopperId = me.loaded ? (me.shopper?.id ?? null) : null;
   229→  const snap = useSyncExternalStore(subscribeMeHub, getMeHubSnapshot, getMeHubServerSnapshot);
   230→
   231→  useEffect(() => {
   232→    if (!enabled || !shopperId) return;
   233→    // 1) paint whatever the persisted snapshot has (same frame, zero network)
   234→    hydrateMeHub(shopperId);
   235→    // 2) revalidate in the background (skipped when a prefetch <45s old is
   236→    //    already fresh) — data-bearing updates never flip a loading flag
   237→    void refreshMeHub(shopperId).catch(() => {});
   238→    // 3) address CRUD (here or during checkout) must reflect instantly
   239→    const sync = () => void refreshMeHub(shopperId, { force: true }).catch(() => {});
   240→    window.addEventListener("nomofee:addresses-changed", sync);
   241→    return () => window.removeEventListener("nomofee:addresses-changed", sync);
   242→  }, [enabled, shopperId]);
   243→
   244→  const refresh = useCallback(async () => {
   245→    if (!shopperId) return;
   246→    await refreshMeHub(shopperId, { force: true }).catch(() => {});
   247→  }, [shopperId]);
   248→
   249→  return {
   250→    profile: snap.profile,
   251→    wallet: snap.wallet,
   252→    counts: snap.counts,
   253→    likeCount: snap.likeCount,
   254→    viewedCount: snap.viewedCount,
   255→    // skeleton only when NOTHING has ever painted (no prefetch, no cache)
   256→    loading: snap.fetchedAt === null,
   257→    refresh,
   258→  };
   259→}

   688→function Avatar({ profile, size = "md" }: { profile: ProfileT | null; size?: "sm" | "md" | "lg" }) {
   689→  // Task 33: the banner avatar shrinks with the mobile banner (−40% height
   690→  // goal) and returns to the full 64px on desktop.
   691→  const px = size === "lg" ? "h-11 w-11 text-base md:h-16 md:w-16 md:text-xl" : size === "md" ? "h-10 w-10 text-sm" : "h-8 w-8 text-xs";
   692→  const gradKey = profile?.avatarUrl?.startsWith("grad:") ? profile.avatarUrl : null;
   693→  const gradClass =
   694→    gradKey === "grad:amber"
   695→      ? "from-amber-400 to-orange-500"
   696→      : gradKey === "grad:rose"
   697→        ? "from-rose-400 to-pink-600"
   698→        : gradKey === "grad:violet"
   699→          ? "from-violet-400 to-purple-600"
   700→          : gradKey === "grad:emerald"
   701→            ? "from-emerald-400 to-teal-600"
   702→            : gradKey === "grad:slate"
   703→              ? "from-slate-400 to-slate-600"
   704→              : "from-teal-500 to-teal-800";
   705→  if (profile?.avatarUrl && !gradKey && /^https?:\/\/|^\//.test(profile.avatarUrl)) {
   706→    return <img src={profile.avatarUrl} alt={`${profile.fullName} avatar`} className={cn(px, "shrink-0 rounded-full object-cover ring-2 ring-white")} />;
   707→  }
   708→  const initial = (profile?.fullName ?? "?").trim().slice(0, 1).toUpperCase();
   709→  return (
   710→    <span className={cn(px, "flex shrink-0 items-center justify-center rounded-full bg-gradient-to-br font-extrabold text-white ring-2 ring-white", gradClass)} aria-hidden="true">
   711→      {initial}
   712→    </span>
   713→  );
   714→}
