    19→CREATE TABLE plans (

    33→CREATE TABLE commission_settings (              -- SINGLETON (id always 1)
    34→  id SMALLINT PRIMARY KEY DEFAULT 1 CHECK (id = 1),
    35→  default_percent NUMERIC(5,2) NOT NULL DEFAULT 8.00,
    36→  applies_to_shipping BOOLEAN NOT NULL DEFAULT FALSE,
    37→  base_price_rule TEXT NOT NULL DEFAULT 'paid_price'

    40→  min_commission_minor BIGINT NOT NULL DEFAULT 0,
    41→  payout_hold_days INT NOT NULL DEFAULT 7,
    42→  hold_release_rule TEXT NOT NULL DEFAULT 'days_after_paid'
    43→      CHECK (hold_release_rule IN ('days_after_paid','days_after_delivered')),
    44→  grace_days_after_delivered INT NOT NULL DEFAULT 2,
    45→  updated_by UUID,                              -- admin_users.id (FK later)
    46→  created_at timestamptz NOT NULL DEFAULT now(),
    47→  updated_at timestamptz NOT NULL DEFAULT now()
    48→);
    49→
    50→CREATE TABLE settings (
    51→  key TEXT PRIMARY KEY,

    55→  description TEXT,
    56→  updated_by UUID,
    57→  updated_at timestamptz NOT NULL DEFAULT now()
