    53→const EMERALD_STATUSES = new Set([
    54→  "active", "live", "paid", "approved", "verified", "completed", "delivered", "success",
    55→  "succeeded", "sent", "processed", "resolved", "issued", "ok", "done", "match",
    56→]);
    57→const AMBER_STATUSES = new Set([
    58→  "pending", "pending_review", "processing", "verifying", "open", "queued", "requires_payment",
    59→  "onboarding", "under_review", "refunding", "partially_refunded", "payment_pending", "requested",
    60→  "running", "partial",
    61→]);
    62→const RED_STATUSES = new Set([
    63→  "suspended", "rejected", "banned", "failed", "dead", "terminated", "urgent", "high",
    64→  "mismatch", "drift",
    65→]);
    66→const TEAL_STATUSES = new Set(["shipped", "medium"]);

    68→export function statusTone(status?: string | null): Tone {
    69→  const s = (status ?? "").toLowerCase();
    70→  if (EMERALD_STATUSES.has(s)) return "emerald";
    71→  if (AMBER_STATUSES.has(s)) return "amber";
    72→  if (RED_STATUSES.has(s)) return "red";
    73→  if (TEAL_STATUSES.has(s)) return "teal";
    74→  return "gray";
    75→}

   133→export function fmtDT(iso?: string | Date | null): string {
   134→  if (!iso) return "—";
   135→  const d = typeof iso === "string" ? new Date(iso) : iso;
   136→  if (Number.isNaN(d.getTime())) return "—";
   137→  return d.toLocaleString("en-MY", {
   138→    day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit", hour12: false,
   139→  });
   140→}
   141→
   142→export function fmtDate(iso?: string | Date | null): string {
   143→  if (!iso) return "—";
   144→  const d = typeof iso === "string" ? new Date(iso) : iso;
   145→  if (Number.isNaN(d.getTime())) return "—";
   146→  return d.toLocaleDateString("en-MY", { day: "2-digit", month: "short", year: "numeric" });
   147→}
   148→
   149→export function timeAgo(iso?: string | Date | null): string {
   150→  if (!iso) return "—";
   151→  const d = typeof iso === "string" ? new Date(iso) : iso;
   152→  if (Number.isNaN(d.getTime())) return "—";
   153→  const secs = Math.floor((Date.now() - d.getTime()) / 1000);
   154→  if (secs < 60) return "just now";
   155→  if (secs < 3600) return `${Math.floor(secs / 60)}m ago`;
   156→  if (secs < 86400) return `${Math.floor(secs / 3600)}h ago`;
   157→  if (secs < 86400 * 30) return `${Math.floor(secs / 86400)}d ago`;
   158→  return fmtDate(d);
   159→}

   228→}: {

   237→  const iconColor =
   238→    tone === "emerald"

   252→      onKeyDown={(e) => {
   253→        if (clickable && (e.key === "Enter" || e.key === " ")) {

   360→export function Pager({
   361→  page,
   362→  size,
   363→  total,
   364→  onPrev,
   365→  onNext,
   366→}: {

   636→/* -------------------------------------------------------------- breakdown */
   637→
   638→const BREAKDOWN_MONEY_KEYS = new Set([
   639→  "base", "percent_part", "fixed_fee_allocated", "discount_allocated", "shipping_allocated",
   640→]);
   641→
   642→function breakdownValue(key: string, v: unknown): string | null {
   643→  if (v === null || v === undefined) return null;
   644→  if (key === "flags") {
   645→    const arr = Array.isArray(v) ? v.map(String) : [];
   646→    return arr.length ? arr.join(", ") : null;
   647→  }
   648→  if (BREAKDOWN_MONEY_KEYS.has(key)) return formatMinor(Number(v));
   649→  if (typeof v === "number") return key === "percent" ? `${v}%` : String(v);
   650→  if (typeof v === "string") return v;
   651→  return JSON.stringify(v);
   652→}

   654→/** Renders a commission breakdownJson payload as key:value chips. */
   655→export function BreakdownChips({ json, className }: { json?: string | null; className?: string }) {
   656→  const data = parseJsonSafe<Record<string, unknown>>(json, {});
   657→  const entries = Object.entries(data).filter(([k, v]) => breakdownValue(k, v) !== null);
   658→  if (entries.length === 0) return <span className="text-xs text-muted-foreground">No breakdown snapshot</span>;
   659→  return (
   660→    <div className={cn("flex flex-wrap gap-1.5", className)}>
   661→      {entries.map(([k, v]) => {
   662→        const val = breakdownValue(k, v) as string;
   663→        const highlight = k === "rule_source";
   664→        return (
   665→          <span
   666→            key={k}
   667→            className={cn(
   668→              "inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[11px]",
   669→              highlight ? TONE_CLASS.teal : "bg-muted/60 text-muted-foreground"
   670→            )}
   671→          >
   672→            <span className="font-medium capitalize">{k.replace(/_/g, " ")}</span>
   673→            <span className="tabular-nums">{val}</span>
   674→          </span>
   675→        );
   676→      })}
   677→    </div>
   678→  );

   681→/* -------------------------------------------------------------- diff view */
   682→
   683→function flatObj(v: unknown): Record<string, unknown> {
   684→  return v && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : {};
   685→}
   686→
   687→function diffVal(key: string, v: unknown): string {
   688→  if (v === null || v === undefined || v === "") return "—";
   689→  if (typeof v === "object") return JSON.stringify(v);
   690→  if (key.endsWith("Minor")) return formatMinor(Number(v));
   691→  if (typeof v === "boolean") return v ? "true" : "false";
   692→  return String(v);
   693→}

   695→/** Field-by-field before/after viewer for audit log rows. */
   696→export function DiffViewer({ beforeJson, afterJson }: { beforeJson?: string | null; afterJson?: string | null }) {
   697→  const before = flatObj(parseJsonSafe<unknown>(beforeJson, {}));
   698→  const after = flatObj(parseJsonSafe<unknown>(afterJson, {}));
   699→  const keys = Array.from(new Set([...Object.keys(before), ...Object.keys(after)])).sort();
   700→  if (keys.length === 0) {
   701→    return <p className="text-xs text-muted-foreground">No structured before/after snapshot recorded for this entry.</p>;
   702→  }
   703→  return (
   704→    <div className="overflow-hidden rounded-lg border">
   705→      <Table>
   706→        <TableHeader>
   707→          <TableRow>
   708→            <TableHead className="w-40">Field</TableHead>
   709→            <TableHead>Before</TableHead>
   710→            <TableHead>After</TableHead>
   711→          </TableRow>
   712→        </TableHeader>
   713→        <TableBody>
   714→          {keys.map((k) => {
   715→            const b = k in before ? diffVal(k, before[k]) : null;
   716→            const a = k in after ? diffVal(k, after[k]) : null;
   717→            const changed = b !== a;
   718→            const added = b === null;
   719→            const removed = a === null;
   720→            return (
   721→              <TableRow key={k} className={cn(changed && !removed && "bg-amber-50/60 dark:bg-amber-950/20")}>
   722→                <TableCell className="font-mono text-[11px] font-medium">{k}</TableCell>
   723→                <TableCell className={cn("text-xs tabular-nums", removed && "text-red-600 line-through dark:text-red-400")}>
   724→                  {b ?? "—"}
   725→                </TableCell>
   726→                <TableCell className={cn("text-xs tabular-nums", added && "font-medium text-emerald-700 dark:text-emerald-400")}>
   727→                  {a ?? "—"}
   728→                </TableCell>
   729→              </TableRow>
   730→            );
   731→          })}
   732→        </TableBody>
   733→      </Table>
   734→    </div>
   735→  );
   736→}
