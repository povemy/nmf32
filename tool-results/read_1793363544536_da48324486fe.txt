   152→/** FIN-3b (3b.3) — payout-status labels shared with the seller wallet
   153→ * (src/components/seller/wallet.tsx) so admin and seller render the SAME
   154→ * wording for a payout state; the admin view stays richer (method,
   155→ * referenceNo, failReason as before). Display-only: raw status strings still

   158→const PAYOUT_STATUS_LABELS: Record<string, string> = {
   159→  requested: "Awaiting approval",
   160→  processing: "Processing",
   161→  paid: "Paid",
   162→  failed: "Failed",
   163→  rejected: "Rejected",
   164→  returned: "Returned",
   165→  reversed: "Reversed",
   166→};
   167→
   168→function PayoutStatusChip({ status, className }: { status: string; className?: string }) {
   169→  const s = (status ?? "").toLowerCase();
   170→  return (
   171→    <Badge variant="outline" className={cn(TONE_CLASS[statusTone(s)], className)}>
   172→      {PAYOUT_STATUS_LABELS[s] ?? s.replace(/_/g, " ")}
   173→    </Badge>
   174→  );
   175→}

   179→export function PayoutsView({ nav }: { nav: AdminNav }) {
   180→  const { data, error, loading, refetch } = useApi<PayoutBatch[]>("/api/admin/payout-batches");
   181→  // Due-now panel: same ledger-derived truth the dashboard card shows, so the
   182→  // two always tally (this used to be the missing piece — the card navigated
   183→  // straight to batch HISTORY, which only shows payouts already run).
   184→  const due = useApi<PayoutsDueBody>("/api/admin/payouts/due");
   185→  const requests = useApi<RequestsBody>("/api/admin/payouts/requests");
   186→  const rows = useMemo(() => data ?? [], [data]);
   187→  const dueRows = useMemo(() => due.data?.due ?? [], [due.data]);
   188→  const openRequests = useMemo(() => requests.data?.open ?? [], [requests.data]);
   189→  const recentRequests = useMemo(() => requests.data?.recent ?? [], [requests.data]);
   190→

   192→  const [busyId, setBusyId] = useState<string | null>(null);
   193→
   194→  const refreshAllPanels = useCallback(() => {
   195→    void requests.refetch();
   196→    void due.refetch();

   199→  const approveRequest = useCallback(
   200→    async (r: RequestRow) => {

   203→        await apiSend(`/api/admin/payouts/${r.id}/approve`, "POST", {});
   204→        toast.success(`${r.referenceNo} approved`, {
   205→          description: `${formatMinor(r.amountMinor)} moved to payout clearing (T3) — mark it paid once the transfer lands.`,
   206→        });

   209→        toast.error("Not approved", { description: e instanceof Error ? e.message : String(e) });

   211→        setBusyId(null);
   212→      }
   213→    },
   214→    [refreshAllPanels]
   215→  );
   216→
   217→  const rejectRequest = useCallback(

   222→        toast.success(`${r.referenceNo} rejected`, { description: "The seller's available balance is untouched (no ledger movement)." });

   224→        refreshAllPanels();
   225→      } catch (e) {
   226→        toast.error("Not rejected", { description: e instanceof Error ? e.message : String(e) });
   227→      } finally {
   228→        setBusyId(null);
   229→      }
   230→    },

   238→  const [running, setRunning] = useState(false);
   239→
   240→  const loadPreview = useCallback(async () => {
   241→    setPreviewLoading(true);
   242→    setPreviewError(null);
   243→    try {
   244→      const body = await apiSend<DryRunBody>("/api/admin/payout-batches", "POST", { dryRun: true });
   245→      setPreview(body);
   246→    } catch (e) {
   247→      setPreview(null);
   248→      setPreviewError(e instanceof Error ? e.message : "Preview failed");
   249→    } finally {
   250→      setPreviewLoading(false);
   251→    }
   252→  }, []);
   253→
   254→  useEffect(() => {
   255→    if (open) void loadPreview();
   256→  }, [open, loadPreview]);
   257→
   258→  const runBatch = async () => {
   259→    setRunning(true);
   260→    try {
   261→      const body = await apiSend<{ batch?: PayoutBatch; payouts?: PayoutRow[] }>("/api/admin/payout-batches", "POST", { dryRun: false });

   263→      toast.success("Payout batch created", {
   264→        description: `${count} payout${count === 1 ? "" : "s"} queued · ${formatMinor(body?.batch?.totalMinor ?? 0)} — ledger T3 posted per wallet.`,
   265→      });
   266→      setOpen(false);
   267→      void refetch();
   268→      void due.refetch(); // sweep happened → the due-now list shrinks
   269→    } catch (e) {
   270→      const msg = e instanceof Error ? e.message : String(e);
   271→      toast.error("Batch not created", { description: msg });
   272→      // A 409 NO_ELIGIBLE_WALLETS leaves the dialog open with a fresh preview.
   273→      void loadPreview();
   274→    } finally {
   275→      setRunning(false);
   276→    }
   277→  };

   279→  return (
   280→    <div className="space-y-4">
   281→      <PageHeader

   286→            <PlusCircle className="h-4 w-4" /> New Payout Batch
   287→          </Button>
   288→        }
   289→      />
   290→
   291→      {/* Due now — mirrors the dashboard "Payouts due" card (same ledger-derived
   292→          truth, same minimum setting). This closes the gap that made the two
   293→          disagree: this page used to show only batch HISTORY. */}
   294→      <Card className="py-0">
   295→        <CardHeader className="px-4 pt-4 sm:px-6 sm:pt-5">
   296→          <CardTitle className="flex flex-wrap items-center gap-x-3 gap-y-1 text-base">
   297→            <Landmark className="h-4 w-4 text-teal-600 dark:text-teal-400" aria-hidden />
   298→            Due now
   299→            {due.data ? (
   300→              <>
   301→                <span className="text-sm font-bold text-teal-700 dark:text-teal-300">
   302→                  <Money minor={due.data.dueTotalMinor} bold />
   303→                </span>
   304→                <Badge variant="outline" className="font-normal text-muted-foreground">
   305→                  across {due.data.dueCount} seller{due.data.dueCount === 1 ? "" : "s"}
   306→                </Badge>
   307→              </>
   308→            ) : null}
   309→            <span className="ml-auto text-[11px] font-normal text-muted-foreground">
   310→              min payout <Money minor={due.data?.minMinor ?? 1000} muted /> · {due.data?.mode ?? "manual_csv"}
   311→            </span>
   312→          </CardTitle>
   313→          <CardDescription>
   314→            Wallet available balances derived from the ledger — exactly what “Run payout batch” will sweep.
   315→          </CardDescription>
   316→        </CardHeader>
   317→        <CardContent className="px-0 pb-2">
   318→          {due.error ? (
   319→            <div className="px-4 sm:px-6">
   320→              <ErrorState message={due.error} onRetry={() => void due.refetch()} />
   321→            </div>
   322→          ) : due.loading ? (
   323→            <TableSkeleton rows={2} cols={4} />
   324→          ) : dueRows.length === 0 ? (
   325→            <div className="px-4 sm:px-6">
   326→              <EmptyState
   327→                title="No payouts due right now"
   328→                hint={
   329→                  due.data
   330→                    ? `${due.data.pendingTotalMinor > 0 ? `${formatMinor(due.data.pendingTotalMinor)} is still in the holding period` : "No funds available"}${due.data.belowMinCount > 0 ? ` · ${formatMinor(due.data.belowMinTotalMinor)} across ${due.data.belowMinCount} wallet${due.data.belowMinCount === 1 ? "" : "s"} below the minimum` : ""}.`
   331→                    : undefined
   332→                }
   333→              />
   334→            </div>
   335→          ) : (
   336→            <Table>
   337→              <TableHeader>
   338→                <TableRow>
   339→                  <TableHead>Seller</TableHead>
   340→                  <TableHead className="text-right">Available</TableHead>
   341→                  <TableHead className="text-right">Requested</TableHead>

   345→              </TableHeader>
   346→              <TableBody>
   347→                {dueRows.map((d) => (

   349→                    <TableCell>
   350→                      <p className="truncate text-sm font-medium">{d.storeName ?? d.sellerName}</p>
   351→                      <p className="truncate text-xs text-muted-foreground">{d.sellerName}</p>

   353→                    <TableCell className="text-right">
   354→                      <Money minor={d.availableMinor} bold />
   355→                    </TableCell>

   359→                          <Money minor={d.requestedMinor} />
   360→                          <p className="text-[10px] text-muted-foreground">seller request</p>

   364→                      )}
   365→                    </TableCell>
   366→                    <TableCell className="hidden md:table-cell">
   367→                      {d.bank ? (
   368→                        <span className="text-xs">
   369→                          {d.bank.bankName} · {d.bank.accountNumberMasked}
   370→                          {d.bank.verified ? (
   371→                            <ShieldCheck className="ml-1 inline h-3 w-3 text-emerald-600" aria-label="Verified" />

   383→                          {formatMinor(d.lastPayout.amountMinor)} · {fmtDT(d.lastPayout.paidAt)}

   386→                        "never"
   387→                      )}
   388→                    </TableCell>
   389→                  </TableRow>
   390→                ))}
   391→              </TableBody>
   392→            </Table>
   393→          )}
   394→          {due.data && (due.data.pendingTotalMinor > 0 || due.data.belowMinCount > 0) ? (
   395→            <p className="px-4 pb-3 pt-2 text-xs text-muted-foreground sm:px-6">
   396→              Where the rest is: {formatMinor(due.data.pendingTotalMinor)} pending in the holding period (released after the hold)

   399→                : " · no wallets below the minimum"}
   400→              .
   401→            </p>
   402→          ) : null}
   403→        </CardContent>
   404→      </Card>
   405→
   406→      {/* Seller payout requests — seller-initiated withdrawals awaiting action.
   407→          Approve posts T3 (available → clearing) exactly like a batch run;

   409→      <Card className="py-0">
   410→        <CardHeader className="px-4 pt-4 sm:px-6 sm:pt-5">
   411→          <CardTitle className="flex flex-wrap items-center gap-x-3 gap-y-1 text-base">

   423→        </CardHeader>
   424→        <CardContent className="px-0 pb-2">

   427→              <ErrorState message={requests.error} onRetry={() => void requests.refetch()} />

   875→function PayoutExpand({
   876→  payout,
   877→  onMarkPaid,
   878→  onMarkFailed,
   879→}: {
   880→  payout: PayoutRow;
   881→  onMarkPaid: () => void;
   882→  onMarkFailed: () => void;
   883→}) {
   884→  const st = parseJsonSafe<Statement>(payout.statementJson, {});
   885→  const lines = st.lines ?? [];
   886→  // Reconciliation audit: statement lines must sum EXACTLY to the net payout.
   887→  // Ledger-truth statements (reconciled: true) always do; legacy demo snapshots
   888→  // with decorated numbers fail this check and are flagged honestly.
   889→  const linesSum = lines.reduce((s, l) => s + l.amountMinor, 0);
   890→  const reconciles = lines.length > 0 && linesSum === payout.amountMinor;
   891→  return (
   892→    <div className="grid grid-cols-1 gap-4 p-3 lg:grid-cols-3">
   893→      <div className="lg:col-span-2">
   894→        <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   895→          Settlement statement{st.sellerName ? ` — ${st.sellerName}` : ""}
   896→        </p>
   897→        {lines.length === 0 ? (
   898→          <p className="text-xs text-muted-foreground">No statement lines recorded for this payout.</p>
   899→        ) : (
   900→          <div className="overflow-hidden rounded-lg border bg-card">
   901→            <div className={cn("flex items-center gap-1.5 border-b px-3 py-2 text-[11px] font-semibold", reconciles ? "bg-emerald-50 text-emerald-800 dark:bg-emerald-950/40 dark:text-emerald-300" : "bg-amber-50 text-amber-800 dark:bg-amber-950/40 dark:text-amber-300")}>
   902→              {reconciles ? (
   903→                <>
   904→                  <ShieldCheck className="h-3.5 w-3.5" /> Ledger-reconciled — every figure is aggregated from ledger lines and Σ(lines) = net payout
   905→                </>
   906→              ) : (
   907→                <>
   908→                  <AlertTriangle className="h-3.5 w-3.5" /> Not reconciled — Σ(lines) {formatMinor(linesSum)} ≠ payout {formatMinor(payout.amountMinor)} (legacy demo snapshot)
   909→                </>
   910→              )}
   911→            </div>
   912→            <Table>
   913→              <TableBody>
   914→                {lines.map((l, i) => (
   915→                  <TableRow key={i}>
   916→                    <TableCell className="text-sm">{l.label}</TableCell>
   917→                    <TableCell className="w-32 text-right"><Money minor={l.amountMinor} /></TableCell>
   918→                  </TableRow>
   919→                ))}
   920→                <TableRow className="border-t-2">
   921→                  <TableCell className="text-sm font-semibold">Net payout</TableCell>
   922→                  <TableCell className="w-32 text-right"><Money minor={payout.amountMinor} bold /></TableCell>
   923→                </TableRow>
   924→              </TableBody>
   925→            </Table>
   926→          </div>
   927→        )}
   928→        {st.orderCount != null || st.periodTo ? (
   929→          <p className="mt-2 text-[11px] text-muted-foreground">
   930→            {st.orderCount != null ? `${st.orderCount} paid order${st.orderCount === 1 ? "" : "s"}` : ""}
   931→            {st.periodFrom && st.periodTo ? ` · ${fmtDT(st.periodFrom)} → ${fmtDT(st.periodTo)}` : ""}
   932→          </p>
   933→        ) : null}
   934→      </div>
   935→
   936→      <div className="space-y-3">
   937→        <div>
   938→          <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank</p>
   939→          {payout.bankAccount ? (
   940→            <div className="mt-1.5 space-y-1 rounded-lg border bg-card p-3 text-sm">
   941→              <p className="flex items-center gap-1.5 font-medium">
   942→                <Landmark className="h-3.5 w-3.5 text-teal-600 dark:text-teal-400" /> {payout.bankAccount.bankName}
   943→              </p>
   944→              <p className="text-muted-foreground">{payout.bankAccount.holderName}</p>
   945→              <p className="font-mono text-xs">{payout.bankAccount.accountNumberMasked}</p>
   946→            </div>
   947→          ) : (
   948→            <p className="mt-1.5 text-xs text-muted-foreground">No bank account linked</p>
   949→          )}
   950→        </div>
   951→        {payout.providerRef ? (
   952→          <div>
   953→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank reference</p>
   954→            <p className="mt-1 font-mono text-xs">{payout.providerRef}</p>
   955→          </div>
   956→        ) : null}
   957→        {payout.status === "processing" ? (
   958→          <div className="flex flex-col gap-2">
   959→            <Button size="sm" className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700" onClick={onMarkPaid}>
   960→              Mark paid…
   961→            </Button>
   962→            <Button size="sm" variant="outline" className="min-h-[44px] border-red-200 text-red-600 hover:bg-red-50 dark:border-red-900 dark:hover:bg-red-950/40" onClick={onMarkFailed}>
   963→              Mark failed…
   964→            </Button>
   965→          </div>
   966→        ) : (
   967→          <p className={cn("text-xs", payout.status === "paid" ? "text-emerald-700 dark:text-emerald-400" : "text-muted-foreground")}>
   968→            {payout.status === "paid" ? `Paid ${fmtDT(payout.paidAt)}` : "No actions available"}
   969→          </p>
   970→        )}
   971→      </div>
   972→    </div>
   973→  );

   977→  payout,
   978→  onClose,
   979→  onDone,
   980→}: {
   981→  payout: PayoutRow | null;
   982→  onClose: () => void;
   983→  onDone: () => void;
   984→}) {
   985→  const [refNo, setRefNo] = useState("");
   986→  const [saving, setSaving] = useState(false);
   987→
   988→  useEffect(() => {
   989→    if (payout) setRefNo("");
   990→  }, [payout]);
   991→
   992→  const submit = async () => {
   993→    if (!payout || !refNo.trim()) return;
   994→    setSaving(true);
   995→    try {
   996→      await apiSend(`/api/admin/payouts/${payout.id}/mark-paid`, "POST", { referenceNo: refNo.trim() });
   997→      toast.success("Payout marked paid", {
   998→        description: `${payout.referenceNo} · bank ref ${refNo.trim()} — ledger T4 posted.`,
   999→      });
  1000→      onDone();
  1001→    } catch (e) {
  1002→      toast.error("Mark paid failed", { description: e instanceof Error ? e.message : String(e) });
  1003→    } finally {
  1004→      setSaving(false);
  1005→    }
  1006→  };
  1007→
  1008→  return (
  1009→    <Dialog open={payout !== null} onOpenChange={(o) => !o && onClose()}>
  1010→      <DialogContent className="sm:max-w-md">
  1011→        <DialogHeader>
  1012→          <DialogTitle>Mark payout as paid</DialogTitle>
  1013→          <DialogDescription>
  1014→            {payout ? `${payout.referenceNo} · ${formatMinor(payout.amountMinor)} to ${payout.sellerName ?? "seller"}.` : ""}
  1015→          </DialogDescription>
  1016→        </DialogHeader>
  1017→        <div className="space-y-1.5">
  1018→          <Label htmlFor="payout-ref">Bank reference number (required)</Label>
  1019→          <Input
  1020→            id="payout-ref"
  1021→            value={refNo}
  1022→            onChange={(e) => setRefNo(e.target.value)}
  1023→            placeholder="e.g. BK-20261008-8817"
  1024→            className="min-h-[44px]"
  1025→            autoFocus
  1026→          />
  1027→          <p className="text-[11px] text-muted-foreground">Recorded on the payout and posted to the ledger as payout_confirmed (T4).</p>
  1028→        </div>
  1029→        <DialogFooter>
  1030→          <Button variant="outline" onClick={onClose} disabled={saving} className="min-h-[44px] sm:min-h-0">Cancel</Button>
  1031→          <Button onClick={() => void submit()} disabled={saving || !refNo.trim()} className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700 sm:min-h-0">
  1032→            {saving ? "Posting…" : "Confirm paid"}
  1033→          </Button>
  1034→        </DialogFooter>
  1035→      </DialogContent>
  1036→    </Dialog>
  1037→  );
  1038→}
