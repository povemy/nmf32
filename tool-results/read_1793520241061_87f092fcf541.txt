     1→// ============================================================================
     2→// Administrator TOTP MFA (Phase 1 / C4) — enrollment storage, login challenges
     3→// and code verification. Stdlib crypto ONLY (no new deps).
     4→//
     5→// Layers:
     6→//   • TOTP (RFC 6238 / RFC 4226): HMAC-SHA1, 30s steps, 6 digits, ±1 step
     7→//     verification window, over a base32 (RFC 4648) 20-byte secret.
     8→//   • Secret at rest: AES-256-GCM, key = sha256(MFA_ENCRYPTION_KEY). Format
     9→//     "base64(iv).base64(tag).base64(ct)". Missing key: dev → one-time-warned
    10→//     default key; production → 503 MFA_NOT_CONFIGURED (fail-closed, mirrors
    11→//     the PSP_WEBHOOK_SECRET pattern).
    12→//   • Recovery codes: 10 × randomToken(5) (10 hex chars), stored as a JSON
    13→//     array of sha256(code) digests, one-time use (removed on use + persisted).
    14→//   • Login challenges: in-memory Map (module-level, single process — same
    15→//     architecture as the in-memory rate limiter), randomToken(32), 5-min TTL,
    16→//     PROGRESSIVE cooldowns (3 fails → 60 s lock; each repeat doubles it,
    17→//     capped at 10 min — Phase 2.1 removed the old 8-fail "permanent" lock so
    18→//     no one can permanently lock out an administrator), PLUS a cross-challenge
    19→//     per-admin throttle that pauses challenge MINTING for admins who keep
    20→//     failing (prevents challenge-cycling to reset the counter).
    21→//   • TOTP replay protection (Phase 2.1): the accepted time-step counter is
    22→//     persisted on the enrollment (lastUsedTotpCounter) and consumed with a
    23→//     CAS updateMany — the same valid code can never verify twice, and the
    24→//     enrollment-confirm code is burned the same way.
    25→//
    26→// DB guard: getMfaEnrollment() treats Prisma P2021/P2022 (table/column does
    27→// not exist) as UNENROLLED, so the app runs BEFORE migration
    28→// 20261101000002 is applied (dev) while migrated production gets full
    29→// enforcement.
    30→//
    31→// NEVER log the TOTP secret, the otpauth URL, recovery codes or submitted
    32→// codes — audit actions carry counts/status only.
    33→// ============================================================================

    45→export const TOTP_DIGITS = 6;

    47→export const TOTP_WINDOW = 1;
    48→export const TOTP_SECRET_BYTES = 20;

    50→/* --------------------------------------------------- base32 (RFC 4648) */
    51→
    52→const B32_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
    53→
    54→/** RFC 4648 base32, no padding. */
    55→export function base32Encode(buf: Buffer): string {
    56→  let bits = 0;
    57→  let value = 0;
    58→  let out = "";
    59→  for (const byte of buf) {
    60→    value = (value << 8) | byte;
    61→    bits += 8;
    62→    while (bits >= 5) {
    63→      out += B32_ALPHABET[(value >>> (bits - 5)) & 31];
    64→      bits -= 5;
    65→    }
    66→  }
    67→  // Final partial group — pad with zero bits on the right.
    68→  if (bits > 0) out += B32_ALPHABET[(value << (5 - bits)) & 31];
    69→  return out;
    70→}
    71→
    72→/** RFC 4648 base32 decode (tolerates padding + whitespace + case). Throws on
    73→ *  characters outside the alphabet. */
    74→export function base32Decode(s: string): Buffer {
    75→  const clean = s.replace(/=+$/g, "").replace(/\s+/g, "").toUpperCase();
    76→  let bits = 0;
    77→  let value = 0;
    78→  const bytes: number[] = [];
    79→  for (const ch of clean) {
    80→    const idx = B32_ALPHABET.indexOf(ch);
    81→    if (idx < 0) throw new Error(`invalid base32 character '${ch}'`);
    82→    value = (value << 5) | idx;
    83→    bits += 5;
    84→    if (bits >= 8) {
    85→      bytes.push((value >>> (bits - 8)) & 255);
    86→      bits -= 8;
    87→    }
    88→  }
    89→  return Buffer.from(bytes);
    90→}

    92→/* ------------------------------------------------------- TOTP (RFC 6238) */
    93→
    94→/** HOTP code (RFC 4226): HMAC-SHA1 over the 8-byte big-endian counter,
    95→ *  dynamic truncation, `digits` decimal digits (zero-padded). */
    96→export function totpCode(secret: Buffer, counter: number, digits = TOTP_DIGITS): string {
    97→  const counterBuf = Buffer.alloc(8);
    98→  counterBuf.writeBigUInt64BE(BigInt(counter));
    99→  const mac = createHmac("sha1", secret).update(counterBuf).digest();
   100→  const offset = mac[mac.length - 1] & 0x0f;
   101→  const bin =
   102→    ((mac[offset] & 0x7f) << 24) | (mac[offset + 1] << 16) | (mac[offset + 2] << 8) | mac[offset + 3];
   103→  return (bin % 10 ** digits).toString().padStart(digits, "0");
   104→}
   105→
   106→/** TOTP counter for a Unix timestamp (seconds). */
   107→export function totpCounterAt(timeSeconds: number, step = TOTP_STEP_SECONDS): number {
   108→  return Math.floor(timeSeconds / step);
   109→}

   111→function safeEqualStr(a: string, b: string): boolean {
   112→  const ab = Buffer.from(a, "utf8");
   113→  const bb = Buffer.from(b, "utf8");
   114→  return ab.length === bb.length && timingSafeEqual(ab, bb);
   115→}

   117→/** Verify a 6-digit code against the secret with a ±window-step window at
   118→ *  `nowMs` (default: current time). Non-digit input simply fails. */
   119→export function verifyTotp(
   120→  secret: Buffer,
   121→  code: string,
   122→  nowMs = Date.now(),
   123→  window = TOTP_WINDOW,
   124→  step = TOTP_STEP_SECONDS
   125→): boolean {
   126→  return verifyTotpDetailed(secret, code, nowMs, window, step).ok;
   127→}
   128→
   129→/** PHASE 2.1 (§5-C): verifyTotp that also reports WHICH time-step counter
   130→ *  matched — required for replay protection (the accepted counter is persisted
   131→ *  and never accepted again). `counter` is present only when ok. */
   132→export function verifyTotpDetailed(
   133→  secret: Buffer,
   134→  code: string,
   135→  nowMs = Date.now(),
   136→  window = TOTP_WINDOW,
   137→  step = TOTP_STEP_SECONDS
   138→): { ok: boolean; counter?: number } {
   139→  const normalized = (code ?? "").replace(/[^0-9]/g, "");
   140→  if (normalized.length !== TOTP_DIGITS) return { ok: false };
   141→  const counter = totpCounterAt(Math.floor(nowMs / 1000), step);
   142→  for (let i = -window; i <= window; i++) {
   143→    if (safeEqualStr(totpCode(secret, counter + i), normalized)) return { ok: true, counter: counter + i };
   144→  }
   145→  return { ok: false };
   146→}

   150→  const secret = randomBytes(TOTP_SECRET_BYTES);
   151→  return { secret, secretBase32: base32Encode(secret) };

   162→/* ------------------------------------------- secret encryption (AES-GCM) */
   163→
   164→const DEV_FALLBACK_KEY = "sm_mfa_dev_key";
   165→let devKeyWarned = false;
   166→
   167→/** True when MFA crypto is usable in this environment: an explicit
   168→ *  MFA_ENCRYPTION_KEY is set, OR we are outside production (dev default key). */
   169→export function mfaEncryptionConfigured(): boolean {
   170→  const key = process.env.MFA_ENCRYPTION_KEY;
   171→  if (key && key.trim() !== "") return true;
   172→  return process.env.NODE_ENV !== "production";
   173→}
   174→
   175→/** AES-256-GCM key = sha256(MFA_ENCRYPTION_KEY). Fail-closed in production
   176→ *  when the key is unset (503 MFA_NOT_CONFIGURED); dev gets a one-time-warned
   177→ *  default so local flows work end-to-end. */
   178→function mfaKey(): Buffer {
   179→  const envKey = process.env.MFA_ENCRYPTION_KEY;
   180→  if (envKey && envKey.trim() !== "") {
   181→    return createHash("sha256").update(envKey, "utf8").digest();
   182→  }
   183→  if (process.env.NODE_ENV === "production") {
   184→    throw new ApiError(503, "MFA_NOT_CONFIGURED", "MFA_ENCRYPTION_KEY is not set — two-factor authentication is unavailable.");
   185→  }
   186→  if (!devKeyWarned) {
   187→    devKeyWarned = true;
   188→    console.warn("[mfa] MFA_ENCRYPTION_KEY not set — using the development default key. Set MFA_ENCRYPTION_KEY in production.");
   189→  }
   190→  return createHash("sha256").update(DEV_FALLBACK_KEY, "utf8").digest();
   191→}

   193→/** Encrypt a secret → "base64(iv).base64(tag).base64(ct)". */
   194→export function encryptSecret(plaintext: string): string {
   195→  const iv = randomBytes(12);
   196→  const cipher = createCipheriv("aes-256-gcm", mfaKey(), iv);
   197→  const ct = Buffer.concat([cipher.update(plaintext, "utf8"), cipher.final()]);
   198→  const tag = cipher.getAuthTag();
   199→  return `${iv.toString("base64")}.${tag.toString("base64")}.${ct.toString("base64")}`;
   200→}
   201→
   202→/** Decrypt "base64(iv).base64(tag).base64(ct)". Throws on tampering (GCM tag
   203→ *  mismatch), malformed input or a changed key. */
   204→export function decryptSecret(enc: string): string {
   205→  const parts = enc.split(".");
   206→  if (parts.length !== 3) throw new Error("malformed MFA secret ciphertext");
   207→  const decipher = createDecipheriv("aes-256-gcm", mfaKey(), Buffer.from(parts[0], "base64"));
   208→  decipher.setAuthTag(Buffer.from(parts[1], "base64"));
   209→  const pt = Buffer.concat([decipher.update(Buffer.from(parts[2], "base64")), decipher.final()]);
   210→  return pt.toString("utf8");
   211→}

   213→/** Verify a TOTP code against an enrollment's ENCRYPTED secret (±1 step).
   214→ *  Tampered/undecryptable secrets fail closed (false); ApiError 503
   215→ *  (unconfigured production key) still propagates. */
   216→export function verifyTotpForSecretEnc(secretEnc: string, code: string, nowMs = Date.now()): boolean {
   217→  try {
   218→    const secret = base32Decode(decryptSecret(secretEnc));
   219→    return verifyTotp(secret, code, nowMs);
   220→  } catch (e) {
   221→    if (e instanceof ApiError) throw e;
   222→    return false;
   223→  }
   224→}

   242→/* ------------------------------------------- TOTP replay guard (§5-C) */
   243→
   244→let replayWarned = false;
   245→
   246→/** CAS-guarded consumption of a TOTP time-step counter: the update lands only
   247→ *  when the stored counter is OLDER than `counter` (or null). Returns false
   248→ *  when the counter was already used (replay) or a concurrent verification
   249→ *  consumed it first — exactly one parallel replay can ever win.
   250→ *  Pre-migration deployments (table OR column missing): protection is skipped
   251→ *  with a loud one-time warning — identical fail-open contract to the Phase 1
   252→ *  table-absence guard (production always deploys migrations first). */
   253→export async function consumeTotpCounter(adminId: string, counter: number): Promise<boolean> {
   254→  try {
   255→    const res = await db.adminMfaEnrollment.updateMany({
   256→      where: {
   257→        adminId,
   258→        OR: [{ lastUsedTotpCounter: null }, { lastUsedTotpCounter: { lt: counter } }],
   259→      },
   260→      data: { lastUsedTotpCounter: counter },
   261→    });
   262→    return res.count === 1;
   263→  } catch (e) {
   264→    if (isPrismaMissingTableError(e)) {
   265→      if (!replayWarned) {
   266→        replayWarned = true;
   267→        console.warn(
   268→          "[mfa] TOTP replay protection unavailable — apply migrations 20261101000002 + 20261103000000 to enable it"
   269→        );
   270→      }
   271→      return true; // column/table absent (unmigrated dev) — fail-open, warn once
   272→    }
   273→    throw e;
   274→  }
   275→}

   279→export const RECOVERY_CODE_COUNT = 10;
   280→
   281→export function sha256Hex(input: string): string {
   282→  return createHash("sha256").update(input, "utf8").digest("hex");
   283→}
   284→
   285→/** Canonical recovery-code form: lowercase alphanumerics only. */
   286→export function normalizeRecoveryCode(raw: string): string {
   287→  return (raw ?? "").toLowerCase().replace(/[^a-z0-9]/g, "");
   288→}
   289→
   290→/** "abcdefghij" → "abcde-fghij" (display grouping only). */
   291→export function formatRecoveryCode(code: string): string {
   292→  return code.length === 10 ? `${code.slice(0, 5)}-${code.slice(5)}` : code;
   293→}
   294→
   295→/** 10 recovery codes (10 hex chars each) + the JSON array of sha256 digests
   296→ *  to persist. Plaintext codes exist ONLY in the caller's response. */
   297→export function generateRecoveryCodes(count = RECOVERY_CODE_COUNT): { codes: string[]; hashesJson: string } {
   298→  const codes = Array.from({ length: count }, () => randomToken(5));
   299→  return { codes, hashesJson: JSON.stringify(codes.map((c) => sha256Hex(c))) };
   300→}
   301→
   302→/** Parse the stored hashesJson → string[] of sha256 hex digests (defensive:
   303→ *  non-string / non-digest entries dropped; malformed JSON → []). */
   304→export function parseRecoveryHashes(hashesJson: string): string[] {
   305→  try {
   306→    const parsed = JSON.parse(hashesJson ?? "[]");
   307→    if (!Array.isArray(parsed)) return [];
   308→    return parsed.filter((h): h is string => typeof h === "string" && /^[0-9a-f]{64}$/.test(h));
   309→  } catch {
   310→    return [];
   311→  }
   312→}
   313→
   314→/** Index of the matching recovery code in the stored hashes, or -1. */
   315→export function matchRecoveryCode(hashesJson: string, rawCode: string): number {
   316→  const normalized = normalizeRecoveryCode(rawCode);
   317→  if (normalized === "") return -1;
   318→  return parseRecoveryHashes(hashesJson).indexOf(sha256Hex(normalized));
   319→}
   320→
   321→/** Pure removal: returns the updated hashesJson with the hash at `index`
   322→ *  removed (one-time-use burn). Out-of-range index → input unchanged. */
   323→export function removeRecoveryHash(hashesJson: string, index: number): string {
   324→  const hashes = parseRecoveryHashes(hashesJson);
   325→  if (index < 0 || index >= hashes.length) return hashesJson;
   326→  return JSON.stringify([...hashes.slice(0, index), ...hashes.slice(index + 1)]);
   327→}

   370→export const MFA_CHALLENGE_TTL_MS = 5 * 60 * 1000;
   371→/** Consecutive failures (since the last lock) that trigger a cooldown lock. */
   372→export const MFA_FAIL_LOCK_THRESHOLD = 3;
   373→/** FIRST cooldown duration; subsequent locks double it (60→120→240→480…). */
   374→export const MFA_FAIL_LOCK_MS = 60 * 1000;
   375→/** Hard cap for a single cooldown lock (10 minutes). */
   376→export const MFA_LOCK_MAX_MS = 10 * 60 * 1000;
   377→/** Total failures that escalate the cooldown straight to the cap. */
   378→export const MFA_TOTAL_FAIL_LIMIT = 8;

   380→export type MfaChallengeState = {
   381→  adminId: string;
   382→  expiresAt: number;
   383→  /** Failures since the last imposed lock (reset whenever a lock fires). */
   384→  failures: number;
   385→  /** Lifetime failures on this challenge (never reset). */
   386→  totalFailures: number;
   387→  /** Cooldown locks imposed so far — drives the escalating duration. */
   388→  lockCount: number;
   389→  /** Challenges are unusable before this timestamp (never "permanent": the
   390→   *  maximum possible cooldown is MFA_LOCK_MAX_MS and the challenge itself
   391→   *  expires at expiresAt). */
   392→  lockedUntil: number;
   393→};
   394→
   395→/** Fresh challenge state (used by tests + createMfaChallenge). */
   396→export function freshChallengeState(adminId: string, expiresAt: number): MfaChallengeState {
   397→  return { adminId, expiresAt, failures: 0, totalFailures: 0, lockCount: 0, lockedUntil: 0 };
   398→}
   399→
   400→/** Pure status check: valid | locked (with retryAfterSeconds) | expired. */
   401→export function challengeStatus(
   402→  state: MfaChallengeState,
   403→  now: number
   404→): { status: "valid" | "locked" | "expired"; retryAfterSeconds?: number } {
   405→  if (now >= state.expiresAt) return { status: "expired" };
   406→  if (now < state.lockedUntil) {
   407→    return { status: "locked", retryAfterSeconds: Math.max(1, Math.ceil((state.lockedUntil - now) / 1000)) };
   408→  }
   409→  return { status: "valid" };
   410→}
   411→
   412→/** Pure failure accounting: after a WRONG code at `now`, what does the
   413→ *  challenge look like and which failure reason applies?
   414→ *
   415→ *  PHASE 2.1 (§5-A): PROGRESSIVE COOLDOWNS — no permanent lockout exists any
   416→ *  more. Every lock is temporary and doubles with each repetition:
   417→ *    • lock #1 (3 consecutive failures) → 60 s cooldown
   418→ *    • lock #2 → 120 s · lock #3 → 240 s · lock #4 → 480 s
   419→ *    • any lock after 8 TOTAL failures, or lock #5+ → 600 s cap
   420→ *  A challenge that hits the cap simply outlives its own 5-minute TTL — the
   421→ *  attacker gains nothing, the legitimate administrator is never permanently
   422→ *  locked out by anyone else's activity (challenges require the password to
   423→ *  obtain in the first place, and a fresh sign-in always starts clean). */
   424→export function recordChallengeFailure(
   425→  state: MfaChallengeState,
   426→  now: number
   427→): { next: MfaChallengeState; outcome: "INVALID" | "LOCKED"; cooldownMs: number; retryAfterSeconds: number } {
   428→  const totalFailures = state.totalFailures + 1;
   429→  const failures = state.failures + 1;
   430→  const lockNow = failures >= MFA_FAIL_LOCK_THRESHOLD || totalFailures >= MFA_TOTAL_FAIL_LIMIT;
   431→  if (lockNow) {
   432→    const lockCount = state.lockCount + 1;
   433→    const escalated = totalFailures >= MFA_TOTAL_FAIL_LIMIT || lockCount >= 5;
   434→    const cooldownMs = escalated
   435→      ? MFA_LOCK_MAX_MS
   436→      : Math.min(MFA_FAIL_LOCK_MS * 2 ** (lockCount - 1), MFA_LOCK_MAX_MS);
   437→    return {
   438→      next: { ...state, failures: 0, totalFailures, lockCount, lockedUntil: Math.max(state.lockedUntil, now + cooldownMs) },
   439→      outcome: "LOCKED",
   440→      cooldownMs,
   441→      retryAfterSeconds: Math.max(1, Math.ceil(cooldownMs / 1000)),
   442→    };
   443→  }
   444→  return { next: { ...state, failures, totalFailures }, outcome: "INVALID", cooldownMs: 0, retryAfterSeconds: 0 };
   445→}

   447→// Module-level in-memory challenge store — single process, consistent with the
   448→// existing in-memory rate limiter architecture (lib/rate-limit.ts).
   449→const challenges = new Map<string, MfaChallengeState>();
   450→let lastChallengeSweep = 0;
   451→
   452→function sweepChallenges(now: number): void {
   453→  if (now - lastChallengeSweep < 60_000) return;
   454→  lastChallengeSweep = now;
   455→  for (const [token, state] of challenges) {
   456→    if (now >= state.expiresAt) challenges.delete(token);
   457→  }
   458→}

   460→/* --------------------------------------- per-admin cross-challenge throttle */
   461→// PHASE 2.1 (§5-A): without this, an attacker holding a stolen password could
   462→// cycle FRESH challenges (each new sign-in mints one) to reset the per-challenge
   463→// failure counter and grind codes indefinitely. The throttle accumulates MFA
   464→// failures PER ADMIN across challenges and applies its own escalating cooldown
   465→// to challenge MINTING. It cannot be armed by unauthenticated traffic: a
   466→// challenge only exists after the password verified, so only someone who
   467→// already passed the password gate can accrue failures here — a legit admin
   468→// who types a wrong code recovers after at most ADMIN_MFA_THROTTLE_MAX_MS.
   469→const adminThrottle = new Map<string, { failures: number; lockCount: number; lockedUntil: number }>();
   470→let lastThrottleSweep = 0;
   471→
   472→/** Cross-challenge failures (within the window) that trigger an admin cooldown. */
   473→export const ADMIN_MFA_THROTTLE_FAIL_LIMIT = 6;
   474→/** Failure accrual window (rolling, refreshed by each failure). */
   475→export const ADMIN_MFA_THROTTLE_WINDOW_MS = 15 * 60 * 1000;
   476→/** Per-admin mint cooldown cap. */
   477→export const ADMIN_MFA_THROTTLE_MAX_MS = 10 * 60 * 1000;
   478→
   479→function sweepAdminThrottle(now: number): void {
   480→  if (now - lastThrottleSweep < 60_000) return;
   481→  lastThrottleSweep = now;
   482→  for (const [adminId, t] of adminThrottle) {
   483→    if (t.lockedUntil < now - ADMIN_MFA_THROTTLE_WINDOW_MS) adminThrottle.delete(adminId);
   484→  }
   485→}
   486→
   487→/** Mint-time gate: is challenge creation currently cooled down for this admin? */
   488→export function adminMfaThrottleStatus(adminId: string, now = Date.now()): { locked: boolean; retryAfterSeconds?: number } {
   489→  sweepAdminThrottle(now);
   490→  const t = adminThrottle.get(adminId);
   491→  if (!t || now >= t.lockedUntil) return { locked: false };
   492→  return { locked: true, retryAfterSeconds: Math.max(1, Math.ceil((t.lockedUntil - now) / 1000)) };
   493→}
   494→
   495→/** Record an MFA failure against the admin (cross-challenge). */
   496→function noteAdminMfaFailure(adminId: string, now: number): void {
   497→  const t = adminThrottle.get(adminId) ?? { failures: 0, lockCount: 0, lockedUntil: 0 };
   498→  t.failures += 1;
   499→  if (t.failures >= ADMIN_MFA_THROTTLE_FAIL_LIMIT) {
   500→    t.lockCount += 1;
   501→    t.failures = 0;
   502→    const cooldown = Math.min(MFA_FAIL_LOCK_MS * 2 ** (t.lockCount - 1), ADMIN_MFA_THROTTLE_MAX_MS);
   503→    t.lockedUntil = Math.max(t.lockedUntil, now + cooldown);
   504→  }
   505→  adminThrottle.set(adminId, t);
   506→}
   507→
   508→/** Mint a login challenge for a (password-verified) admin. TTL 5 min.
   509→ *  PHASE 2.1: refuses (429 MFA_ACCOUNT_COOLDOWN) while the cross-challenge
   510→ *  throttle holds a cooldown for this admin. */
   511→export async function createMfaChallenge(adminId: string): Promise<{ challenge: string; expiresAt: number }> {
   512→  const now = Date.now();
   513→  sweepChallenges(now);
   514→  const throttle = adminMfaThrottleStatus(adminId, now);
   515→  if (throttle.locked) {
   516→    throw new ApiError(
   517→      429,
   518→      "MFA_ACCOUNT_COOLDOWN",
   519→      `Too many incorrect codes — new sign-in attempts are paused for ${throttle.retryAfterSeconds}s`,
   520→      { retryAfterSeconds: throttle.retryAfterSeconds }
   521→    );
   522→  }
   523→  const challenge = randomToken(32);
   524→  const expiresAt = now + MFA_CHALLENGE_TTL_MS;
   525→  challenges.set(challenge, freshChallengeState(adminId, expiresAt));
   526→  return { challenge, expiresAt };
   527→}

   532→  adminId: string,

   534→  meta: Record<string, unknown>,

   541→      actorLabel = admin?.email ?? null;

   546→      actorLabel,
   547→      action,
   548→      entityType: "admin_mfa",
   549→      entityId: adminId,
   550→      meta,
   551→    });
   552→  } catch {
   553→    // Audit is best-effort — a failed log write must never break the auth flow.
   554→  }
   555→}

   557→/* --------------------------------------------------- challenge verify */
   558→
   559→export type MfaVerifyResult =
   560→  | { ok: true; adminId: string; usedRecoveryCode: boolean }
   561→  | {
   562→      ok: false;
   563→      reason: "INVALID" | "EXPIRED" | "LOCKED";
   564→      /** Failure-arm extras (informational, optional): how many wrong codes
   565→       *  remain before the next lock, and how long a lock still runs. */
   566→      attemptsRemaining?: number;
   567→      retryAfterSeconds?: number;
   568→    };
   569→
   570→/** Verify a login challenge: TOTP (±1 step) OR one-time recovery code.
   571→ *  Progressive lockout per challenge; success consumes the challenge.
   572→ *  NEVER logs the submitted code. */
   573→export async function verifyMfaChallenge(challengeToken: string, code: string): Promise<MfaVerifyResult> {
   574→  const token = (challengeToken ?? "").trim();
   575→  const now = Date.now();
   576→  sweepChallenges(now);
   577→  const state = challenges.get(token);
   578→  // Unknown token: already used, swept, or garbage — one uniform answer.
   579→  if (!state) return { ok: false, reason: "EXPIRED" };
   580→
   581→  const status = challengeStatus(state, now);
   582→  if (status.status === "expired") {
   583→    challenges.delete(token);
   584→    return { ok: false, reason: "EXPIRED" };
   585→  }
   586→  if (status.status === "locked") {
   587→    return { ok: false, reason: "LOCKED", retryAfterSeconds: status.retryAfterSeconds };
   588→  }
   589→
   590→  // Login enforcement only applies to CONFIRMED enrollments. An enrollment
   591→  // that vanished (or was never confirmed) can never satisfy this challenge.
   592→  const enrollment = await getMfaEnrollment(state.adminId);
   593→  if (!enrollment || !enrollment.confirmedAt) {
   594→    challenges.delete(token);
   595→    return { ok: false, reason: "EXPIRED" };
   596→  }
   597→
   598→  const attempt = (code ?? "").trim();
   599→
   600→  // 1) TOTP path (6-digit authenticator code) — PHASE 2.1 (§5-C): the matched
   601→  //    time-step counter is CAS-consumed; a replay of the SAME valid code (or
   602→  //    an older still-in-window code) is rejected exactly like a wrong code.
   603→  let secret: Buffer;
   604→  try {
   605→    secret = base32Decode(decryptSecret(enrollment.secretEnc));
   606→  } catch (e) {
   607→    if (e instanceof ApiError) throw e; // 503 MFA_NOT_CONFIGURED — fail-closed
   608→    // Undecryptable secret (corrupt row / rotated key): the challenge can
   609→    // never succeed — retire it and fail closed.
   610→    challenges.delete(token);
   611→    return { ok: false, reason: "INVALID" };
   612→  }
   613→  const totp = verifyTotpDetailed(secret, attempt, now);
   614→  if (totp.ok) {
   615→    const consumed = await consumeTotpCounter(state.adminId, totp.counter!);
   616→    if (consumed) {
   617→      challenges.delete(token);
   618→      return { ok: true, adminId: state.adminId, usedRecoveryCode: false };
   619→    }
   620→    // Replayed time step — uniform INVALID (never reveal WHY it failed).
   621→    await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", { replay: true });
   622→    const replayRec = recordChallengeFailure(state, now);
   623→    challenges.set(token, replayRec.next);
   624→    noteAdminMfaFailure(state.adminId, now);
   625→    if (replayRec.outcome === "LOCKED") {
   626→      return { ok: false, reason: "LOCKED", retryAfterSeconds: replayRec.retryAfterSeconds, attemptsRemaining: 0 };
   627→    }
   628→    return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(replayRec.next) };
   629→  }
   630→
   631→  // 2) Recovery-code path (one-time; hash burned BEFORE success is reported).
   632→  const recoveryIndex = matchRecoveryCode(enrollment.recoveryHashes, attempt);
   633→  if (recoveryIndex >= 0) {
   634→    const remainingHashes = removeRecoveryHash(enrollment.recoveryHashes, recoveryIndex);
   635→    // Optimistic-concurrency burn: the update only lands if the stored hash
   636→    // list is still the one we matched against — two parallel replays of the
   637→    // same code cannot BOTH consume it.
   638→    const burn = await db.adminMfaEnrollment.updateMany({
   639→      where: { adminId: state.adminId, recoveryHashes: enrollment.recoveryHashes },
   640→      data: { recoveryHashes: remainingHashes },
   641→    });
   642→    if (burn.count === 0) {
   643→      // Concurrent verify already burned this code — treat as a wrong code.
   644→      const rec = recordChallengeFailure(state, now);
   645→      challenges.set(token, rec.next);
   646→      noteAdminMfaFailure(state.adminId, now);
   647→      if (rec.outcome === "LOCKED") {
   648→        await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", { cooldown: rec.cooldownMs });
   649→        return { ok: false, reason: "LOCKED", retryAfterSeconds: rec.retryAfterSeconds };
   650→      }
   651→      return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(rec.next) };
   652→    }
   653→    challenges.delete(token);
   654→    await auditMfaEvent(state.adminId, "admin.mfa.recovery_used", { via: "login" });
   655→    return { ok: true, adminId: state.adminId, usedRecoveryCode: true };
   656→  }
   657→
   658→  // 3) Wrong code — progressive lockout (audit only when a lock fires, to
   659→  //    avoid per-attempt spam). Every failure also feeds the cross-challenge
   660→  //    per-admin throttle so challenge-cycling cannot reset the counters.
   661→  const rec = recordChallengeFailure(state, now);
   662→  challenges.set(token, rec.next);
   663→  noteAdminMfaFailure(state.adminId, now);
   664→  if (rec.outcome === "LOCKED") {
   665→    await auditMfaEvent(state.adminId, "admin.mfa.verify_failed", {
   666→      cooldown: rec.cooldownMs,
   667→      totalFailures: rec.next.totalFailures,
   668→    });
   669→    return { ok: false, reason: "LOCKED", retryAfterSeconds: rec.retryAfterSeconds, attemptsRemaining: 0 };
   670→  }
   671→  return { ok: false, reason: "INVALID", attemptsRemaining: remainingAttempts(rec.next) };
   672→}
   673→
   674→function remainingAttempts(state: MfaChallengeState): number {
   675→  return Math.max(0, Math.min(MFA_FAIL_LOCK_THRESHOLD - state.failures, MFA_TOTAL_FAIL_LIMIT - state.totalFailures));
   676→}

   678→/* ------------------------------------- MANDATORY ENROLLMENT GATE (§5-B) */
   679→
   680→let gateWarned = false;
   681→
   682→/** Roles that hold any FINANCIAL capability must have a CONFIRMED TOTP
   683→ *  enrollment before they may exercise privileged admin functionality.
   684→ *  Derived from the ADMIN_CAPS matrix (pure — see admin-caps.ts), so the
   685→ *  privileged set stays true if caps evolve: currently {superadmin, finance,
   686→ *  ops} (ops holds refunds.manage + disputes.manage).
   687→ *
   688→ *  Return values:
   689→ *    enrollmentRequired: false — no restriction (non-privileged role, OR
   690→ *      privileged with a confirmed enrollment).
   691→ *    enrollmentRequired: true — RESTRICTED: the admin may only use the MFA
   692→ *      bootstrap routes (enroll/confirm/status + me/logout) until enrollment
   693→ *      is confirmed. Bootstrap is safe: /api/admin/mfa/enroll needs only the
   694→ *      CURRENT PASSWORD (fresh superadmin on a migrated deployment can always
   695→ *      self-recover into full access — NOMOFEE can never be fully locked out).
   696→ *    Table missing (pre-migration deployment) — no enforcement surface yet;
   697→ *      returns false with a loud one-time warning, identical to the Phase 1
   698→ *      contract (enforcement begins the moment migration 20261101000002 lands).
   699→ */
   700→export async function mfaEnrollmentGate(role: string, adminId: string): Promise<{ enrollmentRequired: boolean }> {
   701→  if (!roleRequiresMfa(role)) return { enrollmentRequired: false };
   702→  let row: { confirmedAt: Date | null } | null;
   703→  try {
   704→    row = await db.adminMfaEnrollment.findUnique({ where: { adminId }, select: { confirmedAt: true } });
   705→  } catch (e) {
   706→    if (isPrismaMissingTableError(e)) {
   707→      if (!gateWarned) {
   708→        gateWarned = true;
   709→        console.warn("[mfa] enrollment table missing — mandatory-enrollment gate inactive until 20261101000002 is applied");
   710→      }
   711→      return { enrollmentRequired: false };
   712→    }
   713→    throw e;
   714→  }
   715→  return { enrollmentRequired: !row?.confirmedAt };
   716→}

   720→// The orchestrator adds `verifyPasswordAsync` to lib/password.ts concurrently

   723→const pwLib = passwordModule as typeof passwordModule & {

   730→export async function verifyAdminPassword(password: string, storedHash: string): Promise<boolean> {
   731→  if (verifyAsync) return verifyAsync(password, storedHash);
   732→  return verifyPassword(password, storedHash);
