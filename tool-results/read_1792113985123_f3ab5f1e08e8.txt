    80→/** Total unread seller messages — powers the "My messages" header badge.
    81→ *  Polls lightly + refreshes when the messages view is opened/left. */
    82→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    83→  const [unread, setUnread] = useState(0);
    84→  useEffect(() => {
    85→    if (!shopperSignedIn) {
    86→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    87→      setUnread(0);
    88→      return;
    89→    }
    90→    let alive = true;
    91→    const load = () =>
    92→      fetchJson<{ total: number }>("/api/chat/unread")
    93→        .then((d) => alive && setUnread(d.total ?? 0))
    94→        .catch(() => {});
    95→    load();
    96→    const t = setInterval(load, 15000);
    97→    return () => {
    98→      alive = false;
    99→      clearInterval(t);
   100→    };
   101→  }, [shopperSignedIn]);
   102→  // instant re-sync when a thread is read inside the messages view (the view
   103→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   104→  useEffect(() => {
   105→    if (!shopperSignedIn) return;
   106→    const sync = () =>
   107→      fetchJson<{ total: number }>("/api/chat/unread")
   108→        .then((d) => setUnread(d.total ?? 0))
   109→        .catch(() => {});
   110→    window.addEventListener("shopmesh:chat-sync", sync);
   111→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   112→  }, [shopperSignedIn]);
   113→  // opening (or leaving) the messages view should immediately re-sync the badge
   114→  useEffect(() => {
   115→    if (!shopperSignedIn) return;
   116→    const t = setTimeout(
   117→      () =>
   118→        fetchJson<{ total: number }>("/api/chat/unread")
   119→          .then((d) => setUnread(d.total ?? 0))
   120→          .catch(() => {}),
   121→      viewName === "messages" ? 600 : 0
   122→    );
   123→    return () => clearTimeout(t);
   124→  }, [shopperSignedIn, viewName]);
   125→  return unread;
   126→}

   128→export default function ShopperShell() {
   129→  const setSurface = useApp((s) => s.setSurface);
   130→  const activeStoreSlug = useApp((s) => s.activeStoreSlug);
   131→  const setActiveStoreSlug = useApp((s) => s.setActiveStoreSlug);
   132→
   133→  const [view, setView] = useState<ShopperView>({ name: "market" });
   134→  const [ready, setReady] = useState(false);
   135→  const [cartOpen, setCartOpen] = useState(false);
   136→  const [headerSearch, setHeaderSearch] = useState("");
   137→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   138→  // intentionally NOT part of ShopperView — a synthetic render override.
   139→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   140→  const restored = useRef(false);
   141→
   142→  // ---- mount: activeStoreSlug override, else restore last view --------------
   143→  useEffect(() => {
   144→    if (restored.current) return;
   145→    restored.current = true;
   146→    // claim deep-link wins over everything
   147→    try {
   148→      const params = new URLSearchParams(window.location.search);
   149→      const cj = params.get("claim_job");
   150→      const ct = params.get("claim_token");
   151→      if (cj && ct) {
   152→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   153→        setClaimLink({ jobId: cj, token: ct });
   154→        setReady(true);
   155→        return;
   156→      }
   157→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   158→      const sf = params.get("sf");
   159→      if (sf) {
   160→        const sp = params.get("sp");
   161→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   162→        setReady(true);
   163→        return;
   164→      }
   165→    } catch {
   166→      /* ignore */
   167→    }
   168→    if (activeStoreSlug) {
   169→      setView({ name: "storefront", slug: activeStoreSlug });
   170→      setActiveStoreSlug(null);
   171→    } else {
   172→      try {
   173→        const raw = window.localStorage.getItem(VIEW_KEY);
   174→        if (raw) {
   175→          const v = JSON.parse(raw) as ShopperView;
   176→          if (v && typeof v.name === "string") setView(v);
   177→        }
   178→      } catch {
   179→        /* ignore */
   180→      }
   181→    }
   182→    setReady(true);
   183→  }, [activeStoreSlug, setActiveStoreSlug]);
   184→
   185→  // ---- nav helper: persist + scroll to top ----------------------------------
   186→  const nav = useCallback((v: ShopperView) => {
   187→    setView(v);
   188→    try {
   189→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   190→    } catch {
   191→      /* ignore */
   192→    }
   193→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   194→  }, []);
   195→
   196→  // refresh storefront/store-page when slug changes externally
   197→  useEffect(() => {
   198→    if (!ready) return;
   199→    if (activeStoreSlug) {
   200→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   201→      setView({ name: "storefront", slug: activeStoreSlug });
   202→      setActiveStoreSlug(null);
   203→    }
   204→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   205→
   206→  const ctx = useMemo<ShopperCtx>(() => ({ nav, openCart: () => setCartOpen(true) }), [nav]);
   207→
   208→  const showChrome = !isStoreChrome(view);
   209→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   210→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   211→
   212→  return (
   213→    <div className="flex-1 flex flex-col min-h-screen">
   214→      <Toaster position="top-center" richColors />
   215→      <ShopperAuthDialog />
   216→
   217→      {showChrome ? (
   218→        <MarketHeader
   219→          view={view}
   220→          nav={nav}
   221→          openCart={() => setCartOpen(true)}
   222→          search={headerSearch}
   223→          setSearch={setHeaderSearch}
   224→        />
   225→      ) : null}
   226→
   227→      <AnimatePresence mode="wait">
   228→        <motion.main
   229→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : view.name === "me" ? view.tab ?? "" : ""}`}
   230→          initial={{ opacity: 0, y: 8 }}
   231→          animate={{ opacity: 1, y: 0 }}
   232→          exit={{ opacity: 0, y: -8 }}
   233→          transition={{ duration: 0.18, ease: "easeOut" }}
   234→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   235→        >
   236→          {claimLink ? null : (
   237→            <>
   238→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   239→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   240→              {view.name === "category" ? (
   241→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   242→              ) : null}
   243→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   244→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   245→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   246→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   247→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   248→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   249→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   250→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   251→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   252→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   253→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}
   254→            </>
   255→          )}
   256→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}
   257→        </motion.main>
   258→      </AnimatePresence>
   259→
   260→      {showBottomNav ? (
   261→        <BottomNav
   262→          view={view}
   263→          nav={nav}
   264→          openCart={() => setCartOpen(true)}
   265→        />
   266→      ) : null}
   267→
   268→      {showChrome ? (
   269→        <ShopperFooter
   270→          onSeller={() => setSurface("seller")}
   271→          onLookup={() => nav({ name: "lookup" })}
   272→          onMarket={() => nav({ name: "market" })}
   273→          onGenerate={() => nav({ name: "generate" })}
   274→        />
   275→      ) : null}
   276→
   277→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   278→    </div>
   279→  );
   280→}

   286→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   287→  const me = useApp((s) => s.me);
   288→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   289→  const signedIn = Boolean(me.loaded && me.shopper);
   290→  const { count } = useCart();
   291→  const unreadChats = useChatUnread(signedIn, view.name);
   292→
   293→  const items = [
   294→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   295→    {
   296→      key: "orders",
   297→      label: "Orders",
   298→      icon: Package,
   299→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   300→      badge: 0,
   301→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   302→    },
   303→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   304→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   305→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   306→  ];
   307→
   308→  return (
   309→    <nav
   310→      aria-label="Bottom navigation"
   311→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   312→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   313→    >
   314→      <div className="mx-auto grid max-w-lg grid-cols-5">
   315→        {items.map((it) => {
   316→          const Icon = it.icon;
   317→          return (
   318→            <button
   319→              key={it.key}
   320→              type="button"
   321→              onClick={it.onClick}
   322→              aria-label={it.label}
   323→              aria-current={it.active ? "page" : undefined}
   324→              className={cn(
   325→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   326→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   327→              )}
   328→            >
   329→              <span className="relative">
   330→                <Icon className="h-5 w-5" aria-hidden="true" />
   331→                {it.badge > 0 ? (
   332→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   333→                    {it.badge > 9 ? "9+" : it.badge}
   334→                  </span>
   335→                ) : null}
   336→              </span>
   337→              {it.label}
   338→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   339→            </button>
   340→          );
   341→        })}
   342→      </div>
   343→    </nav>
   344→  );
   345→}

   795→function timeAgoShort(iso: string): string {
   796→  const diff = Date.now() - new Date(iso).getTime();
   797→  const m = Math.floor(diff / 60000);
   798→  if (m < 1) return "now";
   799→  if (m < 60) return `${m}m`;
   800→  const h = Math.floor(m / 60);
   801→  if (h < 24) return `${h}h`;
   802→  const d = Math.floor(h / 24);
   803→  if (d < 7) return `${d}d`;
   804→  return new Date(iso).toLocaleDateString(undefined, { day: "numeric", month: "short" });
   805→}

   807→function MessagesView({ ctx, initialThreadId }: { ctx: ShopperCtx; initialThreadId?: string }) {
   808→  const me = useApp((s) => s.me);
   809→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   810→  const shopper = me.loaded ? me.shopper : null;
   811→
   812→  const [threads, setThreads] = useState<ThreadRow[] | null>(null);
   813→  const [activeId, setActiveId] = useState<string | null>(initialThreadId ?? null);
   814→  const [messages, setMessages] = useState<ChatMsg[] | null>(null);
   815→  const [activeThread, setActiveThread] = useState<{ storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null } | null>(null);
   816→  const [context, setContext] = useState<ThreadContextCard | null>(null);
   817→  const [draft, setDraft] = useState("");
   818→  const [sending, setSending] = useState(false);
   819→  const socketRef = useRef<Socket | null>(null);
   820→  const scrollRef = useRef<HTMLDivElement>(null);
   821→
   822→  const loadThreads = useCallback(async () => {
   823→    try {
   824→      const rows = await fetchJson<ThreadRow[]>("/api/chat/threads");
   825→      setThreads(rows);
   826→      return rows;
   827→    } catch {
   828→      setThreads([]);
   829→      return [];
   830→    }
   831→  }, []);
   832→
   833→  const loadMessages = useCallback(async (threadId: string) => {
   834→    try {
   835→      const d = await fetchJson<{
   836→        thread: { storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null };
   837→        context: ThreadContextCard | null;
   838→        messages: ChatMsg[];
   839→      }>(`/api/chat/threads/${threadId}`);
   840→      setMessages(d.messages);
   841→      setActiveThread(d.thread);
   842→      setContext(d.context ?? null);
   843→    } catch {
   844→      setMessages([]);
   845→      setContext(null);
   846→    }
   847→  }, []);
   848→
   849→  // initial load
   850→  useEffect(() => {
   851→    if (!shopper) return;
   852→    let alive = true;
   853→    loadThreads().then(() => {
   854→      if (!alive) return;
   855→      // Only auto-open when resuming an intent (contact-seller deep link).
   856→      // Plain entry shows the list first so unread badges stay visible —
   857→      // auto-reading the newest thread would instantly consume the badge.
   858→      if (initialThreadId) setActiveId(initialThreadId);
   859→    });
   860→    return () => {
   861→      alive = false;
   862→    };
   863→  }, [shopper, initialThreadId, loadThreads]);
   864→
   865→  // load conversation when thread changes
   866→  useEffect(() => {
   867→    if (!activeId) return;
   868→    setMessages(null);
   869→    // the detail GET marks the thread read server-side — refresh the list so
   870→    // the unread pill on the row (and the header badge) drops immediately
   871→    loadMessages(activeId).then(() => {
   872→      loadThreads();
   873→      window.dispatchEvent(new Event("shopmesh:chat-sync"));
   874→    });
   875→  }, [activeId, loadMessages, loadThreads]);
   876→
   877→  // live updates via the socket.io chat hub
   878→  useEffect(() => {
   879→    if (!shopper) return;
   880→    const socket = io("/?XTransformPort=3003", {
   881→      transports: ["websocket", "polling"],
   882→      reconnection: true,
   883→      reconnectionAttempts: 5,
   884→      timeout: 10000,
   885→    });
   886→    socketRef.current = socket;
   887→    socket.on("connect", () => {
   888→      // join all thread rooms so the sidebar stays fresh too
   889→      const rooms = (threads ?? []).map((t) => `thread:${t.id}`);
   890→      socket.emit("join", { rooms });
   891→    });
   892→    socket.on("chat:message", (msg: ChatMsg & { threadId?: string }) => {
   893→      if (msg.threadId && msg.threadId === activeId) {
   894→        setMessages((prev) => (prev && !prev.some((m) => m.id === msg.id) ? [...prev, msg] : prev));
   895→        if (msg.senderType === "seller") {
   896→          // message arrived while the shopper is looking at this conversation —
   897→          // clear the badge straight away instead of waiting for the poller
   898→          void postJson(`/api/chat/threads/${activeId}/read`, {}).catch(() => {});
   899→        }
   900→      }
   901→      loadThreads();
   902→    });
   903→    return () => {
   904→      socket.disconnect();
   905→      socketRef.current = null;
   906→    };
   907→  }, [shopper, threads, activeId, loadThreads]);
   908→
   909→  // auto-scroll to latest
   910→  useEffect(() => {
   911→    const el = scrollRef.current;
   912→    if (el) el.scrollTop = el.scrollHeight;
   913→  }, [messages]);
   914→
   915→  const send = async () => {

   918→    setSending(true);
   919→    try {
   920→      const msg = await postJson<ChatMsg>(`/api/chat/threads/${activeId}/messages`, { body: text });
   921→      setMessages((prev) => (prev ? [...prev, msg] : [msg]));
   922→      setDraft("");
   923→      loadThreads();
   924→      window.dispatchEvent(new Event("shopmesh:chat-sync"));
   925→    } catch (e) {
   926→      toast.error(e instanceof Error ? e.message : "Could not send the message");
   927→    } finally {
   928→      setSending(false);
   929→    }
   930→  };
   931→
   932→  if (!shopper) {
   933→    return (
   934→      <div className="mx-auto w-full max-w-3xl flex-1 px-4 py-8">
   935→        <EmptyState
   936→          icon={MessagesSquare}
   937→          title="Sign in to see your messages"
   938→          hint="Create a free SHOPMESH account to chat with sellers about products, orders, deliveries and returns."
   939→          action={
   940→            <Button onClick={() => openShopperAuth(() => ctx.nav({ name: "messages" }))} className="min-h-[44px] bg-brand-gradient-deep text-white hover:brightness-110 gap-2">
   941→              <LogIn className="h-4 w-4" /> Sign in or create account
   942→            </Button>
   943→          }
   944→        />
   945→      </div>

   950→    <div className="mx-auto w-full max-w-5xl flex-1 px-4 py-6 sm:py-8">
   951→      <div className="pb-4">
   952→        <h1 className="flex items-center gap-2 text-2xl font-extrabold tracking-tight">
   953→          <MessagesSquare className="h-6 w-6 text-teal-700" /> Messages
   954→        </h1>
   955→        <p className="text-sm text-muted-foreground">Chat with sellers about your orders — replies arrive in real time.</p>
   956→      </div>
   957→
   958→      <div className="grid gap-4 md:grid-cols-[300px_1fr]">
   959→        {/* thread list */}
   960→        <aside className={cn("rounded-2xl border bg-card p-2 shadow-soft", activeId && "hidden md:block")}>
   961→          {threads === null ? (
