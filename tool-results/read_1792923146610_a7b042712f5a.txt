    45→// Sandbox fallback keeps the mock rails working without env setup; production
    46→// MUST set PSP_WEBHOOK_SECRET (a real PSP secret) — the fallback is intentionally
    47→// public knowledge so it can never be trusted to authenticate real money events.
    48→const WEBHOOK_SECRET = process.env.PSP_WEBHOOK_SECRET ?? "sm_webhook_secret_demo";
    49→
    50→/** Replay-acceptance window: a correct MAC over an out-of-window timestamp is
    51→ *  still a replay — rejected with STALE_EVENT. Simulate endpoints stamp
    52→ *  Date.now() so sandbox flows always fall inside the window. */
    53→export const WEBHOOK_TS_WINDOW_MS = 10 * 60_000;
    54→
    55→export function expectedSignature(providerRef: string, eventType: string, amountMinor: number, ts: number): string {
    56→  return hmacSha256Hex(WEBHOOK_SECRET, `${providerRef}.${eventType}.${amountMinor}.${ts}`);
    57→}

    94→/**
    95→ * Verify signature (caller may pre-compute it, as simulate does), log the raw
    96→ * webhook, then process idempotently. Throws ApiError with proper status codes.
    97→ *
    98→ * RACE SAFETY (why claim-first): the success paths' FIRST write inside the
    99→ * Serializable transaction is a CONDITIONAL updateMany (status flip guarded by
   100→ * the pre-status). Two concurrent deliveries of the same event race on that
   101→ * row lock; the loser matches zero rows, writes nothing and returns the
   102→ * duplicate result — money movement, order transitions and notifications can
   103→ * only ever run for the winner. The ledger dedupKeys (FIN-1) backstop this at
   104→ * the posting layer, and the notification dedupKeys backstop the mail log.
   105→ */
   106→export async function processPaymentWebhook(input: {
   107→  provider: string;
   108→  body: { providerRef?: string; eventType?: string; amountMinor?: number; ts?: number; sig?: string; reason?: string };
   109→  rawBody: string;
   110→  headerSignature?: string | null;
   111→}): Promise<ProcessResult> {
   112→  // (2b.2) Provider sandbox enforcement FIRST — fail closed. resolvePspProvider
   113→  // falls back to the mock provider for unknown ids, so in production an
   114→  // unregistered provider id can never sneak through: mock is sandboxOnly → 403.
   115→  const psp = resolvePspProvider(input.provider);
   116→  if (psp.sandboxOnly && process.env.NODE_ENV === "production") {
   117→    throw new ApiError(403, "PROVIDER_SANDBOX_ONLY", `Provider "${input.provider}" is sandbox-only and cannot process webhooks in production`);
   118→  }
   119→
   120→  const providerRef = String(input.body.providerRef ?? "");
   121→  const eventType = String(input.body.eventType ?? "");
   122→  const amountMinor = Math.round(Number(input.body.amountMinor ?? 0));
   123→  const ts = Math.round(Number(input.body.ts ?? 0));
   124→  if (!providerRef || !eventType) {
   125→    throw new ApiError(400, "BAD_REQUEST", "providerRef and eventType are required");
   126→  }
   127→
   128→  // (2b.1) Timing-safe compare + freshness. signatureValid in the log means
   129→  // "authenticated as a LIVE event": correct MAC AND inside the replay window —
   130→  // a stale-but-correctly-signed delivery is recorded as signatureValid=false
   131→  // (replay protection is part of webhook authentication semantics) and
   132→  // rejected with STALE_EVENT below, AFTER the row is logged.
   133→  const providedSig = input.headerSignature ?? (input.body.sig ? String(input.body.sig) : null);
   134→  const expected = expectedSignature(providerRef, eventType, amountMinor, ts);
   135→  const sigMatches = !!providedSig && safeEqual(providedSig, expected);
   136→  const fresh = Math.abs(Date.now() - ts) <= WEBHOOK_TS_WINDOW_MS;
   137→  const signatureValid = sigMatches && fresh;
   138→
   139→  // OPTION A first: the ref may be a multi-seller payment group.
   140→  const group = await db.paymentGroup.findUnique({
   141→    where: { ref: providerRef },
   142→    include: {
   143→      orders: {
   144→        orderBy: { createdAt: "asc" },
   145→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   146→      },
   147→    },
   148→  });
   149→
   150→  const intent = group ? null : await db.paymentIntent.findUnique({ where: { providerRef } });
   151→
   152→  // ALWAYS log the webhook first (even for invalid signature / unknown ref) —
   153→  // invalid/stale rows stay in the log with their signatureValid flag.
   154→  const log = await db.paymentWebhookLog.create({
   155→    data: {
   156→      provider: input.provider,
   157→      providerRef,
   158→      eventType,
   159→      payloadJson: input.rawBody,
   160→      signatureValid,
   161→      processed: false,
   162→      orderId: group?.orders[0]?.id ?? intent?.orderId ?? null,
   163→    },
   164→  });
   165→
   166→  if (!sigMatches) {
   167→    throw new ApiError(400, "BAD_SIGNATURE", "Webhook signature verification failed");
   168→  }
   169→  if (!fresh) {
   170→    throw new ApiError(400, "STALE_EVENT", `Webhook timestamp is outside the ±${WEBHOOK_TS_WINDOW_MS / 60_000}-minute acceptance window (replay protection)`);
   171→  }
   172→
   173→  // ------------------------------------------------------------------ FIN-2b: inbound payout events
   174→  if (eventType === "payout.succeeded" || eventType === "payout.failed") {
   175→    return processPayoutEvent({ provider: input.provider, providerRef, eventType, amountMinor, reason: input.body.reason ?? null, logId: log.id });
   176→  }
   177→
   178→  // ------------------------------------------------------------------ OPTION A group path
   179→  if (group) {
   180→    if (eventType === "payment.failed") {
   181→      // No money moves on failure — a single conditional update is fine outside
   182→      // a Serializable tx (an already-succeeded group ignores the failed event).
   183→      if (group.status === "succeeded") {
   184→        await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   185→        return {
   186→          ok: true,
   187→          duplicate: true,
   188→          eventType,
   189→          orderNumber: group.orders[0]?.orderNumber ?? "",
   190→          orderNumbers: group.orders.map((o) => o.orderNumber),
   191→          orderStatus: "paid",
   192→          intentStatus: "succeeded",
   193→          isGroup: true,
   194→        };
   195→      }
   196→      await db.paymentGroup.update({ where: { id: group.id }, data: { status: "failed" } });
   197→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   198→      return {
   199→        ok: true,
   200→        duplicate: false,
   201→        eventType,
   202→        orderNumber: group.orders[0]?.orderNumber ?? "",
   203→        orderNumbers: group.orders.map((o) => o.orderNumber),
   204→        orderStatus: group.orders[0]?.status ?? "payment_pending",
   205→        intentStatus: "failed",
   206→        isGroup: true,
   207→      };
   208→    }
   209→
   210→    // Unknown-event + amount checks BEFORE the claim (unchanged order) — a
   211→    // malformed delivery must never even attempt the status flip.
   212→    if (eventType !== "payment.succeeded") {
   213→      throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   214→    }
   215→
   216→    if (amountMinor !== group.totalMinor) {
   217→      throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match payment group ${group.totalMinor}`);
   218→    }
   219→
   220→    const settings = await getCommissionSettings();
   221→    const paidAt = new Date();
   222→
   223→    // SUCCESS PATH — one Serializable transaction around claim + money.
   224→    // The conditional claim is the FIRST write: count 0 ⇒ another delivery
   225→    // already succeeded ⇒ nothing this tx wrote (the claim was the only
   226→    // intended write) ⇒ return the duplicate sentinel; committing an empty tx
   227→    // is equivalent to rollback-by-return.
   228→    const outcome = await db.$transaction(
   229→      async (tx) => {
   230→        const claim = await tx.paymentGroup.updateMany({
   231→          where: { id: group.id, status: { not: "succeeded" } },
   232→          data: { status: "succeeded", paidAt },
   233→        });
   234→        if (claim.count === 0) return { claimed: false as const };
   235→
   236→        for (const order of group.orders) {
   237→          await transitionOrder({
   238→            orderId: order.id,
   239→            toStatus: "paid",
   240→            actor: "system",
   241→            note: `Payment ${providerRef} succeeded via ${group.provider} (group payment)`,
   242→            timestamps: { paidAt },
   243→            tx,
   244→          });
   245→
   246→          // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   247→          const seller = order.store.seller;
   248→          const wallet = seller?.wallet;
   249→          if (seller && wallet) {
   250→            const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   251→            const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   252→            const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   253→            const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   254→            await postOrderPaid({
   255→              orderId: order.id,
   256→              orderNumber: order.orderNumber,
   257→              walletId: wallet.id,
   258→              provider: group.provider,
   259→              totalMinor: order.totalMinor,
   260→              sellerNetMinor,
   261→              commissionMinor,
   262→              holdUntil,
   263→              currency: order.currency,
   264→              occurredAt: paidAt,
   265→              tx,
   266→            });
   267→          }
   268→
   269→          // Store webhooks + notifications run inside the success tx (log-only
   270→          // writes on their own connections). notify carries dedupKeys so even
   271→          // an aborted tx + provider retry cannot multiply seller/customer mail.
   272→          await dispatchStoreWebhooks({
   273→            storeId: order.storeId,
   274→            event: "order.paid",
   275→            payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency, paymentGroupRef: providerRef },
   276→          });
   277→          if (seller) {
   278→            await notify({ event: "order.paid", channel: "email", recipient: seller.email, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   279→            await notify({ event: "order.paid", channel: "inapp", recipient: seller.email, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   280→          }
   281→          await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   282→          // Shopper in-app nudge (Task 10-d) — payment confirmed → "Order confirmed".
   283→          // Left as-is: its own promo-cap rules make it idempotent-ish, and the
   284→          // claim above means it only runs for the winning delivery anyway.
   285→          await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   286→          await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: group.provider, totalMinor: order.totalMinor, paymentGroup: true } });
   287→        }
   288→
   289→        // Server-side cart clearing — only the PURCHASED variants leave the cart,
   290→        // and only on verified payment success (never on failure/pending). Lines
   291→        // the buyer unchecked (kept for later) survive: selection-aware carts.
   292→        // PaymentGroups are account-scoped (shopperId always set). Inside the tx:
   293→        // a rollback never strands paid items out of the cart inconsistently.
   294→        const purchasedVariantIds = [
   295→          ...new Set(group.orders.flatMap((o) => o.items.map((it) => it.variantId)).filter((v): v is string => Boolean(v))),
   296→        ];
   297→        await tx.cartItem.deleteMany({
   298→          where: { cart: { shopperId: group.shopperId }, variantId: { in: purchasedVariantIds } },
   299→        });
   300→
   301→        return { claimed: true as const };
   302→      },
   303→      { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
   304→    );
   305→
   306→    if (!outcome.claimed) {
   307→      // Duplicate delivery — existing duplicate branch shape, processed=true.
   308→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   309→      return {
   310→        ok: true,
   311→        duplicate: true,
   312→        eventType,
   313→        orderNumber: group.orders[0]?.orderNumber ?? "",
   314→        orderNumbers: group.orders.map((o) => o.orderNumber),
   315→        orderStatus: "paid",
   316→        intentStatus: "succeeded",
   317→        isGroup: true,
   318→      };
   319→    }
   320→
   321→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   322→
   323→    // (FIN-1 1.6 wiring) T2 auto-settle — AFTER the success tx commits. The
   324→    // recipe is idempotent (dedupKey gateway_settlement:settlement:{ref}), so a
   325→    // crash here is healed by the next delivery's claim-miss path... which is
   326→    // why the call lives outside the tx: the PSP settles buyer funds to the
   327→    // platform bank account; T2 clears the GATEWAY:{provider} receivable into
   328→    // BANK_MAIN. Default setting false = dormant (documented decision).
   329→    if (await autoSettleEnabled()) {
   330→      await postGatewaySettlement({ provider: group.provider, amountMinor: group.totalMinor, ref: providerRef, occurredAt: paidAt });
   331→    }
   332→
   333→    return {
   334→      ok: true,
   335→      duplicate: false,
   336→      eventType,
   337→      orderNumber: group.orders[0]?.orderNumber ?? "",
   338→      orderNumbers: group.orders.map((o) => o.orderNumber),
   339→      orderStatus: "paid",
   340→      intentStatus: "succeeded",
   341→      isGroup: true,
   342→    };
   343→  }
   344→
   345→  // ------------------------------------------------------------------ legacy single-order path
   346→  if (!intent) {
   347→    throw new ApiError(404, "INTENT_NOT_FOUND", "Payment intent not found");
   348→  }
   349→
   350→  if (eventType === "payment.failed") {
   351→    if (intent.status === "succeeded") {
   352→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   353→      const order = await db.order.findUnique({ where: { id: intent.orderId } });
   354→      return {
   355→        ok: true,
   356→        duplicate: true,
   357→        eventType,
   358→        orderNumber: order?.orderNumber ?? "",
   359→        orderStatus: order?.status ?? "",
   360→        intentStatus: intent.status,
   361→      };
   362→    }
   363→    await db.paymentIntent.update({ where: { id: intent.id }, data: { status: "failed" } });
   364→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   365→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   366→    return {
   367→      ok: true,
   368→      duplicate: false,
   369→      eventType,
   370→      orderNumber: order?.orderNumber ?? "",
   371→      orderStatus: order?.status ?? "",
   372→      intentStatus: "failed",
   373→    };
   374→  }
   375→
   376→  if (eventType !== "payment.succeeded") {
   377→    throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   378→  }
   379→
   380→  if (amountMinor !== intent.amountMinor) {
   381→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match intent ${intent.amountMinor}`);
   382→  }
   383→
   384→  const settings = await getCommissionSettings();
   385→  const paidAt = new Date();
   386→
   387→  // SUCCESS PATH — same claim-first Serializable pattern as the group path.
   388→  const outcome = await db.$transaction(
   389→    async (tx) => {
   390→      const claim = await tx.paymentIntent.updateMany({
   391→        where: { id: intent.id, status: { not: "succeeded" } },
   392→        data: { status: "succeeded" },
   393→      });
   394→      if (claim.count === 0) return { claimed: false as const };
   395→
   396→      const order = await tx.order.findUnique({
   397→        where: { id: intent.orderId },
   398→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   399→      });
   400→      if (!order) throw new ApiError(404, "ORDER_NOT_FOUND", "Order for payment intent not found");
   401→
   402→      await transitionOrder({ orderId: order.id, toStatus: "paid", actor: "system", note: `Payment ${providerRef} succeeded via ${intent.provider}`, timestamps: { paidAt }, tx });
   403→
   404→      // Server-side cart clearing — only the PURCHASED variants leave the cart,
   405→      // and only on verified payment success (never on failure/pending, never
   406→      // client-driven). Account carts clear by shopperId; the token is a legacy/
   407→      // reconciliation fallback for rows that predate account-scoped carts.
   408→      // Selection-aware: unchecked (kept-for-later) lines are never removed here.
   409→      const purchasedVariantIds = [...new Set(order.items.map((it) => it.variantId).filter((v): v is string => Boolean(v)))];
   410→      if (order.shopperId) {
   411→        await tx.cartItem.deleteMany({
   412→          where: { cart: { shopperId: order.shopperId }, variantId: { in: purchasedVariantIds } },
   413→        });
   414→      } else if (order.cartToken) {
   415→        await tx.cartItem.deleteMany({
   416→          where: { cart: { token: order.cartToken }, variantId: { in: purchasedVariantIds } },
   417→        });
   418→      }
   419→
   420→      // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   421→      const seller = order.store.seller;
   422→      const wallet = seller?.wallet;
   423→      if (seller && wallet) {
   424→        const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   425→        const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   426→        const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   427→        const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   428→        await postOrderPaid({
   429→          orderId: order.id,
   430→          orderNumber: order.orderNumber,
   431→          walletId: wallet.id,
   432→          provider: intent.provider,
   433→          totalMinor: intent.amountMinor,
   434→          sellerNetMinor,
   435→          commissionMinor,
   436→          holdUntil,
   437→          currency: order.currency,
   438→          occurredAt: paidAt,
   439→          tx,
   440→        });
   441→      }
   442→
   443→      // Store webhooks + notifications (see group path rationale — dedupKeys).
   444→      await dispatchStoreWebhooks({
   445→        storeId: order.storeId,
   446→        event: "order.paid",
   447→        payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency },
   448→      });
   449→      if (seller) {
   450→        await notify({ event: "order.paid", channel: "email", recipient: seller.email, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   451→        await notify({ event: "order.paid", channel: "inapp", recipient: seller.email, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   452→      }
   453→      await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   454→      // Shopper in-app nudge (Task 10-d) — payment confirmed → "Order confirmed".
   455→      await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   456→      await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: intent.provider, totalMinor: intent.amountMinor } });
   457→
   458→      return { claimed: true as const };
   459→    },
   460→    { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
   461→  );
   462→
   463→  if (!outcome.claimed) {
   464→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   465→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   466→    return {
   467→      ok: true,
   468→      duplicate: true,
   469→      eventType,
   470→      orderNumber: order?.orderNumber ?? "",
   471→      orderStatus: order?.status ?? "",
   472→      intentStatus: "succeeded",
   473→    };
   474→  }
   475→
   476→  await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   477→
   478→  // T2 auto-settle (see group path) — after commit, idempotent recipe.
   479→  if (await autoSettleEnabled()) {
   480→    await postGatewaySettlement({ provider: intent.provider, amountMinor: intent.amountMinor, ref: providerRef, occurredAt: paidAt });
   481→  }
   482→
   483→  return {
   484→    ok: true,
   485→    duplicate: false,
   486→    eventType,
   487→    orderNumber: (await db.order.findUnique({ where: { id: intent.orderId } }))?.orderNumber ?? "",
   488→    orderStatus: "paid",
   489→    intentStatus: "succeeded",
   490→  };
   491→}
   492→
   493→/**
   494→ * FIN-2b (2b.4) — inbound PAYOUT lifecycle events from the PSP.
   495→ *
   496→ * The provider confirms (or fails) a bank transfer we initiated. providerRef
   497→ * resolution: try Payout.referenceNo first (the canonical platform reference —
   498→ * what the mock PSP and payout-simulate echo); fall back to Payout.providerRef
   499→ * (the transfer ref an admin recorded at mark-paid — some providers echo that
   500→ * instead). Both lookups are documented PSP integration surface.
   501→ *
   502→ * Same claim-first race safety as the payment paths: assertTransition guards
   503→ * the semantics, the conditional updateMany serializes concurrent deliveries,
   504→ * and T4/T4f post inside the same Serializable tx as the status flip.
   505→ */
   506→async function processPayoutEvent(args: {
   507→  provider: string;
   508→  providerRef: string;
   509→  eventType: "payout.succeeded" | "payout.failed";
   510→  amountMinor: number;
   511→  reason: string | null;
   512→  logId: string;
   513→}): Promise<ProcessResult> {
   514→  const payout =
   515→    (await db.payout.findUnique({ where: { referenceNo: args.providerRef } })) ??
   516→    (await db.payout.findFirst({ where: { providerRef: args.providerRef } }));
   517→  if (!payout) {
   518→    throw new ApiError(404, "PAYOUT_NOT_FOUND", `No payout matches providerRef ${args.providerRef}`);
   519→  }
   520→
   521→  const toStatus = args.eventType === "payout.succeeded" ? "paid" : "failed";
   522→  const dedupKey = `${toStatus === "paid" ? "payout.paid" : "payout.failed"}:${payout.id}`;
   523→
   524→  // Idempotent replay: the provider re-delivered an event we already applied
   525→  // (e.g. payout.succeeded for an already-paid payout) → ok + conflict flag,
   526→  // never a raw 409 at the provider. Decision is recorded (processed=true).
   527→  if (payout.status === toStatus) {
   528→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   529→    return {
   530→      ok: true,
   531→      duplicate: true,
   532→      payoutConflict: true,
   533→      eventType: args.eventType,
   534→      orderNumber: "",
   535→      orderStatus: payout.status,
   536→      intentStatus: "",
   537→      payoutStatus: payout.status,
   538→    };
   539→  }
   540→
   541→  // State machine guard (cheap, no locks): an illegal transition (e.g.
   542→  // requested→paid — the money was never reserved by T3) is a hard 409. The
   543→  // decision is recorded on the webhook row before throwing.
   544→  assertTransition(payout.status, toStatus);
   545→
   546→  // Amount guard BEFORE the claim (mirror of the payment paths).
   547→  if (args.amountMinor !== payout.amountMinor) {
   548→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${args.amountMinor} does not match payout ${payout.amountMinor}`);
   549→  }
   550→
   551→  const seller = await db.seller.findUnique({ where: { id: payout.sellerId } });
   552→  if (!seller) throw new ApiError(404, "NOT_FOUND", "Seller for payout not found");
   553→
   554→  const at = new Date();
   555→  // T4f reason: use the provider-supplied reason when present.
   556→  const reason = args.reason ?? "provider reported failure";
   557→  const subject = toStatus === "paid" ? `Payout ${payout.referenceNo} on the way` : `Payout ${payout.referenceNo} failed`;
   558→
   559→  const outcome = await db.$transaction(
   560→    async (tx) => {
   561→      // Conditional claim — expectedFrom is "processing" per the state machine
   562→      // (only processing payouts are awaiting provider confirmation). count 0
   563→      // ⇒ lost race / concurrent admin action ⇒ empty-tx-return (rollback-by-
   564→      // return: the claim was the only intended write and it wrote nothing).
   565→      const claim = await tx.payout.updateMany({
   566→        where: { id: payout.id, status: "processing" },
   567→        data: { status: toStatus },
   568→      });
   569→      if (claim.count === 0) return { claimed: false as const };
   570→
   571→      if (toStatus === "paid") {
   572→        await tx.payout.update({ where: { id: payout.id }, data: { paidAt: at } });
   573→        // T4 — clearing → bank; refreshWalletCache runs inside the recipe (tx).
   574→        await postPayoutConfirmed({
   575→          walletId: payout.walletId,
   576→          payoutId: payout.id,
   577→          referenceNo: payout.referenceNo,
   578→          amountMinor: payout.amountMinor,
   579→          tx,
   580→        });
   581→      } else {
   582→        await tx.payout.update({ where: { id: payout.id }, data: { failReason: reason } });
   583→        // T4f — reversal of T3 (clearing → back to seller payable).
   584→        await postPayoutFailed({
   585→          walletId: payout.walletId,
   586→          payoutId: payout.id,
   587→          referenceNo: payout.referenceNo,
   588→          amountMinor: payout.amountMinor,
   589→          reason,
   590→          tx,
   591→        });
   592→      }
   593→
   594→      // Seller notifications (email + inapp) with stable per-event dedupKeys;
   595→      // subjects mirror the admin mark-paid / fail routes.
   596→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "email", recipient: seller.email, subject, dedupKey });
   597→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "inapp", recipient: seller.email, subject, dedupKey });
   598→      await logActivity({
   599→        actorType: "system",
   600→        action: toStatus === "paid" ? "payout.provider_confirmed" : "payout.provider_failed",
   601→        entityType: "payout",
   602→        entityId: payout.id,
   603→        meta: { providerRef: args.providerRef, referenceNo: payout.referenceNo, amountMinor: payout.amountMinor },
   604→      });
   605→
   606→      return { claimed: true as const };
   607→    },
   608→    { isolationLevel: Prisma.TransactionIsolationLevel.Serializable }
   609→  );
   610→
   611→  const freshPayout = await db.payout.findUnique({ where: { id: payout.id } });
   612→  const payoutStatus = freshPayout?.status ?? payout.status;
   613→
   614→  if (!outcome.claimed) {
   615→    // Lost the race (e.g. an admin transitioned the payout concurrently) —
   616→    // answer ok with payoutConflict so the provider's retry converges; never
   617→    // throw raw 5xx at a PSP. Decision recorded (processed=true).
   618→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   619→    return {
   620→      ok: true,
   621→      duplicate: true,
   622→      payoutConflict: true,
   623→      eventType: args.eventType,
   624→      orderNumber: "",
   625→      orderStatus: payoutStatus,
   626→      intentStatus: "",
   627→      payoutStatus,
   628→    };
   629→  }
   630→
   631→  await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   632→  return {
   633→    ok: true,
   634→    duplicate: false,
   635→    eventType: args.eventType,
   636→    orderNumber: "",
   637→    orderStatus: payoutStatus,
   638→    intentStatus: "",
   639→    payoutStatus,
   640→  };
   641→}
   642→
   643→export function webhookPayloadJson(body: unknown): string {
   644→  return toJson(body);
   645→}
