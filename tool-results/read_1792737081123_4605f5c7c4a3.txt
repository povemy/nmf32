    24→async function variantId(slug: string): Promise<string> {
    25→  const v = await db.productVariant.findFirst({
    26→    where: { product: { slug, store: { slug: "lt-main-store" } } },
    27→    select: { id: true },
    28→  });
    29→  if (!v) throw new Error(`missing [LT] variant ${slug} — run seed first`);
    30→  return v.id;
    31→}

    33→async function buyerEmails(n: number): Promise<string[]> {
    34→  const rows = await db.shopperUser.findMany({
    35→    where: { email: { startsWith: "lt-buyer-", endsWith: "@lt.test" } },
    36→    select: { email: true },
    37→    orderBy: { email: "asc" },
    38→    take: n,
    39→  });
    40→  if (rows.length < n) throw new Error(`need ${n} [LT] buyers, found ${rows.length} — run seed`);
    41→  return rows.map((r) => r.email);
    42→}

    45→async function purchaseProfile() {
    46→  const storm = await variantId("lt-perf-storm-item");
    47→  const emails = await buyerEmails(BUYERS);
    48→  const stockBefore = (await db.productVariant.findUnique({ where: { id: storm }, select: { stock: true } }))!.stock;
    49→  const ordersBefore = await db.order.count({ where: { store: { slug: "lt-main-store" } } });
    50→
    51→  // Phase 1 — pre-login (NOT measured; excluded from latency metrics)
    52→  console.log(`[purchase] pre-login ${emails.length} buyers (conc 10)…`);
    53→  const loginT0 = performance.now();
    54→  const cookies = await mapLimit(emails, 10, (email) => loginShopper(email));
    55→  const loginWall = (performance.now() - loginT0) / 1000;
    56→
    57→  // Phase 2 — simultaneous checkout storm (MEASURED)
    58→  console.log(`[purchase] storm: ${emails.length} simultaneous create requests…`);
    59→  const t0 = performance.now();
    60→  const results = await Promise.all(
    61→    emails.map((email, i) =>
    62→      createCheckout({
    63→        cookie: cookies[i],
    64→        variantId: storm,
    65→        qty: 1,
    66→        idemKey: `lt-purchase-${runId}-${i}`,
    67→        email,
    68→      })
    69→    )
    70→  );
    71→  const createWall = (performance.now() - t0) / 1000;
    72→
    73→  const classes = { created: 0, business: 0, infra: 0, network: 0 };
    74→  const codes: Record<string, number> = {};
    75→  const created: { ref: string; email: string }[] = [];
    76→  for (const r of results) {
    77→    classes[classify(r.status)]++;
    78→    const code = r.body.error?.code ?? `${r.status}`;
    79→    codes[code] = (codes[code] ?? 0) + 1;
    80→    if (r.status === 201 && r.body.data?.paymentRef) created.push({ ref: r.body.data.paymentRef, email: r.body.data.paymentRef });
    81→  }
    82→
    83→  // Phase 3 — mock-PSP confirmations (conc 6, MEASURED separately)
    84→  console.log(`[purchase] confirmations: ${created.length} refs (conc 6)…`);
    85→  const confirmResults = await mapLimit(created, 6, (c) => simulatePayment(c.ref));
    86→  const confirmClasses = { ok: 0, business: 0, infra: 0, network: 0 };
    87→  for (const c of confirmResults) {
    88→    if (c.status === 200) confirmClasses.ok++;
    89→    else if (c.status >= 500 || c.status === 0) confirmClasses.infra++;
    90→    else confirmClasses.business++;
    91→  }
    92→
    93→  // Integrity snapshot
    94→  const stockAfter = (await db.productVariant.findUnique({ where: { id: storm }, select: { stock: true } }))!.stock;
    95→  const ordersAfter = await db.order.count({ where: { store: { slug: "lt-main-store" } } });
    96→  const newOrders = ordersAfter - ordersBefore;
    97→  const idemDupes = await db.paymentGroup.groupBy({ by: ["idemKey"], where: { idemKey: { startsWith: `lt-purchase-${runId}` } }, _count: { _all: true } });
    98→  const dupGroups = idemDupes.filter((g) => g._count._all > 1).length;
    99→  const newOrderRows = await db.order.findMany({
   100→    where: { store: { slug: "lt-main-store" }, createdAt: { gte: new Date(Date.now() - 15 * 60 * 1000) } },
   101→    select: { id: true, status: true, paymentGroupId: true },
   102→  });
   103→  const paidCount = newOrderRows.filter((o) => o.status === "paid").length;
   104→
   105→  const evidence = {
   106→    runId, profile: "purchase", label, base: BASE,
   107→    buyers: emails.length,
   108→    loginWallSec: Math.round(loginWall * 10) / 10,
   109→    create: { ...summarize(results), classes, codes, wallSec: Math.round(createWall * 10) / 10, ordersPerMin: Math.round((classes.created / createWall) * 60) },
   110→    confirm: { ...summarize(confirmResults.map((c) => ({ ms: c.ms, status: c.status }))), classes: confirmClasses },
   111→    integrity: {
   112→      stockBefore, stockAfter, stockDelta: stockBefore - stockAfter,
   113→      expectedDelta: classes.created,
   114→      stockDeltaMatches: stockBefore - stockAfter === classes.created,
   115→      newOrders, newOrdersMatchCreated: newOrders === classes.created,
   116→      duplicateIdempotencyGroups: dupGroups,
   117→      paidCount, createdCount: classes.created,
   118→    },
   119→    note: "dev-mode combined-environment Measured; login excluded from create latency; do not extrapolate to production",
   120→  };
   121→
   122→  console.log("\n=== PURCHASE PROFILE ===");
   123→  console.log(`create: ${JSON.stringify(evidence.create, null, 1)}`);
   124→  console.log(`confirm: ${JSON.stringify(evidence.confirm)}`);
   125→  console.log(`integrity: ${JSON.stringify(evidence.integrity)}`);
   126→  return evidence;
   127→}

   130→async function correctnessProfile() {
   131→  const checks: { name: string; ok: boolean; detail: string }[] = [];
   132→  const check = (name: string, ok: boolean, detail: string) => {
   133→    checks.push({ name, ok, detail });
   134→    console.log(`${ok ? "PASS" : "FAIL"}  ${name} — ${detail}`);
   135→  };
   136→
   137→  const emails = await buyerEmails(24);
   138→  const cookies = await mapLimit(emails, 10, (email) => loginShopper(email));
   139→
   140→  // G1 — last-unit race: 8 simultaneous buyers, stock=1
   141→  const lastUnit = await variantId("lt-perf-last-unit");
   142→  await db.productVariant.update({ where: { id: lastUnit }, data: { stock: 1, moq: 1 } });
   143→  const g1 = await Promise.all(emails.slice(0, 8).map((email, i) => createCheckout({ cookie: cookies[i], variantId: lastUnit, qty: 1, idemKey: `lt-g1-${runId}-${i}`, email })));
   144→  const g1Created = g1.filter((r) => r.status === 201).length;
   145→  const g1Rejected = g1.filter((r) => r.status === 409).length;
   146→  const g1Stock = (await db.productVariant.findUnique({ where: { id: lastUnit }, select: { stock: true } }))!.stock;
   147→  check("G1 last-unit race: exactly 1×201 + 7×409", g1Created === 1 && g1Rejected === 7, `${g1.map((r) => r.status).join("/")}`);
   148→  check("G1 stock never negative (0 left)", g1Stock === 0, `stock=${g1Stock}`);
   149→
   150→  // G2 — duplicate idempotency key: 5 concurrent identical requests
   151→  const storm = await variantId("lt-perf-storm-item");
   152→  const g2Key = `lt-g2-dup-${runId}`;
   153→  const g2 = await Promise.all(Array.from({ length: 5 }, () => createCheckout({ cookie: cookies[8], variantId: storm, qty: 1, idemKey: g2Key, email: emails[8] })));
   154→  const g2Refs = new Set(g2.map((r) => r.body.data?.paymentRef).filter(Boolean));
   155→  const g2Group = g2Refs.size === 1 ? await db.paymentGroup.findUnique({ where: { ref: [...g2Refs][0] }, include: { _count: { select: { orders: true } } } }) : null;
   156→  check("G2 same idempotency key ×5 concurrent: exactly 1 payment group", g2Refs.size === 1, `refs=${g2Refs.size}`);
   157→  check("G2 exactly 1 order created", g2Group?._count.orders === 1, `orders=${g2Group?._count.orders ?? 0}`);
   158→
   159→  // G3 — oversell guard: 20 buyers × qty1, stock=10
   160→  const oversell = await variantId("lt-perf-oversell-probe");
   161→  await db.productVariant.update({ where: { id: oversell }, data: { stock: 10, moq: 1 } });
   162→  const g3 = await Promise.all(emails.slice(0, 20).map((email, i) => createCheckout({ cookie: cookies[i], variantId: oversell, qty: 1, idemKey: `lt-g3-${runId}-${i}`, email })));
   163→  const g3Created = g3.filter((r) => r.status === 201).length;
   164→  const g3Rejected = g3.filter((r) => r.status === 409).length;
   165→  const g3Stock = (await db.productVariant.findUnique({ where: { id: oversell }, select: { stock: true } }))!.stock;
   166→  check("G3 oversell 20→10: exactly 10×201 + 10×409", g3Created === 10 && g3Rejected === 10, `${g3.map((r) => r.status).join("/")}`);
   167→  check("G3 stock exactly 0 (never negative)", g3Stock === 0, `stock=${g3Stock}`);
   168→
   169→  // G4 — [LT]-scoped ledger: every group created+confirmed has exactly one T1;
   170→  //       unconfirmed groups have none. (Payment confirmation runs below.)
   171→  const ltGroups = await db.paymentGroup.findMany({
   172→    where: { idemKey: { startsWith: `lt-g1-${runId}` } },
   173→    select: { ref: true, status: true, orders: { select: { id: true, status: true } } },
   174→  });
   175→  const winners = ltGroups.filter((g) => g.orders.some((o) => o.status === "payment_pending"));
   176→  if (winners.length === 1) {
   177→    const sim = await simulatePayment(winners[0].ref);
   178→    const order = winners[0].orders[0];
   179→    const t1 = await db.ledgerTransaction.count({ where: { txnType: "order_paid", refType: "order", refId: order.id } });
   180→    check("G4 confirmed [LT] order has exactly one T1 ledger posting", sim.status === 200 && t1 === 1, `simulate=${sim.status} T1=${t1}`);
   181→    const t1again = await simulatePayment(winners[0].ref);
   182→    const t1count2 = await db.ledgerTransaction.count({ where: { txnType: "order_paid", refType: "order", refId: order.id } });
   183→    check("G4 duplicate confirmation does NOT duplicate T1", t1again.status === 200 && t1count2 === 1, `T1=${t1count2}`);
   184→  } else {
   185→    check("G1 winner pending group found", false, `winners=${winners.length}`);
   186→  }
   187→
   188→  // G5 — global double-entry + wallet-cache truth (existing invariant runner)
   189→  const proc = Bun.spawnSync(["bun", "scripts/verify-ledger.ts"], { cwd: process.cwd() + "/", stdout: "pipe", stderr: "pipe" });
   190→  const out = proc.stdout.toString() + proc.stderr.toString();
   191→  check("G5 verify-ledger (balance + wallet truth==cache) ALL PASS", proc.exitCode === 0 && !out.includes("FAIL"), out.split("\n").filter((l) => l.includes("FAIL")).slice(0, 3).join(" | ") || "all pass");
   192→
   193→  // G6 — no [LT] order without a payment group; no negative stock anywhere [LT]
   194→  const orphan = await db.order.count({ where: { store: { slug: "lt-main-store" }, paymentGroupId: null } });
   195→  check("G6 no [LT] order without payment group", orphan === 0, `orphans=${orphan}`);
   196→  const negStock = await db.productVariant.count({ where: { product: { store: { slug: "lt-main-store" } }, stock: { lt: 0 } } });
   197→  check("G6 no negative [LT] stock", negStock === 0, `negative=${negStock}`);
   198→
   199→  const evidence = { runId, profile: "correctness", label, checks, passed: checks.filter((c) => c.ok).length, failed: checks.filter((c) => !c.ok).length };
   200→  console.log(`\n=== CORRECTNESS PROFILE: ${evidence.passed}/${checks.length} PASS ===`);
   201→  return evidence;
   202→}

   205→// F3 RESERVATION PROFILE — reservation lifecycle & expiry (mission Steps 6.1–15).
   206→// All rows are [LT]-owned; the 30-minute TTL is NEVER shortened — tests
   207→// backdate expiresAt on [LT] payment groups only (isolated-data manipulation).
   208→// ===========================================================================
   209→async function reservationProfile() {
   210→  const { releaseOrderReservation } = await import("../../src/lib/inventory-release");
   211→  const checks: { name: string; ok: boolean; detail: string }[] = [];
   212→  const check = (name: string, ok: boolean, detail: string) => {
   213→    checks.push({ name, ok, detail });
   214→    console.log(`${ok ? "PASS" : "FAIL"}  ${name} — ${detail}`);
   215→  };
   216→  const stockOf = async (id: string) => (await db.productVariant.findUnique({ where: { id }, select: { stock: true } }))!.stock;
   217→  const backdate = async (ref: string) => {
   218→    await db.paymentGroup.update({ where: { ref }, data: { expiresAt: new Date(Date.now() - 31 * 60 * 1000) } });
   219→  };
   220→  const emails = await buyerEmails(24);
   221→  const cookies = await mapLimit(emails, 10, (email) => loginShopper(email));
   222→  const tick = async () => {
   223→    const res = await fetch(`${BASE}/api/orders/expiry/tick`, { method: "POST" });
   224→    return (await res.json()) as { data?: { scanned: number; released: number; skipped: number; errors: { orderId: string; error: string }[] } };
   225→  };
   226→  // The worker's in-process throttle (1.5 s) is a load valve, not correctness —
   227→  // tests wait it out so every tick actually sweeps.
   228→  const tickFresh = async () => {
   229→    await new Promise((r) => setTimeout(r, 1_600));
   230→    return tick();
   231→  };
   232→
   233→  const storm = await variantId("lt-perf-storm-item");
   234→  const multiA = await variantId("lt-perf-multi-a");
   235→  const multiB = await variantId("lt-perf-multi-b");
   236→  const oversellV = await variantId("lt-perf-oversell-probe");
   237→
   238→  // R1 — successful payment commits the reservation without restoring stock
   239→  await db.productVariant.update({ where: { id: oversellV }, data: { stock: 5, moq: 1 } });
   240→  const s0 = await stockOf(oversellV);
   241→  const r1 = await createCheckout({ cookie: cookies[0], variantId: oversellV, qty: 2, idemKey: `lt-r1-${runId}`, email: emails[0] });
   242→  const afterCreate = await stockOf(oversellV);
   243→  const sim1 = await simulatePayment(r1.body.data!.paymentRef!);
   244→  const afterPay = await stockOf(oversellV);
   245→  const r1order = await db.order.findFirst({ where: { paymentGroup: { ref: r1.body.data!.paymentRef! } }, select: { id: true, status: true } });
   246→  check("R1 checkout deducts stock (5→3)", r1.status === 201 && s0 - afterCreate === 2, `${s0}→${afterCreate}`);
   247→  check("R1 payment commit keeps stock deducted (no restock)", sim1.status === 200 && afterPay === afterCreate && r1order?.status === "paid", `stock=${afterPay} status=${r1order?.status}`);
   248→
   249→  // R2 — definitive payment failure releases stock exactly once
   250→  const r2 = await createCheckout({ cookie: cookies[1], variantId: oversellV, qty: 2, idemKey: `lt-r2-${runId}`, email: emails[1] });
   251→  const beforeFail = await stockOf(oversellV);
   252→  const fail1 = await fetch(`${BASE}/api/checkout/payments/${r2.body.data!.paymentRef}/simulate`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ result: "failed" }) });
   253→  const afterFail = await stockOf(oversellV);
   254→  const r2order = await db.order.findFirst({ where: { paymentGroup: { ref: r2.body.data!.paymentRef! } }, select: { status: true } });
   255→  check("R2 payment.failed releases stock (+2) and cancels order", fail1.status === 200 && afterFail === beforeFail + 2 && r2order?.status === "cancelled", `stock ${beforeFail}→${afterFail}, order=${r2order?.status}`);
   256→  const fail2 = await fetch(`${BASE}/api/checkout/payments/${r2.body.data!.paymentRef}/simulate`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ result: "failed" }) });
   257→  const afterFail2 = await stockOf(oversellV);
   258→  check("R2 redelivered failure does NOT double-restock", fail2.status === 200 && afterFail2 === afterFail, `stock=${afterFail2}`);
   259→
   260→  // R3+R4 — expiry sweep releases; repeated sweeps never double-restock
   261→  const r3 = await createCheckout({ cookie: cookies[2], variantId: storm, qty: 3, idemKey: `lt-r3-${runId}`, email: emails[2] });
   262→  const r3StockBefore = await stockOf(storm);
   263→  await backdate(r3.body.data!.paymentRef!);
   264→  const t1 = await tickFresh();
   265→  const r3StockAfter = await stockOf(storm);
   266→  const r3order = await db.order.findFirst({ where: { paymentGroup: { ref: r3.body.data!.paymentRef! } }, select: { status: true } });
   267→  check("R3 sweep releases expired unpaid order (stock +3)", (t1.data?.released ?? 0) >= 1 && r3StockAfter === r3StockBefore + 3 && r3order?.status === "cancelled", `released=${t1.data?.released} stock ${r3StockBefore}→${r3StockAfter}`);
   268→  // R4 — running the sweep again must be a safe no-op (no double restock)
   269→  const t3 = await tickFresh();
   270→  check("R4 repeated sweeps release at most once (2nd sweep: 0 released)", t3.data?.released === 0 && (await stockOf(storm)) === r3StockAfter, `sweep3=${t3.data?.released} stock=${await stockOf(storm)}`);
   271→  check("R4 stock exactly restored, never exceeds capacity", (await stockOf(storm)) === r3StockBefore + 3, `stock=${await stockOf(storm)}`);
   272→
   273→  // R5 — two competing workers: exactly one release wins (direct primitive)
   274→  const r5 = await createCheckout({ cookie: cookies[3], variantId: storm, qty: 1, idemKey: `lt-r5-${runId}`, email: emails[3] });
   275→  const r5Before = await stockOf(storm);
   276→  await backdate(r5.body.data!.paymentRef!);
   277→  const orderId5 = (await db.order.findFirst({ where: { paymentGroup: { ref: r5.body.data!.paymentRef! } }, select: { id: true } }))!.id;
   278→  const [w1, w2] = await Promise.all([
   279→    releaseOrderReservation({ orderId: orderId5, actor: "worker-a", reason: "expired unpaid" }),
   280→    releaseOrderReservation({ orderId: orderId5, actor: "worker-b", reason: "expired unpaid" }),
   281→  ]);
   282→  const wins = [w1.released, w2.released].filter(Boolean).length;
   283→  const r5After = await stockOf(storm);
   284→  check("R5 two workers: exactly one release", wins === 1, `workers released: ${w1.released}/${w2.released}`);
   285→  check("R5 stock restored exactly +1 (no double)", r5After === r5Before + 1, `stock ${r5Before}→${r5After}`);
   286→
   287→  // R6 — confirmation vs expiry race: exactly one valid outcome
   288→  const r6 = await createCheckout({ cookie: cookies[4], variantId: storm, qty: 2, idemKey: `lt-r6-${runId}`, email: emails[4] });
   289→  const r6Ref = r6.body.data!.paymentRef!;
   290→  const r6Before = await stockOf(storm);
   291→  await backdate(r6Ref);
   292→  const [raceTick, racePay] = await Promise.all([tickFresh(), simulatePayment(r6Ref)]);
   293→  const r6order = await db.order.findFirst({ where: { paymentGroup: { ref: r6Ref } }, select: { status: true } });
   294→  const r6Stock = await stockOf(storm);
   295→  const paidOutcome = r6order?.status === "paid" && r6Stock === r6Before; // committed: stays deducted
   296→  const releasedOutcome = r6order?.status === "cancelled" && r6Stock === r6Before + 2; // released: restored
   297→  check("R6 payment/expiry race → exactly one valid outcome", paidOutcome !== releasedOutcome && racePay.status === 200, `order=${r6order?.status} stock=${r6Stock} (before=${r6Before})`);
   298→
   299→  // R8a — late success AFTER release with stock available → re-reserve → paid
   300→  const r8 = await createCheckout({ cookie: cookies[5], variantId: storm, qty: 2, idemKey: `lt-r8-${runId}`, email: emails[5] });
   301→  const r8Ref = r8.body.data!.paymentRef!;
   302→  const r8Base = await stockOf(storm);
   303→  await backdate(r8Ref);
   304→  await tickFresh();
   305→  const r8Released = await stockOf(storm);
   306→  const r8Pay = await simulatePayment(r8Ref);
   307→  const r8order = await db.order.findFirst({ where: { paymentGroup: { ref: r8Ref } }, select: { id: true, status: true } });
   308→  const r8Stock = await stockOf(storm);
   309→  const r8T1 = r8order ? await db.ledgerTransaction.count({ where: { txnType: "order_paid", refType: "order", refId: r8order.id } }) : 0;
   310→  check("R8a late success after release re-reserves → paid, stock net −2, one T1", r8Pay.status === 200 && r8order?.status === "paid" && r8Stock === r8Base && r8T1 === 1, `stock ${r8Base}→(release)${r8Released}→(repay)${r8Stock} T1=${r8T1}`);
   311→
   312→  // R8b — late success AFTER release with stock GONE → refund-required, never silent fulfillment
   313→  const r8b = await createCheckout({ cookie: cookies[6], variantId: oversellV, qty: 2, idemKey: `lt-r8b-${runId}`, email: emails[6] });
   314→  const r8bRef = r8b.body.data!.paymentRef!;
   315→  await backdate(r8bRef);
   316→  await tickFresh();
   317→  await db.productVariant.update({ where: { id: oversellV }, data: { stock: 0 } }); // stock disappears before the late payment
   318→  const r8bPay = await simulatePayment(r8bRef);
   319→  const r8border = await db.order.findFirst({ where: { paymentGroup: { ref: r8bRef } }, select: { id: true, status: true } });
   320→  const r8bRefund = r8border ? await db.refund.findFirst({ where: { orderId: r8border.id, reason: { startsWith: "late payment" } } }) : null;
   321→  const r8bT1 = r8border ? await db.ledgerTransaction.count({ where: { txnType: "order_paid", refType: "order", refId: r8border.id } }) : 0;
   322→  check("R8b late success with stock gone → refund-required (order stays cancelled, no T1, refund row exists)", r8bPay.status === 200 && r8border?.status === "cancelled" && !!r8bRefund && r8bT1 === 0, `order=${r8border?.status} refund=${!!r8bRefund} T1=${r8bT1}`);
   323→  await db.productVariant.update({ where: { id: oversellV }, data: { stock: 3 } }); // restore test baseline (5−2 paid R1)
   324→
   325→  // R10 — multi-item order releases the EXACT reserved quantities
   326→  const m0 = [await stockOf(storm), await stockOf(multiA), await stockOf(multiB)];
   327→  const r10 = await fetch(`${BASE}/api/checkout/create`, {
