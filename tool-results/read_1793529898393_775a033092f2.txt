    58→export function webhookSecret(): string {
    59→  const secret = process.env.PSP_WEBHOOK_SECRET;
    60→  if (secret && secret.length > 0) return secret;
    61→  if (process.env.NODE_ENV === "production") {
    62→    throw new ApiError(
    63→      503,
    64→      "WEBHOOK_SECRET_UNSET",
    65→      "PSP_WEBHOOK_SECRET is not configured — payment webhook processing is disabled (fail closed)"
    66→    );
    67→  }
    68→  return "sm_webhook_secret_demo";
    69→}

    76→export function expectedSignature(providerRef: string, eventType: string, amountMinor: number, ts: number): string {
    77→  return hmacSha256Hex(webhookSecret(), `${providerRef}.${eventType}.${amountMinor}.${ts}`);
    78→}

   115→/**
   116→ * Verify signature (caller may pre-compute it, as simulate does), log the raw
   117→ * webhook, then process idempotently. Throws ApiError with proper status codes.
   118→ *
   119→ * RACE SAFETY (why claim-first): the success paths' FIRST write inside the
   120→ * Serializable transaction is a CONDITIONAL updateMany (status flip guarded by
   121→ * the pre-status). Two concurrent deliveries of the same event race on that
   122→ * row lock; the loser matches zero rows, writes nothing and returns the
   123→ * duplicate result — money movement, order transitions and notifications can
   124→ * only ever run for the winner. The ledger dedupKeys (FIN-1) backstop this at
   125→ * the posting layer, and the notification dedupKeys backstop the mail log.
   126→ */
   127→export async function processPaymentWebhook(input: {
   128→  provider: string;
   129→  body: { providerRef?: string; eventType?: string; amountMinor?: number; ts?: number; sig?: string; reason?: string; paymentId?: string };
   130→  rawBody: string;
   131→  headerSignature?: string | null;
   132→  /** Phase 1 (E3): INTERNAL-ONLY recovery mode. Set exclusively by the
   133→   *  authorized admin payout-ops sweep when re-feeding an event whose
   134→   *  ORIGINAL delivery was signature-verified and persisted
   135→   *  (PaymentWebhookLog.signatureValid = true). The HMAC is STILL fully
   136→   *  verified against the stored payload — recovery NEVER forges or skips
   137→   *  signature verification — only the ±10-minute freshness window is
   138→   *  waived, because the event's age is precisely WHY it needs recovery.
   139→   *  External webhook routes never set this flag. */
   140→  opts?: { recovery?: boolean };
   141→}): Promise<ProcessResult> {
   142→  // (2b.2) Provider sandbox enforcement FIRST — fail closed. resolvePspProvider
   143→  // falls back to the mock provider for unknown ids, so in production an
   144→  // unregistered provider id can never sneak through: mock is sandboxOnly → 403.
   145→  const psp = resolvePspProvider(input.provider);
   146→  if (psp.sandboxOnly && process.env.NODE_ENV === "production") {
   147→    throw new ApiError(403, "PROVIDER_SANDBOX_ONLY", `Provider "${input.provider}" is sandbox-only and cannot process webhooks in production`);
   148→  }
   149→
   150→  const providerRef = String(input.body.providerRef ?? "");
   151→  const eventType = String(input.body.eventType ?? "");
   152→  const amountMinor = Math.round(Number(input.body.amountMinor ?? 0));
   153→  const ts = Math.round(Number(input.body.ts ?? 0));
   154→  if (!providerRef || !eventType) {
   155→    throw new ApiError(400, "BAD_REQUEST", "providerRef and eventType are required");
   156→  }
   157→
   158→  // (2b.1) Timing-safe compare + freshness. signatureValid in the log means
   159→  // "authenticated as a LIVE event": correct MAC AND inside the replay window —
   160→  // a stale-but-correctly-signed delivery is recorded as signatureValid=false
   161→  // (replay protection is part of webhook authentication semantics) and
   162→  // rejected with STALE_EVENT below, AFTER the row is logged.
   163→  // Phase 1 (E3): in recovery mode the row records the HMAC result; the
   164→  // freshness of the ORIGINAL receipt is what authorizes reprocessing.
   165→  const providedSig = input.headerSignature ?? (input.body.sig ? String(input.body.sig) : null);
   166→  const expected = expectedSignature(providerRef, eventType, amountMinor, ts);
   167→  const sigMatches = !!providedSig && safeEqual(providedSig, expected);
   168→  const fresh = Math.abs(Date.now() - ts) <= WEBHOOK_TS_WINDOW_MS;
   169→  const recovery = input.opts?.recovery === true;
   170→  const signatureValid = sigMatches && (fresh || recovery);
   171→
   172→  // OPTION A first: the ref may be a multi-seller payment group.
   173→  const group = await db.paymentGroup.findUnique({
   174→    where: { ref: providerRef },
   175→    include: {
   176→      orders: {
   177→        orderBy: { createdAt: "asc" },
   178→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   179→      },
   180→    },
   181→  });
   182→
   183→  const intent = group ? null : await db.paymentIntent.findUnique({ where: { providerRef } });
   184→
   185→  // ALWAYS log the webhook first (even for invalid signature / unknown ref) —
   186→  // invalid/stale rows stay in the log with their signatureValid flag.
   187→  const log = await db.paymentWebhookLog.create({
   188→    data: {
   189→      provider: input.provider,
   190→      providerRef,
   191→      eventType,
   192→      payloadJson: input.rawBody,
   193→      signatureValid,
   194→      processed: false,
   195→      orderId: group?.orders[0]?.id ?? intent?.orderId ?? null,
   196→    },
   197→  });
   198→
   199→  if (!sigMatches) {
   200→    throw new ApiError(400, "BAD_SIGNATURE", "Webhook signature verification failed");
   201→  }
   202→  if (!fresh && !recovery) {
   203→    throw new ApiError(400, "STALE_EVENT", `Webhook timestamp is outside the ±${WEBHOOK_TS_WINDOW_MS / 60_000}-minute acceptance window (replay protection)`);
   204→  }
   205→
   206→  // ------------------------------------------------------------------ FIN-2b: inbound payout events
   207→  if (eventType === "payout.succeeded" || eventType === "payout.failed") {
   208→    return processPayoutEvent({ provider: input.provider, providerRef, eventType, amountMinor, reason: input.body.reason ?? null, logId: log.id });
   209→  }
   210→
   211→  // ------------------------------------------------------------------ OPTION A group path
   212→  if (group) {
   213→    // LATE PAYMENT AFTER EXPIRY — defined recovery policy (F3 remediation):
   214→    // the expiry sweep has already claimed this group, transitioned its orders
   215→    // to a terminal unpaid state and RESTOCKED the items. Fulfilling now would
   216→    // sell unreserved stock; accepting the money without goods would steal.
   217→    // The ONLY safe response is a definitive refusal: no money moves, no order
   218→    // is resurrected, the PSP sees a permanent failure (mock PSP surfaces it
   219→    // to the buyer as an expired payment window). Never silently fulfill.
   220→    //
   221→    // DURABLE REFUND OBLIGATION (mission: final operational gaps): a REAL PSP
   222→    // has already CAPTURED the buyer's funds when this verified event arrives —
   223→    // the 409 below protects the order/inventory transition, it does not return
   224→    // those funds. Record the obligation BEFORE refusing so a crash between the
   225→    // two leaves at worst "case exists, webhook answered 5xx" → PSP redelivery
   226→    // converges on the unique (provider, providerRef) row. Re-reservation is
   227→    // deliberately NOT attempted (business rule: released stock may be re-sold;
   228→    // fulfilling a stale payment without buyer re-consent is not legitimate).
   229→    if (eventType === "payment.succeeded" && group.status === "expired") {
   230→      await recordLatePaymentObligation({
   231→        provider: input.provider,
   232→        providerRef,
   233→        providerPaymentId: input.body.paymentId ?? null,
   234→        amountMinor,
   235→        groupTotalMinor: group.totalMinor,
   236→        currency: group.currency,
   237→      });
   238→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   239→      throw new ApiError(409, "GROUP_EXPIRED", `Payment group ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   240→    }
   241→    if (eventType === "payment.failed") {
   242→      // No money moves on failure — a single conditional update is fine outside
   243→      // a Serializable tx (an already-succeeded group ignores the failed event).
   244→      if (group.status === "succeeded") {
   245→        await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   246→        return {
   247→          ok: true,
   248→          duplicate: true,
   249→          eventType,
   250→          orderNumber: group.orders[0]?.orderNumber ?? "",
   251→          orderNumbers: group.orders.map((o) => o.orderNumber),
   252→          orderStatus: "paid",
   253→          intentStatus: "succeeded",
   254→          isGroup: true,
   255→        };
   256→      }
   257→      await db.paymentGroup.update({ where: { id: group.id }, data: { status: "failed" } });
   258→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   259→      return {
   260→        ok: true,
   261→        duplicate: false,
   262→        eventType,
   263→        orderNumber: group.orders[0]?.orderNumber ?? "",
   264→        orderNumbers: group.orders.map((o) => o.orderNumber),
   265→        orderStatus: group.orders[0]?.status ?? "payment_pending",
   266→        intentStatus: "failed",
   267→        isGroup: true,
   268→      };
   269→    }
   270→
   271→    // Unknown-event + amount checks BEFORE the claim (unchanged order) — a
   272→    // malformed delivery must never even attempt the status flip.
   273→    if (eventType !== "payment.succeeded") {
   274→      throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   275→    }
   276→
   277→    if (amountMinor !== group.totalMinor) {
   278→      throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match payment group ${group.totalMinor}`);
   279→    }
   280→
   281→    const settings = await getCommissionSettings();
   282→    const paidAt = new Date();
   283→
   284→    // SUCCESS PATH — one Serializable transaction around claim + money.
   285→    // The conditional claim is the FIRST write: count 0 ⇒ another delivery
   286→    // already succeeded ⇒ nothing this tx wrote (the claim was the only
   287→    // intended write) ⇒ return the duplicate sentinel; committing an empty tx
   288→    // is equivalent to rollback-by-return.
   289→    //
   290→    // F1 REMEDIATION (measured): concurrent same-seller confirmations used to
   291→    // mass-abort here (P2034 SSI conflicts on the wallet-cache row — 0/15 in
   292→    // the measured storm). Two fixes, both evidence-driven:
   293→    //   1. the wallet-cache refresh moved OUT of this tx (post-commit,
   294→    //      best-effort) — the wallet row is no longer in this tx's conflict
   295→    //      domain; payout authorization never trusted the cache anyway;
   296→    //   2. bounded P2034 retry of the WHOLE transaction (2 retries, jittered
   297→    //      backoff). SSI conflicts are transient by definition; the claim's
   298→    //      conditional update + ledger dedupKeys make a full retry safe (no
   299→    //      double-posting is possible). PSP calls stay outside — nothing
   300→    //      non-transactional is ever retried.
   301→    const walletIdsTouched = new Set<string>();
   302→    // F1 [Measured]: notification/audit fan-out (notify, dispatchStoreWebhooks,
   303→    // logActivity, shopper nudge) used to run INSIDE the Serializable tx — but
   304→    // on the GLOBAL db client, i.e. on ADDITIONAL pool connections while the tx
   305→    // holds its own. Under concurrent confirmations that multiplies connection
   306→    // demand inside the tx window (15 webhooks × ≥2 connections vs pool of 10)
   307→    // → measured P2024 connection-starvation + P2028 (dev.log). Money + order
   308→    // state stay transactional; the non-transactional side effects fire
   309→    // POST-COMMIT. dedupKeys make a redelivery unable to multiply mail; a
   310→    // crash in the post-commit window can only delay/lose OBSERVABILITY rows,
   311→    // never financial state — the correct trade for a bounded money tx.
   312→    const postedOrders: { order: (typeof group.orders)[number]; sellerEmail: string | null }[] = [];
   313→    let outcome: { claimed: boolean } | undefined;
   314→    for (let attempt = 1; ; attempt++) {
   315→      try {
   316→        outcome = await db.$transaction(
   317→      async (tx) => {
   318→        // Bounded-failure policy (measured): a row lock held by another session
   319→        // stalls statements past Prisma's interactive-tx timer (measured 120 s
   320→        // = the server's 2-min statement_timeout). Session-scoped 10 s bound,
   321→        // self-reverting at commit/rollback — lock contention fails FAST, the
   322→        // PSP redelivers, and no pool connection is burned for 2 minutes.
   323→        await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   324→        const claim = await tx.paymentGroup.updateMany({
   325→          where: { id: group.id, status: { notIn: ["succeeded", "expired"] } },
   326→          data: { status: "succeeded", paidAt },
   327→        });
   328→        if (claim.count === 0) return { claimed: false as const };
   329→
   330→        for (const order of group.orders) {
   331→          await transitionOrder({
   332→            orderId: order.id,
   333→            toStatus: "paid",
   334→            actor: "system",
   335→            note: `Payment ${providerRef} succeeded via ${group.provider} (group payment)`,
   336→            timestamps: { paidAt },
   337→            tx,
   338→          });
   339→
   340→          // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   341→          const seller = order.store.seller;
   342→          const wallet = seller?.wallet;
   343→          if (seller && wallet) {
   344→            const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   345→            const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   346→            const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   347→            const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   348→            // AFFILIATE — seller-funded pool (T1 4th line; 0 ⇒ legacy 3-line
   349→            // posting stays byte-compatible) + the commission rows themselves
   350→            // (exactly once per order item: unique orderItemId idempotency).
   351→            const affiliateMinor = order.items.reduce((s, i) => s + i.affiliatePoolMinor, 0);
   352→            if (wallet) walletIdsTouched.add(wallet.id);
   353→            await postOrderPaid({
   354→              orderId: order.id,
   355→              orderNumber: order.orderNumber,
   356→              walletId: wallet.id,
   357→              provider: group.provider,
   358→              totalMinor: order.totalMinor,
   359→              sellerNetMinor,
   360→              commissionMinor,
   361→              affiliateMinor,
   362→              holdUntil,
   363→              currency: order.currency,
   364→              occurredAt: paidAt,
   365→              tx,
   366→              refreshCache: false, // post-commit below — keeps the wallet row out of this tx's conflict domain
   367→            });
   368→            await postAffiliateCommissionsForOrder(tx, order, paidAt);
   369→          }
   370→          postedOrders.push({ order, sellerEmail: seller?.email ?? null });
   371→        }
   372→
   373→        // Server-side cart clearing — only the PURCHASED variants leave the cart,
   374→        // and only on verified payment success (never on failure/pending). Lines
   375→        // the buyer unchecked (kept for later) survive: selection-aware carts.
   376→        // PaymentGroups are account-scoped (shopperId always set). Inside the tx:
   377→        // a rollback never strands paid items out of the cart inconsistently.
   378→        const purchasedVariantIds = [
   379→          ...new Set(group.orders.flatMap((o) => o.items.map((it) => it.variantId)).filter((v): v is string => Boolean(v))),
   380→        ];
   381→        await tx.cartItem.deleteMany({
   382→          where: { cart: { shopperId: group.shopperId }, variantId: { in: purchasedVariantIds } },
   383→        });
   384→
   385→        return { claimed: true as const };
   386→      },
   387→      // timeout: measured evidence (dev.log) — ~18-query Serializable tx over
   388→      // remote RTT exceeds Prisma's 5 s default under concurrency (P2028 ×13).
   389→      // 15 s is a BOUNDED policy, not the unbounded external-lock stall.
   390→      { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, timeout: 15_000 }
   391→        );
   392→        break;
   393→      } catch (e) {
   394→        if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2034" && attempt < 3) {
   395→          await new Promise((r) => setTimeout(r, 50 + Math.random() * 100));
   396→          continue;
   397→        }
   398→        // Phase 1 (E3): P2034 exhaustion → explicit 503 PAYMENT_TX_RETRY_EXHAUSTED
   399→        // (the provider retries later; the row stays processed=false for the
   400→        // ops sweep) instead of a raw 500 with no recovery contract.
   401→        if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2034") {
   402→          throw new ApiError(503, "PAYMENT_TX_RETRY_EXHAUSTED", "The payment could not be confirmed due to sustained concurrent activity — the provider should retry this delivery", { attempts: attempt });
   403→        }
   404→        throw e;
   405→      }
   406→    }
   407→
   408→    if (!outcome?.claimed) {
   409→      // Claim lost — WHY decides the answer. A concurrent delivery that
   410→      // succeeded → genuine duplicate (ok). But a group the expiry sweep
   411→      // claimed between our load and our claim must NOT be answered "ok":
   412→      // the money must not be captured for goods we already restocked.
   413→      // (Late-payment recovery policy — see the expired check above.)
   414→      const nowStatus = (await db.paymentGroup.findUnique({ where: { id: group.id }, select: { status: true } }))?.status;
   415→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   416→      if (nowStatus === "expired") {
   417→        // The sweep won the claim while this delivery was in flight — same
   418→        // late-payment policy as above, including the durable refund obligation.
   419→        await recordLatePaymentObligation({
   420→          provider: input.provider,
   421→          providerRef,
   422→          providerPaymentId: input.body.paymentId ?? null,
   423→          amountMinor,
   424→          groupTotalMinor: group.totalMinor,
   425→          currency: group.currency,
   426→        });
   427→        throw new ApiError(409, "GROUP_EXPIRED", `Payment group ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   428→      }
   429→      return {
   430→        ok: true,
   431→        duplicate: true,
   432→        eventType,
   433→        orderNumber: group.orders[0]?.orderNumber ?? "",
   434→        orderNumbers: group.orders.map((o) => o.orderNumber),
   435→        orderStatus: "paid",
   436→        intentStatus: "succeeded",
   437→        isGroup: true,
   438→      };
   439→    }
   440→
   441→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   442→
   443→    // POST-COMMIT fan-out (see F1 note above) — non-transactional side effects.
   444→    for (const { order, sellerEmail } of postedOrders) {
   445→      try {
   446→        await dispatchStoreWebhooks({
   447→          storeId: order.storeId,
   448→          event: "order.paid",
   449→          payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency, paymentGroupRef: providerRef },
   450→        });
   451→        if (sellerEmail) {
   452→          await notify({ event: "order.paid", channel: "email", recipient: sellerEmail, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   453→          await notify({ event: "order.paid", channel: "inapp", recipient: sellerEmail, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   454→        }
   455→        await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   456→        await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   457→        await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: group.provider, totalMinor: order.totalMinor, paymentGroup: true } });
   458→      } catch (e) {
   459→        // Observability/notification loss never fails the money path.
   460→        console.error(`[post-commit] fan-out failed for order ${order.orderNumber} (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   461→      }
   462→    }
   463→
   464→    // F1 — post-commit wallet-cache refresh (best-effort, OUTSIDE the money
   465→    // tx). A failure here leaves the display cache stale — classified as
   466→    // benign time-maturity/event staleness by the ledger invariant, invisible
   467→    // to payout authorization (which recomputes truth transactionally), and
   468→    // self-healing on the next wallet event or wallet-page read.
   469→    for (const wid of walletIdsTouched) {
   470→      try {
   471→        await refreshWalletCache(wid);
   472→      } catch (e) {
   473→        console.error(`[wallet-cache] post-commit refresh failed for ${wid} (benign, self-heals):`, e instanceof Error ? e.message : e);
   474→      }
   475→    }
   476→
   477→    // (FIN-1 1.6 wiring) T2 auto-settle — AFTER the success tx commits. The
   478→    // recipe is idempotent (dedupKey gateway_settlement:settlement:{ref}), so a
   479→    // crash here is healed by the next delivery's claim-miss path... which is
   480→    // why the call lives outside the tx: the PSP settles buyer funds to the
   481→    // platform bank account; T2 clears the GATEWAY:{provider} receivable into
   482→    // BANK_MAIN. Default setting false = dormant (documented decision).
   483→    if (await autoSettleEnabled()) {
   484→      await postGatewaySettlement({ provider: group.provider, amountMinor: group.totalMinor, ref: providerRef, occurredAt: paidAt });
   485→    }
   486→
   487→    return {
   488→      ok: true,
   489→      duplicate: false,
   490→      eventType,
   491→      orderNumber: group.orders[0]?.orderNumber ?? "",
   492→      orderNumbers: group.orders.map((o) => o.orderNumber),
   493→      orderStatus: "paid",
   494→      intentStatus: "succeeded",
   495→      isGroup: true,
   496→    };
   497→  }
   498→
   499→  // ------------------------------------------------------------------ legacy single-order path
   500→  if (!intent) {
   501→    throw new ApiError(404, "INTENT_NOT_FOUND", "Payment intent not found");
   502→  }
   503→
   504→  if (eventType === "payment.failed") {
   505→    if (intent.status === "succeeded") {
   506→      await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   507→      const order = await db.order.findUnique({ where: { id: intent.orderId } });
   508→      return {
   509→        ok: true,
   510→        duplicate: true,
   511→        eventType,
   512→        orderNumber: order?.orderNumber ?? "",
   513→        orderStatus: order?.status ?? "",
   514→        intentStatus: intent.status,
   515→      };
   516→    }
   517→    await db.paymentIntent.update({ where: { id: intent.id }, data: { status: "failed" } });
   518→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   519→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   520→    return {
   521→      ok: true,
   522→      duplicate: false,
   523→      eventType,
   524→      orderNumber: order?.orderNumber ?? "",
   525→      orderStatus: order?.status ?? "",
   526→      intentStatus: "failed",
   527→    };
   528→  }
   529→
   530→  if (eventType !== "payment.succeeded") {
   531→    throw new ApiError(400, "UNKNOWN_EVENT", `Unsupported eventType: ${eventType}`);
   532→  }
   533→
   534→  if (amountMinor !== intent.amountMinor) {
   535→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${amountMinor} does not match intent ${intent.amountMinor}`);
   536→  }
   537→
   538→  // Phase 1 (E1) — late-payment guard for LEGACY single-intent orders, mirroring
   539→  // the group path's GROUP_EXPIRED policy: if the order already left
   540→  // payment_pending because the expiry sweep released its stock (cancelled /
   541→  // payment_expired — the sweep now covers legacy intents), the payment can no
   542→  // longer be fulfilled. Refuse, record the durable refund obligation, and let
   543→  // the PSP refund per its own cadence. transitionOrder is unconditional, so
   544→  // WITHOUT this guard a late webhook would resurrect a released order
   545→  // (double-sell). Money never moves on refusal.
   546→  const guardOrder = await db.order.findUnique({
   547→    where: { id: intent.orderId },
   548→    select: { id: true, status: true, currency: true },
   549→  });
   550→  if (guardOrder && (guardOrder.status === "payment_expired" || guardOrder.status === "cancelled")) {
   551→    await recordLatePaymentObligation({
   552→      provider: input.provider,
   553→      providerRef,
   554→      providerPaymentId: input.body.paymentId ?? null,
   555→      amountMinor,
   556→      groupTotalMinor: intent.amountMinor,
   557→      currency: guardOrder.currency,
   558→    });
   559→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   560→    throw new ApiError(409, "ORDER_EXPIRED", `Order for payment ${providerRef} expired before payment completed — no money was moved; the buyer must start a new checkout`);
   561→  }
   562→
   563→  const settings = await getCommissionSettings();
   564→  const paidAt = new Date();
   565→
   566→  // SUCCESS PATH — same claim-first Serializable pattern as the group path,
   567→  // with the same F1 remediation (post-commit wallet refresh + bounded P2034
   568→  // retry of the whole transaction).
   569→  const walletIdsTouched = new Set<string>();
   570→  // NOTE: array holder (same pattern as the group path's postedOrders) — a
   571→  // `let x: T | null = null` assigned inside the $transaction callback gets
   572→  // CFA-narrowed to `null` at post-commit use sites (TS never sees the callback
   573→  // assignment), making `if (postedOrder)` `never` and every property access an
   574→  // error. Array contents are not flow-narrowed, so the post-commit fan-out
   575→  // below reads the real types.
   576→  const postedOrderRef: { order: Prisma.OrderGetPayload<{ include: { items: true; store: { include: { seller: { include: { wallet: true } } } } } }>; sellerEmail: string | null }[] = [];
   577→  let outcome: { claimed: boolean } | undefined;
   578→  for (let attempt = 1; ; attempt++) {
   579→    try {
   580→      outcome = await db.$transaction(
   581→    async (tx) => {
   582→      // Bounded-failure policy (see group path) — 10 s per-statement bound.
   583→      await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   584→      const claim = await tx.paymentIntent.updateMany({
   585→        where: { id: intent.id, status: { not: "succeeded" } },
   586→        data: { status: "succeeded" },
   587→      });
   588→      if (claim.count === 0) return { claimed: false as const };
   589→
   590→      const order = await tx.order.findUnique({
   591→        where: { id: intent.orderId },
   592→        include: { items: true, store: { include: { seller: { include: { wallet: true } } } } },
   593→      });
   594→      if (!order) throw new ApiError(404, "ORDER_NOT_FOUND", "Order for payment intent not found");
   595→
   596→      await transitionOrder({ orderId: order.id, toStatus: "paid", actor: "system", note: `Payment ${providerRef} succeeded via ${intent.provider}`, timestamps: { paidAt }, tx });
   597→
   598→      // Server-side cart clearing — only the PURCHASED variants leave the cart,
   599→      // and only on verified payment success (never on failure/pending, never
   600→      // client-driven). Account carts clear by shopperId; the token is a legacy/
   601→      // reconciliation fallback for rows that predate account-scoped carts.
   602→      // Selection-aware: unchecked (kept-for-later) lines are never removed here.
   603→      const purchasedVariantIds = [...new Set(order.items.map((it) => it.variantId).filter((v): v is string => Boolean(v)))];
   604→      if (order.shopperId) {
   605→        await tx.cartItem.deleteMany({
   606→          where: { cart: { shopperId: order.shopperId }, variantId: { in: purchasedVariantIds } },
   607→        });
   608→      } else if (order.cartToken) {
   609→        await tx.cartItem.deleteMany({
   610→          where: { cart: { token: order.cartToken }, variantId: { in: purchasedVariantIds } },
   611→        });
   612→      }
   613→
   614→      // Ledger T1 — seller wallet credited with net, hold until = paidAt + holdDays.
   615→      const seller = order.store.seller;
   616→      const wallet = seller?.wallet;
   617→      if (seller && wallet) {
   618→        const holdDays = seller.holdDaysOverride ?? settings.payoutHoldDays;
   619→        const holdUntil = new Date(paidAt.getTime() + holdDays * 86400000);
   620→        const commissionMinor = order.items.reduce((s, i) => s + i.commissionMinor, 0);
   621→        const sellerNetMinor = order.items.reduce((s, i) => s + i.sellerNetMinor, 0);
   622→        // AFFILIATE — pool + commission rows (see group-path note).
   623→        const affiliateMinor = order.items.reduce((s, i) => s + i.affiliatePoolMinor, 0);
   624→        walletIdsTouched.add(wallet.id);
   625→        await postOrderPaid({
   626→          orderId: order.id,
   627→          orderNumber: order.orderNumber,
   628→          walletId: wallet.id,
   629→          provider: intent.provider,
   630→          totalMinor: intent.amountMinor,
   631→          sellerNetMinor,
   632→          commissionMinor,
   633→          affiliateMinor,
   634→          holdUntil,
   635→          currency: order.currency,
   636→          occurredAt: paidAt,
   637→          tx,
   638→          refreshCache: false, // post-commit below (F1)
   639→        });
   640→        await postAffiliateCommissionsForOrder(tx, order, paidAt);
   641→      }
   642→
   643→      // Notification/audit fan-out moved POST-COMMIT (group-path F1 note):
   644→      // these ran on the global db client inside the tx window, multiplying
   645→      // pool-connection demand under concurrency (measured P2024/P2028).
   646→      postedOrderRef.push({ order, sellerEmail: seller?.email ?? null });
   647→
   648→      return { claimed: true as const };
   649→    },
   650→    // timeout: same measured P2028 evidence as the group path.
   651→    { isolationLevel: Prisma.TransactionIsolationLevel.Serializable, timeout: 15_000 }
   652→      );
   653→      break;
   654→    } catch (e) {
   655→      if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === "P2034" && attempt < 3) {
   656→        await new Promise((r) => setTimeout(r, 50 + Math.random() * 100));
   657→        continue;
   658→      }
   659→      throw e;
   660→    }
   661→  }
   662→
   663→  if (!outcome?.claimed) {
   664→    await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   665→    const order = await db.order.findUnique({ where: { id: intent.orderId } });
   666→    return {
   667→      ok: true,
   668→      duplicate: true,
   669→      eventType,
   670→      orderNumber: order?.orderNumber ?? "",
   671→      orderStatus: order?.status ?? "",
   672→      intentStatus: "succeeded",
   673→    };
   674→  }
   675→
   676→  await db.paymentWebhookLog.update({ where: { id: log.id }, data: { processed: true } });
   677→
   678→  // POST-COMMIT fan-out (see group-path F1 note) — non-transactional side effects.
   679→  for (const { order, sellerEmail } of postedOrderRef) {
   680→    try {
   681→      await dispatchStoreWebhooks({
   682→        storeId: order.storeId,
   683→        event: "order.paid",
   684→        payload: { orderNumber: order.orderNumber, totalMinor: order.totalMinor, currency: order.currency },
   685→      });
   686→      if (sellerEmail) {
   687→        await notify({ event: "order.paid", channel: "email", recipient: sellerEmail, subject: `You have a new paid order ${order.orderNumber}`, dedupKey: `order.paid.email:${order.id}` });
   688→        await notify({ event: "order.paid", channel: "inapp", recipient: sellerEmail, subject: "You have a new paid order", dedupKey: `order.paid.inapp:${order.id}` });
   689→      }
   690→      await notify({ event: "order.receipt", channel: "email", recipient: order.customerEmail, subject: `Your order ${order.orderNumber} is confirmed`, dedupKey: `order.receipt.email:${order.id}` });
   691→      await notifyShopperOrderConfirmed({ orderNumber: order.orderNumber, shopperId: order.shopperId, storeId: order.storeId, store: order.store });
   692→      await logActivity({ actorType: "system", action: "payment.succeeded", entityType: "order", entityId: order.id, meta: { providerRef, provider: intent.provider, totalMinor: intent.amountMinor } });
   693→    } catch (e) {
   694→      console.error(`[post-commit] fan-out failed for order ${order.orderNumber} (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   695→    }
   696→  }
   697→
   698→  // F1 — post-commit wallet-cache refresh (same rationale as the group path).
   699→  for (const wid of walletIdsTouched) {
   700→    try {
   701→      await refreshWalletCache(wid);
   702→    } catch (e) {
   703→      console.error(`[wallet-cache] post-commit refresh failed for ${wid} (benign, self-heals):`, e instanceof Error ? e.message : e);
   704→    }
   705→  }
   706→
   707→  // T2 auto-settle (see group path) — after commit, idempotent recipe.
   708→  if (await autoSettleEnabled()) {
   709→    await postGatewaySettlement({ provider: intent.provider, amountMinor: intent.amountMinor, ref: providerRef, occurredAt: paidAt });
   710→  }
   711→
   712→  return {
   713→    ok: true,
   714→    duplicate: false,
   715→    eventType,
   716→    orderNumber: (await db.order.findUnique({ where: { id: intent.orderId } }))?.orderNumber ?? "",
   717→    orderStatus: "paid",
   718→    intentStatus: "succeeded",
   719→  };
   720→}
   721→
   722→/**
   723→ * FIN-2b (2b.4) — inbound PAYOUT lifecycle events from the PSP.
   724→ *
   725→ * The provider confirms (or fails) a bank transfer we initiated. providerRef
   726→ * resolution: try Payout.referenceNo first (the canonical platform reference —
   727→ * what the mock PSP and payout-simulate echo); fall back to Payout.providerRef
   728→ * (the transfer ref an admin recorded at mark-paid — some providers echo that
   729→ * instead). Both lookups are documented PSP integration surface.
   730→ *
   731→ * Same claim-first race safety as the payment paths: assertTransition guards
   732→ * the semantics, the conditional updateMany serializes concurrent deliveries,
   733→ * and T4/T4f post inside the same Serializable tx as the status flip.
   734→ */
   735→async function processPayoutEvent(args: {
   736→  provider: string;
   737→  providerRef: string;
   738→  eventType: "payout.succeeded" | "payout.failed";
   739→  amountMinor: number;
   740→  reason: string | null;
   741→  logId: string;
   742→}): Promise<ProcessResult> {
   743→  const payout =
   744→    (await db.payout.findUnique({ where: { referenceNo: args.providerRef } })) ??
   745→    (await db.payout.findFirst({ where: { providerRef: args.providerRef } }));
   746→  if (!payout) {
   747→    throw new ApiError(404, "PAYOUT_NOT_FOUND", `No payout matches providerRef ${args.providerRef}`);
   748→  }
   749→
   750→  const toStatus = args.eventType === "payout.succeeded" ? "paid" : "failed";
   751→  const dedupKey = `${toStatus === "paid" ? "payout.paid" : "payout.failed"}:${payout.id}`;
   752→
   753→  // Idempotent replay: the provider re-delivered an event we already applied
   754→  // (e.g. payout.succeeded for an already-paid payout) → ok + conflict flag,
   755→  // never a raw 409 at the provider. Decision is recorded (processed=true).
   756→  if (payout.status === toStatus) {
   757→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   758→    return {
   759→      ok: true,
   760→      duplicate: true,
   761→      payoutConflict: true,
   762→      eventType: args.eventType,
   763→      orderNumber: "",
   764→      orderStatus: payout.status,
   765→      intentStatus: "",
   766→      payoutStatus: payout.status,
   767→    };
   768→  }
   769→
   770→  // State machine guard (cheap, no locks): an illegal transition (e.g.
   771→  // requested→paid — the money was never reserved by T3) is a hard 409. The
   772→  // decision is recorded on the webhook row before throwing.
   773→  assertTransition(payout.status, toStatus);
   774→
   775→  // Amount guard BEFORE the claim (mirror of the payment paths).
   776→  if (args.amountMinor !== payout.amountMinor) {
   777→    throw new ApiError(409, "AMOUNT_MISMATCH", `Webhook amount ${args.amountMinor} does not match payout ${payout.amountMinor}`);
   778→  }
   779→
   780→  const seller = await db.seller.findUnique({ where: { id: payout.sellerId } });
   781→  if (!seller) throw new ApiError(404, "NOT_FOUND", "Seller for payout not found");
   782→
   783→  const at = new Date();
   784→  // T4f reason: use the provider-supplied reason when present.
   785→  const reason = args.reason ?? "provider reported failure";
   786→  const subject = toStatus === "paid" ? `Payout ${payout.referenceNo} on the way` : `Payout ${payout.referenceNo} failed`;
   787→
   788→  // Phase 1 (E2): the SAME hardening the payment paths already carry —
   789→  //   • withSerializableTxRetry: bounded P2034 retry of the WHOLE tx with
   790→  //     jittered exponential backoff, and 503 PAYMENT_TX_RETRY_EXHAUSTED on
   791→  //     exhaustion (the PSP retries later; the delivery stays
   792→  //     processed=false for this sweep) instead of a raw 500.
   793→  //   • SET LOCAL statement_timeout = 10 s bounds lock waits.
   794→  //   • notification/audit fan-out moved POST-COMMIT (the F1 anti-pattern:
   795→  //     extra round-trips on the global client INSIDE the Serializable tx
   796→  //     multiply pool pressure and hot-row contention).
   797→  //   • timeout/maxWait (10 s/10 s, env-tunable) replace Prisma's 5 s defaults.
   798→  // Retry-safety: the claim is the first write and the only gate; a retried tx
   799→  // whose claim matches 0 rows is a no-op. Ledger dedupKeys backstop T4/T4f.
   800→  const outcome = await withSerializableTxRetry(async (tx) => {
   801→    await tx.$executeRawUnsafe("SET LOCAL statement_timeout = '10s'");
   802→    // Conditional claim — expectedFrom is "processing" per the state machine
   803→    // (only processing payouts are awaiting provider confirmation). count 0
   804→    // ⇒ lost race / concurrent admin action ⇒ empty-tx-return (rollback-by-
   805→    // return: the claim was the only intended write and it wrote nothing).
   806→    const claim = await tx.payout.updateMany({
   807→      where: { id: payout.id, status: "processing" },
   808→      data: { status: toStatus },
   809→    });
   810→    if (claim.count === 0) return { claimed: false as const };
   811→
   812→    if (toStatus === "paid") {
   813→      await tx.payout.update({ where: { id: payout.id }, data: { paidAt: at } });
   814→      // T4 — clearing → bank; refreshWalletCache runs inside the recipe (tx).
   815→      await postPayoutConfirmed({
   816→        walletId: payout.walletId,
   817→        payoutId: payout.id,
   818→        referenceNo: payout.referenceNo,
   819→        amountMinor: payout.amountMinor,
   820→        tx,
   821→      });
   822→    } else {
   823→      await tx.payout.update({ where: { id: payout.id }, data: { failReason: reason } });
   824→      // T4f — reversal of T3 (clearing → back to seller payable).
   825→      await postPayoutFailed({
   826→        walletId: payout.walletId,
   827→        payoutId: payout.id,
   828→        referenceNo: payout.referenceNo,
   829→        amountMinor: payout.amountMinor,
   830→        reason,
   831→        tx,
   832→      });
   833→    }
   834→
   835→    return { claimed: true as const };
   836→  });
   837→
   838→  if (outcome.claimed) {
   839→    // Phase 1 (E2): post-commit fan-out — seller notifications + audit live
   840→    // OUTSIDE the Serializable tx. Stable per-event dedupKeys make a redelivery
   841→    // converge; a failure here is logged and non-financial (never rolls back
   842→    // committed ledger work).
   843→    try {
   844→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "email", recipient: seller.email, subject, dedupKey });
   845→      await notify({ event: toStatus === "paid" ? "payout.paid" : "payout.failed", channel: "inapp", recipient: seller.email, subject, dedupKey });
   846→      await logActivity({
   847→        actorType: "system",
   848→        action: toStatus === "paid" ? "payout.provider_confirmed" : "payout.provider_failed",
   849→        entityType: "payout",
   850→        entityId: payout.id,
   851→        meta: { providerRef: args.providerRef, referenceNo: payout.referenceNo, amountMinor: payout.amountMinor },
   852→      });
   853→    } catch (e) {
   854→      console.error(`[post-commit] payout ${payout.referenceNo} fan-out failed (non-financial, dedupKey-guarded):`, e instanceof Error ? e.message : e);
   855→    }
   856→  }
   857→
   858→  const freshPayout = await db.payout.findUnique({ where: { id: payout.id } });
   859→  const payoutStatus = freshPayout?.status ?? payout.status;
   860→
   861→  if (!outcome.claimed) {
   862→    // Lost the race (e.g. an admin transitioned the payout concurrently) —
   863→    // answer ok with payoutConflict so the provider's retry converges; never
   864→    // throw raw 5xx at a PSP. Decision recorded (processed=true).
   865→    await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   866→    return {
   867→      ok: true,
   868→      duplicate: true,
   869→      payoutConflict: true,
   870→      eventType: args.eventType,
   871→      orderNumber: "",
   872→      orderStatus: payoutStatus,
   873→      intentStatus: "",
   874→      payoutStatus,
   875→    };
   876→  }
   877→
   878→  await db.paymentWebhookLog.update({ where: { id: args.logId }, data: { processed: true } });
   879→  return {
   880→    ok: true,
   881→    duplicate: false,
   882→    eventType: args.eventType,
   883→    orderNumber: "",
   884→    orderStatus: payoutStatus,
   885→    intentStatus: "",
   886→    payoutStatus,
   887→  };
   888→}

   890→export function webhookPayloadJson(body: unknown): string {
   891→  return toJson(body);
