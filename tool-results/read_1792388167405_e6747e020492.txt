    93→    93→/** Total unread seller messages — powers the "My messages" header badge.
    94→    94→ *  Polls lightly + refreshes when the messages view is opened/left. */
    95→    95→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    96→    96→  const [unread, setUnread] = useState(0);
    97→    97→  useEffect(() => {
    98→    98→    if (!shopperSignedIn) {
    99→    99→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
   100→   100→      setUnread(0);
   101→   101→      return;
   102→   102→    }
   103→   103→    let alive = true;
   104→   104→    const load = () =>
   105→   105→      fetchJson<{ total: number }>("/api/chat/unread")
   106→   106→        .then((d) => alive && setUnread(d.total ?? 0))
   107→   107→        .catch(() => {});
   108→   108→    load();
   109→   109→    const t = setInterval(load, 15000);
   110→   110→    return () => {
   111→   111→      alive = false;
   112→   112→      clearInterval(t);
   113→   113→    };
   114→   114→  }, [shopperSignedIn]);
   115→   115→  // instant re-sync when a thread is read inside the messages view (the view
   116→   116→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   117→   117→  useEffect(() => {
   118→   118→    if (!shopperSignedIn) return;
   119→   119→    const sync = () =>
   120→   120→      fetchJson<{ total: number }>("/api/chat/unread")
   121→   121→        .then((d) => setUnread(d.total ?? 0))
   122→   122→        .catch(() => {});
   123→   123→    window.addEventListener("nomofee:chat-sync", sync);
   124→   124→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   125→   125→  }, [shopperSignedIn]);
   126→   126→  // opening (or leaving) the messages view should immediately re-sync the badge
   127→   127→  useEffect(() => {
   128→   128→    if (!shopperSignedIn) return;
   129→   129→    const t = setTimeout(
   130→   130→      () =>
   131→   131→        fetchJson<{ total: number }>("/api/chat/unread")
   132→   132→          .then((d) => setUnread(d.total ?? 0))
   133→   133→          .catch(() => {}),
   134→   134→      viewName === "messages" ? 600 : 0
   135→   135→    );
   136→   136→    return () => clearTimeout(t);
   137→   137→  }, [shopperSignedIn, viewName]);
   138→   138→  return unread;
   139→   139→}
