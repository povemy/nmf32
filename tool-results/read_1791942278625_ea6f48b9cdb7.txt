    79→/** Total unread seller messages — powers the "My messages" header badge.
    80→ *  Polls lightly + refreshes when the messages view is opened/left. */
    81→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    82→  const [unread, setUnread] = useState(0);
    83→  useEffect(() => {
    84→    if (!shopperSignedIn) {
    85→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    86→      setUnread(0);
    87→      return;
    88→    }
    89→    let alive = true;
    90→    const load = () =>
    91→      fetchJson<{ total: number }>("/api/chat/unread")
    92→        .then((d) => alive && setUnread(d.total ?? 0))
    93→        .catch(() => {});
    94→    load();
    95→    const t = setInterval(load, 15000);
    96→    return () => {
    97→      alive = false;
    98→      clearInterval(t);
    99→    };
   100→  }, [shopperSignedIn]);
   101→  // instant re-sync when a thread is read inside the messages view (the view
   102→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   103→  useEffect(() => {
   104→    if (!shopperSignedIn) return;
   105→    const sync = () =>
   106→      fetchJson<{ total: number }>("/api/chat/unread")
   107→        .then((d) => setUnread(d.total ?? 0))
   108→        .catch(() => {});
   109→    window.addEventListener("shopmesh:chat-sync", sync);
   110→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   111→  }, [shopperSignedIn]);
   112→  // opening (or leaving) the messages view should immediately re-sync the badge
   113→  useEffect(() => {
   114→    if (!shopperSignedIn) return;
   115→    const t = setTimeout(
   116→      () =>
   117→        fetchJson<{ total: number }>("/api/chat/unread")
   118→          .then((d) => setUnread(d.total ?? 0))
   119→          .catch(() => {}),
   120→      viewName === "messages" ? 600 : 0
   121→    );
   122→    return () => clearTimeout(t);
   123→  }, [shopperSignedIn, viewName]);
   124→  return unread;
   125→}

   127→export default function ShopperShell() {
   128→  const setSurface = useApp((s) => s.setSurface);
   129→  const activeStoreSlug = useApp((s) => s.activeStoreSlug);
   130→  const setActiveStoreSlug = useApp((s) => s.setActiveStoreSlug);
   131→
   132→  const [view, setView] = useState<ShopperView>({ name: "market" });
   133→  const [ready, setReady] = useState(false);
   134→  const [cartOpen, setCartOpen] = useState(false);
   135→  const [headerSearch, setHeaderSearch] = useState("");
   136→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   137→  // intentionally NOT part of ShopperView — a synthetic render override.
   138→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   139→  const restored = useRef(false);
   140→
   141→  // ---- mount: activeStoreSlug override, else restore last view --------------
   142→  useEffect(() => {
   143→    if (restored.current) return;
   144→    restored.current = true;
   145→    // claim deep-link wins over everything
   146→    try {
   147→      const params = new URLSearchParams(window.location.search);
   148→      const cj = params.get("claim_job");
   149→      const ct = params.get("claim_token");
   150→      if (cj && ct) {
   151→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   152→        setClaimLink({ jobId: cj, token: ct });
   153→        setReady(true);
   154→        return;
   155→      }
   156→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   157→      const sf = params.get("sf");
   158→      if (sf) {
   159→        const sp = params.get("sp");
   160→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   161→        setReady(true);
   162→        return;
   163→      }
   164→    } catch {
   165→      /* ignore */
   166→    }
   167→    if (activeStoreSlug) {
   168→      setView({ name: "storefront", slug: activeStoreSlug });
   169→      setActiveStoreSlug(null);
   170→    } else {
   171→      try {
   172→        const raw = window.localStorage.getItem(VIEW_KEY);
   173→        if (raw) {
   174→          const v = JSON.parse(raw) as ShopperView;
   175→          if (v && typeof v.name === "string") setView(v);
   176→        }
   177→      } catch {
   178→        /* ignore */
   179→      }
   180→    }
   181→    setReady(true);
   182→  }, [activeStoreSlug, setActiveStoreSlug]);
   183→
   184→  // ---- nav helper: persist + scroll to top ----------------------------------
   185→  const nav = useCallback((v: ShopperView) => {
   186→    setView(v);
   187→    try {
   188→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   189→    } catch {
   190→      /* ignore */
   191→    }
   192→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   193→  }, []);
   194→
   195→  // refresh storefront/store-page when slug changes externally
   196→  useEffect(() => {
   197→    if (!ready) return;
   198→    if (activeStoreSlug) {
   199→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   200→      setView({ name: "storefront", slug: activeStoreSlug });
   201→      setActiveStoreSlug(null);
   202→    }
   203→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   204→
   205→  const ctx = useMemo<ShopperCtx>(() => ({ nav, openCart: () => setCartOpen(true) }), [nav]);
   206→
   207→  const showChrome = !isStoreChrome(view);
   208→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   209→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   210→
   211→  return (
   212→    <div className="flex-1 flex flex-col min-h-screen">
   213→      <Toaster position="top-center" richColors />
   214→      <ShopperAuthDialog />
   215→
   216→      {showChrome ? (
   217→        <MarketHeader
   218→          view={view}
   219→          nav={nav}
   220→          openCart={() => setCartOpen(true)}
   221→          search={headerSearch}
   222→          setSearch={setHeaderSearch}
   223→        />
   224→      ) : null}
   225→
   226→      <AnimatePresence mode="wait">
   227→        <motion.main
   228→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : view.name === "me" ? view.tab ?? "" : ""}`}
   229→          initial={{ opacity: 0, y: 8 }}
   230→          animate={{ opacity: 1, y: 0 }}
   231→          exit={{ opacity: 0, y: -8 }}
   232→          transition={{ duration: 0.18, ease: "easeOut" }}
   233→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   234→        >
   235→          {claimLink ? null : (
   236→            <>
   237→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   238→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   239→              {view.name === "category" ? (
   240→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   241→              ) : null}
   242→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   243→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   244→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   245→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   246→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   247→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   248→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   249→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   250→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   251→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   252→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}
   253→            </>
   254→          )}
   255→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}
   256→        </motion.main>
   257→      </AnimatePresence>
   258→
   259→      {showBottomNav ? (
   260→        <BottomNav
   261→          view={view}
   262→          nav={nav}
   263→          openCart={() => setCartOpen(true)}
   264→        />
   265→      ) : null}
   266→
   267→      {showChrome ? (
   268→        <ShopperFooter
   269→          onSeller={() => setSurface("seller")}
   270→          onLookup={() => nav({ name: "lookup" })}
   271→          onMarket={() => nav({ name: "market" })}
   272→          onGenerate={() => nav({ name: "generate" })}
   273→        />
   274→      ) : null}
   275→
   276→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   277→    </div>
   278→  );
   279→}

   285→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   286→  const me = useApp((s) => s.me);
   287→  const signedIn = Boolean(me.loaded && me.shopper);
   288→  const { count } = useCart();
   289→  const unreadChats = useChatUnread(signedIn, view.name);
   290→
   291→  const items = [
   292→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   293→    {
   294→      key: "orders",
   295→      label: "Orders",
   296→      icon: Package,
   297→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   298→      badge: 0,
   299→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : nav({ name: "orders" })),
   300→    },
   301→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   302→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => nav({ name: "messages" }) },
   303→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   304→  ];
   305→
   306→  return (
   307→    <nav
   308→      aria-label="Bottom navigation"
   309→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   310→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   311→    >
   312→      <div className="mx-auto grid max-w-lg grid-cols-5">
   313→        {items.map((it) => {
   314→          const Icon = it.icon;
   315→          return (
   316→            <button
   317→              key={it.key}
   318→              type="button"
   319→              onClick={it.onClick}
   320→              aria-label={it.label}
   321→              aria-current={it.active ? "page" : undefined}
   322→              className={cn(
   323→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   324→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   325→              )}
   326→            >
   327→              <span className="relative">
   328→                <Icon className="h-5 w-5" aria-hidden="true" />
   329→                {it.badge > 0 ? (
   330→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   331→                    {it.badge > 9 ? "9+" : it.badge}
   332→                  </span>
   333→                ) : null}
   334→              </span>
   335→              {it.label}
   336→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   337→            </button>
   338→          );
   339→        })}
   340→      </div>
   341→    </nav>
   342→  );
   343→}

   800→function timeAgoShort(iso: string): string {
   801→  const diff = Date.now() - new Date(iso).getTime();
   802→  const m = Math.floor(diff / 60000);
   803→  if (m < 1) return "now";
   804→  if (m < 60) return `${m}m`;
   805→  const h = Math.floor(m / 60);
   806→  if (h < 24) return `${h}h`;
   807→  const d = Math.floor(h / 24);
   808→  if (d < 7) return `${d}d`;
   809→  return new Date(iso).toLocaleDateString(undefined, { day: "numeric", month: "short" });
   810→}

   812→function MessagesView({ ctx, initialThreadId }: { ctx: ShopperCtx; initialThreadId?: string }) {
   813→  const me = useApp((s) => s.me);
   814→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   815→  const shopper = me.loaded ? me.shopper : null;
   816→
   817→  const [threads, setThreads] = useState<ThreadRow[] | null>(null);
   818→  const [activeId, setActiveId] = useState<string | null>(initialThreadId ?? null);
   819→  const [messages, setMessages] = useState<ChatMsg[] | null>(null);
   820→  const [activeThread, setActiveThread] = useState<{ storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null } | null>(null);
   821→  const [context, setContext] = useState<ThreadContextCard | null>(null);
   822→  const [draft, setDraft] = useState("");
   823→  const [sending, setSending] = useState(false);
   824→  const socketRef = useRef<Socket | null>(null);
   825→  const scrollRef = useRef<HTMLDivElement>(null);
   826→
   827→  const loadThreads = useCallback(async () => {
   828→    try {
   829→      const rows = await fetchJson<ThreadRow[]>("/api/chat/threads");
   830→      setThreads(rows);
   831→      return rows;
   832→    } catch {
   833→      setThreads([]);
   834→      return [];
   835→    }
   836→  }, []);
   837→
   838→  const loadMessages = useCallback(async (threadId: string) => {
   839→    try {
   840→      const d = await fetchJson<{
   841→        thread: { storeName: string; orderNumber: string | null; contextType: string | null; subject: string | null };
   842→        context: ThreadContextCard | null;
   843→        messages: ChatMsg[];
   844→      }>(`/api/chat/threads/${threadId}`);
   845→      setMessages(d.messages);
   846→      setActiveThread(d.thread);
   847→      setContext(d.context ?? null);
   848→    } catch {
   849→      setMessages([]);
   850→      setContext(null);
   851→    }
   852→  }, []);
   853→
   854→  // initial load
   855→  useEffect(() => {
   856→    if (!shopper) return;
   857→    let alive = true;
   858→    loadThreads().then(() => {
   859→      if (!alive) return;
   860→      // Only auto-open when resuming an intent (contact-seller deep link).
   861→      // Plain entry shows the list first so unread badges stay visible —
   862→      // auto-reading the newest thread would instantly consume the badge.
   863→      if (initialThreadId) setActiveId(initialThreadId);
   864→    });
   865→    return () => {
   866→      alive = false;
   867→    };
   868→  }, [shopper, initialThreadId, loadThreads]);
   869→
   870→  // load conversation when thread changes
   871→  useEffect(() => {
   872→    if (!activeId) return;
   873→    setMessages(null);
   874→    // the detail GET marks the thread read server-side — refresh the list so
   875→    // the unread pill on the row (and the header badge) drops immediately
   876→    loadMessages(activeId).then(() => {
   877→      loadThreads();
   878→      window.dispatchEvent(new Event("shopmesh:chat-sync"));
   879→    });
   880→  }, [activeId, loadMessages, loadThreads]);
   881→
   882→  // live updates via the socket.io chat hub
   883→  useEffect(() => {
   884→    if (!shopper) return;
   885→    const socket = io("/?XTransformPort=3003", {
   886→      transports: ["websocket", "polling"],
   887→      reconnection: true,
   888→      reconnectionAttempts: 5,

   890→    });

   898→      if (msg.threadId && msg.threadId === activeId) {
   899→        setMessages((prev) => (prev && !prev.some((m) => m.id === msg.id) ? [...prev, msg] : prev));
   900→        if (msg.senderType === "seller") {
   901→          // message arrived while the shopper is looking at this conversation —
   902→          // clear the badge straight away instead of waiting for the poller
   903→          void postJson(`/api/chat/threads/${activeId}/read`, {}).catch(() => {});
   904→        }
   905→      }
   906→      loadThreads();
   907→    });
   908→    return () => {
   909→      socket.disconnect();
   910→      socketRef.current = null;
   911→    };
   912→  }, [shopper, threads, activeId, loadThreads]);
