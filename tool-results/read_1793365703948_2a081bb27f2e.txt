   123→   123→   123→/** 850 → "8.5" (1 decimal) — seed value for the percent inputs. */
   124→   124→   124→function bpsToPercentInput(bps: number): string {
   125→   125→   125→  return (bps / 100).toFixed(1);
   126→   126→   126→}
   127→   127→   127→
   128→   128→   128→/** Percent text → basis points, or null when blank/unparseable/out of range. */
   129→   129→   129→function percentTextToBps(text: string, maxRateBps: number): number | null {
   130→   130→   130→  const trimmed = text.trim();
   131→   131→   131→  if (trimmed === "") return null;
   132→   132→   132→  const pct = Number(trimmed);
   133→   133→   133→  if (!Number.isFinite(pct)) return null;
   134→   134→   134→  const bps = Math.round(pct * 100);
   135→   135→   135→  if (bps < 0 || bps > maxRateBps) return null;
   136→   136→   136→  return bps;
   137→   137→   137→}
