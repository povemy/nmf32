     2→// AFFILIATE HUB & ONE-LEVEL SPONSOR REFERRAL SYSTEM — core domain library.
     3→// docs/affiliate/*.md are the authoritative design docs.
     4→//
     5→// COMMERCIAL MODEL (fixed, spec §1/§8):
     6→//   • Seller-funded commission pool per eligible OrderItem: P = B × R (bps,
     7→//     integer sen). The pool is carved OUT of seller proceeds at checkout
     8→//     (sellerNet already subtracts it — see computeOrderCommissions).
     9→//   • Split of the POOL: S = floor(P × 10 / 100) to the direct affiliate's
    10→//     ONE-LEVEL sponsor; D = P − S to the direct affiliate. No sponsor ⇒
    11→//     D = P, S = 0. D + S = P ALWAYS (never exceeds the pool).
    12→//   • Attribution travels UP exactly one generation. No recruitment bonuses,
    13→//     no click rewards, no registration rewards — liabilities only ever
    14→//     originate from genuine eligible paid sales.
    15→//
    16→// INTEGRITY CONTRACTS:
    17→//   • Pool snapshots freeze on OrderItem at checkout (commission contract).
    18→//   • AffiliateCommission rows post EXACTLY ONCE per order item at payment
    19→//     confirmation (unique orderItemId; duplicate webhooks are no-ops).
    20→//   • The 90/10 SPLIT is computed at POSTING time from the then-current
    21→//     sponsor-override flag + sponsor eligibility, then never rewritten —
    22→//     so the override can be disabled without corrupting existing records.
    23→//   • Every money mutation rides the immutable ledger (lib/ledger.ts).

    86→function clampInt(v: unknown, min: number, max: number, fallback: number): number {
    87→  const n = typeof v === "number" ? v : Number(v);
    88→  if (!Number.isFinite(n)) return fallback;
    89→  return Math.min(max, Math.max(min, Math.round(n)));
    90→}

   112→/** Effective seller affiliate rate (bps) for a product, honouring every gate.
   113→ *  0 ⇒ not enrolled (no commission). Product override ?? store default,
   114→ *  clamped to the platform max; programme master switch must be on. */
   115→export function effectiveAffiliateRateBps(
   116→  s: AffiliateSettings,
   117→  store: { affiliateOptIn: boolean; affiliateDefaultRateBps: number },
   118→  product: { affiliateEnabled: boolean; affiliateRateBps: number | null }
   119→): number {
   120→  if (!s.enabled || !store.affiliateOptIn || !product.affiliateEnabled) return 0;
   121→  const rate = product.affiliateRateBps ?? store.affiliateDefaultRateBps;
   122→  return Math.max(0, Math.min(rate, s.maxRateBps));
   123→}

   125→// ---------------------------------------------------------------------------
   126→// Commission math — the spec's exact formulas (§8).
   127→// ---------------------------------------------------------------------------
   128→/** Pool P = floor(B × R / 10^4) in integer sen.
   129→ *  POST-IMPLEMENTATION AUDIT (P4): the ONE canonical rounding policy for the
   130→ *  entire affiliate money path is FLOOR (commission-definitions.md §9 is
   131→ *  normative: "Where this document conflicts with prose elsewhere, this
   132→ *  document governs"). Previously half-up (platform percent convention) —
   133→ *  a one-sen drift vs the normative doc. Same floor policy is used at the
   134→ *  checkout snapshot, payment posting, refund reversal, reconciliation and
   135→ *  every report. The platform commission's own percent part keeps its
   136→ *  historical half-up convention (separate ledger line, unchanged). */
   137→export function affiliatePoolMinor(baseMinor: number, rateBps: number): number {
   138→  if (baseMinor <= 0 || rateBps <= 0) return 0;
   139→  return Math.floor((baseMinor * rateBps) / 10000);
   140→}
   141→
   142→/** 90/10 split of the POOL: S = floor(P × 10 / 100); D = P − S (D + S = P). */
   143→export function splitAffiliatePool(poolMinor: number): { directMinor: number; sponsorMinor: number } {
   144→  if (poolMinor <= 0) return { directMinor: 0, sponsorMinor: 0 };
   145→  const sponsorMinor = Math.floor((poolMinor * 10) / 100);
   146→  return { directMinor: poolMinor - sponsorMinor, sponsorMinor };
   147→}

   189→function slugifyAffiliate(name: string): string {
   190→  const base = name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "").slice(0, 24);
   191→  return base || "affiliate";
   192→}

   194→async function uniqueSlug(base: string): Promise<string> {
   195→  let candidate = base;
   196→  for (let i = 0; i < 20; i++) {
   197→    const clash = await db.affiliateProfile.findUnique({ where: { slug: candidate }, select: { id: true } });
   198→    if (!clash) return candidate;
   199→    candidate = `${base}-${randomToken(2)}`;
   200→  }
   201→  return `aff-${randomToken(4)}`;
   202→}

   204→/** Read the FIRST-TOUCH referral cookie and validate it against the touch
   205→ *  table (unexpired, unconsumed). Returns the sponsor profile or null. */
   206→export async function resolveReferralCookie(): Promise<{ affiliateId: string; displayName: string; code: string } | null> {
   207→  try {
   208→    const store = await cookies();
   209→    const raw = store.get(REFERRAL_COOKIE)?.value;
   210→    if (!raw) return null;
   211→    const parsed = parseJson<{ code?: string }>(raw, {});
   212→    if (!parsed?.code) return null;
   213→    const touch = await db.affiliateReferralTouch.findFirst({
   214→      where: { code: parsed.code, consumedAt: null, expiresAt: { gt: new Date() } },
   215→      orderBy: { touchedAt: "desc" },
   216→      select: { affiliateId: true, code: true },
   217→    });
   218→    const profile = touch
   219→      ? await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId }, select: { id: true, displayName: true, code: true, status: true } })
   220→      : null;
   221→    if (!profile || profile.status !== "active") return null;
   222→    return { affiliateId: profile.id, displayName: profile.displayName, code: profile.code };
   223→  } catch {
   224→    return null;
   225→  }
   226→}

   251→/** Activate the current shopper as an affiliate. Idempotent (already-active →
   252→ *  error with the existing profile). Sponsor precedence (spec §5):
   253→ *  1. Valid first-touch referral cookie (30-day window, unconsumed);
   254→ *  2. Else a one-time manual code entry — confirmed explicitly;
   255→ *  3. Else no sponsor (recorded permanently as none).
   256→ *  The relationship locks forever after activation: no click, seller or admin
   257→ *  tool can casually change it (admin corrections are separately audited). */
   258→export async function activateAffiliate(input: ActivationInput) {
   259→  const settings = await getAffiliateSettings();
   260→  if (!settings.enabled) {
   261→    throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   262→  }
   263→
   264→  const existing = await db.affiliateProfile.findUnique({ where: { userId: input.userId } });
   265→  if (existing) {
   266→    throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   267→  }
   268→
   269→  // ---- Sponsor resolution (server-side only) -----------------------------
   270→  let sponsor: { id: string; displayName: string } | null = null;
   271→  let sponsorSource: string | null = null;
   272→
   273→  const touch = await resolveReferralCookie();
   274→  if (touch) {
   275→    const profile = await db.affiliateProfile.findUnique({ where: { id: touch.affiliateId } });
   276→    if (profile && profile.status === "active" && profile.userId !== input.userId) {
   277→      sponsor = { id: profile.id, displayName: profile.displayName };
   278→      sponsorSource = "referral_touch";
   279→    }
   280→  }
   281→  if (!sponsor && input.sponsorCode) {
   282→    const candidate = await findAffiliateByCode(input.sponsorCode);
   283→    if (!candidate) throw new ApiError(400, "INVALID_SPONSOR_CODE", "That referral code was not found");
   284→    if (candidate.userId === input.userId) {
   285→      throw new ApiError(400, "SELF_REFERRAL", "You cannot refer yourself");
   286→    }
   287→    if (candidate.status !== "active") {
   288→      throw new ApiError(400, "SPONSOR_INACTIVE", "That sponsor is not currently eligible");
   289→    }
   290→    if (!input.sponsorConfirmed) {
   291→      throw new ApiError(400, "SPONSOR_CONFIRMATION_REQUIRED", "Confirm the sponsor relationship before activating");
   292→    }
   293→    sponsor = { id: candidate.id, displayName: candidate.displayName };
   294→    sponsorSource = "code_entry";
   295→  }
   296→
   297→  const code = randomToken(5); // 10-hex-char referral code, unpredictable
   298→  const slug = await uniqueSlug(slugifyAffiliate(input.slug || input.displayName || input.fullName));
   299→
   300→  const profile = await db.$transaction(async (tx) => {
   301→    // Re-check inside the tx (concurrent double-submit).
   302→    const dupe = await tx.affiliateProfile.findUnique({ where: { userId: input.userId } });
   303→    if (dupe) throw new ApiError(409, "ALREADY_AFFILIATE", "This account is already an affiliate");
   304→
   305→    const created = await tx.affiliateProfile.create({
   306→      data: {
   307→        userId: input.userId,
   308→        code,
   309→        slug,
   310→        displayName: (input.displayName || input.fullName).slice(0, 60),
   311→        bio: input.bio?.slice(0, 500) ?? null,
   312→        status: "active",
   313→        termsVersion: AFFILIATE_TERMS_VERSION,
   314→        termsAcceptedAt: new Date(),
   315→        activatedAt: new Date(),
   316→        sponsorId: sponsor?.id ?? null,
   317→        sponsorAssignedAt: sponsor ? new Date() : null,
   318→        sponsorSource,
   319→      },
   320→    });
   321→
   322→    // Consume the referral touch exactly once (relationship now recorded).
   323→    if (sponsor && sponsorSource === "referral_touch" && touch) {
   324→      await tx.affiliateReferralTouch.updateMany({
   325→        where: { code: touch.code, consumedAt: null },
   326→        data: { consumedAt: new Date(), consumedUserId: input.userId },
   327→      });
   328→    }
   329→    return created;
   330→  });
   331→
   332→  return { profile, sponsor, sponsorSource };
   333→}

   335→/** Admin-only sponsor correction — audited, with documented reason. Cycle
   336→ *  guard: with one-level relationships a 2-cycle (A↔B) is the only possible
   337→ *  loop; rejecting `newSponsor.sponsorId === affiliateId` kills it. */
   338→export async function correctSponsor(input: {
   339→  affiliateId: string;
   340→  newSponsorId: string | null;
   341→  reason: string;
   342→  adminId: string;
   343→}) {
   344→  if (!input.reason || input.reason.trim().length < 10) {
   345→    throw new ApiError(400, "VALIDATION", "A documented reason (≥10 chars) is required for sponsor corrections");
   346→  }
   347→  const affiliate = await db.affiliateProfile.findUnique({ where: { id: input.affiliateId } });
   348→  if (!affiliate) throw new ApiError(404, "NOT_FOUND", "Affiliate not found");
   349→  if (input.newSponsorId === affiliate.id) {
   350→    throw new ApiError(400, "SELF_REFERRAL", "An affiliate cannot sponsor themselves");
   351→  }
   352→  if (input.newSponsorId) {
   353→    const sponsor = await db.affiliateProfile.findUnique({ where: { id: input.newSponsorId } });
   354→    if (!sponsor) throw new ApiError(404, "NOT_FOUND", "New sponsor not found");
   355→    if (sponsor.sponsorId === affiliate.id) {
   356→      throw new ApiError(400, "SPONSOR_CYCLE", "This correction would create a sponsor cycle");
   357→    }
   358→  }
   359→  return db.$transaction(async (tx) => {
   360→    const updated = await tx.affiliateProfile.update({
   361→      where: { id: input.affiliateId },
   362→      data: {
   363→        sponsorId: input.newSponsorId,
   364→        sponsorAssignedAt: input.newSponsorId ? new Date() : null,
   365→        sponsorSource: "admin_correction",
   366→      },
   367→    });
   368→    await tx.affiliateSponsorAudit.create({
   369→      data: {
   370→        affiliateId: input.affiliateId,
   371→        fromSponsorId: affiliate.sponsorId,
   372→        toSponsorId: input.newSponsorId,
   373→        reason: input.reason.trim(),
   374→        actorType: "admin",
   375→        actorId: input.adminId,
   376→      },
   377→    });
   378→    return updated;
   379→  });
   380→}

   388→/** Post AffiliateCommission rows for one paid order. Idempotent: unique
   389→ *  orderItemId + find-first guard. The 90/10 DECISION comes from the
   390→ *  checkout-time policy snapshot (OrderItem.affiliateSponsorEnabled); the
   391→ *  sponsor's status at posting decides pending vs held only. */
   392→export async function postAffiliateCommissionsForOrder(
   393→  tx: Tx,
   394→  order: { id: string; orderNumber: string; storeId: string; items: { id: string; productId?: string | null; affiliateDirectId: string | null; affiliateSponsorId: string | null; affiliateLinkId: string | null; affiliateRateBps: number | null; affiliatePoolMinor: number; affiliateSponsorEnabled?: boolean | null; breakdownJson: string }[] },
   395→  paidAt: Date
   396→): Promise<number> {
   397→  const attributed = order.items.filter((i) => i.affiliateDirectId && i.affiliatePoolMinor > 0);
   398→  if (attributed.length === 0) return 0;
   399→
   400→  const settings = await getAffiliateSettings();
   401→  if (!settings.enabled) return 0; // kill switch: no NEW liabilities
   402→
   403→  // Sponsor profiles snapshot (status at posting decides hold vs pending).
   404→  const sponsorIds = [...new Set(attributed.map((i) => i.affiliateSponsorId).filter((s): s is string => Boolean(s)))];
   405→  const sponsors = sponsorIds.length
   406→    ? new Map((await tx.affiliateProfile.findMany({ where: { id: { in: sponsorIds } }, select: { id: true, status: true } })).map((p) => [p.id, p.status]))
   407→    : new Map<string, string>();
   408→
   409→  // Link metadata snapshot (linkType + channel for reporting — spec §14).
   410→  const linkIds = [...new Set(attributed.map((i) => i.affiliateLinkId).filter((s): s is string => Boolean(s)))];
   411→  const links = linkIds.length
   412→    ? new Map((await tx.affiliateLink.findMany({ where: { id: { in: linkIds } }, select: { id: true, type: true, channel: true } })).map((l) => [l.id, l]))
   413→    : new Map<string, { type: string; channel: string }>();
   414→
   415→  let posted = 0;
   416→  for (const item of attributed) {
   417→    const existing = await tx.affiliateCommission.findUnique({ where: { orderItemId: item.id }, select: { id: true } });
   418→    if (existing) continue; // idempotent replay
   419→
   420→    const pool = item.affiliatePoolMinor;
   421→    let sponsorMinor = 0;
   422→    let sponsorHeld = false;
   423→    // POST-IMPLEMENTATION AUDIT (P3 — deterministic policy freeze): the
   424→    // sponsor-override decision comes from the CHECKOUT-TIME snapshot
   425→    // (OrderItem.affiliateSponsorEnabled), never from the live setting. A
   426→    // settings change between checkout and payment confirmation must not
   427→    // retroactively alter an existing order's compensation arrangement:
   428→    //   • override ON at checkout → this order keeps the 90/10 split even if
   429→    //     an admin later disables sponsor compensation for NEW orders;
   430→    //   • override OFF at checkout → the direct promoter receives 100% (D = P)
   431→    //     for this order, even if the override is later enabled.
   432→    // NULL (legacy rows created before the snapshot column existed) falls
   433→    // back to the current setting — the historical behaviour those orders
   434→    // were created under. The sponsor's CURRENT status still decides only
   435→    // whether the 10% share posts as pending or held (never who the
   436→    // beneficiary is), matching commission-definitions.md §4.
   437→    const sponsorOverrideOn = item.affiliateSponsorEnabled ?? settings.sponsorEnabled;
   438→    if (item.affiliateSponsorId && sponsorOverrideOn) {
   439→      const status = sponsors.get(item.affiliateSponsorId) ?? "missing";
   440→      if (status === "active") {
   441→        sponsorMinor = Math.floor((pool * 10) / 100);
   442→      } else {
   443→        // Sponsor under review / gone: HOLD the 10% separately — the direct
   444→        // promoter's 90% must NOT absorb it and must NOT be penalised.
   445→        sponsorMinor = Math.floor((pool * 10) / 100);
   446→        sponsorHeld = true;
   447→      }
   448→    }
   449→    const directMinor = pool - sponsorMinor; // D + S = P always
   450→    const bd = parseJson<{ base?: number }>(item.breakdownJson, {});
   451→
   452→    const commission = await tx.affiliateCommission.create({
   453→      data: {
   454→        orderItemId: item.id,
   455→        orderId: order.id,
   456→        orderNumber: order.orderNumber,
   457→        storeId: order.storeId,
   458→        productId: item.productId ?? null,
   459→        affiliateId: item.affiliateDirectId!,
   460→        sponsorId: item.affiliateSponsorId ?? null,
   461→        linkId: item.affiliateLinkId ?? null,
   462→        linkType: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.type ?? null) : null,
   463→        channel: item.affiliateLinkId ? (links.get(item.affiliateLinkId)?.channel ?? null) : null,
   464→        commissionableBaseMinor: Math.max(0, bd.base ?? 0),
   465→        rateBps: item.affiliateRateBps ?? 0,
   466→        poolMinor: pool,
   467→        status: "pending",
   468→        breakdownJson: item.breakdownJson,
   469→      },
   470→    });
   471→
   472→    await tx.affiliateCommissionEntry.create({
   473→      data: {
   474→        commissionId: commission.id,
   475→        beneficiaryAffiliateId: item.affiliateDirectId!,
   476→        role: "direct",
   477→        amountMinor: directMinor,
   478→        status: "pending",
   479→      },
   480→    });
   481→    if (item.affiliateSponsorId && sponsorMinor > 0) {
   482→      await tx.affiliateCommissionEntry.create({
   483→        data: {
   484→          commissionId: commission.id,
   485→          beneficiaryAffiliateId: item.affiliateSponsorId,
   486→          role: "sponsor",
   487→          amountMinor: sponsorMinor,
   488→          status: sponsorHeld ? "held" : "pending",
   489→          holdReason: sponsorHeld ? "sponsor_under_review" : null,
   490→        },
   491→      });
   492→    }
   493→    posted++;
   494→  }
   495→  return posted;
   496→}

   498→// ---------------------------------------------------------------------------
   499→// Lifecycle: pending → approved (order delivered/completed) → available
   500→// (review window elapsed) → paid. Reversals from refunds. Holds from review.
   501→// ---------------------------------------------------------------------------
   502→/** Approve an order's pending entries (authoritative delivered/completed). */
   503→export async function approveAffiliateCommissionsForOrder(orderId: string, tx?: Tx): Promise<number> {
   504→  const client = tx ?? db;
   505→  const commissions = await client.affiliateCommission.findMany({
   506→    where: { orderId, status: "pending" },
   507→    select: { id: true },
   508→  });
   509→  if (commissions.length === 0) return 0;
   510→  const settings = await getAffiliateSettings();
   511→  const now = new Date();
   512→  const availableAt = new Date(now.getTime() + settings.reviewDays * 86400000);
   513→  let approved = 0;
   514→  for (const c of commissions) {
   515→    const res = await client.affiliateCommissionEntry.updateMany({
   516→      where: { commissionId: c.id, status: "pending" },
   517→      data: { status: "approved", availableAt },
   518→    });
   519→    if (res.count > 0) {
   520→      await client.affiliateCommission.update({ where: { id: c.id }, data: { status: "approved", approvedAt: now } });
   521→      approved += res.count;
   522→    }
   523→  }
   524→  return approved;
   525→}

   527→/** Batch sweep (cron): approve commissions on delivered/completed orders +
   528→ *  mature approved entries whose review window elapsed. Bounded batches. */
   529→export async function sweepAffiliateLifecycle(): Promise<{ approved: number; matured: number }> {
   530→  const settings = await getAffiliateSettings();
   531→  if (!settings.enabled) return { approved: 0, matured: 0 };
   532→
   533→  // 1) Orders delivered/completed with pending commissions → approve.
   534→  const pendingCommissions = await db.affiliateCommission.findMany({
   535→    where: { status: "pending" },
   536→    select: { orderId: true },
   537→    distinct: ["orderId"],
   538→    take: 50,
   539→  });
   540→  let approved = 0;
   541→  if (pendingCommissions.length > 0) {
   542→    const orderIds = pendingCommissions.map((c) => c.orderId);
   543→    const eligible = await db.order.findMany({
   544→      where: { id: { in: orderIds }, status: { in: ["delivered", "completed"] } },
   545→      select: { id: true },
   546→    });
   547→    for (const o of eligible) {
   548→      approved += await approveAffiliateCommissionsForOrder(o.id);
   549→    }
   550→  }
   551→
   552→  // 2) Matured approved entries → available (clock transition, CAS-guarded).
   553→  const now = new Date();
   554→  const matured = await db.affiliateCommissionEntry.updateMany({
   555→    where: { status: "approved", availableAt: { lte: now } },
   556→    data: { status: "available" },
   557→  });
   558→  if (matured.count > 0) {
   559→    await db.affiliateCommission.updateMany({
   560→      where: { status: "approved", entries: { some: { status: "available" } } },
   561→      data: { status: "available" },
   562→    });
   563→  }
   564→  return { approved, matured: matured.count };
   565→}
   566→
   567→// ---------------------------------------------------------------------------
   568→// Balances (affiliate-side truth = AffiliateCommissionEntry rows; the ledger
   569→// mirrors totals on PLATFORM:AFFILIATE_LIABILITY for double-entry integrity).
   570→// ---------------------------------------------------------------------------
   571→export type AffiliateBalances = {
   572→  pendingMinor: number;
   573→  approvedMinor: number; // approved but review window not elapsed
   574→  availableMinor: number; // matured, not yet claimed by a payout
   575→  payoutInFlightMinor: number; // claimed by requested/processing payouts
   576→  paidMinor: number;
   577→  reversedMinor: number;
   578→  heldMinor: number;
   579→  recoveryMinor: number; // reversals booked against already-paid entries
   580→  requestableMinor: number; // available − recovery, floored at 0
   581→};
   582→
   583→export async function affiliateBalances(affiliateId: string, tx?: Prisma.TransactionClient): Promise<AffiliateBalances> {
   584→  // C1.2 (Phase 2.2): optional tx client — callers inside a transaction now
   585→  // read balance truth from the SAME snapshot (requestAffiliatePayout
   586→  // previously read balances on the GLOBAL client, outside its transaction).
   587→  const client = tx ?? db;
   588→  const entries = await client.affiliateCommissionEntry.findMany({
   589→    where: { beneficiaryAffiliateId: affiliateId },
   590→    select: { status: true, amountMinor: true, reversedMinor: true, availableAt: true, payoutId: true },
   591→  });
   592→  const now = Date.now();
   593→  const b: AffiliateBalances = {
   594→    pendingMinor: 0, approvedMinor: 0, availableMinor: 0, payoutInFlightMinor: 0,
   595→    paidMinor: 0, reversedMinor: 0, heldMinor: 0, recoveryMinor: 0, requestableMinor: 0,
   596→  };
   597→  for (const e of entries) {
   598→    const matured = (e.availableAt?.getTime() ?? 0) <= now;
   599→    switch (e.status) {
   600→      case "pending":
   601→      case "approved":
   602→        if (e.status === "pending" || !matured) {
   603→          if (e.status === "pending") b.pendingMinor += e.amountMinor - e.reversedMinor;
   604→          else b.approvedMinor += e.amountMinor - e.reversedMinor;
   605→          break;
   606→        }
   607→        // fallthrough: matured approved
   608→      case "available":
   609→        if (e.payoutId) b.payoutInFlightMinor += e.amountMinor - e.reversedMinor;
   610→        else b.availableMinor += e.amountMinor - e.reversedMinor;
   611→        break;
   612→      case "paid":
   613→        b.paidMinor += e.amountMinor;
   614→        b.recoveryMinor += e.reversedMinor; // late refund after payout
   615→        break;
   616→      case "reversed":
   617→        b.reversedMinor += e.amountMinor;
   618→        break;
   619→      case "held":
   620→        b.heldMinor += e.amountMinor - e.reversedMinor;
   621→        break;
   622→    }
   623→  }
   624→  b.requestableMinor = Math.max(0, b.availableMinor - b.recoveryMinor);
   625→  return b;
   626→}
   627→
   628→// ---------------------------------------------------------------------------
   629→// Refund reversals — proportional against the ORIGINAL immutable allocations
   630→// (spec §11). Shipping-only refunds book no product-commission reversal (the
   631→// pool base excluded shipping by construction). Late refunds after payout
   632→// become recovery entries: reversedMinor accumulates on the PAID entry and
   633→// offsets future earnings — the original payout history is never rewritten.
   634→// ---------------------------------------------------------------------------
   635→function fullyReversedFn(newReversed: number, amountMinor: number): boolean {
   636→  return newReversed >= amountMinor;
   637→}
   638→export async function reverseAffiliateForRefund(
   639→  tx: Tx,
   640→  input: {
   641→    refundId: string;
   642→    orderId: string;
   643→    refundAmountMinor: number;
   644→    /** Fresh order items (with affiliate snapshots) under the lock. */
   645→    items: { id: string; lineSubtotalMinor: number; refundedMinor: number; breakdownJson: string; affiliatePoolMinor: number }[];
   646→    /** Σ affiliateReversalMinor of prior PROCESSED refunds on this order. */
   647→    priorAffiliateReversalMinor: number;
   648→  }
   649→): Promise<{ affiliateReversalMinor: number; affiliateClearingMinor: number; rule: string }> {
   650→  const attributed = input.items.filter((i) => i.affiliatePoolMinor > 0);
   651→  if (attributed.length === 0) return { affiliateReversalMinor: 0, affiliateClearingMinor: 0, rule: "no_affiliate" };
   652→
   653→  // ---- C1.2 (Phase 2.2) pre-pass: load commissions + entries, then LOCK the
   654→  // open AffiliatePayout rows BEFORE any entry write. Lock order is
   655→  // payout → entries on every flow that touches both (this function and the
   656→  // admin approve / mark-paid / fail claims, which take the payout row lock
   657→  // first via their conditional updateMany) — no AB-BA deadlock window.
   658→  const commissions = (
   659→    await Promise.all(
   660→      attributed.map((it) => tx.affiliateCommission.findUnique({ where: { orderItemId: it.id }, include: { entries: true } })),
   661→    )
   662→  ).filter((c): c is NonNullable<typeof c> => c !== null);
   663→  const openPayoutIds = [
   664→    ...new Set(commissions.flatMap((c) => c.entries.map((e) => e.payoutId).filter((p): p is string => p !== null))),
   665→  ];
   666→  const openPayouts = openPayoutIds.length
   667→    ? await tx.affiliatePayout.findMany({
   668→        where: { id: { in: openPayoutIds }, status: { in: ["requested", "processing"] } },
   669→        select: { id: true, status: true, amountMinor: true },
   670→      })
   671→    : [];
   672→  // Deterministic lock order (sorted ids) — multiple payouts can be affected
   673→  // when one order's commissions span several payout requests.
   674→  for (const p of [...openPayouts].sort((a, b) => a.id.localeCompare(b.id))) {
   675→    await tx.$queryRaw`SELECT id FROM "AffiliatePayout" WHERE id = ${p.id} FOR UPDATE`;
   676→  }
   677→  const payoutById = new Map(openPayouts.map((p) => [p.id, p]));
   678→
   679→  // ---- C1.2 verification fix (Phase 2.2): the pre-pass snapshot above is
   680→  // stale by definition. Every writer that mutates an entry claimed by an
   681→  // open payout (approve / mark-paid / fail / reject) takes the payout row
   682→  // lock FIRST, and a concurrent requestAffiliatePayout can claim
   683→  // payoutId:null entries between our pre-pass read and the locks above.
   684→  // Re-reading AFTER the locks therefore yields the authoritative entry +
   685→  // payout state for the remainder of this tx (we hold the locks — writers
   686→  // serialize behind us):
   687→  //   • a stale entry.status/reversedMinor would misclassify paid entries
   688→  //     (recovery vs reversed flip) and could over-reverse past the true
   689→  //     remaining pool of a concurrently-refunded entry;
   690→  //   • a stale payout.status would book the reversal against the WRONG
   691→  //     ledger side (AFFILIATE_LIABILITY vs PAYOUT_CLEARING) — the T5 split
   692→  //     must match the payout's REAL state at approval/transfer time.
   693→  const commissionIds = commissions.map((c) => c.id);
   694→  const freshEntries = commissionIds.length
   695→    ? await tx.affiliateCommissionEntry.findMany({ where: { commissionId: { in: commissionIds } } })
   696→    : [];
   697→  for (const c of commissions) {
   698→    c.entries = freshEntries.filter((e) => e.commissionId === c.id);
   699→  }
   700→  // A payout claimed AFTER our pre-pass (its entries showed payoutId:null in
   701→  // the pre-pass read) must also be locked and coordinated — same open-set
   702→  // definition, same deterministic lock order.
   703→  const freshOpenIds = [
   704→    ...new Set(freshEntries.map((e) => e.payoutId).filter((p): p is string => p !== null)),
   705→  ].filter((id) => !payoutById.has(id));
   706→  if (freshOpenIds.length) {
   707→    const newlyOpen = (
   708→      await tx.affiliatePayout.findMany({
   709→        where: { id: { in: freshOpenIds }, status: { in: ["requested", "processing"] } },
   710→        select: { id: true, status: true, amountMinor: true },
   711→      })
   712→    ).sort((a, b) => a.id.localeCompare(b.id));
   713→    for (const p of newlyOpen) {
   714→      await tx.$queryRaw`SELECT id FROM "AffiliatePayout" WHERE id = ${p.id} FOR UPDATE`;
   715→    }
   716→    openPayouts.push(...newlyOpen);
   717→  }
   718→  // Authoritative payout state — the rows are locked, so this read is stable
   719→  // until commit. Rebuild the map from it (drops payouts that left the open
   720→  // set, e.g. mark-paid committed before our lock: their entries are recovery
   721→  // reversals only, no clearing-side coordination).
   722→  const lockedOpenPayouts = openPayouts.length
   723→    ? await tx.affiliatePayout.findMany({
   724→        where: { id: { in: openPayouts.map((p) => p.id) }, status: { in: ["requested", "processing"] } },
   725→        select: { id: true, status: true, amountMinor: true },
   726→      })
   727→    : [];
   728→  payoutById.clear();
   729→  for (const p of lockedOpenPayouts) payoutById.set(p.id, p);
   730→
   731→  // Customer-paid gross per item (breakdown-proportional, mirroring
   732→  // deriveCommissionReversal's allocation family).
   733→  const grossArr = attributed.map((it) => {
   734→    const bd = parseJson<{ discount_allocated?: number; shipping_allocated?: number }>(it.breakdownJson, {});
   735→    return Math.max(0, it.lineSubtotalMinor - (bd.discount_allocated ?? 0) + (bd.shipping_allocated ?? 0));
   736→  });
   737→  const allocs: number[] = [];
   738→  {
   739→    // floor + remainder-to-largest (same as allocateProportionally)
   740→    const totalGross = grossArr.reduce((s, v) => s + v, 0);
   741→    if (totalGross <= 0) {
   742→      attributed.forEach(() => allocs.push(0));
   743→    } else {
   744→      let allocated = 0;
   745→      for (let i = 0; i < attributed.length; i++) {
   746→        const a = Math.floor((input.refundAmountMinor * grossArr[i]) / totalGross);
   747→        allocs.push(a);
   748→        allocated += a;
   749→      }
   750→      let rem = input.refundAmountMinor - allocated;
   751→      while (rem > 0) {
   752→        let maxIdx = 0;
   753→        for (let i = 1; i < grossArr.length; i++) if (grossArr[i] > grossArr[maxIdx]) maxIdx = i;
   754→        allocs[maxIdx] += 1;
   755→        rem -= 1;
   756→      }
   757→    }
   758→  }
   759→
   760→  // Per-item affiliate reversal, clamped to the remaining reversible pool.
   761→  const orderPool = attributed.reduce((s, it) => s + it.affiliatePoolMinor, 0);
   762→  const remainingReversible = Math.max(0, orderPool - input.priorAffiliateReversalMinor);
   763→  let total = 0;
   764→  const perItem: { item: (typeof attributed)[number]; reversal: number }[] = [];
   765→  for (let i = 0; i < attributed.length; i++) {
   766→    if (grossArr[i] <= 0 || allocs[i] <= 0) {
   767→      perItem.push({ item: attributed[i], reversal: 0 });
   768→      continue;
   769→    }
   770→    const r = Math.min(Math.floor((allocs[i] * attributed[i].affiliatePoolMinor) / grossArr[i]), attributed[i].affiliatePoolMinor);
   771→    perItem.push({ item: attributed[i], reversal: r });
   772→    total += r;
   773→  }
   774→  total = Math.min(Math.max(0, total), input.refundAmountMinor, remainingReversible);
   775→
   776→  // C1.2: cumulative per-payout adjustments for THIS refund.
   777→  //   requested  → liability-side: T5 keeps debiting AFFILIATE_LIABILITY and
   778→  //                the payout's frozen amount shrinks (no ledger posted for a
   779→  //                requested payout yet — fully consistent).
   780→  //   processing → clearing-side: T-A3 already moved the frozen amount from
   781→  //                AFFILIATE_LIABILITY into PAYOUT_CLEARING, so the reversal
   782→  //                must debit PAYOUT_CLEARING (returned as
   783→  //                affiliateClearingMinor for the T5 split) and the payout
   784→  //                amount shrinks so T-A4 pays the reduced figure — the
   785→  //                clearing account nets to exactly zero across T-A3, T5's
   786→  //                split line and T-A4/T-A4f at the adjusted amount.
   787→  const adjByPayout = new Map<string, { liability: number; clearing: number }>();
   788→  let clearingTotal = 0;
   789→
   790→  // Apply to the commission + entries (proportional split within each pool).
   791→  if (total > 0) {
   792→    // Scale per-item reversals down proportionally if the clamp reduced total.
   793→    const rawTotal = perItem.reduce((s, p) => s + p.reversal, 0);
   794→    const scale = rawTotal > 0 ? total / rawTotal : 0;
   795→    for (const p of perItem) {
   796→      const itemReversal = p.reversal === 0 ? 0 : Math.min(p.reversal, Math.floor(p.reversal * scale));
   797→      if (itemReversal <= 0) continue;
   798→      const commission = commissions.find((c) => c.orderItemId === p.item.id);
   799→      if (!commission) continue; // pre-affiliate legacy rows: nothing to reverse
   800→      const direct = commission.entries.find((e) => e.role === "direct");
   801→      const sponsor = commission.entries.find((e) => e.role === "sponsor");
   802→      // Split proportionally to the frozen allocation shares (P4: floor policy
   803→      // — same rounding as the pool computation). The SPONSOR share is the
   804→      // floor-proportional part and the DIRECT share takes the residual
   805→      // (commission-definitions.md §7 step 3: "direct absorbs the rounding
   806→      // residual"), mirroring the forward split where D = P − S. Invariant:
   807→      // Σ(dRev+sRev) === itemReversal whenever caps allow, and never more.
   808→      const sponsorShare = sponsor ? Math.min(sponsor.amountMinor - sponsor.reversedMinor, Math.floor((itemReversal * sponsor.amountMinor) / commission.poolMinor)) : 0;
   809→      const directShare = direct ? Math.min(direct.amountMinor - direct.reversedMinor, itemReversal - sponsorShare) : 0;
   810→      for (const [entry, share] of [[direct, directShare], [sponsor, sponsorShare]] as const) {
   811→        if (!entry || share <= 0) continue;
   812→        const newReversed = Math.min(entry.amountMinor, entry.reversedMinor + share);
   813→        const actualApplied = newReversed - entry.reversedMinor;
   814→        await tx.affiliateCommissionEntry.update({
   815→          where: { id: entry.id },
   816→          data: {
   817→            reversedMinor: newReversed,
   818→            // PAID entries stay paid — the reversal is a recovery entry that
   819→            // offsets future earnings (spec §11 late refunds); pending/
   820→            // approved/available entries flip to reversed.
   821→            ...(entry.status !== "paid" && fullyReversedFn(newReversed, entry.amountMinor) ? { status: "reversed" as const } : {}),
   822→          },
   823→        });
   824→        // ---- C1.2 payout coordination: shrink the frozen amount of the open
   825→        // payout that claimed this entry, by exactly the share applied. A
   826→        // refund and a payout can no longer both consume the same commission
   827→        // liability.
   828→        if (actualApplied > 0 && entry.payoutId) {
   829→          const payout = payoutById.get(entry.payoutId);
   830→          if (payout && (payout.status === "requested" || payout.status === "processing")) {
   831→            const adj = adjByPayout.get(payout.id) ?? { liability: 0, clearing: 0 };
   832→            if (payout.status === "processing") { adj.clearing += actualApplied; clearingTotal += actualApplied; }
   833→            else adj.liability += actualApplied;
   834→            adjByPayout.set(payout.id, adj);
   835→          }
   836→        }
   837→      }
   838→      const anyLive = commission.entries.some((e) => e.status !== "reversed");
   839→      if (!anyLive) {
   840→        await tx.affiliateCommission.update({
   841→          where: { id: commission.id },
   842→          data: { status: "reversed", reversedAt: new Date() },
   843→        });
   844→      }
   845→    }
   846→  }
   847→
   848→  // ---- C1.2: apply the per-payout amount reductions (the payout rows are
   849→  // already locked). A REQUESTED payout whose frozen amount is fully consumed
   850→  // is auto-released (rejected + entries unclaimed) — no ledger was posted
   851→  // for it, so this is clean; the affiliate simply re-requests later.
   852→  for (const [payoutId, adj] of adjByPayout) {
   853→    const payout = payoutById.get(payoutId);
   854→    if (!payout) continue;
   855→    const reduce = adj.liability + adj.clearing;
   856→    if (reduce <= 0) continue;
   857→    const newAmount = Math.max(0, payout.amountMinor - reduce);
   858→    if (payout.status === "requested" && newAmount <= 0) {
   859→      await tx.affiliatePayout.update({
   860→        where: { id: payoutId, status: "requested" },
   861→        data: {
   862→          status: "rejected",
   863→          amountMinor: 0,
   864→          failReason: "Auto-released: commissions fully reversed by refunds before approval",
   865→          processedAt: new Date(),
   866→        },
   867→      });
   868→      // Release the still-live claimed entries back to the available pool.
   869→      await tx.affiliateCommissionEntry.updateMany({
   870→        where: { payoutId, status: { in: ["available", "approved"] } },
   871→        data: { payoutId: null },
   872→      });
   873→    } else if (newAmount !== payout.amountMinor) {
   874→      const shrunk = await tx.affiliatePayout.updateMany({
   875→        where: { id: payoutId, status: { in: ["requested", "processing"] } },
   876→        data: { amountMinor: newAmount },
   877→      });
   878→      // Verification fix: we hold the row lock — a 0-count shrink means the
   879→      // open-set membership changed underneath us, which must be impossible.
   880→      // Fail the refund tx (rolled back atomically) rather than silently
   881→      // leaving the frozen payout amount overstated against consumed liability.
   882→      if (shrunk.count !== 1) {
   883→        throw new ApiError(409, "PAYOUT_STATE_CONFLICT", "Affiliate payout state changed during refund reconciliation — refund aborted, safe to retry");
   884→      }
   885→    }
   886→  }
   887→
   888→  return { affiliateReversalMinor: total, affiliateClearingMinor: clearingTotal, rule: "breakdown_snapshot_proportional" };
   889→}
   890→export async function requestAffiliatePayout(affiliateId: string): Promise<{ payoutId: string; amountMinor: number }> {
   891→  const settings = await getAffiliateSettings();
   892→  if (!settings.enabled) throw new ApiError(403, "AFFILIATE_DISABLED", "The affiliate programme is currently paused");
   893→
   894→  // C1.2 (Phase 2.2) — brought to the SAME standard as the seller payout
   895→  // system (store/payouts route): SERIALIZABLE isolation + bounded P2034
   896→  // retry of the whole transaction, balance truth read via the transaction
   897→  // client, and a CONDITIONAL entry claim that re-asserts `payoutId: null`.
   898→  //
   899→  // The previous implementation ran at READ COMMITTED with a check-then-act
   900→  // one-open-request guard and a NON-conditional `updateMany({ id IN ... })`
   901→  // claim: two concurrent requests both read the same `payoutId: null`
   902→  // entries, both froze the FULL requestable amount into two AffiliatePayout
   903→  // rows, and the second write overwrote the first's claim (last-write-wins)
   904→  // — a double-payment with per-row dedupKeys that the ledger could NOT
   905→  // catch. Serializable + conditional claims + the partial unique index
   906→  // (migration 20261105000000) close every layer of that window.
   907→  return withSerializableTxRetry(async (tx) => {
   908→    // One open request per affiliate — re-checked INSIDE the Serializable
   909→    // transaction (a concurrent duplicate submit now aborts one of the two
   910→    // transactions instead of both passing the check).
   911→    const open = await tx.affiliatePayout.findFirst({
   912→      where: { affiliateId, status: { in: ["requested", "processing"] } },
   913→      select: { id: true },
   914→    });
   915→    if (open) throw new ApiError(409, "PAYOUT_OPEN", "You already have a payout request in progress");
   916→
   917→    // Balance truth from THIS transaction's snapshot (was the global client).
   918→    const balances = await affiliateBalances(affiliateId, tx);
   919→    if (balances.requestableMinor < settings.minPayoutMinor) {
   920→      throw new ApiError(400, "BELOW_MIN_PAYOUT", `Minimum payout is RM ${(settings.minPayoutMinor / 100).toFixed(2)}`);
   921→    }
   922→
   923→    const amountMinor = balances.requestableMinor;
   924→    const now = new Date();
   925→    // Claim every matured, unclaimed entry; bind to the payout row we create.
   926→    const toClaim = await tx.affiliateCommissionEntry.findMany({
   927→      where: {
   928→        beneficiaryAffiliateId: affiliateId,
   929→        status: { in: ["approved", "available"] },
   930→        availableAt: { lte: now },
   931→        payoutId: null,
   932→      },
   933→      select: { id: true, amountMinor: true, reversedMinor: true, commissionId: true },
   934→    });
   935→    const payout = await tx.affiliatePayout.create({
   936→      data: { affiliateId, amountMinor, status: "requested", method: "manual_bank" },
   937→    });
   938→    if (toClaim.length > 0) {
   939→      // CONDITIONAL claim — `payoutId: null` re-asserted in the WHERE so a
   940→      // concurrent claimant's binding can never be silently overwritten.
   941→      // (Under Serializable any concurrent writer of these rows aborts this
   942→      // transaction first; the conditional guard is belt-and-braces against
   943→      // non-transactional writers and keeps the count assertion honest.)
   944→      const claim = await tx.affiliateCommissionEntry.updateMany({
   945→        where: { id: { in: toClaim.map((e) => e.id) }, payoutId: null },
   946→        data: { payoutId: payout.id },
   947→      });
   948→      if (claim.count !== toClaim.length) {
   949→        throw new ApiError(409, "PAYOUT_CLAIM_CONFLICT", "Commission entries changed concurrently — please retry the payout request");
   950→      }
   951→    }
   952→    // Ledger-derived statement (schema contract: built from entries at
   953→    // request time, never hand-edited) — order references + balance snapshot.
   954→    const orderRefs = toClaim.length
   955→      ? await tx.affiliateCommission.findMany({
   956→          where: { id: { in: [...new Set(toClaim.map((e) => e.commissionId))] } },
   957→          select: { id: true, orderNumber: true, productId: true, storeId: true },
   958→        })
   959→      : [];
   960→    const refMap = new Map(orderRefs.map((c) => [c.id, c]));
   961→    await tx.affiliatePayout.update({
   962→      where: { id: payout.id },
   963→      data: {
   964→        statementJson: toJson({
   965→          builtAt: now.toISOString(),
   966→          balances: balances,
   967→          entries: toClaim.map((e) => ({
   968→            entryId: e.id,
   969→            commissionId: e.commissionId,
   970→            orderNumber: refMap.get(e.commissionId)?.orderNumber ?? null,
   971→            amountMinor: e.amountMinor,
   972→            reversedMinor: e.reversedMinor,
   973→          })),
   974→        }),
   975→      },
   976→    });
   977→    return { payoutId: payout.id, amountMinor };
   978→  });
   979→}
