   116→/** Total unread seller messages — powers the "My messages" header badge.
   117→ *  Polls lightly + refreshes when the messages view is opened/left. */
   118→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
   119→  const [unread, setUnread] = useState(0);
   120→  useEffect(() => {
   121→    if (!shopperSignedIn) {
   122→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
   123→      setUnread(0);
   124→      return;
   125→    }
   126→    let alive = true;
   127→    const load = () =>
   128→      fetchJson<{ total: number }>("/api/chat/unread")
   129→        .then((d) => alive && setUnread(d.total ?? 0))
   130→        .catch(() => {});
   131→    load();
   132→    const t = setInterval(load, 15000);
   133→    return () => {
   134→      alive = false;
   135→      clearInterval(t);
   136→    };
   137→  }, [shopperSignedIn]);
   138→  // instant re-sync when a thread is read inside the messages view (the view
   139→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   140→  useEffect(() => {
   141→    if (!shopperSignedIn) return;
   142→    const sync = () =>
   143→      fetchJson<{ total: number }>("/api/chat/unread")
   144→        .then((d) => setUnread(d.total ?? 0))
   145→        .catch(() => {});
   146→    window.addEventListener("nomofee:chat-sync", sync);
   147→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   148→  }, [shopperSignedIn]);
   149→  // opening (or leaving) the messages view should immediately re-sync the badge
   150→  useEffect(() => {
   151→    if (!shopperSignedIn) return;
   152→    const t = setTimeout(
   153→      () =>
   154→        fetchJson<{ total: number }>("/api/chat/unread")
   155→          .then((d) => setUnread(d.total ?? 0))
   156→          .catch(() => {}),
   157→      viewName === "messages" ? 600 : 0
   158→    );
   159→    return () => clearTimeout(t);
   160→  }, [shopperSignedIn, viewName]);
   161→  return unread;
   162→}

   164→// ---- platform chrome config (global footer visibility) ---------------------
   165→// Admin → Settings → "Show global footer" (Setting ui.footer_visible) governs
   166→// the shopper footer. Public endpoint + 60 s module cache: toggling takes
   167→// effect on the next app load without per-navigation refetch chatter.
   168→let appConfigCache: { at: number; footerVisible: boolean } | null = null;
   169→
   170→function useFooterVisible() {
   171→  const [visible, setVisible] = useState(true);
   172→  useEffect(() => {
   173→    if (appConfigCache && Date.now() - appConfigCache.at < 60_000) {
   174→      // eslint-disable-next-line react-hooks/set-state-in-effect -- cache hit syncs an external module cache; the fresh-fetch path below sets state in a callback
   175→      setVisible(appConfigCache.footerVisible);
   176→      return;
   177→    }
   178→    let cancelled = false;
   179→    fetch("/api/sf/app-config", { cache: "no-store" })
   180→      .then((r) => (r.ok ? r.json() : null))
   181→      .then((b) => {
   182→        const v = (b as { data?: { footerVisible?: boolean } } | null)?.data?.footerVisible;
   183→        if (typeof v === "boolean" && !cancelled) {
   184→          appConfigCache = { at: Date.now(), footerVisible: v };
   185→          setVisible(v);
   186→        }
   187→      })
   188→      .catch(() => undefined);
   189→    return () => {
   190→      cancelled = true;
   191→    };
   192→  }, []);
   193→  return visible;
   194→}

   196→export default function ShopperShell() {

   203→  const [cartOpen, setCartOpen] = useState(false);
   204→  // Which store's customized cart experience to render in the drawer — set by
   205→  // PDP add-to-cart / storefront header; null = platform defaults.
   206→  const [cartStoreSlug, setCartStoreSlug] = useState<string | null>(null);
   207→  const [headerSearch, setHeaderSearch] = useState("");
   208→  // Back-navigation history stack (user request: back button on every page).
   209→  // viewRef mirrors `view` so nav() can push the PREVIOUS view without reading
   210→  // state inside the setState updater (StrictMode-safe).
   211→  const viewStack = useRef<ShopperView[]>([]);
   212→  const viewRef = useRef<ShopperView>({ name: "market" });
   213→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   214→  // intentionally NOT part of ShopperView — a synthetic render override.
   215→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   216→  const restored = useRef(false);
   217→
   218→  // ---- mount: activeStoreSlug override, else restore last view --------------
   219→  useEffect(() => {
   220→    if (restored.current) return;
   221→    restored.current = true;
   222→    const params = new URLSearchParams(window.location.search);
   223→
   224→    // Marketing attribution capture (once per landing): when the landing URL
   225→    // carries utm_* / nmcl parameters, fire the same-origin beacon so the
   226→    // server validates them and lands the `nm_ins_attr` cookie BEFORE the
   227→    // storefront fetch records the first store_view event (bounded wait — on
   228→    // beacon failure the visit proceeds untagged, never blocked).
   229→    const ATTR_PARAM_KEYS = ["utm_source", "utm_medium", "utm_campaign", "utm_term", "utm_content", "nmcl"] as const;
   230→    const hasAttrParams = ATTR_PARAM_KEYS.some((k) => params.get(k) != null);
   231→    const fireLandingBeacon = async (): Promise<void> => {
   232→      try {
   233→        const payload: Record<string, string> = {};
   234→        for (const k of [...ATTR_PARAM_KEYS, "sf"]) {
   235→          const v = params.get(k);
   236→          if (v != null && v !== "") payload[k] = v.slice(0, 200);
   237→        }
   238→        await fetch("/api/analytics/landing", {
   239→          method: "POST",
   240→          // X-SM-Client marker: proves app-origin to the CSRF gate in contexts
   241→          // where the proxy chain rewrites Host (preview panels / iframes).
   242→          headers: { "content-type": "application/json", ...webHeaders() },
   243→          body: JSON.stringify(payload),
   244→          keepalive: true,
   245→        });
   246→      } catch {
   247→        /* attribution is best-effort — never block the landing */
   248→      }
   249→    };
   250→
   251→    const proceed = () => {
   252→      // claim deep-link wins over everything
   253→      try {
   254→        const cj = params.get("claim_job");
   255→        const ct = params.get("claim_token");
   256→        if (cj && ct) {
   257→          setClaimLink({ jobId: cj, token: ct });
   258→          setReady(true);
   259→          return;
   260→        }
   261→        // ---- affiliate public landings (Task 2-A) — BEFORE ?sf= -------------
   262→        // ?al={code}: promotional link landing. The server resolves the code,
   263→        // records the money-grade attribution touch (visitor-cookie keyed) and
   264→        // returns the navigation target. Invalid codes fail closed → we fall
   265→        // back to the default restore below, never a fake page.
   266→        const al = params.get("al");
   267→        if (al) {
   268→          setReady(true);
   269→          void (async () => {
   270→            try {
   271→              const r = await postJson<{
   272→                target:
   273→                  | { kind: "product"; storeSlug: string; productSlug: string }
   274→                  | { kind: "store"; storeSlug: string }
   275→                  | { kind: "collection"; collectionSlug: string }
   276→                  | { kind: "showcase"; affiliateSlug: string };
   277→              }>("/api/affiliate/track", { code: al });
   278→              const t = r.target;
   279→              if (t.kind === "product") setView({ name: "pdp", slug: t.storeSlug, pSlug: t.productSlug });
   280→              else if (t.kind === "store") setView({ name: "storefront", slug: t.storeSlug });
   281→              else if (t.kind === "collection") setView({ name: "aff-collection", slug: t.collectionSlug, linkCode: al });
   282→              else setView({ name: "aff-showcase", slug: t.affiliateSlug, linkCode: al });
   283→            } catch {
   284→              /* invalid/tampered code — stay on the default view (fail closed) */
   285→            }
   286→          })();
   287→          return;
   288→        }
   289→        // ?ref={code}: referral join landing — fire the first-touch beacon
   290→        // (best-effort, non-blocking; the server sets the nm_aff_ref cookie),
   291→        // then open the affiliate hub (gate/activation reads “invited by @X”).
   292→        const refCode = params.get("ref");
   293→        if (refCode) {
   294→          const channel = params.get("src");
   295→          void postJson("/api/affiliate/track-referral", { code: refCode, channel: channel ?? undefined }).catch(() => {
   296→            /* best-effort */
   297→          });
   298→          setView({ name: "affiliate" });
   299→          setReady(true);
   300→          return;
   301→        }
   302→        // ?aff={slug}: public affiliate showcase page
   303→        const aff = params.get("aff");
   304→        if (aff) {
   305→          setView({ name: "aff-showcase", slug: aff });
   306→          setReady(true);
   307→          return;
   308→        }
   309→        // ?affc={slug}: public affiliate collection page
   310→        const affc = params.get("affc");
   311→        if (affc) {
   312→          setView({ name: "aff-collection", slug: affc });
   313→          setReady(true);
   314→          return;
   315→        }
   316→        // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   317→        const sf = params.get("sf");
   318→        if (sf) {
   319→          const sp = params.get("sp");
   320→          setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   321→          setReady(true);
   322→          return;
   323→        }
   324→      } catch {
   325→        /* ignore */
   326→      }
   327→      if (activeStoreSlug) {
   328→        setView({ name: "storefront", slug: activeStoreSlug });
   329→        setActiveStoreSlug(null);
   330→      } else {
   331→        try {
   332→          const raw = window.localStorage.getItem(VIEW_KEY);
   333→          if (raw) {
   334→            const v = JSON.parse(raw) as ShopperView;
   335→            if (v && typeof v.name === "string") setView(v);
   336→          }
   337→        } catch {
   338→          /* ignore */
   339→        }
   340→      }
   341→      setReady(true);
   342→    };
   343→
   344→    if (hasAttrParams) {
   345→      void Promise.race([fireLandingBeacon(), new Promise<void>((r) => setTimeout(r, 1500))]).then(() => proceed());
   346→    } else {
   347→      proceed();
   348→    }
   349→  }, [activeStoreSlug, setActiveStoreSlug]);
   350→
   351→  // ---- nav helper: persist + scroll to top ----------------------------------
   352→  useEffect(() => {
   353→    viewRef.current = view;
   354→  }, [view]);

   358→      viewStack.current.push(viewRef.current);

   363→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));

   365→      /* ignore */

   376→      /* ignore */

   455→
   456→  return (

   460→      {showChrome ? (

"" : view.name === "me" || view.name === "affiliate" ? view.tab ?? "" : ""}`}
   473→          initial={{ opacity: 0, y: 8 }}
   474→          animate={{ opacity: 1, y: 0 }}
   475→          exit={{ opacity: 0, y: -8 }}

  1840→
  1841→  // ---- urgency: reservation countdown (ticks once per second while open) ------
  1842→  const [openedAtMs, setOpenedAtMs] = useState(() => Date.now());
  1843→  const [nowMs, setNowMs] = useState(() => Date.now());
  1844→  useEffect(() => {

  1923→  function bumpQty(itemId: string, nextQty: number) {
  1924→    const it = cartSnapshot().items.find((x) => x.id === itemId);
  1925→    if (!it) return;

  1927→    // checkout/create re-validates authoritatively either way).

  1929→    if (clamped === it.qty) return;
  1930→    if (!qtyTimers.current.has(itemId)) qtyRevertPoint.current.set(itemId, it.qty);
  1931→    mutateCartItems((items) =>
  1932→      items.map((x) => (x.id === itemId ? { ...x, qty: clamped, lineTotalMinor: x.unitPriceMinor * clamped } : x))
  1933→    );
  1934→    const t = qtyTimers.current.get(itemId);
  1935→    if (t) clearTimeout(t);
  1936→    qtyTimers.current.set(
  1937→      itemId,
  1938→      setTimeout(() => {
  1939→        qtyTimers.current.delete(itemId);
  1940→        const intended = cartSnapshot().items.find((x) => x.id === itemId)?.qty ?? clamped;
  1941→        patchJson<{ cart: CartT }>(`/api/cart/items/${itemId}`, { qty: intended })
  1942→          .then((res) => {

  1944→            if (qtyTimers.current.has(itemId)) return; // newer taps pending — they will settle

  1946→            const localItem = cartSnapshot().items.find((x) => x.id === itemId);
  1947→            // The server clamps on stock changes (another device bought the
  1948→            // last units mid-burst). Adopt its number if we're still pristine.
  1949→            if (serverItem && localItem && serverItem.qty !== localItem.qty) {
  1950→              mutateCartItems((items) =>
  1951→                items.map((x) =>
  1952→                  x.id === itemId

  1954→                    : x

  1958→            }

  1960→          .catch(() => {
