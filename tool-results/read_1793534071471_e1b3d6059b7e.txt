    22→// F-14: single source of truth — the buyer-facing method list lives in
    23→// lib/payment-methods.ts and is shared with the shopper checkout UI, so the
    24→// server-accepted set and the buyer-selectable set can never drift again.

    26→// POST /api/checkout/create — header Idempotency-Key REQUIRED.
    27→// OPTION B (one licensed-PSP charge per checkout; no platform custody): the
    28→// buyer pays ONCE even when the cart spans N sellers. Server recomputes ALL
    29→// money math, groups cart lines per seller, and creates ONE PaymentGroup + N
    30→// per-seller Orders in a single transaction. Stock + salesCount move here (NOT
    31→// in the webhook). Each order carries its own commission lines → ledger →
    32→// seller wallet payout hold.
    33→//
    34→// PHASE 2.2 GUEST CHECKOUT RESTORATION — purchase WITHOUT registering:
    35→//   • Signed-in shoppers: the account path is BYTE-IDENTICAL to the previous
    36→//     requireShopper() behaviour (same session gate, same 401s, same fields).
    37→//   • Guests (no shopper session): the buyer MUST first verify the checkout
    38→//     email with a one-time code (POST /api/checkout/guest-verify/*) — the
    39→//     short-lived "gcv:v1:" token from that flow is REQUIRED here
    40→//     (fail-closed without it) and body.customer.email must EQUAL the token's
    41→//     verified email. Orders are created with shopperId NULL + customerEmail =
    42→//     the verified guest email (+ customerName from the body — no account
    43→//     fallback); the guest cart TOKEN (sm_cart cookie) is recorded on every
    44→//     order (Order.cartToken) so the payment webhook clears the purchased
    45→//     items from the guest cart.
    46→//   • Guests get NO wallet vouchers (entries are ignored — the account-bound
    47→//     UserVoucher machinery is skipped entirely), NO saved addresses (the
    48→//     inline shippingAddress body is the only destination), NO shopper
    49→//     notifications (webhook no-ops for guest groups). Affiliate attribution
    50→//     runs with shopperId null (self-purchase exclusion simply doesn't apply).
    51→//   • Restored guest orders become claimable into an account later through the
    52→//     EXISTING email-ownership-gated claim path (shopper-merge.ts) — unchanged.
    53→// Task 10-b claimable vouchers: body.vouchers = [{storeId, userVoucherId}] per
    54→// payment group. Verdicts run per seller group (lib/vouchers); a valid discount
    55→// voucher adds to the group discount (capped at the subtotal), freeshipping
    56→// zeroes that group's parcel cost. Redemption — UserVoucher→USED (usedAt,
    57→// usedOrderId) + Voucher.claimedCount+1 + Order.voucherId + discount applied —
    58→// happens INSIDE the same transaction as the orders (atomic). Idempotent per
    59→// order: the Idempotency-Key replay path above returns before any redemption,
    60→// and the status-guarded updateMany prevents concurrent double-redemption.
    61→// Redemption follows the existing coupon semantics: recorded at order CREATE,
    62→// not at payment.
    63→//
    64→// PHASE 2 / A2 — checkout round-trip diet (NO change to transaction
    65→// correctness; every financial/inventory guard is untouched):
    66→//   • idempotency replays run in ONE parallel batch (were 2 sequential reads)
    67→//   • commission context is loaded ONCE for ALL seller groups via the batched
    68→//     loadRateContext (was (5+2·D) sequential queries per LINE per group)
    69→//   • affiliate settings are fetched once and shared with the attribution
    70→//     resolver (was a duplicate setting.findMany)
    71→//   • the cart is read token-only (was a full deep include of every cart line)
    72→//   • in-tx fast-fail re-validation is ONE findMany (was per-line findUnique)
    73→//   • inventory logs are one createMany per seller group (was per-line create)
    74→//   • salesCount increments are aggregated per distinct product
    75→//   • the post-commit orders re-read is GONE — the summary is returned from
    76→//     inside the transaction (strictly more consistent), and notify/logActivity
    77→//     fan out in parallel
    78→// The AUTHORITATIVE oversell guard stays EXACTLY where it was: the conditional
    79→// `updateMany … WHERE stock >= qty` decrement, per line, inside the tx.
    80→export const POST = handle(async (req: Request) => {
    81→  const idemKey = req.headers.get("Idempotency-Key")?.trim();
    82→  if (!idemKey) {
    83→    throw new ApiError(400, "IDEMPOTENCY_KEY_REQUIRED", "Idempotency-Key header is required");
    84→  }
    85→
    86→  // PHASE 2 / A2: latency-anatomy instrumentation (src/lib/checkout-timing.ts)
    87→  // — one stdout JSON line per checkout when CHECKOUT_TIMING=1, zero overhead
    88→  // otherwise. Emitted on success AND error paths (status 0 = error path). The
    89→  // success emit is DEFERRED until the admission gate returns so the measured
    90→  // queue wait (gateWaitMs) is included.
    91→  const timing = startCheckoutTiming(idemKey);
    92→  // Holder object dodges TS's closure-assignment narrowing (pendingEmit is set
    93→  // inside runCheckout, consumed after withCheckoutAdmission).
    94→  const emitHolder: { pending: { status: number; extra: Partial<CheckoutTimingRecord> } | null } = { pending: null };
    95→  // checkoutGateStats() → the timing record's field names (gateActive/gateQueued).
    96→  const gateStats = (): Pick<CheckoutTimingRecord, "gateActive" | "gateQueued"> => {
    97→    const s = checkoutGateStats();
    98→    return { gateActive: s.inflight, gateQueued: s.waiting };
    99→  };
   100→
   101→  try {
   102→    // PHASE 2.2 GUEST CHECKOUT RESTORATION — account OR verified guest.
   103→    // A shopper session keeps EXACTLY the requireShopper() contract (same
   104→    // lookups, same 401 for missing/suspended accounts — byte-identical account
   105→    // path); NO session (or a non-shopper session) means the GUEST path, which
   106→    // is gated below on the "gcv:v1:" verified-email token.
   107→    const sess = await getSession();
   108→    let me: ShopperUserDTO | null = null;
   109→    if (sess?.userType === "shopper") {
   110→      const shopper = await db.shopperUser.findUnique({ where: { id: sess.userId } });
   111→      if (!shopper || shopper.status !== "active") {
   112→        throw new ApiError(401, "UNAUTHORIZED", "Shopper account unavailable");
   113→      }
   114→      me = { id: shopper.id, email: shopper.email, fullName: shopper.fullName, phone: shopper.phone };
   115→    }
   116→    timing.mark("auth");
   117→
   118→    // Idempotent replays — both unique-key lookups, fetched in ONE parallel
   119→    // batch (replays never consume admission slots, unchanged).
   120→    const [existingGroup, existing] = await Promise.all([
   121→      db.paymentGroup.findUnique({
   122→        where: { idemKey },
   123→        include: { orders: { orderBy: { createdAt: "asc" } } },
   124→      }),
   125→      db.order.findUnique({
   126→        where: { idemKey },
   127→        include: { paymentIntents: { orderBy: { createdAt: "desc" }, take: 1 } },
   128→      }),
   129→    ]);
   130→    timing.mark("idem");
   131→
   132→    if (existingGroup) {
   133→      // PHASE 2.2: guest groups carry shopperId NULL — a guest replay matches
   134→      // on the SAME null binding (the idemKey is a client-minted UUID, exactly
   135→      // as on the account path).
   136→      if (existingGroup.shopperId !== (me?.id ?? null)) throw new ApiError(403, "FORBIDDEN", "This checkout session belongs to a different account");
   137→      timing.emit(200, { gateWaitMs: 0, ...gateStats() });
   138→      return ok({
   139→        orderNumber: existingGroup.orders[0]?.orderNumber ?? "",
   140→        orderNumbers: existingGroup.orders.map((o) => o.orderNumber),
   141→        orderPaymentRef: existingGroup.ref,
   142→        paymentRef: existingGroup.ref,
   143→        totalMinor: existingGroup.totalMinor,
   144→        status: existingGroup.orders[0]?.status ?? "payment_pending",
   145→        duplicate: true,
   146→      });
   147→    }
   148→
   149→    // Idempotent replay #2 — legacy single-order checkout with this key.
   150→    if (existing) {
   151→      if (existing.shopperId !== (me?.id ?? null)) throw new ApiError(403, "FORBIDDEN", "This checkout session belongs to a different account");
   152→      timing.emit(200, { gateWaitMs: 0, ...gateStats() });
   153→      return ok({
   154→        orderNumber: existing.orderNumber,
   155→        orderNumbers: [existing.orderNumber],
   156→        orderPaymentRef: existing.paymentIntents[0]?.providerRef ?? null,
   157→        paymentRef: existing.paymentIntents[0]?.providerRef ?? null,
   158→        totalMinor: existing.totalMinor,
   159→        status: existing.status,
   160→        duplicate: true,
   161→      });
   162→    }
   163→
   164→    // PHASE 2.2: guest checkout create is UNAUTHENTICATED — apply a guest-only
   165→    // IP throttle (the account path keeps its exact prior posture: no IP
   166→    // throttle, admission gate + idempotency only). Placed AFTER the replay
   167→    // checks so a busy-retry replay never burns the limiter.
   168→    if (!me) rateLimit(req, "checkout-guest-create", 10);
   169→
   170→    // F2 REMEDIATION — bounded admission control. Everything below (pricing
   171→    // reads + the create transaction) runs under a process-level semaphore:
   172→    // excess arrivals WAIT (bounded FIFO, queue time measured separately) and,
   173→    // when the wait budget is exhausted, receive a CLEAN retryable
   174→    // `503 CHECKOUT_BUSY` + `Retry-After` instead of a raw 500. The gate sits
   175→    // AFTER the idempotency replays (key replays never consume admission
   176→    // slots) and BEFORE any write (a busy rejection leaves the cart untouched).
   177→    // The buyer retries with the SAME Idempotency-Key, so a post-busy retry
   178→    // converges on this checkout instead of duplicating it.
   179→    const runCheckout = async () => {
   180→
   181→      const body = (await req.json().catch(() => ({}))) as Record<string, unknown>;
   182→
   183→      // Contact details — PHASE 2.2 branches here:
   184→      // • ACCOUNT path (byte-identical to before): prefill from the account; a
   185→      //   shopper may still override per-order (e.g. sending a gift), blanks
   186→      //   fall back to the account.
   187→      // • GUEST path: the email must have been VERIFIED through the OTP flow —
   188→      //   body.guestVerification carries the short-lived "gcv:v1:" token, which
   189→      //   is verified constant-time and MUST bind the exact checkout email.
   190→      //   customerName comes from the body (no account fallback). FAIL-CLOSED
   191→      //   without a valid, unexpired token.
   192→      let customerName: string;
   193→      let customerEmail: string;
   194→      let customerPhone: string | null;
   195→      if (me) {
   196→        const customer = (body.customer ?? {}) as Record<string, unknown>;
   197→        customerName = String(customer.name ?? "").trim() || me.fullName;
   198→        customerEmail = (String(customer.email ?? "").trim() || me.email).toLowerCase();
   199→        customerPhone = customer.phone != null ? String(customer.phone).trim() || null : me.phone;
   200→      } else {
   201→        const guestVerification = body.guestVerification != null ? String(body.guestVerification) : null;
   202→        const gcv = verifyGuestCheckoutToken(guestVerification); // constant-time MAC + absolute expiry
   203→        if (!gcv) {
   204→          throw new ApiError(
   205→            401,
   206→            "GUEST_VERIFICATION_REQUIRED",
   207→            "Verify your email address with the one-time code before placing this order"
   208→          );
   209→        }
   210→        const customer = (body.customer ?? {}) as Record<string, unknown>;
   211→        customerEmail = String(customer.email ?? "").trim().toLowerCase();
   212→        if (!customerEmail || customerEmail !== gcv.email) {
   213→          throw new ApiError(400, "VALIDATION", "customer.email must match the email you verified");
   214→        }
   215→        customerName = String(customer.name ?? "").trim(); // required from the body — no account fallback for guests
   216→        customerPhone = customer.phone != null ? String(customer.phone).trim() || null : null;
   217→      }
   218→      if (!customerName) throw new ApiError(400, "VALIDATION", "customer.name is required");
   219→      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(customerEmail)) throw new ApiError(400, "VALIDATION", "customer.email is invalid");
   220→      if (customerPhone && !/^[+()\d\s-]{6,20}$/.test(customerPhone)) throw new ApiError(400, "VALIDATION", "customer.phone is invalid");
   221→
   222→      const provider = String(body.provider ?? "fpx_toyyibpay");
   223→      if (!KNOWN_PAYMENT_METHODS.includes(provider)) {
   224→        throw new ApiError(400, "INVALID_PROVIDER", `provider must be one of: ${KNOWN_PAYMENT_METHODS.join(", ")}`);
   225→      }
   226→
   227→      // Shipping address — full destination is REQUIRED (drives delivery eligibility).
   228→      const shippingAddress = (body.shippingAddress ?? {}) as Record<string, unknown>;
   229→      const addr = {
   230→        line1: String(shippingAddress.line1 ?? "").trim(),
   231→        city: String(shippingAddress.city ?? "").trim(),
   232→        state: String(shippingAddress.state ?? "").trim(),
   233→        postcode: String(shippingAddress.postcode ?? "").trim(),
   234→      };
   235→      if (!addr.line1 || !addr.city || !addr.state || !addr.postcode) {
   236→        throw new ApiError(400, "VALIDATION", "shippingAddress.line1, city, state and postcode are required");
   237→      }
   238→      if (!/^\d{5}$/.test(addr.postcode)) throw new ApiError(400, "VALIDATION", "shippingAddress.postcode must be a 5-digit Malaysian postcode");
   239→      const shippingAddressJson = toJson(addr);
   240→      timing.mark("body");
   241→
   242→      // One clock for the whole checkout: commission context, order timestamps,
   243→      // voucher redemption stamps and the expiry window all read the SAME
   244→      // `now` (previously each helper minted its own — ms-level drift between
   245→      // groups; a single snapshot is strictly more deterministic).
   246→      const now = new Date();
   247→      const expiresAt = new Date(now.getTime() + 30 * 60 * 1000);
   248→
   249→      // Recompute totals EXACTLY like quote — never trust client numbers.
   250→      const pricing = await computeGroupPricing({
   251→        items: (body.items ?? []) as { variantId: string; qty: number }[],
   252→        couponCode: body.couponCode != null ? String(body.couponCode) : null,
   253→        shippingRateId: body.shippingRateId != null ? String(body.shippingRateId) : null,
   254→        state: addr.state,
   255→      });
   256→
   257→      // Destination-aware delivery rule, enforced server-side (applies to every parcel).
   258→      assertShippingAvailable(pricing.shippingRateId, addr.state);
   259→      timing.mark("pricing");
   260→
   261→      // Claimable vouchers per seller group — hard mode: any unusable selection
   262→      // fails the checkout with 409 so the buyer can re-quote (never silently
   263→      // drop a voucher). One voucher per seller group (first entry wins).
   264→      // PHASE 2.2: wallet vouchers are ACCOUNT-bound (UserVoucher.userId) —
   265→      // guests have no wallet, so guest submissions are dropped here and the
   266→      // whole voucher-outcome path runs empty (applyVoucherOutcomesToGroups is
   267→      // a no-op with zero outcomes; redemption below never fires).
   268→      const voucherEntriesRaw = !me
   269→        ? []
   270→        : Array.isArray(body.vouchers)
   271→          ? (body.vouchers as VoucherEntryInput[]).filter((e) => e && typeof e === "object")
   272→          : [];
   273→      const seenStores = new Set<string>();
   274→      const voucherEntries = voucherEntriesRaw.filter((e) => {
   275→        const k = String(e?.storeId ?? "");
   276→        if (!k || seenStores.has(k)) return false;
   277→        seenStores.add(k);
   278→        return true;
   279→      });
   280→      const voucherOutcomes =
   281→        voucherEntries.length > 0
   282→          ? await buildVoucherOutcomes({
   283→              entries: voucherEntries,
   284→              shopperId: me?.id ?? null, // guests never reach here (entries dropped above)
   285→              groups: pricing.groups.map((g) => ({ storeId: g.storeId, subtotalMinor: g.subtotalMinor })),
   286→              addressState: addr.state,
   287→              hard: true,
   288→            })
   289→          : [];
   290→      const outcomeByStore = new Map(voucherOutcomes.map((o) => [o.storeId, o]));
   291→      const { groups: adjustedGroups, totals: adjustedTotals } = applyVoucherOutcomesToGroups(pricing.groups, voucherOutcomes);
   292→      timing.mark("voucher");
   293→
   294→      // AFFILIATE (Hub): money-grade per-item attribution is resolved server-side
   295→      // FIRST (last eligible click per product from the visitor-cookie touch
   296→      // table, 30-day window, self-purchases excluded, enrollment re-checked),
   297→      // then fed into the commission computation so the seller-funded affiliate
   298→      // pool is carved from seller proceeds in the SAME snapshot math. The client
   299→      // never supplies affiliate ids.
   300→      //
   301→      // POST-IMPLEMENTATION AUDIT (P3): freeze the sponsor-override policy TOGETHER
   302→      // with the attribution snapshot so the compensation arrangement of THIS order
   303→      // is fully determined at checkout. Later affiliate.sponsor_enabled changes
   304→      // affect new orders only — the payment webhook posts from this frozen value
   305→      // (OrderItem.affiliateSponsorEnabled), never from the live setting.
   306→      //
   307→      // PHASE 2 / A2: settings are fetched ONCE here and passed into the
   308→      // attribution resolver (previously resolveAffiliateAttributionForCart
   309→      // re-read them internally — a duplicate setting.findMany per checkout).
   310→      const affiliateSettings = await getAffiliateSettings();
   311→      const affiliateMap = await resolveAffiliateAttributionForCart({
   312→        shopperId: me?.id ?? null, // PHASE 2.2: guests — self-purchase exclusion not applicable
   313→        lines: pricing.groups.flatMap((g) => g.lines.map((l) => ({ variantId: l.variantId, productId: l.productId, storeId: g.storeId }))),
   314→        preloadedSettings: affiliateSettings,
   315→      }).catch(() => new Map<string, { affiliateId: string; sponsorId: string | null; linkId: string; rateBps: number; poolBaseMinor: number }>());
   316→
   317→      // Commission preview per seller group (percent + fixed fee parts per line).
   318→      // Computed on the VOUCHER-ADJUSTED totals so the discount reaches the ledger.
   319→      // A live store always has an owner — the ?? "" is a defensive fallback that
   320→      // resolves to default commission settings instead of throwing.
   321→      //
   322→      // PHASE 2 / A2: ONE batched rate context for EVERY seller group (fixed ~6
   323→      // round-trips, item-count independent) shared across the per-group
   324→      // computeOrderCommissions calls — previously each group resolved rates
   325→      // line-by-line with (5+2·D) sequential queries per line. Parity with the
   326→      // sequential resolver is byte-proven by scripts/test-commission-parity.ts.
   327→      const commissionCtx = await loadRateContext({
   328→        sellerIds: adjustedGroups.map((g) => g.sellerId ?? "").filter(Boolean),
   329→        productIds: adjustedGroups.flatMap((g) => g.lines.map((l) => l.productId ?? "")).filter(Boolean),
   330→        categoryIds: adjustedGroups.flatMap((g) => g.lines.map((l) => l.categoryId ?? "")).filter(Boolean),
   331→        now,
   332→      });
   333→      const groupCommissions = await Promise.all(
   334→        adjustedGroups.map((g) =>
   335→          computeOrderCommissions(
   336→            {
   337→              sellerId: g.sellerId ?? "",
   338→              items: g.lines.map((l) => ({
   339→                productId: l.productId,
   340→                categoryId: l.categoryId,
   341→                title: l.title,
   342→                variantLabel: l.variantLabel,
   343→                unitPriceMinor: l.unitPriceMinor,
   344→                qty: l.qty,
   345→                affiliate: affiliateMap.get(l.variantId) ?? null,
   346→              })),
   347→              orderDiscountMinor: g.discountMinor,
   348→              shippingMinorTotal: g.shippingMinor,
   349→              now,
   350→            },
   351→            commissionCtx
   352→          )
   353→        )
   354→      );
   355→      timing.mark("commission");
   356→
   357→      const providerRef = `ref_${randomToken(8)}`;
   358→
   359→      // The orders record the shopper's cart token so the webhook can clear the
   360→      // purchased items from the account cart once payment is verified.
   361→      // PHASE 2 / A2: token-only read (the previous deep include dragged every
   362→      // cart line's variant→product→store graph over the wire for one string).
   363→      // PHASE 2.2 GUEST CHECKOUT: guests have no account cart — the cart is
   364→      // token-keyed (sm_cart cookie), and the token read HERE is what gets
   365→      // recorded on the orders (Order.cartToken) so the payment webhook's guest
   366→      // path can clear the purchased items from that guest cart.
   367→      const cartToken = me
   368→        ? (await db.cart.findUnique({ where: { shopperId: me.id }, select: { token: true } }))?.token ?? null
   369→        : await getCartToken();
   370→      timing.mark("cart");
   371→
   372→      // Marketing Intelligence — resolve the shopper's validated attribution
   373→      // (last-touch cookie) BEFORE the transaction so it can be frozen onto every
   374→      // order as constant snapshot columns (same pattern as the country snapshot).
   375→      // Best-effort by design: an unreadable/absent cookie simply means untagged.
   376→      const attribution = await resolveAttribution().catch(() => null);
   377→
   378→      const orderNumbers = await Promise.all(pricing.groups.map(() => nextOrderNumber()));
   379→      timing.mark("counter");
   380→
   381→      // --- Transaction anatomy marks (see src/lib/checkout-timing.ts) ---------
   382→      let txAcquireMs = 0;
   383→      let txStatements = 0;
   384→      const txCallAt = performance.now();
   385→
   386→      // Per-order summary collected INSIDE the transaction (replaces the
   387→      // post-commit orders re-read — the data is exactly what this request
   388→      // wrote, with no re-read window).
   389→      const orderSummaries: { id: string; orderNumber: string; storeId: string; storeSlug: string; storeName: string; totalMinor: number }[] = [];
   390→
   391→      const group = await db.$transaction(async (tx) => {
   392→        txAcquireMs = performance.now() - txCallAt; // callback entry = connection acquired (the P2028 window)
   393→
   394→        // Friendly fast-fail re-validation inside the transaction (stock snapshot;
   395→        // the AUTHORITATIVE race-proof guard is the conditional decrement below).
   396→        // PHASE 2 / A2: one findMany for every line (was per-line findUnique).
   397→        const variantIds = pricing.items.map((l) => l.variantId);
   398→        const variantRows = await tx.productVariant.findMany({
   399→          where: { id: { in: variantIds } },
   400→          select: { id: true, isActive: true, stock: true, moq: true },
   401→        });
   402→        txStatements++;
   403→        const variantById = new Map(variantRows.map((v) => [v.id, v]));
   404→        for (const line of pricing.items) {
   405→          const v = variantById.get(line.variantId);
   406→          if (!v || !v.isActive || v.stock < line.qty) {
   407→            throw new ApiError(409, "OUT_OF_STOCK", `Insufficient stock for "${line.title}"${line.variantLabel ? ` (${line.variantLabel})` : ""}`);
   408→          }
   409→          if (line.qty < Math.max(1, v.moq)) {
   410→            throw new ApiError(409, "MOQ_NOT_MET", `Seller minimum for "${line.title}" is ${Math.max(1, v.moq)}`);
   411→          }
   412→        }
   413→
   414→        // Voucher rows for the redeemed selection (userVoucherId → voucherId/quota).
   415→        // PHASE 2.2: guests never reach here (redeemedIds is always empty — voucher

   447→        });

   449→
   450→        for (let gi = 0; gi < adjustedGroups.length; gi++) {

   452→          const comm = groupCommissions[gi];

   455→          const uvRow = voucherOutcome ? uvById.get(voucherOutcome.userVoucherId) : undefined;
   456→
   457→          const order = await tx.order.create({
   458→            data: {
   459→              storeId: g.storeId,
   460→              orderNumber,
   461→              customerName,
   462→              customerEmail,
   463→              customerPhone,
   464→              // PHASE 2.2: guest orders carry shopperId NULL + the VERIFIED
   465→              // customerEmail — they become claimable into an account later via
   466→              // the existing email-ownership-gated claim (shopper-merge.ts).
   467→              shopperId: me?.id ?? null,
   468→              status: "payment_pending",
   469→              channel: "storefront",
   470→              currency: g.currency,
   471→              subtotalMinor: g.subtotalMinor,
   472→              discountMinor: g.discountMinor,
   473→              shippingMinor: g.shippingMinor,
   474→              totalMinor: g.totalMinor,
   475→              couponCode: g.coupon.valid ? g.coupon.code : null,

   479→              // enforces the Malaysian 5-digit postcode + MY state rules above, so

   482→              shippingCountryCode: "MY",
   483→              shippingCountrySource: "checkout",

   487→              // country snapshot above — never recomputed, never backfilled.
   488→              attrCampaignId: attribution?.campaignId ?? null,
   489→              attrSource: attribution?.utmSource ?? null,
   490→              attrMedium: attribution?.utmMedium ?? null,
   491→              attrCampaign: attribution?.utmCampaign ?? null,
   492→              attrTerm: attribution?.utmTerm ?? null,
   493→              attrContent: attribution?.utmContent ?? null,
   494→              customerNote: body.customerNote != null ? String(body.customerNote) : null,
   495→              paymentProvider: provider,
   496→              cartToken,
   497→              idemKey: `${idemKey}:${g.storeId}`,
   498→              paymentGroupId: paymentGroup.id,
   499→            },
   500→          });
   501→          txStatements++;
   502→          orderSummaries.push({ id: order.id, orderNumber, storeId: g.storeId, storeSlug: g.storeSlug, storeName: g.storeName, totalMinor: g.totalMinor });
   503→

   505→          // with the same voucher can never double-redeem; quota-guarded so the
   506→          // redeemable pool cannot be exceeded.
   507→          if (voucherOutcome && uvRow) {
   508→            const redeemed = await tx.userVoucher.updateMany({
   509→              where: { id: voucherOutcome.userVoucherId, userId: me?.id ?? "", status: "AVAILABLE" },
   510→              data: { status: "USED", usedAt: now, usedOrderId: order.id },
   511→            });
   512→            txStatements++;
   513→            if (redeemed.count === 0) {
   514→              throw new ApiError(409, "VOUCHER_ALREADY_USED", "One of your selected vouchers has just been used — please review your voucher selection");

   518→                id: uvRow.voucher.id,
   519→                OR: [{ quota: null }, { claimedCount: { lt: uvRow.voucher.quota ?? Number.MAX_SAFE_INTEGER } }],
   520→              },
   521→              data: { claimedCount: { increment: 1 } },

   529→          await tx.orderItem.createMany({

   535→                productId: l.productId,

   538→                variantLabel: l.variantLabel,

   564→          // Stock + sales move AT CHECKOUT (per contract) — never repeated in webhook.
   565→          // LAUNCH-RUSH OVERSELL GUARD: the decrement is CONDITIONAL (UPDATE …
   566→          // WHERE stock >= qty). Under concurrent checkouts — think an influencer
   567→          // drop where hundreds of buyers hit Pay at once — the read above cannot
   568→          // be trusted (two transactions both see 1 left); Postgres evaluates the
   569→          // row-locked condition atomically, so EXACTLY ONE of the racing
   570→          // transactions matches. The loser gets count=0 → 409 → the WHOLE
   571→          // transaction (orders, items, vouchers, ledger) rolls back and the buyer
   572→          // is asked to adjust, while the winner takes the goods. Stock can never
   573→          // go negative, no reservation tables, no extra services — one UPDATE per

   730→  } catch (e) {

   732→    timing.emit(0, { gateWaitMs: 0, ...gateStats() });
