     3→// ---------------------------------------------------------------------------
     4→// Conversion Funnel — interactive 3D funnel diagram (HTML5 Canvas) + stage
     5→// streamgraph (recharts, centered "silhouette" stream) for the Insights page.
     6→//
     7→// Data: GET /api/store/insights/funnel?window=today|7d|30d|1y (seller-session
     8→// scoped; stage numbers are server-computed — this component never re-derives
     9→// step rates or drop-off).
    10→//
    11→// Canvas rendering notes:
    12→//   • The funnel is a tapered horizontal ribbon: thickness ∝ stage count
    13→//     (min clamp keeps small stages visible/hoverable). Sections WIDEN when a
    14→//     later stage exceeds an earlier one — the honest mixed-basis case
    15→//     (sessions vs orders), never clamped to fake a funnel shape.
    16→//   • Pseudo-3D: each section gets an extruded top face (lighter tint) and the
    17→//     last section a right cap (darker shade).
    18→//   • Icons are white-stroke SVG data-URL <Image>s drawn on the slab faces.
    19→//   • Callout pointer chips sit above the ribbon (wide viewports); on narrow
    20→//     viewports the count pills on the ribbon + the DOM legend carry the info.
    21→//   • Hover: hit-test rects stored per section → lift + glow + DOM tooltip
    22→//     with count / prev / step rate / drop-off-or-gained.
    23→//   • Animations: thickness/height ease toward targets via requestAnimationFrame
    24→//     (prefers-reduced-motion snaps instantly).
    25→//   • The canvas is aria-hidden; a visually-hidden table mirrors the stage data
    26→//     for screen readers.
    27→// ---------------------------------------------------------------------------

   109→function svgIcon(paths: string): string {
   110→  return `data:image/svg+xml;charset=utf-8,${encodeURIComponent(
   111→    `<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 24 24' fill='none' stroke='white' stroke-width='2' stroke-linecap='round' stroke-linejoin='round'>${paths}</svg>`
   112→  )}`;

   115→/** Hex mix toward white (t>0) or black (t<0) — for the 3D face shading. */
   116→function shade(hex: string, t: number): string {
   117→  const n = parseInt(hex.slice(1), 16);
   118→  const r = (n >> 16) & 255;
   119→  const g = (n >> 8) & 255;
   120→  const b = n & 255;
   121→  const m = t >= 0 ? 255 : 0;
   122→  const k = Math.abs(t);
   123→  const c = (v: number) => Math.round(v + (m - v) * k);
   124→  return `rgb(${c(r)},${c(g)},${c(b)})`;
   125→}
   126→function withAlpha(hex: string, a: number): string {
   127→  const n = parseInt(hex.slice(1), 16);
   128→  return `rgba(${(n >> 16) & 255},${(n >> 8) & 255},${n & 255},${a})`;
   129→}

   229→type StageRow = FunnelData["stages"][number];
   230→type HitRect = { i: number; x0: number; x1: number; y0: number; y1: number };
   231→
   232→function FunnelCanvas({ stages }: { stages: StageRow[] }) {
   233→  const wrapRef = useRef<HTMLDivElement | null>(null);
   234→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   235→  const [width, setWidth] = useState(0);
   236→  const [hover, setHover] = useState<number | null>(null);
   237→  const [tip, setTip] = useState<{ i: number; x: number; y: number } | null>(null);
   238→  const [iconTick, setIconTick] = useState(0);
   239→
   240→  const iconsRef = useRef<Map<string, HTMLImageElement>>(new Map());
   241→  const hitRef = useRef<HitRect[]>([]);
   242→  const animRef = useRef<{ thick: number[]; lift: number[]; raf: number | null }>({ thick: [], lift: [], raf: null });
   243→  const hoverRef = useRef<number | null>(null);
   244→  useEffect(() => {
   245→    hoverRef.current = hover;
   246→  }, [hover]);
   247→
   248→  const reducedMotion = useMemo(
   249→    () => typeof window !== "undefined" && window.matchMedia("(prefers-reduced-motion: reduce)").matches,
   250→    []
   251→  );
   252→
   253→  // Load the SVG icons once; each completed load triggers a redraw.
   254→  useEffect(() => {
   255→    for (const m of STAGE_META) {
   256→      if (iconsRef.current.has(m.key)) continue;
   257→      const img = new Image();
   258→      img.onload = () => setIconTick((t) => t + 1);
   259→      img.src = m.icon;
   260→      iconsRef.current.set(m.key, img);
   261→    }
   262→  }, []);
   263→
   264→  // Responsive width.
   265→  useEffect(() => {
   266→    const el = wrapRef.current;
   267→    if (!el) return;
   268→    const ro = new ResizeObserver((entries) => {
   269→      for (const e of entries) setWidth(Math.max(280, Math.round(e.contentRect.width)));
   270→    });
   271→    ro.observe(el);
   272→    setWidth(Math.max(280, Math.round(el.getBoundingClientRect().width)));
   273→    return () => ro.disconnect();
   274→  }, []);
   275→
   276→  const draw = useCallback(() => {
   277→    const canvas = canvasRef.current;
   278→    if (!canvas || width < 100) return;
   279→    const wide = width >= 640;
   280→    const H = wide ? 340 : 300;
   281→    const dpr = Math.min(2, window.devicePixelRatio || 1);
   282→    canvas.width = Math.round(width * dpr);
   283→    canvas.height = Math.round(H * dpr);
   284→    canvas.style.height = `${H}px`;
   285→    const ctx = canvas.getContext("2d");
   286→    if (!ctx) return;
   287→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   288→    ctx.clearRect(0, 0, width, H);
   289→
   290→    // ----- geometry -------------------------------------------------------
   291→    const DX = 24;
   292→    const DY = 12;
   293→    const topPad = wide ? 92 : 16;
   294→    const bottomPad = wide ? 34 : 26;
   295→    const fx0 = 14;
   296→    const fx1 = width - 14 - DX;
   297→    const fy = topPad + (H - topPad - bottomPad) / 2;
   298→    const maxThick = Math.max(60, H - topPad - bottomPad - 20);
   299→    const minThick = 18;
   300→
   301→    const counts = stages.map((s) => s.count);
   302→    const maxCount = Math.max(1, ...counts);
   303→    const targets = stages.map((s) => minThick + (s.count / maxCount) * (maxThick - minThick));
   304→    const anim = animRef.current;
   305→    if (anim.thick.length !== stages.length) {
   306→      anim.thick = reducedMotion ? [...targets] : stages.map(() => 0);
   307→      anim.lift = stages.map(() => 0);
   308→    }
   309→
   310→    const n = stages.length;
   311→    const cx = (i: number) => fx0 + ((i + 0.5) * (fx1 - fx0)) / n;
   312→    const bound = (i: number) => (i === 0 ? fx0 : i === n ? fx1 : (cx(i - 1) + cx(i)) / 2);
   313→
   314→    const empty = counts.every((c) => c === 0);
   315→    const thick = anim.thick;
   316→    const lift = anim.lift;
   317→
   318→    // Thickness at a boundary: flat over the owning stage's centre, linear
   319→    // between adjacent stage centres.
   320→    const thAt = (i: number) => {
   321→      if (i <= 0) return thick[0] ?? minThick;
   322→      if (i >= n) return thick[n - 1] ?? minThick;
   323→      return ((thick[i - 1] ?? minThick) + (thick[i] ?? minThick)) / 2;
   324→    };
   325→
   326→    hitRef.current = [];
   327→
   328→    // ----- draw ------------------------------------------------------------
   329→    for (let i = 0; i < n; i++) {
   330→      const bL = bound(i);
   331→      const bR = bound(i + 1);
   332→      const thL = Math.max(minThick * 0.6, thAt(i));
   333→      const thR = Math.max(minThick * 0.6, thAt(i + 1));
   334→      const meta = STAGE_META[i]!;
   335→      const color = empty ? "#cbd5e1" : meta.color;
   336→      const hov = hoverRef.current === i;
   337→      const ly = -lift[i]!;
   338→
   339→      // hit rect (generous: front face + the extruded depth column)
   340→      const maxTh = Math.max(thL, thR);
   341→      hitRef.current.push({ i, x0: bL, x1: bR + DX, y0: fy - maxTh / 2 - 8 + ly, y1: fy + maxTh / 2 + 6 + ly });
   342→
   343→      const frontPath = new Path2D();
   344→      frontPath.moveTo(bL, fy - thL / 2 + ly);
   345→      frontPath.lineTo(bR, fy - thR / 2 + ly);
   346→      frontPath.lineTo(bR, fy + thR / 2 + ly);
   347→      frontPath.lineTo(bL, fy + thL / 2 + ly);
   348→      frontPath.closePath();
   349→
   350→      const topPath = new Path2D();
   351→      topPath.moveTo(bL, fy - thL / 2 + ly);
   352→      topPath.lineTo(bR, fy - thR / 2 + ly);
   353→      topPath.lineTo(bR + DX, fy - thR / 2 - DY + ly);
   354→      topPath.lineTo(bL + DX, fy - thL / 2 - DY + ly);
   355→      topPath.closePath();
   356→
   357→      // shadow under hovered slab
   358→      if (hov) {
   359→        ctx.save();
   360→        ctx.shadowColor = "rgba(15,23,42,0.28)";
   361→        ctx.shadowBlur = 20;
   362→        ctx.shadowOffsetY = 10;
   363→        ctx.fillStyle = shade(color, hov ? 0.08 : -0.12);
   364→        ctx.fill(frontPath);
   365→        ctx.restore();
   366→      }
   367→
   368→      // top (3D) face — lighter tint; hovered pops brighter
   369→      const topFill = ctx.createLinearGradient(bL, fy - thL / 2 - DY + ly, bL, fy - thL / 2 + ly);
   370→      topFill.addColorStop(0, shade(color, hov ? 0.62 : 0.45));
   371→      topFill.addColorStop(1, shade(color, hov ? 0.2 : 0.08));
   372→      ctx.fillStyle = topFill;
   373→      ctx.fill(topPath);
   374→
   375→      // front face — the stage's colour gradient
   376→      const frontFill = ctx.createLinearGradient(0, fy - thL / 2 + ly, 0, fy + thL / 2 + ly);
   377→      frontFill.addColorStop(0, shade(color, hov ? 0.18 : 0.04));
   378→      frontFill.addColorStop(0.5, color);
   379→      frontFill.addColorStop(1, shade(color, hov ? -0.1 : -0.22));
   380→      ctx.fillStyle = frontFill;
   381→      ctx.fill(frontPath);
   382→
   383→      // right cap on the last section (depth shading)
   384→      if (i === n - 1) {
   385→        const capPath = new Path2D();
   386→        capPath.moveTo(bR, fy - thR / 2 + ly);
   387→        capPath.lineTo(bR + DX, fy - thR / 2 - DY + ly);
   388→        capPath.lineTo(bR + DX, fy + thR / 2 - DY + ly);
   389→        capPath.lineTo(bR, fy + thR / 2 + ly);
   390→        capPath.closePath();
   391→        ctx.fillStyle = shade(color, -0.32);
   392→        ctx.fill(capPath);
   393→      }
   394→
   395→      // hover outline
   396→      if (hov) {
   397→        ctx.strokeStyle = withAlpha("#ffffff", 0.85);
   398→        ctx.lineWidth = 1.5;
   399→        ctx.stroke(frontPath);
   400→        ctx.stroke(topPath);
   401→      }
   402→
   403→      // ----- icon + count on the slab ---------------------------------------
   404→      const icx = cx(i);
   405→      const t = thick[i] ?? minThick;
   406→      const stage = stages[i]!;
   407→      const cnt = stage.count.toLocaleString("en-MY");
   408→      const icon = iconsRef.current.get(meta.key);
   409→
   410→      if (t >= 30) {
   411→        ctx.font = FONT(700, 14);
   412→        const tw = ctx.measureText(cnt).width;
   413→        const hasIcon = t >= 38 && icon?.complete;
   414→        const iconW = hasIcon ? 20 : 0;
   415→        const startX = icx - (tw + iconW + (iconW ? 5 : 0)) / 2;
   416→        if (hasIcon && icon) {
   417→          ctx.drawImage(icon, startX, fy - 9 + ly, 18, 18);
   418→        }
   419→        ctx.fillStyle = "#ffffff";
   420→        ctx.textAlign = "left";
   421→        ctx.textBaseline = "middle";
   422→        ctx.fillText(cnt, startX + iconW + (iconW ? 5 : 0), fy + ly + 1);
   423→      } else {
   424→        // small slab → count pill floating above the ribbon
   425→        ctx.font = FONT(700, 12);
   426→        const tw = ctx.measureText(cnt).width;
   427→        const pillW = tw + 16;
   428→        const pillH = 20;
   429→        const px = icx - pillW / 2;
   430→        const py = fy - t / 2 - pillH - 5 + ly;
   431→        roundRect(ctx, px, py, pillW, pillH, 10);
   432→        ctx.fillStyle = "rgba(255,255,255,0.96)";
   433→        ctx.fill();
   434→        ctx.strokeStyle = withAlpha(color, 0.55);
   435→        ctx.lineWidth = 1;
   436→        ctx.stroke();
   437→        ctx.fillStyle = shade(color, -0.18);
   438→        ctx.textAlign = "left";
   439→        ctx.textBaseline = "middle";
   440→        ctx.fillText(cnt, px + 8, py + pillH / 2 + 1);
   441→        if (icon?.complete) ctx.drawImage(icon, icx - 9, fy - 9 + ly, 18, 18);
   442→      }
   443→
   444→      // ----- callout pointer chips (wide viewports) --------------------------
   445→      if (wide && !empty) {
   446→        const level = i % 2 === 0 ? 12 : 45;
   447→        const label = stage.label.replace(" (sessions)", "").replace(" (orders)", "");
   448→        ctx.font = FONT(600, 10.5);
   449→        const lw = ctx.measureText(label).width;
   450→        ctx.font = FONT(700, 12);
   451→        const vw = ctx.measureText(cnt).width;
   452→        ctx.font = FONT(500, 10);
   453→        const stepTxt = stage.stepRate != null ? `${stage.stepRate}%` : "—";
   454→        const sw = ctx.measureText(stepTxt).width;
   455→        const chipW = Math.max(lw, vw + 3 + sw) + 26;
   456→        const chipH = 27;
   457→        let chipX = icx - chipW / 2;
   458→        chipX = Math.max(4, Math.min(width - chipW - 4, chipX));
   459→        const chipY = level;
   460→
   461→        // pointer line: chip → slab top edge
   462→        const slabTop = fy - (thick[i] ?? minThick) / 2 + ly;
   463→        ctx.strokeStyle = withAlpha(color, 0.6);
   464→        ctx.lineWidth = 1;
   465→        ctx.beginPath();
   466→        const pxLine = chipX + chipW / 2;
   467→        ctx.moveTo(pxLine, chipY + chipH);
   468→        ctx.lineTo(pxLine, slabTop - 7);
   469→        ctx.stroke();
   470→        // arrowhead
   471→        ctx.beginPath();
   472→        ctx.moveTo(pxLine - 3.5, slabTop - 8);
   473→        ctx.lineTo(pxLine, slabTop - 3);
   474→        ctx.lineTo(pxLine + 3.5, slabTop - 8);
   475→        ctx.closePath();
   476→        ctx.fillStyle = withAlpha(color, 0.75);
   477→        ctx.fill();
   478→
   479→        // chip
   480→        roundRect(ctx, chipX, chipY, chipW, chipH, 8);
   481→        ctx.fillStyle = hov ? "#ffffff" : "rgba(255,255,255,0.94)";
   482→        ctx.fill();
   483→        ctx.strokeStyle = withAlpha(color, hov ? 0.95 : 0.5);
   484→        ctx.lineWidth = hov ? 1.6 : 1;
   485→        ctx.stroke();
   486→        // colour dot
   487→        ctx.beginPath();
   488→        ctx.arc(chipX + 10, chipY + chipH / 2, 3.5, 0, Math.PI * 2);
   489→        ctx.fillStyle = color;
   490→        ctx.fill();
   491→        // label + value
   492→        ctx.textBaseline = "middle";
   493→        ctx.textAlign = "left";
   494→        ctx.fillStyle = "#374151";
   495→        ctx.font = FONT(600, 10.5);
   496→        ctx.fillText(label, chipX + 18, chipY + chipH / 2 - 4.5);
   497→        ctx.fillStyle = shade(color, -0.25);
   498→        ctx.font = FONT(700, 12);
   499→        ctx.fillText(cnt, chipX + 18, chipY + chipH / 2 + 6.5);
   500→        ctx.fillStyle = "#6b7280";
   501→        ctx.font = FONT(500, 10);
   502→        ctx.fillText(stepTxt, chipX + 19 + ctx.measureText(cnt).width + 3, chipY + chipH / 2 + 6.5);
   503→      }
   504→    }
   505→
   506→    // empty-state caption
   507→    if (empty) {
   508→      ctx.font = FONT(500, 12.5);
   509→      ctx.fillStyle = "#94a3b8";
   510→      ctx.textAlign = "center";
   511→      ctx.textBaseline = "middle";
   512→      ctx.fillText("No funnel activity in this window yet", width / 2, fy + maxThick / 2 + 16);
   513→    }
   514→  }, [stages, width, reducedMotion, iconTick]);
   515→
   516→  // Animation loop — eases thickness toward targets and lift toward hover state.
   517→  useEffect(() => {
   518→    const targets = stages.map((s) => {
   519→      const counts = stages.map((x) => x.count);
   520→      const maxCount = Math.max(1, ...counts);
   521→      const wide = width >= 640;
   522→      const H = wide ? 340 : 300;
   523→      const topPad = wide ? 92 : 16;
   524→      const bottomPad = wide ? 34 : 26;
   525→      const maxThick = Math.max(60, H - topPad - bottomPad - 20);
   526→      return 18 + (s.count / maxCount) * (maxThick - 18);
   527→    });
   528→    const anim = animRef.current;
   529→    if (anim.thick.length !== stages.length) anim.thick = reducedMotion ? [...targets] : stages.map(() => 0);
   530→    if (anim.lift.length !== stages.length) anim.lift = stages.map(() => 0);
   531→
   532→    if (reducedMotion) {
   533→      anim.thick = [...targets];
   534→      anim.lift = stages.map((_, i) => (hoverRef.current === i ? 6 : 0));
   535→      draw();
   536→      return;
   537→    }
   538→
   539→    const step = () => {
   540→      let moving = false;
   541→      for (let i = 0; i < targets.length; i++) {
   542→        const dt = targets[i]! - (anim.thick[i] ?? 0);
   543→        const dl = (hoverRef.current === i ? 6 : 0) - (anim.lift[i] ?? 0);
   544→        if (Math.abs(dt) > 0.4) {
   545→          anim.thick[i] = (anim.thick[i] ?? 0) + dt * 0.16;
   546→          moving = true;
   547→        } else anim.thick[i] = targets[i]!;
   548→        if (Math.abs(dl) > 0.3) {
   549→          anim.lift[i] = (anim.lift[i] ?? 0) + dl * 0.28;
   550→          moving = true;
   551→        } else anim.lift[i] = hoverRef.current === i ? 6 : 0;
   552→      }
   553→      draw();
   554→      anim.raf = moving ? requestAnimationFrame(step) : null;
   555→    };
   556→    anim.raf = requestAnimationFrame(step);
   557→    return () => {
   558→      if (anim.raf != null) cancelAnimationFrame(anim.raf);
   559→      anim.raf = null;
   560→    };
   561→  }, [stages, width, hover, reducedMotion, draw, iconTick]);
   562→
   563→  // Pointer events → hover hit-testing + tooltip.
   564→  const onPointer = useCallback((e: React.PointerEvent<HTMLCanvasElement>) => {
   565→    const canvas = canvasRef.current;
   566→    if (!canvas) return;
   567→    const rect = canvas.getBoundingClientRect();
   568→    const x = e.clientX - rect.left;
   569→    const y = e.clientY - rect.top;
   570→    let found: number | null = null;
   571→    for (const h of hitRef.current) {
   572→      if (x >= h.x0 && x <= h.x1 && y >= h.y0 && y <= h.y1) {
   573→        found = h.i;
   574→        break;
   575→      }
   576→    }
   577→    setHover(found);
   578→    setTip(found != null ? { i: found, x: e.clientX - rect.left, y: e.clientY - rect.top } : null);
   579→  }, []);
   580→
   581→  const stage = tip != null ? stages[tip.i] : null;
   582→  const meta = tip != null ? STAGE_META[tip.i] : null;
   583→
   584→  return (
   585→    <div ref={wrapRef} className="relative w-full">
   586→      <canvas
   587→        ref={canvasRef}
   588→        className="block w-full touch-none select-none"
   589→        style={{ cursor: hover != null ? "pointer" : "default" }}
   590→        aria-hidden="true"
   591→        onPointerMove={onPointer}
   592→        onPointerLeave={() => {
   593→          setHover(null);
   594→          setTip(null);
   595→        }}
   596→      />
   597→      {/* screen-reader mirror of the diagram */}
   598→      <table className="sr-only">
   599→        <caption>Conversion funnel stages for the selected window</caption>
   600→        <thead>
   601→          <tr>
   602→            <th scope="col">Stage</th>
   603→            <th scope="col">Count</th>
   604→            <th scope="col">Previous window</th>
   605→            <th scope="col">Step rate</th>
   606→            <th scope="col">Drop-off</th>
   607→          </tr>
   608→        </thead>
   609→        <tbody>
   610→          {stages.map((s) => (
   611→            <tr key={s.key}>
   612→              <th scope="row">{s.label}</th>
   613→              <td>{s.count}</td>
   614→              <td>{s.prev}</td>
   615→              <td>{s.stepRate == null ? "not computable" : `${s.stepRate}%`}</td>
   616→              <td>{s.dropOff == null ? "—" : s.dropOff > 0 ? `${s.dropOff} (${s.dropOffPct ?? "—"}%)` : s.dropOff < 0 ? `gained ${-s.dropOff}` : "0"}</td>
   617→            </tr>
   618→          ))}
   619→        </tbody>
   620→      </table>
   621→
   622→      {stage && meta && tip ? (
   623→        <div
   624→          role="tooltip"
   625→          className="pointer-events-none absolute z-10 w-52 rounded-xl border bg-popover/95 p-3 text-xs shadow-lg backdrop-blur"
   626→          style={{
   627→            left: Math.max(6, Math.min(Math.max(280, width) - 216, tip.x + 16)),
   628→            top: Math.max(6, tip.y - 64),
   629→          }}
   630→        >
   631→          <p className="flex items-center gap-1.5 font-semibold">
   632→            <span className="h-2 w-2 rounded-full" style={{ backgroundColor: meta.color }} aria-hidden="true" />
   633→            {stage.label}
   634→          </p>
   635→          <p className="mt-1 tabular-nums">
   636→            <span className="font-bold" style={{ color: shade(meta.color, -0.15) }}>{stage.count.toLocaleString("en-MY")}</span>
   637→            <span className="text-muted-foreground"> in window · prev {stage.prev.toLocaleString("en-MY")}</span>
   638→          </p>
   639→          <p className="mt-0.5 tabular-nums text-muted-foreground">
   640→            Step {stage.stepRate == null ? "—" : `${stage.stepRate}%`}
   641→            {stage.dropOff != null ? (
   642→              <span className={stage.dropOff > 0 ? "text-rose-600" : stage.dropOff < 0 ? "text-emerald-600" : ""}>
   643→                {stage.dropOff > 0
   644→                  ? ` · drop ${stage.dropOff.toLocaleString("en-MY")}${stage.dropOffPct != null ? ` (${stage.dropOffPct}%)` : ""}`
   645→                  : stage.dropOff < 0
   646→                    ? ` · gained ${(-stage.dropOff).toLocaleString("en-MY")}`
   647→                    : " · drop 0"}
   648→              </span>
   649→            ) : null}
   650→          </p>
   651→        </div>
   652→      ) : null}
   653→    </div>
   654→  );
   655→}

   657→function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   658→  ctx.beginPath();
   659→  ctx.moveTo(x + r, y);
   660→  ctx.lineTo(x + w - r, y);
   661→  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
   662→  ctx.lineTo(x + w, y + h - r);
   663→  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
   664→  ctx.lineTo(x + r, y + h);
   665→  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
   666→  ctx.lineTo(x, y + r);
   667→  ctx.quadraticCurveTo(x, y, x + r, y);
   668→  ctx.closePath();
   669→}

   673→/** Stage flow over time. Weekly/Monthly/Yearly → centred streamgraph (recharts
   674→ *  stacked areas over an invisible negative-offset base — the "silhouette"
   675→ *  stream). Daily → the window is a single day, so a stacked bar of today's
   676→ *  stages replaces the degenerate one-point stream. */
   677→function FunnelStream({ data, view }: { data: FunnelData; view: FunnelViewKey }) {
   678→  const series = useMemo(
   679→    () =>
   680→      data.byDay.map((d) => {
   681→        const vals = [d.visits, d.productViews, d.cartAdds, d.checkouts, d.paidOrders];
   682→        const total = vals.reduce((a, b) => a + b, 0);
   683→        return { day: d.day, off: -total / 2, v0: d.visits, v1: d.productViews, v2: d.cartAdds, v3: d.checkouts, v4: d.paidOrders, total };
   684→      }),
   685→    [data.byDay]
   686→  );
   687→  const hasFlow = series.some((p) => p.total > 0);
   688→
   689→  if (view === "daily") {
   690→    const p = series[series.length - 1];
   691→    const vals = p ? [p.v0, p.v1, p.v2, p.v3, p.v4] : [0, 0, 0, 0, 0];
   692→    const total = vals.reduce((a, b) => a + b, 0);
   693→    return (
   694→      <div>
   695→        <p className="mb-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Today&apos;s stages</p>
   696→        {total === 0 ? (
   697→          <p className="rounded-xl border border-dashed py-6 text-center text-xs text-muted-foreground">No activity today yet.</p>
   698→        ) : (
   699→          <div className="space-y-2">
   700→            <div className="flex h-9 w-full overflow-hidden rounded-lg" role="img" aria-label="Today's funnel stages stacked bar">
   701→              {vals.map((v, i) =>
   702→                v <= 0 ? null : (
   703→                  <div
   704→                    key={i}
   705→                    className="h-full min-w-[4px] transition-[flex-grow] duration-500"
   706→                    style={{ flexGrow: v, backgroundColor: STAGE_META[i]!.color }}
   707→                    title={`${STAGE_META[i]!.label}: ${v}`}
   708→                  />
   709→                )
   710→              )}
   711→            </div>
   712→            <p className="text-[11px] text-muted-foreground">
   713→              {total.toLocaleString("en-MY")} stage events today · segments scale to their share.
   714→            </p>
   715→          </div>
   716→        )}
   717→      </div>
   718→    );
   719→  }
   720→
   721→  return (
   722→    <div>
   723→      <p className="mb-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Stage flow over time</p>
   724→      {hasFlow ? (
   725→        <div className="h-[190px] w-full" role="img" aria-label="Funnel stage flow streamgraph by day">
   726→          <ResponsiveContainer width="100%" height="100%">
   727→            <AreaChart data={series} margin={{ top: 4, right: 8, left: 8, bottom: 0 }}>
   728→              {STAGE_META.map((m, i) => (
   729→                <defs key={m.key}>
   730→                  <linearGradient id={`fsGrad${i}`} x1="0" y1="0" x2="0" y2="1">
   731→                    <stop offset="0%" stopColor={m.color} stopOpacity={0.72} />
   732→                    <stop offset="100%" stopColor={m.color} stopOpacity={0.28} />
   733→                  </linearGradient>
   734→                </defs>
   735→              ))}
   736→              <XAxis
   737→                dataKey="day"
   738→                tickFormatter={(v: string) => String(v).slice(5)}
   739→                fontSize={10}
   740→                tickLine={false}
   741→                axisLine={false}
   742→                minTickGap={30}
   743→                stroke="currentColor"
   744→                className="text-muted-foreground"
   745→              />
   746→              <Tooltip content={<StreamTip />} />
   747→              {/* invisible negative base centred around zero → silhouette stream */}
   748→              <Area dataKey="off" stackId="s" type="monotone" stroke="none" fill="transparent" isAnimationActive={false} />
   749→              {STAGE_META.map((m, i) => (
   750→                <Area
   751→                  key={m.key}
   752→                  dataKey={`v${i}`}
   753→                  stackId="s"
   754→                  type="monotone"
   755→                  stroke={m.color}
   756→                  strokeWidth={1.4}
   757→                  fill={`url(#fsGrad${i})`}
   758→                  isAnimationActive={false}
   759→                />
   760→              ))}
   761→            </AreaChart>
   762→          </ResponsiveContainer>
   763→        </div>
   764→      ) : (
   765→        <p className="rounded-xl border border-dashed py-6 text-center text-xs text-muted-foreground">
   766→          No stage activity in this window yet — the stream appears as visitor tracking and orders accrue.
   767→        </p>
   768→      )}
   769→    </div>
   770→  );
   771→}

   773→function StreamTip(props: { active?: boolean; payload?: Array<{ payload: { day: string; v0: number; v1: number; v2: number; v3: number; v4: number; total: number } }>; label?: string }) {
   774→  const { active, payload } = props;
   775→  if (!active || !payload?.length) return null;
   776→  const p = payload[0]!.payload;
   777→  const rows = [p.v0, p.v1, p.v2, p.v3, p.v4];
   778→  return (
   779→    <div className="rounded-xl border bg-popover/95 p-3 text-xs shadow-lg backdrop-blur">
   780→      <p className="mb-1.5 font-semibold">{p.day}</p>
   781→      <ul className="space-y-0.5">
   782→        {STAGE_META.map((m, i) => (
   783→          <li key={m.key} className="flex items-center justify-between gap-4 tabular-nums">
   784→            <span className="flex items-center gap-1.5 text-muted-foreground">
   785→              <span className="h-2 w-2 rounded-full" style={{ backgroundColor: m.color }} aria-hidden="true" />
   786→              {m.label}
   787→            </span>
   788→            <span className="font-medium">{rows[i]!.toLocaleString("en-MY")}</span>
   789→          </li>
   790→        ))}
   791→      </ul>
   792→    </div>
   793→  );
   794→}
