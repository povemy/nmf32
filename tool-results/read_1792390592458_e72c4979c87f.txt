   101→export function PayoutsView({ nav }: { nav: AdminNav }) {

   104→

   108→  const [previewError, setPreviewError] = useState<string | null>(null);
   109→  const [running, setRunning] = useState(false);
   110→
   111→  const loadPreview = useCallback(async () => {
   112→    setPreviewLoading(true);
   113→    setPreviewError(null);
   114→    try {
   115→      const body = await apiSend<DryRunBody>("/api/admin/payout-batches", "POST", { dryRun: true });
   116→      setPreview(body);
   117→    } catch (e) {
   118→      setPreview(null);
   119→      setPreviewError(e instanceof Error ? e.message : "Preview failed");
   120→    } finally {
   121→      setPreviewLoading(false);
   122→    }
   123→  }, []);
   124→
   125→  useEffect(() => {
   126→    if (open) void loadPreview();

   354→        </Card>

   356→        <Card className="py-0">
   357→          <CardContent className="p-4">
   358→            <p className="text-sm text-muted-foreground">Batch not found — it may have been removed.</p>

   360→        </Card>

   363→      <Card className="py-0">
   364→        <CardHeader className="px-4 pt-4 sm:px-6 sm:pt-5">

   453→            .then(() => {
   454→              toast.success("Payout marked failed", { description: "Ledger reversal posted; funds returned to wallet." });
   455→              setFailTarget(null);
   456→              done();
   457→            })
   458→            .catch((e: unknown) => toast.error("Failed to update payout", { description: e instanceof Error ? e.message : String(e) }));
   459→        }}
   460→      />
   461→    </div>
   462→  );

   464→
   465→function PayoutExpand({
   466→  payout,
   467→  onMarkPaid,
   468→  onMarkFailed,
   469→}: {
   470→  payout: PayoutRow;
   471→  onMarkPaid: () => void;
   472→  onMarkFailed: () => void;
   473→}) {
   474→  const st = parseJsonSafe<Statement>(payout.statementJson, {});
   475→  const lines = st.lines ?? [];
   476→  // Reconciliation audit: statement lines must sum EXACTLY to the net payout.
   477→  // Ledger-truth statements (reconciled: true) always do; legacy demo snapshots
   478→  // with decorated numbers fail this check and are flagged honestly.
   479→  const linesSum = lines.reduce((s, l) => s + l.amountMinor, 0);
   480→  const reconciles = lines.length > 0 && linesSum === payout.amountMinor;
   481→  return (
   482→    <div className="grid grid-cols-1 gap-4 p-3 lg:grid-cols-3">
   483→      <div className="lg:col-span-2">
   484→        <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   485→          Settlement statement{st.sellerName ? ` — ${st.sellerName}` : ""}
   486→        </p>
   487→        {lines.length === 0 ? (
   488→          <p className="text-xs text-muted-foreground">No statement lines recorded for this payout.</p>
   489→        ) : (
   490→          <div className="overflow-hidden rounded-lg border bg-card">
   491→            <div className={cn("flex items-center gap-1.5 border-b px-3 py-2 text-[11px] font-semibold", reconciles ? "bg-emerald-50 text-emerald-800 dark:bg-emerald-950/40 dark:text-emerald-300" : "bg-amber-50 text-amber-800 dark:bg-amber-950/40 dark:text-amber-300")}>
   492→              {reconciles ? (
   493→                <>
   494→                  <ShieldCheck className="h-3.5 w-3.5" /> Ledger-reconciled — every figure is aggregated from ledger lines and Σ(lines) = net payout
   495→                </>
   496→              ) : (
   497→                <>
   498→                  <AlertTriangle className="h-3.5 w-3.5" /> Not reconciled — Σ(lines) {formatMinor(linesSum)} ≠ payout {formatMinor(payout.amountMinor)} (legacy demo snapshot)
   499→                </>
   500→              )}
   501→            </div>
   502→            <Table>
   503→              <TableBody>
   504→                {lines.map((l, i) => (
   505→                  <TableRow key={i}>
   506→                    <TableCell className="text-sm">{l.label}</TableCell>
   507→                    <TableCell className="w-32 text-right"><Money minor={l.amountMinor} /></TableCell>
   508→                  </TableRow>
   509→                ))}
   510→                <TableRow className="border-t-2">
   511→                  <TableCell className="text-sm font-semibold">Net payout</TableCell>
   512→                  <TableCell className="w-32 text-right"><Money minor={payout.amountMinor} bold /></TableCell>
   513→                </TableRow>
   514→              </TableBody>
   515→            </Table>
   516→          </div>
   517→        )}
   518→        {st.orderCount != null || st.periodTo ? (
   519→          <p className="mt-2 text-[11px] text-muted-foreground">
   520→            {st.orderCount != null ? `${st.orderCount} paid order${st.orderCount === 1 ? "" : "s"}` : ""}
   521→            {st.periodFrom && st.periodTo ? ` · ${fmtDT(st.periodFrom)} → ${fmtDT(st.periodTo)}` : ""}
   522→          </p>
   523→        ) : null}
   524→      </div>
   525→
   526→      <div className="space-y-3">
   527→        <div>
   528→          <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank</p>
   529→          {payout.bankAccount ? (
   530→            <div className="mt-1.5 space-y-1 rounded-lg border bg-card p-3 text-sm">
   531→              <p className="flex items-center gap-1.5 font-medium">
   532→                <Landmark className="h-3.5 w-3.5 text-teal-600 dark:text-teal-400" /> {payout.bankAccount.bankName}
   533→              </p>
   534→              <p className="text-muted-foreground">{payout.bankAccount.holderName}</p>
   535→              <p className="font-mono text-xs">{payout.bankAccount.accountNumberMasked}</p>
   536→            </div>
   537→          ) : (
   538→            <p className="mt-1.5 text-xs text-muted-foreground">No bank account linked</p>
   539→          )}
   540→        </div>
   541→        {payout.providerRef ? (
   542→          <div>
   543→            <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Bank reference</p>
   544→            <p className="mt-1 font-mono text-xs">{payout.providerRef}</p>
   545→          </div>
   546→        ) : null}
   547→        {payout.status === "processing" ? (
   548→          <div className="flex flex-col gap-2">
   549→            <Button size="sm" className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700" onClick={onMarkPaid}>
   550→              Mark paid…
   551→            </Button>
   552→            <Button size="sm" variant="outline" className="min-h-[44px] border-red-200 text-red-600 hover:bg-red-50 dark:border-red-900 dark:hover:bg-red-950/40" onClick={onMarkFailed}>
   553→              Mark failed…
   554→            </Button>
   555→          </div>
   556→        ) : (
   557→          <p className={cn("text-xs", payout.status === "paid" ? "text-emerald-700 dark:text-emerald-400" : "text-muted-foreground")}>
   558→            {payout.status === "paid" ? `Paid ${fmtDT(payout.paidAt)}` : "No actions available"}
   559→          </p>
   560→        )}
   561→      </div>
   562→    </div>
   563→  );
   564→}
   565→
   566→function MarkPaidDialog({
   567→  payout,
   568→  onClose,
   569→  onDone,
   570→}: {
   571→  payout: PayoutRow | null;
   572→  onClose: () => void;
   573→  onDone: () => void;
   574→}) {
   575→  const [refNo, setRefNo] = useState("");
   576→  const [saving, setSaving] = useState(false);
   577→
   578→  useEffect(() => {
   579→    if (payout) setRefNo("");
   580→  }, [payout]);
   581→
   582→  const submit = async () => {
   583→    if (!payout || !refNo.trim()) return;
   584→    setSaving(true);
   585→    try {
   586→      await apiSend(`/api/admin/payouts/${payout.id}/mark-paid`, "POST", { referenceNo: refNo.trim() });
   587→      toast.success("Payout marked paid", {
   588→        description: `${payout.referenceNo} · bank ref ${refNo.trim()} — ledger T4 posted.`,
   589→      });
   590→      onDone();
   591→    } catch (e) {
   592→      toast.error("Mark paid failed", { description: e instanceof Error ? e.message : String(e) });
   593→    } finally {
   594→      setSaving(false);
   595→    }
   596→  };
   597→
   598→  return (
   599→    <Dialog open={payout !== null} onOpenChange={(o) => !o && onClose()}>
   600→      <DialogContent className="sm:max-w-md">
   601→        <DialogHeader>
   602→          <DialogTitle>Mark payout as paid</DialogTitle>
   603→          <DialogDescription>
   604→            {payout ? `${payout.referenceNo} · ${formatMinor(payout.amountMinor)} to ${payout.sellerName ?? "seller"}.` : ""}
   605→          </DialogDescription>
   606→        </DialogHeader>
   607→        <div className="space-y-1.5">
   608→          <Label htmlFor="payout-ref">Bank reference number (required)</Label>
   609→          <Input
   610→            id="payout-ref"
   611→            value={refNo}
   612→            onChange={(e) => setRefNo(e.target.value)}
   613→            placeholder="e.g. BK-20261008-8817"
   614→            className="min-h-[44px]"
   615→            autoFocus
   616→          />
   617→          <p className="text-[11px] text-muted-foreground">Recorded on the payout and posted to the ledger as payout_confirmed (T4).</p>
   618→        </div>
   619→        <DialogFooter>
   620→          <Button variant="outline" onClick={onClose} disabled={saving} className="min-h-[44px] sm:min-h-0">Cancel</Button>
   621→          <Button onClick={() => void submit()} disabled={saving || !refNo.trim()} className="min-h-[44px] bg-emerald-600 hover:bg-emerald-700 sm:min-h-0">
   622→            {saving ? "Posting…" : "Confirm paid"}
   623→          </Button>
   624→        </DialogFooter>
   625→      </DialogContent>
   626→    </Dialog>
   627→  );
   628→}
