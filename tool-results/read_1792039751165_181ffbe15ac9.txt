   185→// ---------------------------------------------------------------------------
   186→// data hook — profile + wallet + purchase counts + like/viewed counters
   187→// ---------------------------------------------------------------------------
   188→
   189→function useMeData(enabled: boolean) {
   190→  const [profile, setProfile] = useState<ProfileT | null>(null);
   191→  const [wallet, setWallet] = useState<WalletT | null>(null);
   192→  const [counts, setCounts] = useState<BucketCounts | null>(null);
   193→  const [likeCount, setLikeCount] = useState(0);
   194→  const [viewedCount, setViewedCount] = useState(0);
   195→  const [loading, setLoading] = useState(enabled);
   196→
   197→  const refresh = useCallback(async () => {
   198→    if (!enabled) return;
   199→    const [p, w, o, lk, vw] = await Promise.allSettled([
   200→      fetchJson<ProfileT>("/api/me/profile"),
   201→      fetchJson<WalletT>("/api/me/wallet"),
   202→      fetchJson<{ bucketCounts: BucketCounts; orders: OrderRowT[] }>("/api/me/orders?size=1"),
   203→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=LIKE&size=1"),
   204→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=VIEWED&size=1"),
   205→    ]);
   206→    if (p.status === "fulfilled") setProfile(p.value);
   207→    if (w.status === "fulfilled") {
   208→      // DORMANT SCAFFOLD — MESH FINANCE: while MESH_FINANCE_ENABLED is false
   209→      // the wallet API answers a `disabled` envelope (wallet/checkIn null).
   210→      // Normalize it into the UI shape so the kept scaffold cards render
   211→      // inert zeroes instead of crashing; the cards stay hidden behind the flag.
   212→      const raw = w.value as Partial<WalletT> & { disabled?: boolean };
   213→      setWallet({
   214→        wallet: raw.wallet ?? { balanceMinor: 0, coinBalance: 0, spayLaterLimitMinor: 0, spayLaterUsedMinor: 0, spayLaterAvailableMinor: 0, checkInStreak: 0 },
   215→        vouchers: raw.vouchers ?? { available: 0, used: 0 },
   216→        checkIn: raw.checkIn ?? { canClaim: false, nextClaimAt: null },
   217→      });
   218→    }
   219→    if (o.status === "fulfilled") setCounts(o.value.bucketCounts);
   220→    if (lk.status === "fulfilled") setLikeCount(lk.value.meta?.total ?? 0);
   221→    if (vw.status === "fulfilled") setViewedCount(vw.value.meta?.total ?? 0);
   222→    setLoading(false);
   223→  }, [enabled]);
   224→
   225→  useEffect(() => {
   226→    // eslint-disable-next-line react-hooks/set-state-in-effect -- initial dashboard data load (async fetch inside refresh)
   227→    void refresh();
   228→  }, [refresh]);
   229→
   230→  // Address CRUD (in this hub's sub-view or during checkout) must reflect on
   231→  // the dashboard teaser instantly — "N saved addresses" never goes stale.
   232→  useEffect(() => {
   233→    if (!enabled) return;
   234→    const sync = () => void refresh();
   235→    window.addEventListener("shopmesh:addresses-changed", sync);
   236→    return () => window.removeEventListener("shopmesh:addresses-changed", sync);
   237→  }, [enabled, refresh]);
   238→
   239→  return { profile, wallet, counts, likeCount, viewedCount, loading, refresh };
   240→}

   246→export default function MeHubView({ ctx, initialTab }: { ctx: ShopperCtx; initialTab?: string }) {
   247→  const meState = useApp((s) => s.me);
   248→  const openShopperAuth = useApp((s) => s.openShopperAuth);

   251→  const [purchaseTab, setPurchaseTab] = useState<string>(initialTab && initialTab.startsWith("bucket:") ? initialTab.slice(7) : "all");
   252→  const data = useMeData(Boolean(shopper));
   253→
   254→  if (!shopper) {
   255→    return (
   256→      <div className="mx-auto w-full max-w-3xl flex-1 px-4 py-10">
   257→        <EmptyState
   258→          icon={User}
   259→          title="Your dashboard awaits"
   260→          hint="Sign in to see your purchases, vouchers and personalised picks — all in one place."
   261→          action={
   262→            <Button onClick={() => openShopperAuth()} className="min-h-[44px] bg-brand-gradient-deep hover:brightness-110">
   263→              Sign in / Create account
   264→            </Button>
   265→          }
   266→        />
   267→      </div>
   268→    );
   269→  }
   270→
   271→  if (sub) {
   272→    const back = () => setSub(null);
   273→    switch (sub) {
   274→      case "purchases":
   275→        return <PurchasesView ctx={ctx} onBack={back} initialTab={purchaseTab} />;
   276→      case "wallet":
   277→        // DORMANT SCAFFOLD — MESH FINANCE: WalletView is kept intact for the
   278→        // roadmap; until MESH_FINANCE_ENABLED flips, this entry renders a
   279→        // coming-soon view instead of any stored-value UI.
   280→        return MESH_FINANCE_ENABLED ? (
   281→          <WalletView onBack={back} wallet={data.wallet} onRefresh={data.refresh} />
   282→        ) : (
   283→          <FinanceComingSoonView onBack={back} />
   284→        );
   285→      case "vouchers":
   286→        return <VouchersView onBack={back} onOpenMarket={() => ctx.nav({ name: "market" })} />;
   287→      case "likes":
   288→        return <ActivityGridView key="likes" type="LIKE" title="My Likes" onBack={back} ctx={ctx} />;
   289→      case "viewed":
   290→        return <ActivityGridView key="viewed" type="VIEWED" title="Recently Viewed" onBack={back} ctx={ctx} />;
   291→      case "addresses":
   292→        return <AddressesView onBack={back} />;
   293→    }
   294→  }
   295→
   296→  return (
   297→    <Dashboard
   298→      ctx={ctx}
   299→      data={data}
   300→      onSub={(s) => setSub(s)}
   301→      onPurchaseTab={(tab) => {
   302→        setPurchaseTab(tab);
   303→        setSub("purchases");
   304→      }}
   305→    />
   306→  );
   307→}

   613→function Avatar({ profile, size = "md" }: { profile: ProfileT | null; size?: "sm" | "md" | "lg" }) {
   614→  const px = size === "lg" ? "h-16 w-16 text-xl" : size === "md" ? "h-10 w-10 text-sm" : "h-8 w-8 text-xs";
   615→  const gradKey = profile?.avatarUrl?.startsWith("grad:") ? profile.avatarUrl : null;
   616→  const gradClass =
   617→    gradKey === "grad:amber"
   618→      ? "from-amber-400 to-orange-500"
   619→      : gradKey === "grad:rose"
   620→        ? "from-rose-400 to-pink-600"
   621→        : gradKey === "grad:violet"
   622→          ? "from-violet-400 to-purple-600"
   623→          : gradKey === "grad:emerald"
   624→            ? "from-emerald-400 to-teal-600"
   625→            : gradKey === "grad:slate"
   626→              ? "from-slate-400 to-slate-600"
   627→              : "from-teal-500 to-teal-800";
   628→  if (profile?.avatarUrl && !gradKey && /^https?:\/\/|^\//.test(profile.avatarUrl)) {
   629→    return <img src={profile.avatarUrl} alt={`${profile.fullName} avatar`} className={cn(px, "shrink-0 rounded-full object-cover ring-2 ring-white/70")} />;
   630→  }
   631→  const initial = (profile?.fullName ?? "?").trim().slice(0, 1).toUpperCase();
   632→  return (
   633→    <span className={cn(px, "flex shrink-0 items-center justify-center rounded-full bg-gradient-to-br font-extrabold text-white ring-2 ring-white/70", gradClass)} aria-hidden="true">
   634→      {initial}
   635→    </span>
   636→  );
   637→}

   838→function CoinCard({ wallet, onRefresh }: { wallet: WalletT | null; onRefresh: () => Promise<void> }) {
   839→  const [busy, setBusy] = useState(false);
   840→  const canClaim = wallet?.checkIn?.canClaim ?? false;
   841→
   842→  const claim = async () => {
   843→    setBusy(true);
   844→    try {
   845→      const res = await postJson<{ awarded: number; coinBalance: number; checkInStreak: number }>("/api/me/coins/claim", {});
   846→      toast.success(`+${res.awarded} coins claimed!`, { description: `Streak ${res.checkInStreak} day${res.checkInStreak === 1 ? "" : "s"} · balance ${res.coinBalance} coins` });
   847→      await onRefresh();
   848→    } catch (e) {
   849→      toast.error(e instanceof Error ? e.message : "Could not check in");
   850→    } finally {
   851→      setBusy(false);
   852→    }
   853→  };
   854→
   855→  return (
   856→    <div className="rounded-2xl border bg-card p-4 shadow-soft">
   857→      <div className="flex items-center gap-2 pb-1.5">
   858→        <Coins className="h-4 w-4 text-amber-500" aria-hidden="true" />

   862→      <div className="flex items-center gap-2 pt-1">
   863→        <Button
   864→          size="sm"
   865→          disabled={!canClaim || busy}
   866→          onClick={() => void claim()}
   867→          className={cn(
   868→            "relative h-7 min-h-[28px] px-2.5 text-[11px] font-bold",
   869→            canClaim ? "bg-amber-500 text-white hover:bg-amber-600" : "bg-muted text-muted-foreground hover:bg-muted"

  1125→  const [total, setTotal] = useState(0);
  1126→  const [loading, setLoading] = useState(true);
  1127→  const [more, setMore] = useState(false);
  1128→  // Task 10-a — orderNumber currently being mutated (receive / reorder)
  1129→  const [busyOrder, setBusyOrder] = useState<string | null>(null);
  1130→
  1131→  const load = useCallback(async (t: string, p: number, replace: boolean) => {
  1132→    if (replace) setLoading(true);
  1133→    else setMore(true);
  1134→    try {
  1135→      const q = t === "all" ? "" : `&bucket=${t}`;
  1136→      const res = await fetchEnvelope<{ bucketCounts: BucketCounts; orders: OrderRowT[] }>(`/api/me/orders?page=${p}&size=8${q}`);
  1137→      setTotal(res.meta?.total ?? 0);
  1138→      setOrders((prev) => (replace || p === 1 ? res.data.orders : [...(prev ?? []), ...res.data.orders]));
  1139→    } catch {
  1140→      if (replace) setOrders([]);

  1142→      setLoading(false);
  1143→      setMore(false);

  1146→
  1147→  useEffect(() => {
  1148→    setPage(1);

  1770→      const d = await deleteJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`);
  1771→      setRows(d.addresses);
  1772→      setConfirmDelete(null);
  1773→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
  1774→      toast.success("Address removed");
  1775→    } catch (e) {
  1776→      toast.error(e instanceof Error ? e.message : "Could not remove the address");
  1777→    }
  1778→  };
  1779→
  1780→  const setDefault = async (id: string) => {
  1781→    try {
  1782→      const d = await patchJson<{ addresses: AddressT[] }>(`/api/checkout/addresses/${id}`, { isDefault: true });
  1783→      setRows(d.addresses);
  1784→      window.dispatchEvent(new Event("shopmesh:addresses-changed"));
  1785→      toast.success("Default address updated");
  1786→    } catch (e) {
  1787→      toast.error(e instanceof Error ? e.message : "Could not update the default");
  1788→    }
  1789→  };
  1790→
  1791→  const field = (key: keyof typeof EMPTY_ADDR, label: string, opts?: { type?: string; placeholder?: string; wide?: boolean }) => (
  1792→    <div className={opts?.wide ? "space-y-1.5 sm:col-span-2" : "space-y-1.5"}>
  1793→      <Label htmlFor={`addr-${key}`}>{label}</Label>
  1794→      <Input
  1795→        id={`addr-${key}`}
  1796→        type={opts?.type ?? "text"}
  1797→        value={form?.[key] ?? ""}
  1798→        onChange={(e) => setForm((f) => (f ? { ...f, [key]: e.target.value } : f))}
  1799→        placeholder={opts?.placeholder}
  1800→        className="min-h-[44px]"
  1801→      />
  1802→    </div>
  1803→  );
  1804→
  1805→  return (
  1806→    <div className="mx-auto w-full max-w-3xl flex-1 px-3 py-4 sm:px-4 sm:py-6">
  1807→      <SubHeader
  1808→        title="Delivery Addresses"
  1809→        onBack={onBack}
  1810→        action={
  1811→          <Button
  1812→            size="sm"
  1813→            onClick={() => { setEditId(null); setForm(EMPTY_ADDR); }}
  1814→            className="min-h-[36px] bg-brand-gradient-deep text-white hover:brightness-110"
  1815→          >
  1816→            <Plus className="h-3.5 w-3.5" /> New
  1817→          </Button>
  1818→        }
  1819→      />
  1820→
  1821→      {rows === null ? (
  1822→        <div className="space-y-3">
  1823→          <Skeleton className="h-24 rounded-2xl" />
  1824→          <Skeleton className="h-24 rounded-2xl" />
  1825→        </div>
  1826→      ) : rows.length === 0 && !form ? (
  1827→        <EmptyState
  1828→          icon={MapPin}
  1829→          title="No saved addresses"
  1830→          hint="Save your delivery address once — checkout will fill it in for you automatically."
  1831→          action={<Button onClick={() => { setEditId(null); setForm(EMPTY_ADDR); }} className="min-h-[44px] bg-brand-gradient-deep hover:brightness-110">Add address</Button>}
  1832→        />
  1833→      ) : (
  1834→        <ul className="space-y-3">
  1835→          {rows.map((a) => (
  1836→            <li key={a.id} className="rounded-2xl border bg-card p-4 shadow-sm" aria-label={`${a.label} address for ${a.name}`}>
  1837→              <div className="flex items-center gap-2 pb-1">
  1838→                <span className="rounded-full bg-muted px-2 py-0.5 text-[10px] font-extrabold uppercase tracking-wide text-muted-foreground">{a.label}</span>
  1839→                {a.isDefault ? <Badge className="bg-teal-700 hover:bg-teal-700 text-[10px]">Default</Badge> : null}
  1840→              </div>
  1841→              <p className="text-sm font-bold">{a.name} {a.phone ? <span className="font-normal text-muted-foreground">· {a.phone}</span> : null}</p>
  1842→              <p className="text-sm text-muted-foreground">{a.line1}, {a.city}, {a.state} {a.postcode}</p>
  1843→              <div className="flex flex-wrap gap-2 pt-3">
  1844→                {!a.isDefault ? (
  1845→                  <Button size="sm" variant="outline" className="h-8 min-h-[32px]" onClick={() => void setDefault(a.id)}>Set default</Button>
  1846→                ) : null}
  1847→                <Button size="sm" variant="outline" className="h-8 min-h-[32px]" onClick={() => { setEditId(a.id); setForm({ label: a.label, name: a.name, phone: a.phone ?? "", line1: a.line1, city: a.city, state: a.state, postcode: a.postcode }); }}>
  1848→                  <Pencil className="h-3 w-3" /> Edit
  1849→                </Button>
  1850→                {confirmDelete === a.id ? (
  1851→                  <Button size="sm" className="h-8 min-h-[32px] bg-rose-600 hover:bg-rose-700" onClick={() => void remove(a.id)} onBlur={() => setConfirmDelete(null)}>
  1852→                    Confirm delete
  1853→                  </Button>
  1854→                ) : (
  1855→                  <Button size="sm" variant="outline" className="h-8 min-h-[32px] text-rose-600 hover:bg-rose-50" onClick={() => setConfirmDelete(a.id)}>
  1856→                    <Trash2 className="h-3 w-3" /> Delete
  1857→                  </Button>
  1858→                )}
  1859→              </div>
  1860→            </li>
  1861→          ))}
  1862→        </ul>
  1863→      )}
  1864→
  1865→      <Dialog open={form !== null} onOpenChange={(v) => { if (!v) { setForm(null); setEditId(null); } }}>
  1866→        <DialogContent className="max-w-md rounded-2xl">
  1867→          <DialogHeader>
  1868→            <DialogTitle>{editId ? "Edit address" : "New address"}</DialogTitle>
  1869→            <DialogDescription>Saved to your account — used to prefill checkout.</DialogDescription>
  1870→          </DialogHeader>
  1871→          {form ? (
  1872→            <div className="grid gap-3 sm:grid-cols-2">
  1873→              {field("name", "Recipient name", { placeholder: "Nur Aisyah" })}
  1874→              {field("phone", "Phone", { type: "tel", placeholder: "+60 12-345 6789" })}
  1875→              {field("line1", "Street address", { wide: true, placeholder: "12 Jalan Melati, Taman Melawati" })}
  1876→              {field("city", "City", { placeholder: "Kuala Lumpur" })}
  1877→              {field("state", "State", { placeholder: "Selangor" })}
  1878→              {field("postcode", "Postcode", { placeholder: "53100" })}
  1879→              {field("label", "Label", { placeholder: "Home / Office" })}
  1880→              <div className="sm:col-span-2">
  1881→                <Button onClick={() => void save()} disabled={busy} className="min-h-[44px] w-full bg-brand-gradient-deep text-white hover:brightness-110">
  1882→                  {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : editId ? "Save changes" : "Save address"}
  1883→                </Button>
  1884→              </div>
  1885→            </div>
  1886→          ) : null}
  1887→        </DialogContent>
  1888→      </Dialog>
  1889→    </div>
  1890→  );
  1891→}

  1893→// cart count (lightweight local hook so the header badge stays live)
  1894→function useCartCount() {
  1895→  const [count, setCount] = useState(0);
  1896→  useEffect(() => {
  1897→    let alive = true;
  1898→    const load = () =>
  1899→      fetchJson<{ cart: { items: { qty: number }[] } }>("/api/cart")
  1900→        .then((d) => alive && setCount(d.cart?.items.reduce((s, i) => s + i.qty, 0) ?? 0))
  1901→        .catch(() => {});
  1902→    load();
  1903→    window.addEventListener("sm-cart-updated", load);
  1904→    return () => {
  1905→      alive = false;
  1906→      window.removeEventListener("sm-cart-updated", load);
  1907→    };
  1908→  }, []);
  1909→  return { count };
  1910→}
