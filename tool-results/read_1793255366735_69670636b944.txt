   337→// FIN-3-c — seller in-app notification bell. Feeds from the NotificationLog
   338→// rows (channel "inapp") the finance flows already write for the seller email:
   339→// order.paid (FIN-2-b) + payout requested/approved/paid/failed/rejected
   340→// (FIN-2-a/FIN-3-b). The shopper bell is a different subsystem (Notification
   341→// model via /api/me/notifications) and is intentionally untouched.
   342→
   343→type SellerNotificationRow = { id: string; subject: string; createdAt: string; event: string };
   344→
   345→type SellerNotificationsPayload = { notifications: SellerNotificationRow[]; recentCount: number };
   346→
   347→/** House scrollbar style (same treatment as the shopper bell panel). */
   348→const SELLER_BELL_SCROLL =
   349→  "max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent";
   350→
   351→/** 24h recency = client-side "new" heuristic: NotificationLog has no read /
   352→ *  seen column (adding fake read-tracking was out of scope), so "within the
   353→ *  last 24h" is the honest approximation of "unread". */
   354→const RECENT_WINDOW_MS = 24 * 60 * 60 * 1000;
   355→
   356→function SellerNotificationBell() {
   357→  const [open, setOpen] = useState(false);
   358→  const [data, setData] = useState<SellerNotificationsPayload | null>(null);
   359→  const [loading, setLoading] = useState(false);
   360→  const [errored, setErrored] = useState(false);
   361→
   362→  const load = useCallback(async () => {
   363→    setLoading(true);
   364→    try {
   365→      const d = await apiGet<SellerNotificationsPayload>("/api/store/notifications");
   366→      setData(d);
   367→      setErrored(false);
   368→    } catch (e) {
   369→      // Silent by design — a failed bell must never break the seller shell:
   370→      // keep the last data (no badge flicker), warn for debugging only.
   371→      setErrored(true);
   372→      console.warn("[seller-bell] notifications fetch failed", e instanceof Error ? e.message : e);
   373→    } finally {
   374→      setLoading(false);
   375→    }
   376→  }, []);
   377→
   378→  // Badge freshness without aggressive polling: fetch once on mount (dot only
   379→  // needs recentCount) and refetch whenever the popover opens.
   380→  useEffect(() => {
   381→    void load();
   382→  }, [load]);
   383→
   384→  const onOpenChange = (v: boolean) => {
   385→    setOpen(v);
   386→    if (v) void load();
   387→  };
   388→
   389→  const recent = data?.recentCount ?? 0;
   390→  const rows = data?.notifications ?? null;
   391→
   392→  return (
   393→    <Popover open={open} onOpenChange={onOpenChange}>
   394→      <PopoverTrigger asChild>
   395→        <Button
   396→          variant="ghost"
   397→          size="icon"
   398→          className="relative text-muted-foreground"
   399→          aria-label="Seller notifications"
   400→          aria-haspopup="dialog"
   401→          aria-expanded={open}
   402→        >
   403→          <Bell className="h-4 w-4" aria-hidden="true" />
   404→          {recent > 0 ? (
   405→            <span
   406→              className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-amber-500 ring-2 ring-background"
   407→              aria-hidden="true"
   408→            />
   409→          ) : null}
   410→        </Button>
   411→      </PopoverTrigger>
   412→      <PopoverContent align="end" className="w-80 rounded-2xl p-0 sm:w-96" role="dialog" aria-label="Seller notifications panel">
   413→        <div className="flex items-center gap-2 border-b px-4 py-3">
   414→          <h2 className="text-sm font-semibold">Notifications</h2>
   415→          {recent > 0 ? (
   416→            <span className="rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 tabular-nums">
   417→              {recent} new
   418→            </span>
   419→          ) : null}
   420→        </div>
   421→
   422→        {loading && !rows ? (
   423→          <div className="space-y-3 p-4" aria-label="Loading notifications">
   424→            {Array.from({ length: 3 }).map((_, i) => (
   425→              <div key={i} className="flex items-center gap-3">
   426→                <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   427→                <div className="flex-1 space-y-1.5">
   428→                  <Skeleton className="h-3.5 w-3/4" />
   429→                  <Skeleton className="h-3 w-1/3" />
   430→                </div>
   431→              </div>
   432→            ))}
   433→          </div>
   434→        ) : !rows || rows.length === 0 ? (
   435→          <div className="flex flex-col items-center gap-2 px-6 py-10 text-center" aria-label="No notifications">
   436→            <span className="rounded-full bg-muted p-4">
   437→              <Bell className="h-6 w-6 text-muted-foreground" aria-hidden="true" />
   438→            </span>
   439→            <p className="text-sm font-semibold">{errored && !data ? "Notifications unavailable" : "No notifications yet"}</p>
   440→            <p className="max-w-[16rem] text-xs text-muted-foreground">
   441→              {errored && !data
   442→                ? "Couldn't load right now — it will retry next time you open the bell."
   443→                : "Payout and order updates will appear here."}
   444→            </p>
   445→          </div>
   446→        ) : (
   447→          <ul className={cn("divide-y", SELLER_BELL_SCROLL)} aria-label="Notification list">
   448→            {rows.map((n) => {
   449→              const isNew = Date.now() - new Date(n.createdAt).getTime() < RECENT_WINDOW_MS;
   450→              return (
   451→                <li key={n.id} className="flex items-start gap-3 px-4 py-3">
   452→                  <div className="min-w-0 flex-1">
   453→                    <p className="truncate text-[13px] font-medium" title={n.subject}>
   454→                      {n.subject}
   455→                    </p>
   456→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   457→                      <span title={new Date(n.createdAt).toLocaleString("en-MY")}>{ageOf(n.createdAt)}</span>
   458→                      <span aria-hidden="true"> · </span>
   459→                      <span className="lowercase">{n.event}</span>
   460→                    </p>
   461→                  </div>
   462→                  {isNew ? (
   463→                    <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-amber-500" role="img" aria-label="Recent" />
   464→                  ) : null}
   465→                </li>
   466→              );
   467→            })}
   468→          </ul>
   469→        )}
   470→      </PopoverContent>
   471→    </Popover>
   472→  );
   473→}

   477→/** Total unread shopper messages — powers the "Messages" nav badge. */
   478→function useSellerChatUnread(active: SellerNav["view"]) {
   479→  const [unread, setUnread] = useState(0);
   480→  useEffect(() => {
   481→    let alive = true;
   482→    const load = () =>
   483→      apiGet<{ total: number }>("/api/store/chat/unread")
   484→        .then((d) => alive && setUnread(d.total ?? 0))
   485→        .catch(() => {});
   486→    load();
   487→    const t = setInterval(load, 15000);
   488→    return () => {
   489→      alive = false;
   490→      clearInterval(t);
   491→    };
   492→  }, [active]);
   493→  // instant re-sync when a thread is read inside the messages view (the view
   494→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   495→  useEffect(() => {
   496→    const sync = () =>
   497→      apiGet<{ total: number }>("/api/store/chat/unread")
   498→        .then((d) => setUnread(d.total ?? 0))
   499→        .catch(() => {});
   500→    window.addEventListener("nomofee:chat-sync", sync);
   501→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   502→  }, []);
   503→  return unread;
   504→}
