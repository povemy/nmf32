   113→/** Total unread seller messages — powers the "My messages" header badge.
   114→ *  Polls lightly + refreshes when the messages view is opened/left. */
   115→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
   116→  const [unread, setUnread] = useState(0);
   117→  useEffect(() => {
   118→    if (!shopperSignedIn) {
   119→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
   120→      setUnread(0);
   121→      return;
   122→    }
   123→    let alive = true;
   124→    const load = () =>
   125→      fetchJson<{ total: number }>("/api/chat/unread")
   126→        .then((d) => alive && setUnread(d.total ?? 0))
   127→        .catch(() => {});
   128→    load();
   129→    const t = setInterval(load, 15000);
   130→    return () => {
   131→      alive = false;
   132→      clearInterval(t);
   133→    };
   134→  }, [shopperSignedIn]);
   135→  // instant re-sync when a thread is read inside the messages view (the view
   136→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   137→  useEffect(() => {
   138→    if (!shopperSignedIn) return;
   139→    const sync = () =>
   140→      fetchJson<{ total: number }>("/api/chat/unread")
   141→        .then((d) => setUnread(d.total ?? 0))
   142→        .catch(() => {});
   143→    window.addEventListener("nomofee:chat-sync", sync);
   144→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   145→  }, [shopperSignedIn]);
   146→  // opening (or leaving) the messages view should immediately re-sync the badge
   147→  useEffect(() => {
   148→    if (!shopperSignedIn) return;
   149→    const t = setTimeout(
   150→      () =>
   151→        fetchJson<{ total: number }>("/api/chat/unread")
   152→          .then((d) => setUnread(d.total ?? 0))
   153→          .catch(() => {}),
   154→      viewName === "messages" ? 600 : 0
   155→    );
   156→    return () => clearTimeout(t);
   157→  }, [shopperSignedIn, viewName]);
   158→  return unread;
   159→}

   161→// ---- platform chrome config (global footer visibility) ---------------------
   162→// Admin → Settings → "Show global footer" (Setting ui.footer_visible) governs
   163→// the shopper footer. Public endpoint + 60 s module cache: toggling takes
   164→// effect on the next app load without per-navigation refetch chatter.
   165→let appConfigCache: { at: number; footerVisible: boolean } | null = null;
   166→
   167→function useFooterVisible() {
   168→  const [visible, setVisible] = useState(true);
   169→  useEffect(() => {
   170→    if (appConfigCache && Date.now() - appConfigCache.at < 60_000) {
   171→      // eslint-disable-next-line react-hooks/set-state-in-effect -- cache hit syncs an external module cache; the fresh-fetch path below sets state in a callback
   172→      setVisible(appConfigCache.footerVisible);
   173→      return;
   174→    }
   175→    let cancelled = false;
   176→    fetch("/api/sf/app-config", { cache: "no-store" })
   177→      .then((r) => (r.ok ? r.json() : null))
   178→      .then((b) => {
   179→        const v = (b as { data?: { footerVisible?: boolean } } | null)?.data?.footerVisible;
   180→        if (typeof v === "boolean" && !cancelled) {
   181→          appConfigCache = { at: Date.now(), footerVisible: v };
   182→          setVisible(v);
   183→        }
   184→      })
   185→      .catch(() => undefined);
   186→    return () => {
   187→      cancelled = true;
   188→    };
   189→  }, []);
   190→  return visible;
   191→}

   193→export default function ShopperShell() {
   194→  const setSurface = useApp((s) => s.setSurface);
   195→  const activeStoreSlug = useApp((s) => s.activeStoreSlug);
   196→  const setActiveStoreSlug = useApp((s) => s.setActiveStoreSlug);
   197→
   198→  const [view, setView] = useState<ShopperView>({ name: "market" });
   199→  const [ready, setReady] = useState(false);
   200→  const [cartOpen, setCartOpen] = useState(false);
   201→  // Which store's customized cart experience to render in the drawer — set by
   202→  // PDP add-to-cart / storefront header; null = platform defaults.
   203→  const [cartStoreSlug, setCartStoreSlug] = useState<string | null>(null);
   204→  const [headerSearch, setHeaderSearch] = useState("");
   205→  // Back-navigation history stack (user request: back button on every page).
   206→  // viewRef mirrors `view` so nav() can push the PREVIOUS view without reading
   207→  // state inside the setState updater (StrictMode-safe).
   208→  const viewStack = useRef<ShopperView[]>([]);
   209→  const viewRef = useRef<ShopperView>({ name: "market" });
   210→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   211→  // intentionally NOT part of ShopperView — a synthetic render override.
   212→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   213→  const restored = useRef(false);
   214→
   215→  // ---- mount: activeStoreSlug override, else restore last view --------------
   216→  useEffect(() => {
   217→    if (restored.current) return;
   218→    restored.current = true;
   219→    const params = new URLSearchParams(window.location.search);
   220→
   221→    // Marketing attribution capture (once per landing): when the landing URL
   222→    // carries utm_* / nmcl parameters, fire the same-origin beacon so the
   223→    // server validates them and lands the `nm_ins_attr` cookie BEFORE the
   224→    // storefront fetch records the first store_view event (bounded wait — on
   225→    // beacon failure the visit proceeds untagged, never blocked).
   226→    const ATTR_PARAM_KEYS = ["utm_source", "utm_medium", "utm_campaign", "utm_term", "utm_content", "nmcl"] as const;
   227→    const hasAttrParams = ATTR_PARAM_KEYS.some((k) => params.get(k) != null);
   228→    const fireLandingBeacon = async (): Promise<void> => {
   229→      try {
   230→        const payload: Record<string, string> = {};
   231→        for (const k of [...ATTR_PARAM_KEYS, "sf"]) {
   232→          const v = params.get(k);
   233→          if (v != null && v !== "") payload[k] = v.slice(0, 200);
   234→        }
   235→        await fetch("/api/analytics/landing", {
   236→          method: "POST",
   237→          // X-SM-Client marker: proves app-origin to the CSRF gate in contexts
   238→          // where the proxy chain rewrites Host (preview panels / iframes).
   239→          headers: { "content-type": "application/json", ...webHeaders() },
   240→          body: JSON.stringify(payload),
   241→          keepalive: true,
   242→        });
   243→      } catch {
   244→        /* attribution is best-effort — never block the landing */
   245→      }
   246→    };
   247→
   248→    const proceed = () => {
   249→      // claim deep-link wins over everything
   250→      try {
   251→        const cj = params.get("claim_job");
   252→        const ct = params.get("claim_token");
   253→        if (cj && ct) {
   254→          setClaimLink({ jobId: cj, token: ct });
   255→          setReady(true);
   256→          return;
   257→        }
   258→        // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   259→        const sf = params.get("sf");
   260→        if (sf) {
   261→          const sp = params.get("sp");
   262→          setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   263→          setReady(true);
   264→          return;
   265→        }
   266→      } catch {
   267→        /* ignore */
   268→      }
   269→      if (activeStoreSlug) {
   270→        setView({ name: "storefront", slug: activeStoreSlug });
   271→        setActiveStoreSlug(null);
   272→      } else {
   273→        try {
   274→          const raw = window.localStorage.getItem(VIEW_KEY);
   275→          if (raw) {
   276→            const v = JSON.parse(raw) as ShopperView;
   277→            if (v && typeof v.name === "string") setView(v);
   278→          }
   279→        } catch {
   280→          /* ignore */
   281→        }
   282→      }
   283→      setReady(true);
   284→    };
   285→
   286→    if (hasAttrParams) {
   287→      void Promise.race([fireLandingBeacon(), new Promise<void>((r) => setTimeout(r, 1500))]).then(() => proceed());
   288→    } else {
   289→      proceed();
   290→    }
   291→  }, [activeStoreSlug, setActiveStoreSlug]);
   292→
   293→  // ---- nav helper: persist + scroll to top ----------------------------------
   294→  useEffect(() => {
   295→    viewRef.current = view;
   296→  }, [view]);
   297→
   298→  const nav = useCallback((v: ShopperView) => {
   299→    if (viewRef.current.name !== v.name || JSON.stringify(viewRef.current) !== JSON.stringify(v)) {
   300→      viewStack.current.push(viewRef.current);
   301→      if (viewStack.current.length > 25) viewStack.current.shift();
   302→    }
   303→    setView(v);
   304→    try {
   305→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   306→    } catch {
   307→      /* ignore */
   308→    }
   309→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   310→  }, []);
   311→
   312→  const back = useCallback(() => {
   313→    const prev = viewStack.current.pop() ?? { name: "market" as const };
   314→    setView(prev);
   315→    try {
   316→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(prev));
   317→    } catch {
   318→      /* ignore */
   319→    }
   320→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   321→  }, []);
   322→
   323→  // refresh storefront/store-page when slug changes externally
   324→  useEffect(() => {
   325→    if (!ready) return;
   326→    if (activeStoreSlug) {
   327→      // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional data-fetch reset
   328→      setView({ name: "storefront", slug: activeStoreSlug });
   329→      setActiveStoreSlug(null);
   330→    }
   331→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   332→
   333→  // ---- sign-out redirect: reset to the marketplace home --------------------
   334→  // The shared app-store logout() clears the persisted view and switches
   335→  // seller/admin surfaces here (fresh mount → default market view). When the
   336→  // SHOPPER surface is already mounted (buyer sign-out) the in-memory view
   337→  // survives — this listener drops it (and the back-stack) back to home so
   338→  // every portal lands on the homepage after signing out.
   339→  useEffect(() => {
   340→    const onSignedOut = () => {
   341→      viewStack.current = [];
   342→      viewRef.current = { name: "market" };
   343→      setView({ name: "market" });
   344→      try {
   345→        window.localStorage.setItem(VIEW_KEY, JSON.stringify({ name: "market" }));
   346→      } catch {
   347→        /* storage unavailable — in-memory reset already applied */
   348→      }
   349→      window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   350→    };
   351→    window.addEventListener(SIGNED_OUT_EVENT, onSignedOut);
   352→    return () => window.removeEventListener(SIGNED_OUT_EVENT, onSignedOut);
   353→  }, []);
   354→
   355→  const ctx = useMemo<ShopperCtx>(
   356→    () => ({
   357→      nav,
   358→      back,
   359→      openCart: (storeSlug?: string) => {
   360→        setCartStoreSlug(storeSlug ?? null);
   361→        setCartOpen(true);
   362→      },
   363→    }),
   364→    [nav, back]
   365→  );
   366→
   367→  // ---- global inbox presence (ONE socket on every view) ---------------------
   368→  // Away from the messages page nothing used to listen for pushes, so the
   369→  // Message/Chat tab badges only moved on a 15s poll — incoming messages felt
   370→  // silent. This keeps the user's inbox room joined on every view: a push
   371→  // fires "nomofee:chat-sync" (all badge hooks re-sync instantly) and, when
   372→  // the user is NOT on the messages page, raises a toast with a jump-in link.
   373→  const meForPresence = useApp((s) => s.me);
   374→  const shopperForPresence = meForPresence.loaded ? meForPresence.shopper : null;
   375→  useChatUserRoom(shopperForPresence?.id ?? null, {
   376→    onMessage: (msg) => {
   377→      if (msg.senderType === "shopper") return; // own echo / auto-ack paths
   378→      window.dispatchEvent(new Event("nomofee:chat-sync"));
   379→      if (viewRef.current.name !== "messages") {
   380→        const preview = msg.body.length > 90 ? `${msg.body.slice(0, 90)}…` : msg.body;
   381→        toast(`${msg.senderName} · New message`, {
   382→          description: preview,
   383→          action: msg.threadId
   384→            ? { label: "Open chat", onClick: () => nav({ name: "messages", threadId: msg.threadId }) }
   385→            : undefined,
   386→        });
   387→      }
   388→    },
   389→    onActivity: () => window.dispatchEvent(new Event("nomofee:chat-sync")),
   390→  });
   391→
   392→  const showChrome = !isStoreChrome(view);
   393→  // App-like bottom navigation — mobile only, hidden during checkout focus mode.
   394→  const showBottomNav = showChrome && view.name !== "checkout" && view.name !== "pay";
   395→  // Admin-controlled global footer switch (default ON; checkout never shows it).
   396→  const footerVisible = useFooterVisible();
   397→
   398→  return (
   399→    <div className="flex-1 flex flex-col min-h-screen">
   400→      <ShopperAuthDialog />
   401→
   402→      {showChrome ? (
   403→        <MarketHeader
   404→          view={view}
   405→          nav={nav}
   406→          openCart={() => setCartOpen(true)}
   407→          search={headerSearch}
   408→          setSearch={setHeaderSearch}
   409→        />
   410→      ) : null}
   411→
   412→      <AnimatePresence mode="wait">
   413→        <motion.main
   414→          key={`${view.name}:${"slug" in view ? view.slug : ""}:${"pSlug" in view ? view.pSlug : ""}:${"ref" in view ? view.ref : ""}:${view.name === "search" ? view.q : view.name === "category" ? view.categorySlug : view.name === "order-status" ? view.orderNumber : view.name === "store-page" ? view.pageSlug : view.name === "messages" ? view.threadId ?? "" : view.name === "me" ? view.tab ?? "" : ""}`}
   415→          initial={{ opacity: 0, y: 8 }}
   416→          animate={{ opacity: 1, y: 0 }}
   417→          exit={{ opacity: 0, y: -8 }}
   418→          transition={{ duration: 0.18, ease: "easeOut" }}
   419→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   420→        >
   421→          {claimLink ? null : (
   422→            <>
   423→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   424→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   425→              {view.name === "category" ? (
   426→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   427→              ) : null}
   428→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   429→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   430→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}
   431→              {view.name === "checkout" ? <CheckoutView ctx={ctx} /> : null}
   432→              {view.name === "pay" ? <PayView ctx={ctx} paymentRef={view.ref} /> : null}
   433→              {view.name === "order-status" ? <OrderStatusView ctx={ctx} orderNumber={view.orderNumber} /> : null}
   434→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   435→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   436→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   437→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   438→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}
   439→            </>
   440→          )}
   441→          {claimLink ? <ClaimView jobId={claimLink.jobId} token={claimLink.token} onExit={() => setClaimLink(null)} /> : null}
   442→        </motion.main>
   443→      </AnimatePresence>
   444→
   445→      {showBottomNav ? (
   446→        <BottomNav
   447→          view={view}
   448→          nav={nav}
   449→          openCart={() => setCartOpen(true)}
   450→        />
   451→      ) : null}
   452→
   453→      {showChrome && footerVisible ? (
   454→        <ShopperFooter
   455→          onSeller={() => setSurface("seller")}
   456→          onLookup={() => nav({ name: "lookup" })}
   457→          onMarket={() => nav({ name: "market" })}
   458→          onGenerate={() => nav({ name: "generate" })}
   459→        />
   460→      ) : null}
   461→
   462→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} storeSlug={cartStoreSlug} />
   463→    </div>
   464→  );
   465→}

   471→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   472→  const me = useApp((s) => s.me);
   473→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   474→  const signedIn = Boolean(me.loaded && me.shopper);
   475→  const { count } = useCart();
   476→  const unreadChats = useChatUnread(signedIn, view.name);
   477→
   478→  const items = [
   479→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   480→    {
   481→      key: "orders",
   482→      label: "Orders",
   483→      icon: Package,
   484→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   485→      badge: 0,
   486→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   487→    },
   488→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   489→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   490→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   491→  ];
   492→
   493→  return (
   494→    <nav
   495→      aria-label="Bottom navigation"
   496→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   497→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   498→    >
   499→      <div className="mx-auto grid max-w-lg grid-cols-5">
   500→        {items.map((it) => {
   501→          const Icon = it.icon;
   502→          return (
   503→            <button
   504→              key={it.key}
   505→              type="button"
   506→              onClick={it.onClick}
   507→              aria-label={it.label}
   508→              aria-current={it.active ? "page" : undefined}

   941→function ShopperAuthDialog() {
   942→  const open = useApp((s) => s.shopperAuthOpen);
   943→  const close = useApp((s) => s.closeShopperAuth);
   944→  const after = useApp((s) => s.afterShopperAuth);
   945→  const refreshMe = useApp((s) => s.refreshMe);
   946→  const [tab, setTab] = useState("login");
   947→  const [busy, setBusy] = useState(false);
   948→
   949→  useEffect(() => {
   950→    if (open) setTab("login");
   951→  }, [open]);
   952→
   953→  const finish = async () => {
   954→    await refreshMe();
   955→    close();
   956→    after?.();
   957→  };
   958→
   959→  const submitLogin = async (e: React.FormEvent<HTMLFormElement>) => {
   960→    e.preventDefault();
   961→    if (busy) return;
   962→    const f = new FormData(e.currentTarget);
   963→    setBusy(true);
   964→    try {
   965→      const res = await postJson<{ token?: string }>("/api/auth/shopper/login", { email: f.get("email"), password: f.get("password") });
   966→      setSessionToken(res?.token); // localStorage bridge for cookie-less contexts
   967→      toast.success("Welcome back!");
   968→      await finish();
   969→    } catch (err) {
   970→      toast.error(err instanceof Error ? err.message : "Sign in failed");
   971→    } finally {
   972→      setBusy(false);
   973→    }
   974→  };
   975→
   976→  const submitRegister = async (e: React.FormEvent<HTMLFormElement>) => {
   977→    e.preventDefault();
   978→    if (busy) return;
   979→    const f = new FormData(e.currentTarget);
   980→    setBusy(true);
