   327→// FIN-3-c — seller in-app notification bell. Feeds from the NotificationLog
   328→// rows (channel "inapp") the finance flows already write for the seller email:
   329→// order.paid (FIN-2-b) + payout requested/approved/paid/failed/rejected
   330→// (FIN-2-a/FIN-3-b). The shopper bell is a different subsystem (Notification
   331→// model via /api/me/notifications) and is intentionally untouched.
   332→
   333→type SellerNotificationRow = { id: string; subject: string; createdAt: string; event: string };
   334→
   335→type SellerNotificationsPayload = { notifications: SellerNotificationRow[]; recentCount: number };
   336→
   337→/** House scrollbar style (same treatment as the shopper bell panel). */
   338→const SELLER_BELL_SCROLL =
   339→  "max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-border [&::-webkit-scrollbar-track]:bg-transparent";
   340→
   341→/** 24h recency = client-side "new" heuristic: NotificationLog has no read /
   342→ *  seen column (adding fake read-tracking was out of scope), so "within the
   343→ *  last 24h" is the honest approximation of "unread". */
   344→const RECENT_WINDOW_MS = 24 * 60 * 60 * 1000;
   345→
   346→function SellerNotificationBell() {
   347→  const [open, setOpen] = useState(false);
   348→  const [data, setData] = useState<SellerNotificationsPayload | null>(null);
   349→  const [loading, setLoading] = useState(false);
   350→  const [errored, setErrored] = useState(false);
   351→
   352→  const load = useCallback(async () => {
   353→    setLoading(true);
   354→    try {
   355→      const d = await apiGet<SellerNotificationsPayload>("/api/store/notifications");
   356→      setData(d);
   357→      setErrored(false);
   358→    } catch (e) {
   359→      // Silent by design — a failed bell must never break the seller shell:
   360→      // keep the last data (no badge flicker), warn for debugging only.
   361→      setErrored(true);
   362→      console.warn("[seller-bell] notifications fetch failed", e instanceof Error ? e.message : e);
   363→    } finally {
   364→      setLoading(false);
   365→    }
   366→  }, []);
   367→
   368→  // Badge freshness without aggressive polling: fetch once on mount (dot only
   369→  // needs recentCount) and refetch whenever the popover opens.
   370→  useEffect(() => {
   371→    void load();
   372→  }, [load]);
   373→
   374→  const onOpenChange = (v: boolean) => {
   375→    setOpen(v);
   376→    if (v) void load();
   377→  };
   378→
   379→  const recent = data?.recentCount ?? 0;
   380→  const rows = data?.notifications ?? null;
   381→
   382→  return (
   383→    <Popover open={open} onOpenChange={onOpenChange}>
   384→      <PopoverTrigger asChild>
   385→        <Button
   386→          variant="ghost"
   387→          size="icon"
   388→          className="relative text-muted-foreground"
   389→          aria-label="Seller notifications"
   390→          aria-haspopup="dialog"
   391→          aria-expanded={open}
   392→        >
   393→          <Bell className="h-4 w-4" aria-hidden="true" />
   394→          {recent > 0 ? (
   395→            <span
   396→              className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-amber-500 ring-2 ring-background"
   397→              aria-hidden="true"
   398→            />
   399→          ) : null}
   400→        </Button>
   401→      </PopoverTrigger>
   402→      <PopoverContent align="end" className="w-80 rounded-2xl p-0 sm:w-96" role="dialog" aria-label="Seller notifications panel">
   403→        <div className="flex items-center gap-2 border-b px-4 py-3">
   404→          <h2 className="text-sm font-semibold">Notifications</h2>
   405→          {recent > 0 ? (
   406→            <span className="rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-bold text-amber-700 tabular-nums">
   407→              {recent} new
   408→            </span>
   409→          ) : null}
   410→        </div>
   411→
   412→        {loading && !rows ? (
   413→          <div className="space-y-3 p-4" aria-label="Loading notifications">
   414→            {Array.from({ length: 3 }).map((_, i) => (
   415→              <div key={i} className="flex items-center gap-3">
   416→                <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   417→                <div className="flex-1 space-y-1.5">
   418→                  <Skeleton className="h-3.5 w-3/4" />
   419→                  <Skeleton className="h-3 w-1/3" />
   420→                </div>
   421→              </div>
   422→            ))}
   423→          </div>
   424→        ) : !rows || rows.length === 0 ? (
   425→          <div className="flex flex-col items-center gap-2 px-6 py-10 text-center" aria-label="No notifications">
   426→            <span className="rounded-full bg-muted p-4">
   427→              <Bell className="h-6 w-6 text-muted-foreground" aria-hidden="true" />
   428→            </span>
   429→            <p className="text-sm font-semibold">{errored && !data ? "Notifications unavailable" : "No notifications yet"}</p>
   430→            <p className="max-w-[16rem] text-xs text-muted-foreground">
   431→              {errored && !data
   432→                ? "Couldn't load right now — it will retry next time you open the bell."
   433→                : "Payout and order updates will appear here."}
   434→            </p>
   435→          </div>
   436→        ) : (
   437→          <ul className={cn("divide-y", SELLER_BELL_SCROLL)} aria-label="Notification list">
   438→            {rows.map((n) => {
   439→              const isNew = Date.now() - new Date(n.createdAt).getTime() < RECENT_WINDOW_MS;
   440→              return (
   441→                <li key={n.id} className="flex items-start gap-3 px-4 py-3">
   442→                  <div className="min-w-0 flex-1">
   443→                    <p className="truncate text-[13px] font-medium" title={n.subject}>
   444→                      {n.subject}
   445→                    </p>
   446→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   447→                      <span title={new Date(n.createdAt).toLocaleString("en-MY")}>{ageOf(n.createdAt)}</span>
   448→                      <span aria-hidden="true"> · </span>
   449→                      <span className="lowercase">{n.event}</span>
   450→                    </p>
   451→                  </div>
   452→                  {isNew ? (
   453→                    <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-amber-500" role="img" aria-label="Recent" />
   454→                  ) : null}
   455→                </li>
   456→              );
   457→            })}
   458→          </ul>
   459→        )}
   460→      </PopoverContent>
   461→    </Popover>
   462→  );
   463→}

   467→/** Total unread shopper messages — powers the "Messages" nav badge. */
   468→function useSellerChatUnread(active: SellerNav["view"]) {
   469→  const [unread, setUnread] = useState(0);
   470→  useEffect(() => {
   471→    let alive = true;
   472→    const load = () =>
   473→      apiGet<{ total: number }>("/api/store/chat/unread")
   474→        .then((d) => alive && setUnread(d.total ?? 0))
   475→        .catch(() => {});
   476→    load();
   477→    const t = setInterval(load, 15000);
   478→    return () => {
   479→      alive = false;
   480→      clearInterval(t);
   481→    };
   482→  }, [active]);
   483→  // instant re-sync when a thread is read inside the messages view (the view
   484→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   485→  useEffect(() => {
   486→    const sync = () =>
   487→      apiGet<{ total: number }>("/api/store/chat/unread")
   488→        .then((d) => setUnread(d.total ?? 0))
   489→        .catch(() => {});
   490→    window.addEventListener("nomofee:chat-sync", sync);
   491→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   492→  }, []);
   493→  return unread;
   494→}
