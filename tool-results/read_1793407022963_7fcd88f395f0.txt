    82→export type AffiliateTabKey =
    83→  | "overview"
    84→  | "discover"

    92→const TABS: { key: AffiliateTabKey; label: string; icon: typeof LayoutDashboard }[] = [
    93→  { key: "overview", label: "Overview", icon: LayoutDashboard },
    94→  { key: "discover", label: "Discover", icon: Search },
    95→  { key: "links", label: "Links", icon: Link2 },
    96→  { key: "collections", label: "Collections", icon: LayoutGrid },
    97→  { key: "showcase", label: "Showcase", icon: Megaphone },
    98→  { key: "network", label: "Network", icon: Users },
    99→  { key: "earnings", label: "Earnings", icon: CircleDollarSign },
   100→  { key: "payouts", label: "Payouts", icon: Wallet },
   101→];

   103→function sanitizeTab(t: string | undefined): AffiliateTabKey {
   104→  return (TABS.find((x) => x.key === t)?.key ?? "overview") as AffiliateTabKey;

   111→export default function AffiliateHubView({
   112→  ctx,
   113→  initialTab,
   114→  onBack,
   115→  onTabChange,
   116→}: {
   117→  ctx: ShopperCtx;
   118→  initialTab?: string;
   119→  /** When embedded (e.g. inside the Me hub) a dedicated back handler renders. */
   120→  onBack?: () => void;
   121→  /** Persist the active tab (shopper-shell writes the view cache key). */
   122→  onTabChange?: (tab: string) => void;
   123→}) {
   124→  const meState = useApp((s) => s.me);
   125→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   126→  const shopper = meState.loaded ? meState.shopper : null;
   127→
   128→  const [data, setData] = useState<AffiliateMeT | null>(null);
   129→  const [loading, setLoading] = useState(true);
   130→  const [loadErr, setLoadErr] = useState<string | null>(null);
   131→  // Public activation context for the NOT-signed-in gate (referral banner).
   132→  const [gateCtx, setGateCtx] = useState<AffiliateActivateCtxT | null>(null);
   133→  const [tab, setTab] = useState<AffiliateTabKey>(sanitizeTab(initialTab));
   134→  // Cross-tab "create a link for this product" hand-off from Discover.
   135→  const [linkDraft, setLinkDraft] = useState<DiscoverProductT | null>(null);
   136→  const [linkOpen, setLinkOpen] = useState(false);
   137→
   138→  const refresh = useCallback(async () => {
   139→    if (!shopper) {
   140→      setData(null);
   141→      setLoading(false);
   142→      return;
   143→    }
   144→    setLoading(true);
   145→    setLoadErr(null);
   146→    try {
   147→      const d = await affGet<AffiliateMeT>("/api/affiliate/me");
   148→      setData(d);
   149→    } catch (e) {
   150→      setData(null);
   151→      setLoadErr(e instanceof Error ? e.message : "Could not load your affiliate account");
   152→    } finally {
   153→      setLoading(false);
   154→    }
   155→  }, [shopper]);
   156→
   157→  useEffect(() => {
   158→    void refresh();
   159→  }, [refresh]);
   160→
   161→  // Guest: fetch the public pre-activation context (invited-by banner, pause
   162→  // state) — this endpoint is public by design for the landing experience.
   163→  useEffect(() => {
   164→    if (!meState.loaded || shopper) return;
   165→    let alive = true;
   166→    affGet<AffiliateActivateCtxT>("/api/affiliate/activate")
   167→      .then((d) => {
   168→        if (alive) setGateCtx(d);
   169→      })
   170→      .catch(() => {
   171→        /* banner is best-effort */
   172→      });
   173→    return () => {
   174→      alive = false;
   175→    };
   176→  }, [meState.loaded, shopper]);
   177→
   178→  const switchTab = useCallback(
   179→    (t: AffiliateTabKey) => {
   180→      setTab(t);
   181→      onTabChange?.(t);
   182→    },
   183→    [onTabChange]
   184→  );
   185→
   186→  const openLinkDialog = useCallback((p: DiscoverProductT | null) => {
   187→    setLinkDraft(p);
   188→    setLinkOpen(true);
   189→  }, []);
   190→
   191→  // ---- phases --------------------------------------------------------------
   192→
   193→  let body: React.ReactNode;
   194→  if (!meState.loaded || (loading && !data)) {
   195→    body = (
   196→      <div className="space-y-3" aria-busy="true">
   197→        <Skeleton className="h-24 rounded-2xl" />
   198→        <Skeleton className="h-10 w-72 rounded-full" />
   199→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   200→          {Array.from({ length: 4 }).map((_, i) => (
   201→            <Skeleton key={i} className="h-20 rounded-xl" />
   202→          ))}
   203→        </div>
   204→      </div>
   205→    );
   206→  } else if (!shopper) {
   207→    body = (
   208→      <SignInGate
   209→        invitedBy={gateCtx?.invitedBy ?? null}
   210→        enabled={gateCtx ? gateCtx.enabled : true}
   211→        onSignIn={() => openShopperAuth()}
   212→      />
   213→    );
   214→  } else if (loadErr || !data) {
   215→    body = (
   216→      <div className="space-y-4">
   217→        <TabError message={loadErr ?? "Could not load your affiliate account."} onRetry={() => void refresh()} />
   218→      </div>
   219→    );
   220→  } else if (!data.activated) {
   221→    body = data.settings.enabled ? (
   222→      <ActivationPanel settings={data.settings} invitedBy={data.invitedBy} onActivated={() => void refresh()} />
   223→    ) : (
   224→      <PausedCard />
   225→    );
   226→  } else if (data.profile.status !== "active") {
   227→    body = <AccountStatusCard status={data.profile.status} />;
   228→  } else {
   229→    const { profile, settings } = data;
   230→    body = (
   231→      <>
   232→        {/* tab bar */}
   233→        <div className="sticky top-[57px] z-20 -mx-3 border-b bg-background/95 px-3 backdrop-blur supports-[backdrop-filter]:bg-background/85 sm:-mx-4 sm:px-4">
   234→          <div className="flex gap-1 overflow-x-auto py-2" role="tablist" aria-label="Affiliate hub sections">
   235→            {TABS.map(({ key, label, icon: Icon }) => (
   236→              <button
   237→                key={key}
   238→                type="button"
   239→                role="tab"
   240→                aria-selected={tab === key}
   241→                onClick={() => switchTab(key)}
   242→                className={cn(
   243→                  "inline-flex min-h-[44px] shrink-0 items-center gap-1.5 rounded-full px-3.5 text-xs font-bold transition-colors sm:text-[13px]",
   244→                  tab === key ? "bg-brand-gradient text-white shadow-glow-cyan-sm" : "text-muted-foreground hover:bg-muted hover:text-foreground"
   245→                )}
   246→              >
   247→                <Icon className="h-4 w-4" aria-hidden />
   248→                {label}
   249→              </button>
   250→            ))}
   251→          </div>
   252→        </div>
   253→
   254→        <div className="pt-4">
   255→          {tab === "overview" ? (
   256→            <OverviewTab settings={settings} onNavigate={switchTab} />
   257→          ) : tab === "discover" ? (
   258→            <DiscoverTab onCreateLink={(p) => openLinkDialog(p)} />
   259→          ) : tab === "links" ? (
   260→            <LinksTab onCreate={() => openLinkDialog(null)} />
   261→          ) : tab === "collections" ? (
   262→            <CollectionsTab ctx={ctx} />
   263→          ) : tab === "showcase" ? (
   264→            <ShowcaseTab ctx={ctx} profile={profile} />
   265→          ) : tab === "network" ? (
   266→            <NetworkTab settings={settings} joinCode={profile.code} />
   267→          ) : tab === "earnings" ? (
   268→            <EarningsTab settings={settings} />
   269→          ) : (
   270→            <PayoutsTab />
   271→          )}
   272→        </div>
   273→
   274→        {/* link creation dialog — shared by Links tab + Discover cards */}
   275→        <LinkCreateDialog
   276→          open={linkOpen}
   277→          onOpenChange={setLinkOpen}
   278→          preset={linkDraft}
   279→          onCreated={() => {
   280→            if (tab !== "links") switchTab("links");
   281→          }}
   282→        />
   283→      </>
   284→    );
   285→  }
   286→
   287→  return (
   288→    <div className="mx-auto w-full max-w-7xl flex-1 px-3 py-4 sm:px-4 sm:py-6">
   289→      {/* page header */}
   290→      <div className="flex items-center gap-3 pb-4">
   291→        {onBack ? (
   292→          <Button type="button" variant="ghost" size="icon" className="h-10 w-10 shrink-0" onClick={onBack} aria-label="Back to dashboard">
   293→            <ArrowLeft className="h-5 w-5" />
   294→          </Button>
   295→        ) : null}
   296→        <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-brand-gradient text-white shadow-glow-cyan-sm" aria-hidden>
   297→          <HandCoins className="h-5 w-5" />
   298→        </span>
   299→        <div className="min-w-0">
   300→          <h1 className="truncate text-lg font-extrabold tracking-tight text-slate-900 sm:text-xl">Affiliate Hub</h1>
   301→          <p className="truncate text-xs text-muted-foreground">Earn from products you genuinely recommend</p>
   302→        </div>
   303→        {data?.activated && data.profile.status === "active" ? (
   304→          <div className="ml-auto hidden shrink-0 items-center gap-2 sm:flex">
   305→            <CopyButton text={referralJoinUrl(data.profile.code)} label="Join link" />
   306→          </div>
   307→        ) : null}
   308→      </div>
   309→      {body}
   310→    </div>
   311→  );
   312→}

   407→function ActivationPanel({
   408→  settings,
   409→  invitedBy,
   410→  onActivated,
   411→}: {
   412→  settings: AffiliateSettingsT;
   413→  invitedBy: string | null;
   414→  onActivated: () => void;
   415→}) {
   416→  const meState = useApp((s) => s.me);
   417→  const shopper = meState.loaded ? meState.shopper : null;
   418→  const [displayName, setDisplayName] = useState("");
   419→  const [bio, setBio] = useState("");
   420→  const [slug, setSlug] = useState("");
   421→  const [sponsorCode, setSponsorCode] = useState("");
   422→  const [confirmSponsor, setConfirmSponsor] = useState(false);
   423→  const [terms, setTerms] = useState(false);
   424→  const [busy, setBusy] = useState(false);
   425→  const [error, setError] = useState<string | null>(null);
   426→  const hydrated = useRef(false);
   427→
   428→  // Prefill display name from the signed-in shopper (once).
   429→  useEffect(() => {
   430→    if (hydrated.current || !shopper) return;
   431→    hydrated.current = true;
   432→    setDisplayName(shopper.fullName);
   433→  }, [shopper]);
   434→
   435→  const submit = async () => {
   436→    setError(null);
   437→    if (!terms) {
   438→      setError("Please accept the affiliate programme terms first.");
   439→      return;
   440→    }
   441→    if (invitedBy && !confirmSponsor) {
   442→      setError("Confirm the sponsor relationship shown above, or continue without it by leaving the confirmation unticked (the invitation cannot be re-applied later).");
   443→      return;
   444→    }
   445→    if (!invitedBy && sponsorCode.trim() && !confirmSponsor) {
   446→      setError("Tick the confirmation to use this sponsor code — the relationship is permanent.");
   447→      return;
   448→    }
   449→    setBusy(true);
   450→    try {
   451→      await affPost("/api/affiliate/activate", {
   452→        displayName: displayName.trim() || undefined,
   453→        bio: bio.trim() || undefined,
   454→        slug: slug.trim() || undefined,
   455→        sponsorCode: !invitedBy && sponsorCode.trim() ? sponsorCode.trim() : undefined,
   456→        sponsorConfirmed: confirmSponsor,
   457→        termsAccepted: true,
   458→      });
   459→      toast.success("Welcome to the NOMOFEE Affiliate Programme!");
   460→      onActivated();
   461→    } catch (e) {
   462→      const af = e as { status?: number; code?: string; message?: string };
   463→      if (af.code === "ALREADY_AFFILIATE") {
   464→        toast.info("You're already an affiliate");
   465→        onActivated();
   466→        return;
   467→      }
   468→      if (af.code === "AFFILIATE_DISABLED") {
   469→        setError(af.message ?? "The affiliate programme is currently paused.");
   470→      } else {
   471→        setError(af.message ?? "Activation failed — please try again.");
   472→      }
   473→    } finally {
   474→      setBusy(false);
   475→    }
   476→  };
   477→
   478→  return (
   479→    <div className="mx-auto w-full max-w-2xl space-y-4 pb-6">
   480→      <div className="rounded-2xl border bg-card p-5 shadow-soft-card sm:p-6">
   481→        <h2 className="flex items-center gap-2 text-base font-extrabold text-slate-900">
   482→          <BadgeCheck className="h-5 w-5 text-teal-700" aria-hidden /> Activate your affiliate account
   483→        </h2>
   484→        <p className="pt-1 text-xs text-muted-foreground">
   485→          Free to join. You can update how your name appears later — but the sponsor decision below is permanent.
   486→        </p>
   487→
   488→        {invitedBy ? (
   489→          <div className="mt-4 rounded-xl border border-teal-200 bg-teal-50/70 p-4" role="note">
   490→            <p className="text-sm font-bold text-teal-900">You were invited by @{invitedBy}</p>
   491→            <p className="pt-1 text-xs leading-relaxed text-teal-900/80">
   492→              Accepting makes this person your permanent sponsor: they earn 10% of the commission pool on your future
   493→              eligible sales, and you keep 90%. This cannot be changed later.
   494→            </p>
   495→            <label className="mt-3 flex min-h-[44px] cursor-pointer items-center gap-2.5 rounded-lg bg-white/70 px-3 text-xs font-semibold text-teal-900">
   496→              <Checkbox checked={confirmSponsor} onCheckedChange={(v) => setConfirmSponsor(v === true)} aria-label="Confirm permanent sponsor relationship" />
   497→              Confirm permanent sponsor relationship with @{invitedBy}
   498→            </label>
   499→          </div>
   500→        ) : (
   501→          <div className="mt-4 space-y-3">
   502→            <div className="space-y-1.5">
   503→              <Label htmlFor="aff-sponsor-code" className="text-xs font-bold text-slate-700">
   504→                Sponsor code (optional)
   505→              </Label>
   506→              <Input
   507→                id="aff-sponsor-code"
   508→                value={sponsorCode}
   509→                onChange={(e) => setSponsorCode(e.target.value)}
   510→                placeholder="e.g. a1b2c3d4e5 — only if a friend shared their code"
   511→                className="min-h-[44px]"
   512→                maxLength={32}
   513→                autoComplete="off"
   514→              />
   515→              <p className="text-[11px] text-muted-foreground">Leave empty if nobody invited you.</p>
   516→            </div>
   517→            {sponsorCode.trim() ? (
   518→              <label className="flex min-h-[44px] cursor-pointer items-center gap-2.5 rounded-lg border border-teal-200 bg-teal-50/70 px-3 text-xs font-semibold text-teal-900">
   519→                <Checkbox checked={confirmSponsor} onCheckedChange={(v) => setConfirmSponsor(v === true)} aria-label="Confirm permanent sponsor relationship" />
   520→                Make this my permanent sponsor (cannot be changed later)
   521→              </label>
   522→            ) : null}
   523→          </div>
   524→        )}
