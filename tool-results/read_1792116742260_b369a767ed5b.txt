   181→   181→// data hook — profile + wallet + purchase counts + like/viewed counters
   182→   182→// ---------------------------------------------------------------------------
   183→   183→
   184→   184→function useMeData(enabled: boolean) {
   185→   185→  const [profile, setProfile] = useState<ProfileT | null>(null);
   186→   186→  const [wallet, setWallet] = useState<WalletT | null>(null);
   187→   187→  const [counts, setCounts] = useState<BucketCounts | null>(null);
   188→   188→  const [likeCount, setLikeCount] = useState(0);
   189→   189→  const [viewedCount, setViewedCount] = useState(0);
   190→   190→  const [loading, setLoading] = useState(enabled);
   191→   191→
   192→   192→  const refresh = useCallback(async () => {
   193→   193→    if (!enabled) return;
   194→   194→    const [p, w, o, lk, vw] = await Promise.allSettled([
   195→   195→      fetchJson<ProfileT>("/api/me/profile"),
   196→   196→      fetchJson<WalletT>("/api/me/wallet"),
   197→   197→      fetchJson<{ bucketCounts: BucketCounts; orders: OrderRowT[] }>("/api/me/orders?size=1"),
   198→   198→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=LIKE&size=1"),
   199→   199→      fetchEnvelope<{ activities: unknown[] }>("/api/me/activity?type=VIEWED&size=1"),
   200→   200→    ]);
   201→   201→    if (p.status === "fulfilled") setProfile(p.value);
   202→   202→    if (w.status === "fulfilled") {
   203→   203→      // DORMANT SCAFFOLD — MESH FINANCE: while MESH_FINANCE_ENABLED is false
   204→   204→      // the wallet API answers a `disabled` envelope (wallet/checkIn null).
   205→   205→      // Normalize it into the UI shape so the kept scaffold cards render
   206→   206→      // inert zeroes instead of crashing; the cards stay hidden behind the flag.
   207→   207→      const raw = w.value as Partial<WalletT> & { disabled?: boolean };
   208→   208→      setWallet({
   209→   209→        wallet: raw.wallet ?? { balanceMinor: 0, coinBalance: 0, spayLaterLimitMinor: 0, spayLaterUsedMinor: 0, spayLaterAvailableMinor: 0, checkInStreak: 0 },
   210→   210→        vouchers: raw.vouchers ?? { available: 0, used: 0 },
   211→   211→        checkIn: raw.checkIn ?? { canClaim: false, nextClaimAt: null },
   212→   212→      });
   213→   213→    }
   214→   214→    if (o.status === "fulfilled") setCounts(o.value.bucketCounts);
   215→   215→    if (lk.status === "fulfilled") setLikeCount(lk.value.meta?.total ?? 0);
   216→   216→    if (vw.status === "fulfilled") setViewedCount(vw.value.meta?.total ?? 0);
   217→   217→    setLoading(false);
   218→   218→  }, [enabled]);
   219→   219→
   220→   220→  useEffect(() => {
   221→   221→    // eslint-disable-next-line react-hooks/set-state-in-effect -- initial dashboard data load (async fetch inside refresh)
   222→   222→    void refresh();
   223→   223→  }, [refresh]);
   224→   224→
   225→   225→  // Address CRUD (in this hub's sub-view or during checkout) must reflect on
   226→   226→  // the dashboard teaser instantly — "N saved addresses" never goes stale.
   227→   227→  useEffect(() => {
   228→   228→    if (!enabled) return;
   229→   229→    const sync = () => void refresh();
   230→   230→    window.addEventListener("shopmesh:addresses-changed", sync);
   231→   231→    return () => window.removeEventListener("shopmesh:addresses-changed", sync);
   232→   232→  }, [enabled, refresh]);
   233→   233→
   234→   234→  return { profile, wallet, counts, likeCount, viewedCount, loading, refresh };
   235→   235→}

   608→   608→function Avatar({ profile, size = "md" }: { profile: ProfileT | null; size?: "sm" | "md" | "lg" }) {
   609→   609→  const px = size === "lg" ? "h-16 w-16 text-xl" : size === "md" ? "h-10 w-10 text-sm" : "h-8 w-8 text-xs";
   610→   610→  const gradKey = profile?.avatarUrl?.startsWith("grad:") ? profile.avatarUrl : null;
   611→   611→  const gradClass =
   612→   612→    gradKey === "grad:amber"
   613→   613→      ? "from-amber-400 to-orange-500"
   614→   614→      : gradKey === "grad:rose"
   615→   615→        ? "from-rose-400 to-pink-600"
   616→   616→        : gradKey === "grad:violet"
   617→   617→          ? "from-violet-400 to-purple-600"
   618→   618→          : gradKey === "grad:emerald"
   619→   619→            ? "from-emerald-400 to-teal-600"
   620→   620→            : gradKey === "grad:slate"
   621→   621→              ? "from-slate-400 to-slate-600"
   622→   622→              : "from-teal-500 to-teal-800";
   623→   623→  if (profile?.avatarUrl && !gradKey && /^https?:\/\/|^\//.test(profile.avatarUrl)) {
   624→   624→    return <img src={profile.avatarUrl} alt={`${profile.fullName} avatar`} className={cn(px, "shrink-0 rounded-full object-cover ring-2 ring-white/70")} />;
   625→   625→  }
   626→   626→  const initial = (profile?.fullName ?? "?").trim().slice(0, 1).toUpperCase();
   627→   627→  return (
   628→   628→    <span className={cn(px, "flex shrink-0 items-center justify-center rounded-full bg-gradient-to-br font-extrabold text-white ring-2 ring-white/70", gradClass)} aria-hidden="true">
   629→   629→      {initial}
   630→   630→    </span>
   631→   631→  );
   632→   632→}
