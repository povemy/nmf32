   311→/** Total unread shopper messages — powers the "Messages" nav badge. */
   312→function useSellerChatUnread(active: SellerNav["view"]) {
   313→  const [unread, setUnread] = useState(0);
   314→  useEffect(() => {
   315→    let alive = true;
   316→    const load = () =>
   317→      apiGet<{ total: number }>("/api/store/chat/unread")
   318→        .then((d) => alive && setUnread(d.total ?? 0))
   319→        .catch(() => {});
   320→    load();
   321→    const t = setInterval(load, 15000);
   322→    return () => {
   323→      alive = false;
   324→      clearInterval(t);
   325→    };
   326→  }, [active]);
   327→  // instant re-sync when a thread is read inside the messages view (the view
   328→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   329→  useEffect(() => {
   330→    const sync = () =>
   331→      apiGet<{ total: number }>("/api/store/chat/unread")
   332→        .then((d) => setUnread(d.total ?? 0))
   333→        .catch(() => {});
   334→    window.addEventListener("nomofee:chat-sync", sync);
   335→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   336→  }, []);
   337→  return unread;
   338→}

   480→function AuthScreen({ onAuthed }: { onAuthed: () => Promise<void> }) {
   481→  const setSurface = useApp((s) => s.setSurface);
   482→  const [loginEmail, setLoginEmail] = useState("");
   483→  const [loginPassword, setLoginPassword] = useState("");
   484→  const [busy, setBusy] = useState(false);
   485→  const [reg, setReg] = useState({ fullName: "", displayName: "", email: "", password: "", phone: "", storeName: "" });
   486→
   487→  const goMarket = () => setSurface("shopper");
   488→
   489→  const login = async () => {
   490→    if (!loginEmail || !loginPassword) {
   491→      toast.error("Enter your email and password");
   492→      return;
   493→    }
   494→    setBusy(true);
   495→    const ok = await runMutation(async () => {
   496→      const body = await apiSend<{ token?: string }>("POST", "/api/auth/login", {
   497→        email: loginEmail,
   498→        password: loginPassword,
   499→        portal: "seller",
   500→      });
   501→      setSessionToken(body?.token); // localStorage bridge for cookie-less contexts
   502→    }, "Welcome back!");
   503→    setBusy(false);
   504→    if (ok) await onAuthed();
   505→  };
   506→
   507→  const register = async () => {
   508→    if (!reg.fullName.trim() || !reg.email.includes("@") || reg.password.length < 8) {
   509→      toast.error("Name, valid email and a password of at least 8 characters are required");
   510→      return;
