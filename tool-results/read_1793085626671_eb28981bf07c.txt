    49→/* ------------------------------------------------------------- SettingsView */
    50→
    51→type SettingRow = { key: string; valueJson: string; description?: string | null; updatedAt?: string };
    52→
    53→function prettyValue(raw: string): string {
    54→  try {
    55→    return JSON.stringify(JSON.parse(raw), null, 0);
    56→  } catch {
    57→    return raw;
    58→  }
    59→}
    60→
    61→export function SettingsView() {
    62→  const { data, error, loading, refetch } = useApi<SettingRow[]>("/api/admin/settings");
    63→  const rows = data ?? [];
    64→
    65→  const [editing, setEditing] = useState<SettingRow | null>(null);
    66→  const [draft, setDraft] = useState("");
    67→  const [saving, setSaving] = useState(false);
    68→
    69→  const openEdit = (r: SettingRow) => {
    70→    setEditing(r);
    71→    setDraft(r.valueJson ?? "");
    72→  };
    73→
    74→  const save = async () => {
    75→    if (!editing) return;
    76→    try {
    77→      JSON.parse(draft);
    78→    } catch {
    79→      toast.error("Invalid JSON", {
    80→        description: "Use quoted strings, numbers, booleans or objects — e.g. \"friday\" or 1000.",
    81→      });
    82→      return;
    83→    }
    84→    setSaving(true);
    85→    try {
    86→      await apiSend("/api/admin/settings", "PUT", { settings: [{ key: editing.key, valueJson: draft }] });
    87→      toast.success("Setting saved", { description: editing.key });
    88→      setEditing(null);
    89→      void refetch();
    90→    } catch (e) {
    91→      toast.error("Save failed", { description: e instanceof Error ? e.message : String(e) });
    92→    } finally {
    93→      setSaving(false);
    94→    }
    95→  };
    96→
    97→  return (
    98→    <div className="space-y-4">
    99→      <PageHeader
   100→        title="Settings"
   101→        description="Platform configuration — JSON values consumed across commission, payouts, moderation and payments."
   102→      />
   103→
   104→      <Card className="py-0">
   105→        <CardContent className="p-4">
   106→          <div className="overflow-x-auto">
   107→            {error ? (
   108→              <ErrorState message={error} onRetry={() => void refetch()} />
   109→            ) : loading ? (
   110→              <TableSkeleton rows={6} cols={3} />
   111→            ) : rows.length === 0 ? (
   112→              <EmptyState title="No settings" />
   113→            ) : (
   114→              <Table>
   115→                <TableHeader>
   116→                  <TableRow>
   117→                    <TableHead>Key</TableHead>
   118→                    <TableHead>Value</TableHead>
   119→                    <TableHead className="hidden md:table-cell">Description</TableHead>
   120→                    <TableHead className="hidden text-right lg:table-cell">Updated</TableHead>
   121→                    <TableHead className="text-right">Action</TableHead>
   122→                  </TableRow>
   123→                </TableHeader>
   124→                <TableBody>
   125→                  {rows.map((r) => (
   126→                    <TableRow key={r.key}>
   127→                      <TableCell className="font-mono text-xs font-semibold">{r.key}</TableCell>
   128→                      <TableCell>
   129→                        <code className="block max-w-[240px] truncate rounded bg-muted px-1.5 py-0.5 font-mono text-xs" title={prettyValue(r.valueJson)}>
   130→                          {prettyValue(r.valueJson)}
   131→                        </code>
   132→                      </TableCell>
   133→                      <TableCell className="hidden max-w-[280px] text-sm text-muted-foreground md:table-cell">{r.description ?? "—"}</TableCell>
   134→                      <TableCell className="hidden text-right text-xs text-muted-foreground lg:table-cell">{timeAgo(r.updatedAt)}</TableCell>
   135→                      <TableCell className="text-right">
   136→                        <Button size="sm" variant="outline" className="min-h-[44px] sm:min-h-0" onClick={() => openEdit(r)}>
   137→                          Edit
   138→                        </Button>
   139→                      </TableCell>
   140→                    </TableRow>
   141→                  ))}
   142→                </TableBody>
   143→              </Table>
   144→            )}
   145→          </div>
   146→        </CardContent>
   147→      </Card>
   148→
   149→      <Dialog open={editing !== null} onOpenChange={(o) => !o && setEditing(null)}>
   150→        <DialogContent className="sm:max-w-md">
   151→          <DialogHeader>
   152→            <DialogTitle className="font-mono text-sm">{editing?.key}</DialogTitle>
   153→            <DialogDescription>{editing?.description ?? "Edit the JSON value for this setting."}</DialogDescription>
   154→          </DialogHeader>
   155→          <div className="space-y-1.5">
   156→            <Textarea
   157→              value={draft}
   158→              onChange={(e) => setDraft(e.target.value)}
   159→              rows={5}
   160→              className="resize-none font-mono text-xs"
   161→              aria-label="Setting value (JSON)"
   162→              placeholder='e.g. "friday" or 1000 or {"weights":{}}'
   163→            />
   164→            <p className="text-[11px] text-muted-foreground">Must be valid JSON — strings need quotes, numbers/booleans stay bare.</p>
   165→          </div>
   166→          <DialogFooter>
   167→            <Button variant="outline" onClick={() => setEditing(null)} disabled={saving} className="min-h-[44px] sm:min-h-0">Cancel</Button>
   168→            <Button onClick={() => void save()} disabled={saving} className="min-h-[44px] sm:min-h-0">
   169→              {saving ? "Saving…" : "Save setting"}
   170→            </Button>
   171→          </DialogFooter>
   172→        </DialogContent>
   173→      </Dialog>
   174→    </div>
   175→  );
   176→}

   180→type FlagRow = { key: string; description?: string | null; isEnabled: boolean; rolloutPercent: number };
   181→
   182→export function FlagsView() {
   183→  const { data, error, loading, refetch } = useApi<FlagRow[]>("/api/admin/feature-flags");
   184→  const rows = data ?? [];
   185→  const [busyKey, setBusyKey] = useState<string | null>(null);
   186→
   187→  const toggle = async (f: FlagRow) => {
   188→    setBusyKey(f.key);
   189→    try {
   190→      await apiSend("/api/admin/feature-flags", "PATCH", { key: f.key, isEnabled: !f.isEnabled });
   191→      toast.success(`Flag ${f.isEnabled ? "disabled" : "enabled"}`, { description: f.key });
   192→      void refetch();
   193→    } catch (e) {
   194→      toast.error("Toggle failed", { description: e instanceof Error ? e.message : String(e) });
   195→    } finally {
   196→      setBusyKey(null);
   197→    }
   198→  };
   199→
   200→  return (
   201→    <div className="space-y-4">
   202→      <PageHeader title="Feature Flags" description="Runtime switches for progressive delivery — changes are audited." />
   203→
   204→      <Card className="py-0">
   205→        <CardHeader className="px-4 pt-4 sm:px-6 sm:pt-5">
   206→          <CardTitle className="flex items-center gap-2 text-base">
   207→            <Flag className="h-4 w-4 text-teal-600 dark:text-teal-400" /> Flags ({rows.length})
   208→          </CardTitle>
   209→        </CardHeader>
   210→        <CardContent className="p-0">
   211→          {error ? (
   212→            <ErrorState message={error} onRetry={() => void refetch()} />
   213→          ) : loading ? (
   214→            <TableSkeleton rows={5} cols={4} />
   215→          ) : rows.length === 0 ? (
   216→            <EmptyState title="No feature flags" />
   217→          ) : (
   218→            <Table>
   219→              <TableHeader>
   220→                <TableRow>
   221→                  <TableHead>Flag</TableHead>
   222→                  <TableHead className="hidden md:table-cell">Description</TableHead>
   223→                  <TableHead>Rollout</TableHead>
   224→                  <TableHead className="text-center">Enabled</TableHead>
   225→                </TableRow>
   226→              </TableHeader>
   227→              <TableBody>
   228→                {rows.map((f) => (
   229→                  <TableRow key={f.key}>
   230→                    <TableCell className="font-mono text-xs font-semibold">{f.key}</TableCell>
   231→                    <TableCell className="hidden max-w-[320px] text-sm text-muted-foreground md:table-cell">{f.description ?? "—"}</TableCell>
   232→                    <TableCell>
   233→                      <Badge
   234→                        variant="outline"
   235→                        className={cn(
   236→                          f.rolloutPercent >= 100
   237→                            ? "border-emerald-200 bg-emerald-50 text-emerald-800 dark:border-emerald-900 dark:bg-emerald-950/60 dark:text-emerald-300"
   238→                            : "border-amber-200 bg-amber-50 text-amber-900 dark:border-amber-900 dark:bg-amber-950/60 dark:text-amber-300"
   239→                        )}
   240→                      >
   241→                        {f.rolloutPercent}% rollout
   242→                      </Badge>
   243→                    </TableCell>
   244→                    <TableCell className="text-center">
   245→                      <Switch checked={f.isEnabled} disabled={busyKey === f.key} onCheckedChange={() => void toggle(f)} aria-label={`Toggle ${f.key}`} />
   246→                    </TableCell>
   247→                  </TableRow>
   248→                ))}
   249→              </TableBody>
   250→            </Table>
   251→          )}
   252→        </CardContent>
   253→      </Card>
   254→    </div>
   255→  );
   256→}

   260→type AuditRow = {
   261→  id: string;
   262→  adminEmail: string;

   271→/** Audit rows deliver parsed objects; DiffViewer expects JSON strings. */
   272→function asJsonStr(v: unknown): string | null {
   273→  if (v === null || v === undefined) return null;
   274→  if (typeof v === "string") return v;
   275→  try {
   276→    return JSON.stringify(v);
   277→  } catch {
   278→    return null;
   279→  }
   280→}

   288→  const url = useMemo(
   289→    () => `/api/admin/audit-logs${qs({ adminEmail, action, entityType, page, size: 20 })}`,
   290→    [adminEmail, action, entityType, page]
   291→  );
   292→  const { data, meta, error, loading, refetch } = useApi<AuditRow[]>(url);
   293→  const rows = data ?? [];

   320→              value={action}
   321→              onChange={(e) => {
   322→                setAction(e.target.value);
   323→                setPage(1);
   324→              }}
   325→              placeholder="Action contains… e.g. payout"
   326→              className="min-h-[44px]"
   327→              aria-label="Filter by action"
   328→            />
   329→            <Input
   330→              value={entityType}
   331→              onChange={(e) => {

   341→          <div className="mt-4 overflow-x-auto">

   343→              <ErrorState message={error} onRetry={() => void refetch()} />

   349→              <>
   350→                <Table>
   351→                  <TableHeader>
   352→                    <TableRow>
   353→                      <TableHead>Admin</TableHead>
   354→                      <TableHead>Action</TableHead>
   355→                      <TableHead className="hidden md:table-cell">Entity</TableHead>
   356→                      <TableHead className="hidden lg:table-cell">Entity ID</TableHead>

   359→                  </TableHeader>
   360→                  <TableBody>
   361→                    {rows.map((r) => (
