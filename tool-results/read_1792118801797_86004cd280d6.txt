   473→
   474→type CouponRow = {
   475→  id: string;
   476→  scope: string;
   477→  code: string;
   478→  description?: string | null;
   479→  type: "percent" | "fixed" | "freeship" | string;
   480→  percent?: number | null;
   481→  amountMinor?: number | null;
   482→  minSpendMinor?: number;
   483→  maxRedemptions?: number | null;
   484→  perCustomerLimit?: number;
   485→  startsAt?: string | null;
   486→  endsAt?: string | null;
   487→  isActive: boolean;
   488→  timesUsed?: number;
   489→  createdAt?: string;
   490→  store?: { id: string; name: string } | null;
   491→};
   492→
   493→function couponValue(c: CouponRow): string {
   494→  if (c.type === "percent") return `${c.percent ?? 0}% off`;
   495→  if (c.type === "fixed") return formatMinor(c.amountMinor ?? 0) + " off";
   496→  return "Free shipping";
   497→}
   498→

   501→  const rows = data ?? [];
   502→
   503→  const [createOpen, setCreateOpen] = useState(false);
   504→  const [code, setCode] = useState("");
   505→  const [description, setDescription] = useState("");
   506→  const [type, setType] = useState<"percent" | "fixed" | "freeship">("percent");
   507→  const [percent, setPercent] = useState("10");
   508→  const [amount, setAmount] = useState("10.00");
   509→  const [minSpend, setMinSpend] = useState("0");
   510→  const [perCustomerLimit, setPerCustomerLimit] = useState("1");
   511→  const [startsAt, setStartsAt] = useState("");
   512→  const [endsAt, setEndsAt] = useState("");
   513→  const [creating, setCreating] = useState(false);
   514→  const [busyId, setBusyId] = useState<string | null>(null);
   515→  const [deleteTarget, setDeleteTarget] = useState<CouponRow | null>(null);
   516→
   517→  const resetForm = () => {
   518→    setCode("");
   519→    setDescription("");
   520→    setType("percent");
   521→    setPercent("10");
   522→    setAmount("10.00");
   523→    setMinSpend("0");
   524→    setPerCustomerLimit("1");
   525→    setStartsAt("");
   526→    setEndsAt("");
   527→  };
   528→
   529→  const create = async () => {
   530→    setCreating(true);
   531→    try {
   532→      const body: Record<string, unknown> = {
   533→        code: code.trim().toUpperCase(),
   534→        description: description.trim() || undefined,
   535→        type,
   536→        minSpendMinor: toMinor(minSpend || "0"),
   537→        perCustomerLimit: Math.max(1, Number(perCustomerLimit) || 1),
   538→      };
   539→      if (type === "percent") body.percent = Number(percent);
   540→      if (type === "fixed") body.amountMinor = toMinor(amount || "0");
   541→      if (startsAt) body.startsAt = new Date(startsAt).toISOString();
   542→      if (endsAt) body.endsAt = new Date(`${endsAt}T23:59:59`).toISOString();
   543→      await apiSend("/api/admin/coupons", "POST", body);
   544→      toast.success("Coupon created", { description: body.code as string });
   545→      setCreateOpen(false);
   546→      resetForm();
   547→      void refetch();
   548→    } catch (e) {
   549→      toast.error("Create failed", { description: e instanceof Error ? e.message : String(e) });
   550→    } finally {
   551→      setCreating(false);
   552→    }
   553→  };
   554→
   555→  const toggle = async (c: CouponRow) => {
   556→    setBusyId(c.id);
   557→    try {
   558→      await apiSend(`/api/admin/coupons/${c.id}`, "PATCH", { isActive: !c.isActive });
   559→      toast.success(`Coupon ${c.isActive ? "disabled" : "enabled"}`, { description: c.code });
   560→      void refetch();
   561→    } catch (e) {
   562→      toast.error("Update failed", { description: e instanceof Error ? e.message : String(e) });
   563→    } finally {
   564→      setBusyId(null);
   565→    }
   566→  };
   567→
   568→  const doDelete = async () => {
   569→    if (!deleteTarget) return;
   570→    setBusyId(deleteTarget.id);
   571→    try {
   572→      await apiSend(`/api/admin/coupons/${deleteTarget.id}`, "DELETE");
   573→      toast.success("Coupon deleted", { description: deleteTarget.code });
   574→      setDeleteTarget(null);
   575→      void refetch();
   576→    } catch (e) {
   577→      toast.error("Delete failed", { description: e instanceof Error ? e.message : String(e) });
   578→    } finally {
   579→      setBusyId(null);
   580→    }
   581→  };
   582→
   583→  return (
   584→    <div className="space-y-4">
   585→      <PageHeader
   586→        title="Coupons"
   587→        description="Platform-wide promo codes — store-scoped coupons are managed in the seller portal."
   588→        actions={
   589→          <Button size="sm" className="min-h-[44px]" onClick={() => setCreateOpen(true)}>
   590→            <Ticket className="h-4 w-4" /> New coupon
   591→          </Button>
   592→        }
   593→      />
   594→
   595→      <Card className="py-0">
   596→        <CardContent className="p-4">
   597→          <div className="overflow-x-auto">
   598→            {error ? (
   599→              <ErrorState message={error} onRetry={() => void refetch()} />
   600→            ) : loading ? (
   601→              <TableSkeleton rows={4} cols={7} />
   602→            ) : rows.length === 0 ? (
   603→              <EmptyState title="No coupons" hint="Create a platform coupon to attract first orders." />
   604→            ) : (
   605→              <Table>
   606→                <TableHeader>
   607→                  <TableRow>
   608→                    <TableHead>Code</TableHead>
   609→                    <TableHead>Value</TableHead>
   610→                    <TableHead className="hidden md:table-cell">Scope</TableHead>
   611→                    <TableHead className="hidden text-right sm:table-cell">Min spend</TableHead>
   612→                    <TableHead className="hidden text-center lg:table-cell">Used</TableHead>
   613→                    <TableHead className="hidden xl:table-cell">Window</TableHead>
   614→                    <TableHead className="text-center">Active</TableHead>
   615→                    <TableHead className="text-right">Actions</TableHead>
   616→                  </TableRow>
   617→                </TableHeader>
   618→                <TableBody>
   619→                  {rows.map((c) => (
   620→                    <TableRow key={c.id}>
   621→                      <TableCell>
   622→                        <p className="font-mono text-sm font-semibold">{c.code}</p>
   623→                        <p className="max-w-[200px] truncate text-xs text-muted-foreground">{c.description ?? "—"}</p>
   624→                      </TableCell>
   625→                      <TableCell>
   626→                        <Badge variant="outline" className="capitalize">{c.type}</Badge>
   627→                        <p className="mt-1 text-sm">{couponValue(c)}</p>
   628→                      </TableCell>
   629→                      <TableCell className="hidden text-sm md:table-cell">
   630→                        {c.scope === "store" ? c.store?.name ?? "store" : <Badge variant="outline">platform</Badge>}
   631→                      </TableCell>
   632→                      <TableCell className="hidden text-right sm:table-cell"><Money minor={c.minSpendMinor ?? 0} muted /></TableCell>
   633→                      <TableCell className="hidden text-center tabular-nums lg:table-cell">{c.timesUsed ?? 0}</TableCell>
   634→                      <TableCell className="hidden text-xs text-muted-foreground xl:table-cell">
   635→                        {c.startsAt || c.endsAt ? `${c.startsAt ? fmtDT(c.startsAt) : "∞"} → ${c.endsAt ? fmtDT(c.endsAt) : "∞"}` : "always"}
   636→                      </TableCell>
   637→                      <TableCell className="text-center">
   638→                        <Switch checked={c.isActive} disabled={busyId === c.id} onCheckedChange={() => void toggle(c)} aria-label={`Toggle coupon ${c.code}`} />
   639→                      </TableCell>
   640→                      <TableCell className="text-right">
   641→                        <Button
   642→                          size="sm"
   643→                          variant="ghost"
   644→                          className="min-h-[44px] text-red-600 hover:bg-red-50 hover:text-red-700 sm:min-h-0 dark:hover:bg-red-950/40"
   645→                          onClick={() => setDeleteTarget(c)}
   646→                        >
   647→                          Delete
   648→                        </Button>
   649→                      </TableCell>
   650→                    </TableRow>
   651→                  ))}
   652→                </TableBody>
   653→              </Table>
   654→            )}
   655→          </div>
   656→        </CardContent>
   657→      </Card>
   658→
   659→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   660→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg [scrollbar-width:thin]">
   661→          <DialogHeader>
   662→            <DialogTitle>New platform coupon</DialogTitle>
   663→            <DialogDescription>Applies at checkout across all storefronts (unless a store scope is chosen server-side).</DialogDescription>
   664→          </DialogHeader>
   665→          <div className="space-y-3">
   666→            <div className="grid grid-cols-2 gap-3">
   667→              <div className="space-y-1.5">
   668→                <Label htmlFor="coupon-code">Code</Label>
   669→                <Input
   670→                  id="coupon-code"
   671→                  value={code}
   672→                  onChange={(e) => setCode(e.target.value.toUpperCase())}
   673→                  placeholder="RAYA25"
   674→                  className="min-h-[44px] font-mono uppercase"
   675→                  autoComplete="off"
   676→                />
   677→              </div>
   678→              <div className="space-y-1.5">
   679→                <Label>Type</Label>
   680→                <Select value={type} onValueChange={(v) => setType(v as typeof type)}>
   681→                  <SelectTrigger className="min-h-[44px]" aria-label="Coupon type"><SelectValue /></SelectTrigger>
   682→                  <SelectContent>
   683→                    <SelectItem value="percent">Percent off</SelectItem>
   684→                    <SelectItem value="fixed">Fixed amount off</SelectItem>
   685→                    <SelectItem value="freeship">Free shipping</SelectItem>
   686→                  </SelectContent>
   687→                </Select>
   688→              </div>
   689→            </div>
   690→            <div className="space-y-1.5">
   691→              <Label htmlFor="coupon-desc">Description</Label>
   692→              <Input id="coupon-desc" value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Shown at checkout" className="min-h-[44px]" />
   693→            </div>
   694→            <div className="grid grid-cols-2 gap-3">
   695→              {type === "percent" ? (
   696→                <div className="space-y-1.5">
   697→                  <Label htmlFor="coupon-percent">Percent (%)</Label>
   698→                  <Input id="coupon-percent" inputMode="decimal" value={percent} onChange={(e) => setPercent(e.target.value)} className="min-h-[44px]" />
   699→                </div>
   700→              ) : null}
   701→              {type === "fixed" ? (
   702→                <div className="space-y-1.5">
   703→                  <Label htmlFor="coupon-amount">Amount (RM)</Label>
   704→                  <Input id="coupon-amount" inputMode="decimal" value={amount} onChange={(e) => setAmount(e.target.value)} className="min-h-[44px]" />
   705→                </div>
   706→              ) : null}
   707→              <div className="space-y-1.5">
   708→                <Label htmlFor="coupon-min">Min spend (RM)</Label>
   709→                <Input id="coupon-min" inputMode="decimal" value={minSpend} onChange={(e) => setMinSpend(e.target.value)} className="min-h-[44px]" />
   710→              </div>
   711→              <div className="space-y-1.5">
   712→                <Label htmlFor="coupon-limit">Per-customer limit</Label>
   713→                <Input id="coupon-limit" inputMode="numeric" value={perCustomerLimit} onChange={(e) => setPerCustomerLimit(e.target.value)} className="min-h-[44px]" />
   714→              </div>
   715→            </div>
   716→            <div className="grid grid-cols-2 gap-3">
   717→              <div className="space-y-1.5">
   718→                <Label htmlFor="coupon-start">Starts at (optional)</Label>
   719→                <Input id="coupon-start" type="date" value={startsAt} onChange={(e) => setStartsAt(e.target.value)} className="min-h-[44px]" />
   720→              </div>
   721→              <div className="space-y-1.5">
   722→                <Label htmlFor="coupon-end">Ends at (optional)</Label>
   723→                <Input id="coupon-end" type="date" value={endsAt} onChange={(e) => setEndsAt(e.target.value)} className="min-h-[44px]" />
   724→              </div>
   725→            </div>
   726→          </div>
   727→          <DialogFooter>
   728→            <Button variant="outline" onClick={() => setCreateOpen(false)} disabled={creating} className="min-h-[44px] sm:min-h-0">Cancel</Button>
   729→            <Button
   730→              onClick={() => void create()}
   731→              disabled={
   732→                creating ||
   733→                !code.trim() ||
   734→                (type === "percent" && !(Number(percent) >= 0)) ||
   735→                (type === "fixed" && !(Number(amount) >= 0))
   736→              }
   737→              className="min-h-[44px] sm:min-h-0"
   738→            >
   739→              {creating ? "Creating…" : "Create coupon"}
   740→            </Button>
   741→          </DialogFooter>
   742→        </DialogContent>
   743→      </Dialog>
   744→
   745→      <ConfirmDialog
   746→        open={deleteTarget !== null}
   747→        onOpenChange={(o) => !o && setDeleteTarget(null)}
   748→        title="Delete coupon?"
   749→        description={deleteTarget ? `Code ${deleteTarget.code} will stop working immediately.` : ""}
   750→        confirmLabel="Delete"
   751→        destructive
   752→        loading={busyId !== null}
   753→        onConfirm={() => void doDelete()}
   754→      />
   755→    </div>
   756→  );
   757→}
