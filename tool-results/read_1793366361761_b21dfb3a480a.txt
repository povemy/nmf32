   215→   215→export class ApiFail extends Error {
   216→   216→  status: number;
   217→   217→  code: string;
   218→   218→  constructor(status: number, code: string, message: string) {
   219→   219→    super(message);
   220→   220→    this.status = status;
   221→   221→    this.code = code;
   222→   222→  }

   225→   225→export async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   226→   226→  const res = await fetch(url, {
   227→   227→    cache: "no-store",
   228→   228→    ...init,
   229→   229→    // Device identity headers (cart + shopper token) — keeps guest state working
   230→   230→    // where cookies are unavailable (cross-site iframe preview panels).
   231→   231→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   232→   232→  });
   233→   233→  let body: unknown = null;
   234→   234→  try {
   235→   235→    body = await res.json();
   236→   236→  } catch {
   237→   237→    /* non-json */
   238→   238→  }
   239→   239→  const b = body as { data?: T; error?: { code?: string; message?: string } } | null;
   240→   240→  if (!res.ok || !b || b.error) {
   241→   241→    // 401 → the shopper session is gone (signed out elsewhere / expired /
   242→   242→    // revoked). Signal app-root to re-check identity; guests stay guests.
   243→   243→    if (res.status === 401) signalSessionExpired();
   244→   244→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   245→   245→  }
   246→   246→  return (b.data ?? ({} as T)) as T;
   247→   247→}

   249→   249→/** Like fetchJson but also returns the paged meta envelope. */
   250→   250→export async function fetchEnvelope<T>(
   251→   251→  url: string,
   252→   252→  init?: RequestInit
   253→   253→): Promise<{ data: T; meta?: { page: number; size: number; total: number } }> {
   254→   254→  const res = await fetch(url, {
   255→   255→    cache: "no-store",
   256→   256→    ...init,
   257→   257→    headers: { ...deviceHeaders(), ...(init?.headers ?? {}) },
   258→   258→  });
   259→   259→  let body: unknown = null;
   260→   260→  try {
   261→   261→    body = await res.json();
   262→   262→  } catch {
   263→   263→    /* non-json */
   264→   264→  }
   265→   265→  const b = body as { data?: T; meta?: { page: number; size: number; total: number }; error?: { code?: string; message?: string } } | null;
   266→   266→  if (!res.ok || !b || b.error) {
   267→   267→    if (res.status === 401) signalSessionExpired();
   268→   268→    throw new ApiFail(res.status, b?.error?.code ?? "ERROR", b?.error?.message ?? `Request failed (${res.status})`);
   269→   269→  }
   270→   270→  return { data: (b.data ?? ({} as T)) as T, meta: b.meta };
   271→   271→}

   333→   333→/** Fetch the account cart and live-refresh on sm-cart-updated events + on any
   334→   334→ *  sign-in / sign-out change (carts are account-scoped — guests have none).
   335→   335→ *
   336→   336→ *  Task 28: cart state lives in a MODULE-LEVEL shared store, so every useCart()
   337→   337→ *  instance (drawer, header badge, checkout) renders from the same snapshot —
   338→   338→ *  one fetch per event instead of one per instance — and checkout-selection
   339→   339→ *  toggles can mutate it optimistically (mutateCartItems): the UI updates in
   340→   340→ *  the same frame the user clicks while the PATCH rides in the background.
   341→   341→ *  Zero network in the render path — same philosophy as the client-side
   342→   342→ *  Messages list filter. */
   343→   343→const EMPTY_CART: CartT = { id: null, storeId: null, storeSlug: null, storeName: null, items: [] };
   344→   344→
   345→   345→type CartState = { cart: CartT; loading: boolean };
   346→   346→let cartState: CartState = { cart: EMPTY_CART, loading: true };
   347→   347→const CART_SERVER_STATE: CartState = { cart: EMPTY_CART, loading: true };
   348→   348→const cartSubs = new Set<() => void>();
   349→   349→
   350→   350→function emitCart() {
   351→   351→  for (const fn of cartSubs) fn();
   352→   352→}
   353→   353→
   354→   354→function subscribeCart(fn: () => void) {
   355→   355→  cartSubs.add(fn);
   356→   356→  return () => {
   357→   357→    cartSubs.delete(fn);
   358→   358→  };
   359→   359→}
   360→   360→
   361→   361→/** Optimistically mutate the shared cart state — no network, no refetch.
   362→   362→ *  Consumers must keep the failure path honest: revert via mutateCartItems
   363→   363→ *  (guarded so it only undoes its own change) and surface a toast. */
   364→   364→export function mutateCartItems(mutate: (items: CartItemT[]) => CartItemT[]) {
   365→   365→  const cur = cartState.cart;
   366→   366→  if (!cur) return;
   367→   367→  cartState = { ...cartState, cart: { ...cur, items: mutate(cur.items) } };
   368→   368→  emitCart();
   369→   369→}
   370→   370→
   371→   371→/** Adopt a server cart payload directly into the shared store (no refetch).
   372→   372→ *  Add-to-cart responses already carry the fresh cart — feeding it here keeps
   373→   373→ *  every consumer (badge, drawer, checkout) in sync in the same frame without
   374→   374→ *  the event-driven /api/cart refetch storm. */
   375→   375→export function applyServerCart(cart: CartT | null | undefined) {
   376→   376→  if (!cart) return;
   377→   377→  cartState = { cart, loading: false };
   378→   378→  emitCart();
   379→   379→}

   382→   382→ *  drawer's stock probe building the variant-id list). */
   383→   383→export function cartSnapshot(): CartT {
   384→   384→  return cartState.cart;
   385→   385→}

   387→   387→export function useCart() {
   388→   388→  const me = useApp((s) => s.me);
   389→   389→  const authed = Boolean(me.loaded && me.shopper);
   390→   390→  const { cart, loading } = useSyncExternalStore(
   391→   391→    subscribeCart,
   392→   392→    () => cartState,
   393→   393→    () => CART_SERVER_STATE
   394→   394→  );
   395→   395→
   396→   396→  const refresh = useCallback(async () => {
   397→   397→    try {
   398→   398→      const data = await fetchJson<{ cart: CartT }>("/api/cart");
   399→   399→      cartState = { cart: data.cart ?? EMPTY_CART, loading: false };
   400→   400→    } catch {
   401→   401→      cartState = { cart: EMPTY_CART, loading: false };
   402→   402→    }
   403→   403→    emitCart();
   404→   404→  }, []);
   405→   405→
   406→   406→  useEffect(() => {
   407→   407→    // Show the loader only when there is nothing to paint yet — data-bearing
   408→   408→    // re-syncs (mounts, sign-in flips, sm-cart-updated events) stay silent so
   409→   409→    // no surface ever flashes a skeleton over live data.
   410→   410→    if (cartState.cart.items.length === 0) {
   411→   411→      cartState = { ...cartState, loading: true };
   412→   412→      emitCart();
   413→   413→    }
   414→   414→    void refresh();
   415→   415→    const onEvt = () => void refresh();
   416→   416→    window.addEventListener(CART_EVENT, onEvt);
   417→   417→    return () => window.removeEventListener(CART_EVENT, onEvt);
   418→   418→  }, [refresh, authed]);
   419→   419→
   420→   420→  const count = cart.items.reduce((s, i) => s + i.qty, 0);
   421→   421→  const subtotal = cart.items.reduce((s, i) => s + i.lineTotalMinor, 0);
   422→   422→  // Selection-aware figures — only checked lines proceed to checkout.
   423→   423→  const selectedItems = cart.items.filter((i) => i.selected);
   424→   424→  const selectedCount = selectedItems.reduce((s, i) => s + i.qty, 0);
   425→   425→  const selectedSubtotal = selectedItems.reduce((s, i) => s + i.lineTotalMinor, 0);
   426→   426→  return { cart, count, subtotal, selectedCount, selectedSubtotal, loading, refresh };
   427→   427→}

   438→   438→export function ensureFontsLoaded(ids: (string | null | undefined)[]) {

   440→   440→  const href = fontLinkHref(ids);

   442→   442→  if (!document.querySelector(`link[data-sm-fonts="${href}"]`)) {
   443→   443→    const link = document.createElement("link");
   444→   444→    link.rel = "stylesheet";
   445→   445→    link.href = href;

   455→   455→export function Stars({ rating, count, className }: { rating: number; count?: number; className?: string }) {
   456→   456→  const full = Math.floor(rating);
   457→   457→  const half = rating - full >= 0.25 && rating - full < 0.75;
   458→   458→  const rounded = rating - full >= 0.75 ? full + 1 : full;
   459→   459→  return (
   460→   460→    <span className={cn("inline-flex items-center gap-0.5", className)} aria-label={`Rated ${rating.toFixed(1)} out of 5`}>
   461→   461→      {[0, 1, 2, 3, 4].map((i) =>
   462→   462→        i < full || (i === full && rounded > full && !half) ? (
   463→   463→          <Star key={i} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   464→   464→        ) : i === full && half ? (
