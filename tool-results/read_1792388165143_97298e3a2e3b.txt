    93→/** Total unread seller messages — powers the "My messages" header badge.
    94→ *  Polls lightly + refreshes when the messages view is opened/left. */
    95→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    96→  const [unread, setUnread] = useState(0);
    97→  useEffect(() => {
    98→    if (!shopperSignedIn) {
    99→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
   100→      setUnread(0);
   101→      return;
   102→    }
   103→    let alive = true;
   104→    const load = () =>
   105→      fetchJson<{ total: number }>("/api/chat/unread")
   106→        .then((d) => alive && setUnread(d.total ?? 0))
   107→        .catch(() => {});
   108→    load();
   109→    const t = setInterval(load, 15000);
   110→    return () => {
   111→      alive = false;
   112→      clearInterval(t);
   113→    };
   114→  }, [shopperSignedIn]);
   115→  // instant re-sync when a thread is read inside the messages view (the view
   116→  // dispatches "nomofee:chat-sync" instead of prop-drilling a callback)
   117→  useEffect(() => {
   118→    if (!shopperSignedIn) return;
   119→    const sync = () =>
   120→      fetchJson<{ total: number }>("/api/chat/unread")
   121→        .then((d) => setUnread(d.total ?? 0))
   122→        .catch(() => {});
   123→    window.addEventListener("nomofee:chat-sync", sync);
   124→    return () => window.removeEventListener("nomofee:chat-sync", sync);
   125→  }, [shopperSignedIn]);
   126→  // opening (or leaving) the messages view should immediately re-sync the badge
   127→  useEffect(() => {
   128→    if (!shopperSignedIn) return;
   129→    const t = setTimeout(
   130→      () =>
   131→        fetchJson<{ total: number }>("/api/chat/unread")
   132→          .then((d) => setUnread(d.total ?? 0))
   133→          .catch(() => {}),
   134→      viewName === "messages" ? 600 : 0
   135→    );
   136→    return () => clearTimeout(t);
   137→  }, [shopperSignedIn, viewName]);
   138→  return unread;
   139→}

   148→  const [cartOpen, setCartOpen] = useState(false);
   149→  const [headerSearch, setHeaderSearch] = useState("");
   150→  // Back-navigation history stack (user request: back button on every page).
   151→  // viewRef mirrors `view` so nav() can push the PREVIOUS view without reading
   152→  // state inside the setState updater (StrictMode-safe).

   154→  const viewRef = useRef<ShopperView>({ name: "market" });
   155→  // Deep-link override for claim landing (?claim_job=…&claim_token=…):
   156→  // intentionally NOT part of ShopperView — a synthetic render override.
   157→  const [claimLink, setClaimLink] = useState<{ jobId: string; token: string } | null>(null);
   158→  const restored = useRef(false);
   159→
   160→  // ---- mount: activeStoreSlug override, else restore last view --------------
   161→  useEffect(() => {
   162→    if (restored.current) return;
   163→    restored.current = true;
   164→    // claim deep-link wins over everything
   165→    try {
   166→      const params = new URLSearchParams(window.location.search);
   167→      const cj = params.get("claim_job");
   168→      const ct = params.get("claim_token");
   169→      if (cj && ct) {
   170→        // eslint-disable-next-line react-hooks/set-state-in-effect -- intentional deep-link reset
   171→        setClaimLink({ jobId: cj, token: ct });
   172→        setReady(true);
   173→        return;
   174→      }
   175→      // SEO deep links (sitemap): /?sf={storeSlug} and /?sf={storeSlug}&sp={productSlug}
   176→      const sf = params.get("sf");
   177→      if (sf) {
   178→        const sp = params.get("sp");
   179→        setView(sp ? { name: "pdp", slug: sf, pSlug: sp } : { name: "storefront", slug: sf });
   180→        setReady(true);
   181→        return;
   182→      }
   183→    } catch {
   184→      /* ignore */
   185→    }
   186→    if (activeStoreSlug) {
   187→      setView({ name: "storefront", slug: activeStoreSlug });
   188→      setActiveStoreSlug(null);
   189→    } else {
   190→      try {
   191→        const raw = window.localStorage.getItem(VIEW_KEY);
   192→        if (raw) {
   193→          const v = JSON.parse(raw) as ShopperView;
   194→          if (v && typeof v.name === "string") setView(v);
   195→        }
   196→      } catch {
   197→        /* ignore */
   198→      }
   199→    }
   200→    setReady(true);
   201→  }, [activeStoreSlug, setActiveStoreSlug]);
   202→
   203→  // ---- nav helper: persist + scroll to top ----------------------------------
   204→  useEffect(() => {
   205→    viewRef.current = view;
   206→  }, [view]);
   207→
   208→  const nav = useCallback((v: ShopperView) => {
   209→    if (viewRef.current.name !== v.name || JSON.stringify(viewRef.current) !== JSON.stringify(v)) {
   210→      viewStack.current.push(viewRef.current);
   211→      if (viewStack.current.length > 25) viewStack.current.shift();
   212→    }
   213→    setView(v);
   214→    try {
   215→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(v));
   216→    } catch {
   217→      /* ignore */
   218→    }
   219→    window.scrollTo({ top: 0, behavior: "instant" as ScrollBehavior });
   220→  }, []);
   221→
   222→  const back = useCallback(() => {
   223→    const prev = viewStack.current.pop() ?? { name: "market" as const };
   224→    setView(prev);
   225→    try {
   226→      window.localStorage.setItem(VIEW_KEY, JSON.stringify(prev));

   238→      setView({ name: "storefront", slug: activeStoreSlug });
   239→      setActiveStoreSlug(null);
   240→    }
   241→  }, [activeStoreSlug, ready, setActiveStoreSlug]);
   242→
   243→  const ctx = useMemo<ShopperCtx>(() => ({ nav, back, openCart: () => setCartOpen(true) }), [nav, back]);

   245→  const showChrome = !isStoreChrome(view);

"" : view.name === "me" ? view.tab ?? "" : ""}`}

   271→          className={cn("flex-1 flex flex-col", showBottomNav && "pb-[calc(4.25rem+env(safe-area-inset-bottom))] md:pb-0")}
   272→        >
   273→          {claimLink ? null : (
   274→            <>
   275→              {view.name === "market" ? <MarketHome ctx={ctx} /> : null}
   276→              {view.name === "search" ? <ResultsView ctx={ctx} mode={{ kind: "search", q: view.q }} /> : null}
   277→              {view.name === "category" ? (
   278→                <ResultsView ctx={ctx} mode={{ kind: "category", categorySlug: view.categorySlug, label: view.label }} />
   279→              ) : null}
   280→              {view.name === "storefront" ? <StorefrontView ctx={ctx} slug={view.slug} /> : null}
   281→              {view.name === "store-page" ? <StorePageView ctx={ctx} slug={view.slug} pageSlug={view.pageSlug} /> : null}
   282→              {view.name === "pdp" ? <PdpView ctx={ctx} slug={view.slug} pSlug={view.pSlug} /> : null}

   286→              {view.name === "orders" ? <OrdersView ctx={ctx} /> : null}
   287→              {view.name === "generate" ? <GenerateStoreView ctx={ctx} /> : null}
   288→              {view.name === "lookup" ? <LookupView ctx={ctx} /> : null}
   289→              {view.name === "messages" ? <MessagesView ctx={ctx} initialThreadId={view.threadId} /> : null}
   290→              {view.name === "me" ? <MeHubView ctx={ctx} initialTab={view.tab} /> : null}

   312→      ) : null}
   313→
   314→      <CartDrawer open={cartOpen} onOpenChange={setCartOpen} ctx={ctx} />
   315→    </div>

   323→function BottomNav({ view, nav, openCart }: { view: ShopperView; nav: (v: ShopperView) => void; openCart: () => void }) {
   324→  const me = useApp((s) => s.me);
   325→  const openShopperAuth = useApp((s) => s.openShopperAuth);
   326→  const signedIn = Boolean(me.loaded && me.shopper);
   327→  const { count } = useCart();
   328→  const unreadChats = useChatUnread(signedIn, view.name);
   329→
   330→  const items = [
   331→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },
   332→    {
   333→      key: "orders",
   334→      label: "Orders",
   335→      icon: Package,
   336→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   337→      badge: 0,
   338→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   339→    },
   340→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   341→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   342→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },
   343→  ];
   344→
   345→  return (
   346→    <nav
   347→      aria-label="Bottom navigation"
   348→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   349→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   350→    >
   351→      <div className="mx-auto grid max-w-lg grid-cols-5">
   352→        {items.map((it) => {
   353→          const Icon = it.icon;
   354→          return (
   355→            <button
   356→              key={it.key}
   357→              type="button"
   358→              onClick={it.onClick}
   359→              aria-label={it.label}
   360→              aria-current={it.active ? "page" : undefined}
   361→              className={cn(
   362→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   363→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   364→              )}
   365→            >
   366→              <span className="relative">
   367→                <Icon className="h-5 w-5" aria-hidden="true" />
   368→                {it.badge > 0 ? (
   369→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   370→                    {it.badge > 9 ? "9+" : it.badge}
   371→                  </span>
   372→                ) : null}
   373→              </span>
   374→              {it.label}
   375→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   376→            </button>
   377→          );
   378→        })}
   379→      </div>
   380→    </nav>
   381→  );
   382→}
