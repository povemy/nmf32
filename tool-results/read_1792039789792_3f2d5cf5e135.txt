    81→/** Total unread seller messages — powers the "My messages" header badge.
    82→ *  Polls lightly + refreshes when the messages view is opened/left. */
    83→function useChatUnread(shopperSignedIn: boolean, viewName: ShopperView["name"]) {
    84→  const [unread, setUnread] = useState(0);
    85→  useEffect(() => {
    86→    if (!shopperSignedIn) {
    87→      // eslint-disable-next-line react-hooks/set-state-in-effect -- reset the badge on sign-out (no fetch involved)
    88→      setUnread(0);
    89→      return;
    90→    }
    91→    let alive = true;
    92→    const load = () =>
    93→      fetchJson<{ total: number }>("/api/chat/unread")
    94→        .then((d) => alive && setUnread(d.total ?? 0))
    95→        .catch(() => {});
    96→    load();
    97→    const t = setInterval(load, 15000);
    98→    return () => {
    99→      alive = false;
   100→      clearInterval(t);
   101→    };
   102→  }, [shopperSignedIn]);
   103→  // instant re-sync when a thread is read inside the messages view (the view
   104→  // dispatches "shopmesh:chat-sync" instead of prop-drilling a callback)
   105→  useEffect(() => {
   106→    if (!shopperSignedIn) return;
   107→    const sync = () =>
   108→      fetchJson<{ total: number }>("/api/chat/unread")
   109→        .then((d) => setUnread(d.total ?? 0))
   110→        .catch(() => {});
   111→    window.addEventListener("shopmesh:chat-sync", sync);
   112→    return () => window.removeEventListener("shopmesh:chat-sync", sync);
   113→  }, [shopperSignedIn]);
   114→  // opening (or leaving) the messages view should immediately re-sync the badge
   115→  useEffect(() => {
   116→    if (!shopperSignedIn) return;
   117→    const t = setTimeout(
   118→      () =>
   119→        fetchJson<{ total: number }>("/api/chat/unread")
   120→          .then((d) => setUnread(d.total ?? 0))
   121→          .catch(() => {}),
   122→      viewName === "messages" ? 600 : 0
   123→    );
   124→    return () => clearTimeout(t);
   125→  }, [shopperSignedIn, viewName]);
   126→  return unread;
   127→}

   326→  useEffect(() => {

   334→      });
   335→    return () => {
   336→      alive = false;
   337→    };
   338→  }, []);
   339→  return campaign;

   433→

   435→    { key: "home", label: "Home", icon: ShoppingBag, active: view.name === "market", badge: 0, onClick: () => nav({ name: "market" }) },

   438→      label: "Orders",
   439→      icon: Package,
   440→      active: view.name === "orders" || (view.name === "me" && view.tab === "purchases"),
   441→      badge: 0,
   442→      onClick: () => (signedIn ? nav({ name: "me", tab: "purchases" }) : openShopperAuth()),
   443→    },
   444→    { key: "cart", label: "Cart", icon: ShoppingCart, active: false, badge: count, onClick: openCart },
   445→    { key: "chat", label: "Chat", icon: MessagesSquare, active: view.name === "messages", badge: unreadChats, onClick: () => (signedIn ? nav({ name: "messages" }) : openShopperAuth()) },
   446→    { key: "me", label: "Me", icon: User, active: view.name === "me", badge: 0, onClick: () => nav({ name: "me" }) },

   449→  return (
   450→    <nav
   451→      aria-label="Bottom navigation"
   452→      className="fixed inset-x-0 bottom-0 z-40 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85 md:hidden"
   453→      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
   454→    >
   455→      <div className="mx-auto grid max-w-lg grid-cols-5">
   456→        {items.map((it) => {
   457→          const Icon = it.icon;
   458→          return (
   459→            <button
   460→              key={it.key}
   461→              type="button"
   462→              onClick={it.onClick}
   463→              aria-label={it.label}
   464→              aria-current={it.active ? "page" : undefined}
   465→              className={cn(
   466→                "relative flex h-16 min-h-[56px] flex-col items-center justify-center gap-0.5 text-[10px] font-semibold transition-colors",
   467→                it.active ? "text-teal-800" : "text-muted-foreground hover:text-foreground"
   468→              )}
   469→            >
   470→              <span className="relative">
   471→                <Icon className="h-5 w-5" aria-hidden="true" />
   472→                {it.badge > 0 ? (
   473→                  <span className="absolute -right-2 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-500 px-1 text-[9px] font-extrabold text-white shadow" aria-hidden>
   474→                    {it.badge > 9 ? "9+" : it.badge}

   476→                ) : null}
   477→              </span>
   478→              {it.label}
   479→              {it.active ? <span className="absolute inset-x-5 top-0 h-0.5 rounded-full bg-brand-gradient" aria-hidden="true" /> : null}
   480→            </button>
   481→          );
   482→        })}
   483→      </div>
   484→    </nav>
   485→  );
   486→}

   492→function MarketHeader({

   502→  search: string;

   509→  const shopperSignedIn = Boolean(me.loaded && me.shopper);
   510→  const unreadChats = useChatUnread(shopperSignedIn, view.name);
   511→

   513→    let alive = true;

   515→      .then((d) => alive && setCategories(d.categories ?? []))

   522→  function submitSearch() {

   936→function timeAgoShort(iso: string): string {
   937→  const diff = Date.now() - new Date(iso).getTime();
   938→  const m = Math.floor(diff / 60000);
   939→  if (m < 1) return "now";
   940→  if (m < 60) return `${m}m`;
   941→  const h = Math.floor(m / 60);
   942→  if (h < 24) return `${h}h`;
   943→  const d = Math.floor(h / 24);
   944→  if (d < 7) return `${d}d`;
   945→  return new Date(iso).toLocaleDateString(undefined, { day: "numeric", month: "short" });
   946→}
