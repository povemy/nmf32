   185→/** Deterministic parcel estimate: 150g packaging + 120g per ordered unit. */
   186→export function parcelWeightForItems(items: { qty: number }[]): number {
   187→  const units = items.reduce((s, i) => s + Math.max(0, Math.floor(i.qty)), 0);
   188→  return 150 + units * 120;
   189→}

   191→function bandPriceMinor(spec: CourierSpec, weightGrams: number, zone: ShippingZone): number {
   192→  let base = spec.bandsPen[spec.bandsPen.length - 1][1];
   193→  for (const [ceiling, price] of spec.bandsPen) {
   194→    if (weightGrams <= ceiling) {
   195→      base = price;
   196→      break;
   197→    }
   198→  }
   199→  const lastCeiling = spec.bandsPen[spec.bandsPen.length - 1][0];
   200→  if (weightGrams > lastCeiling) {
   201→    const extraKg = Math.ceil((weightGrams - lastCeiling) / 1000);
   202→    base += extraKg * spec.perKgBeyondPen;
   203→  }
   204→  return Math.round(base * (zone === "east" ? spec.eastSurcharge : 1));
   205→}

   207→/** Quote 3 Malaysian courier options — deterministic by (courier, weight, zone). */
   208→export function quoteRates(parcel: ShippingParcel, dest: ShippingDestination): ShippingQuote[] {
   209→  const zone = zoneOf(dest.state);
   210→  return COURIERS.map((spec) => ({
   211→    rateId: `${MOCK_PROVIDER_ID}:${spec.code}:${spec.serviceLevel}`,
   212→    providerId: MOCK_PROVIDER_ID,
   213→    courierName: spec.name,
   214→    serviceLevel: spec.serviceLevel,
   215→    priceMinor: bandPriceMinor(spec, Math.max(1, Math.round(parcel.weightGrams)), zone),
   216→    etaFromDays: spec.etaFromDays + (zone === "east" ? 1 : 0),
   217→    etaToDays: spec.etaToDays + (zone === "east" ? 3 : 0),
   218→  }));
   219→}

   229→const AWB_DIGITS = 10;

   231→/** `PLJ` + 10 digits — per-courier prefix (Pos Laju style) + hashed digits. */
   232→export function awbFor(orderNumber: string, courierCode: string): string {
   233→  const prefix =
   234→    {
   235→      poslaju: "PLJ",
   236→      jt: "JTE",
   237→      citylink: "CLX",
   238→    }[courierCode] ?? "MYX";
   239→  const h = hashStr(`${orderNumber}:${courierCode}:${Math.floor(Date.now() / 1000)}`);
   240→  const digits = String(h % 10_000_000_000).padStart(AWB_DIGITS, "0");
   241→  return `${prefix}${digits}`;
   242→}
   243→
   244→/** Barcode-ish deterministic vertical bars for an AWB number. */
   245→function barcodeBars(awb: string, x0: number, y0: number, h: number): string {
   246→  const rects: string[] = [];
   247→  let x = x0;
   248→  for (let i = 0; i < 44; i++) {
   249→    const bit = hashStr(`${awb}:${i}`) % 3; // 0 thin, 1 mid, 2 thick
   250→    const w = 2 + bit * 2;
   251→    if (i % 2 === 0) rects.push(`<rect x="${x}" y="${y0}" width="${w}" height="${h}" fill="#111111"/>`);
   252→    x += w + 2;
   253→  }
   254→  return rects.join("");
   255→}

   368→/**
   369→ * Book the chosen quote for an order: creates/refreshes the Shipment (BOOKED),
   370→ * stamps the AWB, generates the SVG label (inline data-URL) and writes the
   371→ * first BOOKED ShipmentEvent. Deterministic + offline.
   372→ */
   373→export async function bookShipment(
   374→  order: BookingOrder,
   375→  quote: ShippingQuote
   376→): Promise<ShipmentSnapshot> {
   377→  const dest = parseAddress(order.shippingAddressJson);
   378→  const now = new Date();
   379→  const awb = awbFor(order.orderNumber, quote.rateId.split(":")[1] ?? "myx");
   380→  const labelSvg = labelSvgFor({
   381→    awbNumber: awb,
   382→    courierName: quote.courierName,
   383→    orderNumber: order.orderNumber,
   384→    storeName: order.storeName,
   385→    destCity: cityForState(dest.state),
   386→    destState: dest.state || "Malaysia",
   387→    recipient: order.customerName,
   388→    etaText: etaTextFor(quote),
   389→  });
   390→
   391→  const data = {
   392→    storeId: order.storeId,
   393→    providerId: MOCK_PROVIDER_ID,
   394→    providerRef: awb,
   395→    status: "BOOKED",
   396→    awbNumber: awb,
   397→    courierName: quote.courierName,
   398→    trackingUrl: null,
   399→    labelUrl: svgDataUrl(labelSvg),
   400→    bookedAt: now,
   401→    pickedUpAt: null,
   402→    etaFrom: new Date(now.getTime() + quote.etaFromDays * 24 * 3600 * 1000),
   403→    etaTo: new Date(now.getTime() + quote.etaToDays * 24 * 3600 * 1000),
   404→    deliveredAt: null,
   405→    lastSyncedAt: now,
   406→  };
   407→
   408→  const shipment = await db.shipment.upsert({
   409→    where: { orderId: order.id },
   410→    create: { orderId: order.id, ...data },
   411→    update: { ...data },
   412→    include: { order: { select: { shopperId: true } } },
   413→  });
   414→
   415→  // Fresh tracking history for a new booking (upsert replaces stale rows).
   416→  await db.shipmentEvent.deleteMany({ where: { shipmentId: shipment.id } });
   417→  await db.shipmentEvent.create({
   418→    data: {
   419→      shipmentId: shipment.id,
   420→      status: "BOOKED",
   421→      description: `Shipping label created — parcel registered by ${quote.courierName} for ${order.storeName}`,
   422→      location: `${order.storeName}, ${cityForState(dest.state)}`,
   423→      occurredAt: now,
   424→    },
   425→  });
   426→
   427→  return shapeShipment(shipment);
   428→}

   500→    include: { order: { select: { id: true, orderNumber: true, status: true, customerEmail: true, shopperId: true, storeId: true, shippingAddressJson: true, store: { select: { slug: true } } } } },
   501→  });
   502→  if (!shipment) return null;
   503→  if (!ACTIVE_SHIPMENT_STATUSES.includes(shipment.status as (typeof ACTIVE_SHIPMENT_STATUSES)[number])) {
   504→    return shapeShipment(shipment);
   505→  }
   506→
   507→  const next = TRACK_FLOW[shipment.status];
   508→  const now = new Date();
   509→  const awb = shipment.awbNumber ?? "UNKNOWN";
   510→  const destState = parseAddress(shipment.order.shippingAddressJson ?? "{}").state ?? "";
   511→  const destCity = cityForState(destState);
   512→
   513→  const steps: Record<string, { description: string; location: string; patch: Record<string, unknown> }> = {
   514→    PICKED_UP: {
   515→      description: `Parcel collected by ${shipment.courierName ?? "courier"} from the seller`,
   516→      location: pick(zoneOf(destState) === "east" ? HUBS_EAST : HUBS_PENINSULAR, `${awb}:pickup`),
   517→      patch: { pickedUpAt: now },
   518→    },
   519→    IN_TRANSIT: {
   520→      description: "Arrived at transit facility — departing on the next scheduled trailer",
   521→      location: pick(HUBS_TRANSIT, `${awb}:transit`),
   522→      patch: {},
   523→    },
   524→    OUT_FOR_DELIVERY: {
   525→      description: "Out for delivery — your courier is on the way",
   526→      location: `${destCity} Delivery Base`,
   527→      patch: {},
   528→    },
   529→    DELIVERED: {
   530→      description: "Delivered — signed by recipient at the delivery address",
   531→      location: destCity,
   532→      patch: { deliveredAt: now },
   533→    },
   534→  };
   535→  const step = steps[next];
   536→
   537→  await db.shipmentEvent.create({
   538→    data: {
   539→      shipmentId: shipment.id,
   540→      status: next,
   541→      description: step.description,
   542→      location: step.location,
   543→      occurredAt: now,
   544→      ...(next === "DELIVERED" ? { proofImageUrl: mockPodUrl(awb) } : {}),
   545→    },
   546→  });
   547→
   548→  const updated = await db.shipment.update({
   549→    where: { id: shipment.id },
   550→    data: { status: next, lastSyncedAt: now, ...step.patch },
   551→    include: { order: { select: { id: true, orderNumber: true, status: true, customerEmail: true, shopperId: true, storeId: true, store: { select: { slug: true } } } } },
   552→  });
   553→
   554→  if (next === "DELIVERED") {
   555→    if (["shipped", "processing"].includes(updated.order.status)) {
   556→      await transitionOrder({
   557→        orderId: updated.order.id,
   558→        toStatus: "delivered",
   559→        actor: "courier",
   560→        note: `Delivered by ${updated.courierName ?? "courier"} (mock network) · AWB ${awb}`,
   561→        timestamps: { deliveredAt: now },
   562→      });
   563→    }
   564→    await deliverySideEffects({
   565→      order: updated.order,
   566→      storeSlug: updated.order.store?.slug ?? null,
   567→      courierName: updated.courierName,
   568→      awbNumber: awb,
   569→      note: `Delivered by ${updated.courierName ?? "courier"}`,
   570→      deliveredAt: now,
   571→    });
   572→  }
   573→
   574→  return shapeShipment(updated);
   575→}

   577→/**
   578→ * Lazy sync for read endpoints: only advances when the shipment has been idle
   579→ * for longer than `minIntervalMs` (default 45s — the dev demo cadence).
   580→ * Never throws: lazy side-roads must not break the primary read.
   581→ */
   582→export async function maybeSyncTracking(
   583→  shipmentId: string,
   584→  minIntervalMs = 45_000
   585→): Promise<ShipmentSnapshot | null> {
   586→  try {
   587→    const s = await db.shipment.findUnique({
   588→      where: { id: shipmentId },
   589→      select: { id: true, status: true, lastSyncedAt: true },
   590→    });
   591→    if (!s || !ACTIVE_SHIPMENT_STATUSES.includes(s.status as (typeof ACTIVE_SHIPMENT_STATUSES)[number])) return null;
   592→    if (s.lastSyncedAt && Date.now() - s.lastSyncedAt.getTime() < minIntervalMs) return null;
   593→    return await syncTracking(shipmentId);
   594→  } catch (e) {
   595→    console.error("[shipping] lazy sync failed", e);
   596→    return null;
   597→  }
   598→}

   600→// ---------------------------------------------------------------------------
   601→// Tick — advance ALL active mock shipments one step (cron-ish sweep)
   602→// ---------------------------------------------------------------------------
   603→
   604→export async function tickShipments(limit = 100): Promise<{ advanced: number; results: { shipmentId: string; orderId: string; awbNumber: string | null; status: string }[] }> {
   605→  const rows = await db.shipment.findMany({
   606→    where: { providerId: MOCK_PROVIDER_ID, status: { in: [...ACTIVE_SHIPMENT_STATUSES] } },
   607→    select: { id: true },
   608→    orderBy: { createdAt: "asc" },
   609→    take: limit,
   610→  });
   611→  const results: { shipmentId: string; orderId: string; awbNumber: string | null; status: string }[] = [];
   612→  for (const r of rows) {
   613→    const s = await syncTracking(r.id);
   614→    if (s) results.push({ shipmentId: s.id, orderId: s.orderId, awbNumber: s.awbNumber, status: s.status });
   615→  }
   616→  return { advanced: results.length, results };
   617→}

   619→// ---------------------------------------------------------------------------
   620→// Fast-forward — manual "mark delivered" backfill
   621→// ---------------------------------------------------------------------------
   622→
   623→/**
   624→ * Seller marks delivered manually while a platform shipment exists: backfill
   625→ * the skipped milestone events between bookedAt and deliveredAt (strictly
   626→ * increasing, never OUT_FOR_DELIVERY after DELIVERED), append the DELIVERED
   627→ * event with the proof photo, flip shipment + order states.
   628→ */
   629→export async function fastForwardDelivery(
   630→  orderId: string,
   631→  deliveredAt: Date,
   632→  proofUrl?: string | null
   633→): Promise<ShipmentSnapshot | null> {
   634→  const shipment = await db.shipment.findUnique({
   635→    where: { orderId },
   636→    include: {
   637→      order: {
   638→        select: {
   639→          id: true, orderNumber: true, status: true, customerEmail: true, shopperId: true, storeId: true,
   640→          shippingAddressJson: true,
   641→          store: { select: { slug: true } },
   642→        },
   643→      },
   644→    },
   645→  });
   646→  if (!shipment) return null;
   647→
   648→  const awb = shipment.awbNumber ?? "UNKNOWN";
   649→  const destState = parseAddress(shipment.order.shippingAddressJson ?? "{}").state ?? "";
   650→  const destCity = cityForState(destState);
   651→  const booked = shipment.bookedAt ?? shipment.createdAt;
   652→  const span = Math.max(60_000, deliveredAt.getTime() - booked.getTime());
   653→  const at = (frac: number) => new Date(booked.getTime() + Math.round(span * frac));
   654→
   655→  const existing = await db.shipmentEvent.findMany({
   656→    where: { shipmentId: shipment.id },
   657→    select: { status: true },
   658→  });
   659→  const have = new Set(existing.map((e) => e.status));
   660→
   661→  // Backfill plan, ordered, timestamps clamped to [booked, deliveredAt].
   662→  const plan: { status: string; frac: number; description: string; location: string; proof?: boolean }[] = [];
   663→  if (!have.has("BOOKED"))
   664→    plan.push({
   665→      status: "BOOKED",
   666→      frac: 0,
   667→      description: `Shipping label created — parcel registered by ${shipment.courierName ?? "courier"}`,
   668→      location: `Seller warehouse, ${destCity}`,
   669→    });
   670→  if (!have.has("PICKED_UP"))
   671→    plan.push({
   672→      status: "PICKED_UP",
   673→      frac: 0.12,
   674→      description: `Parcel collected by ${shipment.courierName ?? "courier"} from the seller`,
   675→      location: pick(zoneOf(destState) === "east" ? HUBS_EAST : HUBS_PENINSULAR, `${awb}:pickup`),
   676→    });
   677→  if (!have.has("IN_TRANSIT"))
   678→    plan.push({
   679→      status: "IN_TRANSIT",
   680→      frac: 0.4,
   681→      description: "Arrived at transit facility — line-haul departed on time",
   682→      location: pick(HUBS_TRANSIT, `${awb}:transit`),
   683→    });
   684→  if (!have.has("OUT_FOR_DELIVERY"))
   685→    plan.push({
   686→      status: "OUT_FOR_DELIVERY",
   687→      frac: 0.9,
   688→      description: "Out for delivery — courier completing the neighbourhood route",
   689→      location: `${destCity} Delivery Base`,
   690→    });
   691→  plan.push({
   692→    status: "DELIVERED",
   693→    frac: 1,
   694→    description: "Delivered — signed by recipient at the delivery address",
   695→    location: destCity,
   696→    proof: true,
   697→  });
   698→
   699→  let prev = booked.getTime() - 1;
   700→  for (const p of plan) {
   701→    let ts = Math.min(Math.max(at(p.frac).getTime(), prev + 1), deliveredAt.getTime());
   702→    if (p.status === "DELIVERED") ts = deliveredAt.getTime();
   703→    await db.shipmentEvent.create({
   704→      data: {
   705→        shipmentId: shipment.id,
   706→        status: p.status,
   707→        description: p.description,
   708→        location: p.location,
   709→        occurredAt: new Date(ts),
   710→        ...(p.proof ? { proofImageUrl: proofUrl ?? mockPodUrl(awb) } : {}),
   711→      },
   712→    });
   713→    prev = ts;
   714→  }
   715→
   716→  const updated = await db.shipment.update({
   717→    where: { id: shipment.id },
   718→    data: {
   719→      status: "DELIVERED",
   720→      deliveredAt,
   721→      lastSyncedAt: new Date(),
   722→      pickedUpAt: shipment.pickedUpAt ?? at(0.12),
   723→    },
   724→    include: { order: { select: { id: true, orderNumber: true, status: true, customerEmail: true, shopperId: true, storeId: true, store: { select: { slug: true } } } } },
   725→  });
   726→
   727→  if (["shipped", "processing"].includes(updated.order.status)) {
   728→    await transitionOrder({
   729→      orderId: updated.order.id,
   730→      toStatus: "delivered",
   731→      actor: "seller",
   732→      note: "Seller marked the parcel delivered",
   733→      timestamps: { deliveredAt },
   734→    });
   735→  }
   736→  await deliverySideEffects({
   737→    order: updated.order,
   738→    storeSlug: updated.order.store?.slug ?? null,
   739→    courierName: updated.courierName,
   740→    awbNumber: awb,
   741→    note: `Delivered by ${updated.courierName ?? "the seller"}`,
   742→    deliveredAt,
   743→  });
   744→
   745→  return shapeShipment(updated);
   746→}

   752→/** Ship-guard parity helper: throws ApiError when the order cannot be shipped. */
   753→export function assertShippable(order: { status: string; orderNumber: string }): void {
   754→  if (!["paid", "processing"].includes(order.status)) {
   755→    throw new ApiError(409, "INVALID_STATUS", `Cannot ship order ${order.orderNumber} in status '${order.status}'`);
   756→  }
   757→}

   759→/** Validate a seller-uploaded proof-of-delivery data-URL (≤1.2MB). */
   760→export function validateProofDataUrl(proof: unknown): string | null {
   761→  if (proof == null) return null;
   762→  const s = String(proof);
   763→  if (!s.startsWith("data:image/")) {
   764→    throw new ApiError(400, "VALIDATION", "proof must be a data:image/* URL");
   765→  }
   766→  if (s.length > 1_200_000) {
   767→    throw new ApiError(413, "PROOF_TOO_LARGE", "proof image must be 1.2MB or smaller");
   768→  }
   769→  return s;
   770→}
